66 template<class TGeometryType>
67 static inline typename TGeometryType::CoordinatesArrayType& PointLocalCoordinatesPlanarFaceTetrahedra(
68 const TGeometryType& rGeometry,
69 typename TGeometryType::CoordinatesArrayType& rResult,
70 const typename TGeometryType::CoordinatesArrayType& rPoint
71 )
72 {
73 // Debug check that it is at least a tetrahedra
74 KRATOS_DEBUG_ERROR_IF_NOT(rGeometry.GetGeometryFamily() == GeometryData::KratosGeometryFamily::Kratos_Tetrahedra) << "Geometry should be a tetrahedra in order to use PointLocalCoordinatesPlanarFaceTetrahedra" << std::endl;
75
76 // Compute RHS
77 array_1d<double,4> X;
78 X[0] = 1.0;
79 X[1] = rPoint[0];
80 X[2] = rPoint[1];
81 X[3] = rPoint[2];
82
83 // Auxiliary coordinates
84 const auto& r_coordinates_0 = rGeometry[0].Coordinates();
85 const auto& r_coordinates_1 = rGeometry[1].Coordinates();
86 const auto& r_coordinates_2 = rGeometry[2].Coordinates();
87 const auto& r_coordinates_3 = rGeometry[3].Coordinates();
88 const double x1 = r_coordinates_0[0];
89 const double y1 = r_coordinates_0[1];
90 const double z1 = r_coordinates_0[2];
91 const double x2 = r_coordinates_1[0];
92 const double y2 = r_coordinates_1[1];
93 const double z2 = r_coordinates_1[2];
94 const double x3 = r_coordinates_2[0];
95 const double y3 = r_coordinates_2[1];
96 const double z3 = r_coordinates_2[2];
97 const double x4 = r_coordinates_3[0];
98 const double y4 = r_coordinates_3[1];
99 const double z4 = r_coordinates_3[2];
100
101 // Auxiliary diff
102 const double x12 = x1 - x2;
103 const double x13 = x1 - x3;
104 const double x14 = x1 - x4;
105 const double x21 = x2 - x1;
106 const double x24 = x2 - x4;
107 const double x31 = x3 - x1;
108 const double x32 = x3 - x2;
109 const double x34 = x3 - x4;
110 const double x42 = x4 - x2;
111 const double x43 = x4 - x3;
112 const double y12 = y1 - y2;
113 const double y13 = y1 - y3;
114 const double y14 = y1 - y4;
115 const double y21 = y2 - y1;
116 const double y24 = y2 - y4;
117 const double y31 = y3 - y1;
118 const double y32 = y3 - y2;
119 const double y34 = y3 - y4;
120 const double y42 = y4 - y2;
121 const double y43 = y4 - y3;
122 const double z12 = z1 - z2;
123 const double z13 = z1 - z3;
124 const double z14 = z1 - z4;
125 const double z21 = z2 - z1;
126 const double z24 = z2 - z4;
127 const double z31 = z3 - z1;
128 const double z32 = z3 - z2;
129 const double z34 = z3 - z4;
130 const double z42 = z4 - z2;
131 const double z43 = z4 - z3;
132
133 // Compute LHS
134 BoundedMatrix<double, 4,4> invJ;
135 const double aux_volume = 1.0/(6.0*rGeometry.Volume());
136 invJ(0,0) = aux_volume * (x2*(y3*z4-y4*z3)+x3*(y4*z2-y2*z4)+x4*(y2*z3-y3*z2));
137 invJ(1,0) = aux_volume * (x1*(y4*z3-y3*z4)+x3*(y1*z4-y4*z1)+x4*(y3*z1-y1*z3));
138 invJ(2,0) = aux_volume * (x1*(y2*z4-y4*z2)+x2*(y4*z1-y1*z4)+x4*(y1*z2-y2*z1));
139 invJ(3,0) = aux_volume * (x1*(y3*z2-y2*z3)+x2*(y1*z3-y3*z1)+x3*(y2*z1-y1*z2));
140 invJ(0,1) = aux_volume * (y42*z32 - y32*z42);
141 invJ(1,1) = aux_volume * (y31*z43 - y34*z13);
142 invJ(2,1) = aux_volume * (y24*z14 - y14*z24);
143 invJ(3,1) = aux_volume * (y13*z21 - y12*z31);
144 invJ(0,2) = aux_volume * (x32*z42 - x42*z32);
145 invJ(1,2) = aux_volume * (x43*z31 - x13*z34);
146 invJ(2,2) = aux_volume * (x14*z24 - x24*z14);
147 invJ(3,2) = aux_volume * (x21*z13 - x31*z12);
148 invJ(0,3) = aux_volume * (x42*y32 - x32*y42);
149 invJ(1,3) = aux_volume * (x31*y43 - x34*y13);
150 invJ(2,3) = aux_volume * (x24*y14 - x14*y24);
151 invJ(3,3) = aux_volume * (x13*y21 - x12*y31);
152
153 // Compute result
154 const array_1d<double,4> result = prod(invJ, X);
155
156 // Resize if needed
157 if (rResult.size() != 3) {
158 rResult.resize(3,false);
159 }
160
161 // Copy result
162 rResult[0] = result[1];
163 rResult[1] = result[2];
164 rResult[2] = result[3];
165
166 return rResult;
167 }

176 template<class TGeometryType>
177 static inline typename TGeometryType::CoordinatesArrayType& PointLocalCoordinatesStraightEdgesTriangle(
178 const TGeometryType& rGeometry,
179 typename TGeometryType::CoordinatesArrayType& rResult,
180 const typename TGeometryType::CoordinatesArrayType& rPoint
181 )
182 {
183 KRATOS_DEBUG_ERROR_IF_NOT(rGeometry.GetGeometryFamily() == GeometryData::KratosGeometryFamily::Kratos_Triangle) <<
184 "Geometry should be a triangle in order to use PointLocalCoordinatesStraightEdgesTriangle" << std::endl;
185
186 noalias(rResult) = ZeroVector(3);
187
188 array_1d<double, 3> tangent_xi = rGeometry.GetPoint(1) - rGeometry.GetPoint(0);
189 tangent_xi /= norm_2(tangent_xi);
190 array_1d<double, 3> tangent_eta = rGeometry.GetPoint(2) - rGeometry.GetPoint(0);
191 tangent_eta /= norm_2(tangent_eta);
192
193 const auto center = rGeometry.Center();
194
195 BoundedMatrix<double, 3, 3> rotation_matrix = ZeroMatrix(3, 3);
196 for (IndexType i = 0; i < 3; ++i) {
197 rotation_matrix(0, i) = tangent_xi[i];
198 rotation_matrix(1, i) = tangent_eta[i];
199 }
200
201 typename TGeometryType::CoordinatesArrayType aux_point_to_rotate, destination_point_rotated;
202 noalias(aux_point_to_rotate) = rPoint - center.Coordinates();
203 noalias(destination_point_rotated) = prod(rotation_matrix, aux_point_to_rotate) + center.Coordinates();
204
205 array_1d<typename TGeometryType::CoordinatesArrayType, 3> points_rotated;
206 for (IndexType i = 0; i < 3; ++i) {
207 noalias(aux_point_to_rotate) = rGeometry.GetPoint(i).Coordinates() - center.Coordinates();
208 noalias(points_rotated[i]) = prod(rotation_matrix, aux_point_to_rotate) + center.Coordinates();
209 }
210
211 // Compute the Jacobian matrix and its determinant
212 BoundedMatrix<double, 2, 2> J;
213 J(0,0) = points_rotated[1][0] - points_rotated[0][0];
214 J(0,1) = points_rotated[2][0] - points_rotated[0][0];
215 J(1,0) = points_rotated[1][1] - points_rotated[0][1];
216 J(1,1) = points_rotated[2][1] - points_rotated[0][1];
217 const double det_J = J(0,0)*J(1,1) - J(0,1)*J(1,0);
218
219 const double eta = (J(1,0)*(points_rotated[0][0] - destination_point_rotated[0]) +
220 J(0,0)*(destination_point_rotated[1] - points_rotated[0][1])) / det_J;
221 const double xi = (J(1,1)*(destination_point_rotated[0] - points_rotated[0][0]) +
222 J(0,1)*(points_rotated[0][1] - destination_point_rotated[1])) / det_J;
223
224 rResult(0) = xi;
225 rResult(1) = eta;
226
227 return rResult;
228 }

237 static inline void CalculateGeometryData(
238 const GeometryType& rGeometry,
239 BoundedMatrix<double,4,3>& rDN_DX,
240 array_1d<double,4>& rN,
241 double& rVolume
242 )
243 {
244 const double x10 = rGeometry[1].X() - rGeometry[0].X();
245 const double y10 = rGeometry[1].Y() - rGeometry[0].Y();
246 const double z10 = rGeometry[1].Z() - rGeometry[0].Z();
247
248 const double x20 = rGeometry[2].X() - rGeometry[0].X();
249 const double y20 = rGeometry[2].Y() - rGeometry[0].Y();
250 const double z20 = rGeometry[2].Z() - rGeometry[0].Z();
251
252 const double x30 = rGeometry[3].X() - rGeometry[0].X();
253 const double y30 = rGeometry[3].Y() - rGeometry[0].Y();
254 const double z30 = rGeometry[3].Z() - rGeometry[0].Z();
255
256 const double detJ = x10 * y20 * z30 - x10 * y30 * z20 + y10 * z20 * x30 - y10 * x20 * z30 + z10 * x20 * y30 - z10 * y20 * x30;
257
258 rDN_DX(0,0) = -y20 * z30 + y30 * z20 + y10 * z30 - z10 * y30 - y10 * z20 + z10 * y20;
259 rDN_DX(0,1) = -z20 * x30 + x20 * z30 - x10 * z30 + z10 * x30 + x10 * z20 - z10 * x20;
260 rDN_DX(0,2) = -x20 * y30 + y20 * x30 + x10 * y30 - y10 * x30 - x10 * y20 + y10 * x20;
261 rDN_DX(1,0) = y20 * z30 - y30 * z20;
262 rDN_DX(1,1) = z20 * x30 - x20 * z30;
263 rDN_DX(1,2) = x20 * y30 - y20 * x30;
264 rDN_DX(2,0) = -y10 * z30 + z10 * y30;
265 rDN_DX(2,1) = x10 * z30 - z10 * x30;
266 rDN_DX(2,2) = -x10 * y30 + y10 * x30;
267 rDN_DX(3,0) = y10 * z20 - z10 * y20;
268 rDN_DX(3,1) = -x10 * z20 + z10 * x20;
269 rDN_DX(3,2) = x10 * y20 - y10 * x20;
270
271 rDN_DX /= detJ;
272
273 rN[0] = 0.25;
274 rN[1] = 0.25;
275 rN[2] = 0.25;
276 rN[3] = 0.25;
277
278 rVolume = detJ*0.1666666666666666666667;
279 }

312 static inline void CalculateGeometryData(
313 const GeometryType& rGeometry,
314 BoundedMatrix<double,3,2>& DN_DX,
315 array_1d<double,3>& N,
316 double& rArea
317 )
318 {
319 const double x10 = rGeometry[1].X() - rGeometry[0].X();
320 const double y10 = rGeometry[1].Y() - rGeometry[0].Y();
321
322 const double x20 = rGeometry[2].X() - rGeometry[0].X();
323 const double y20 = rGeometry[2].Y() - rGeometry[0].Y();
324
325 //Jacobian is calculated:
326 // |dx/dxi dx/deta| |x1-x0 x2-x0|
327 //J=| |= | |
328 // |dy/dxi dy/deta| |y1-y0 y2-y0|
329
330 double detJ = x10 * y20-y10 * x20;
331
332 DN_DX(0,0) = -y20 + y10;
333 DN_DX(0,1) = x20 - x10;
334 DN_DX(1,0) = y20 ;
335 DN_DX(1,1) = -x20 ;
336 DN_DX(2,0) = -y10 ;
337 DN_DX(2,1) = x10 ;
338
339 DN_DX /= detJ;
340 N[0] = 0.333333333333333;
341 N[1] = 0.333333333333333;
342 N[2] = 0.333333333333333;
343
344 rArea = 0.5*detJ;
345 }

372 static inline void SideLenghts2D(
373 const GeometryType& rGeometry,
374 double& hmin,
375 double& hmax
376 )
377 {
378 const double x10 = rGeometry[1].X() - rGeometry[0].X();
379 const double y10 = rGeometry[1].Y() - rGeometry[0].Y();
380
381 const double x20 = rGeometry[2].X() - rGeometry[0].X();
382 const double y20 = rGeometry[2].Y() - rGeometry[0].Y();
383
384 double l = std::pow(x20, 2) + std::pow(y20, 2);
385 hmax = l;
386 hmin = l;
387
388 if(l>hmax) hmax = l;
389 else if(l<hmin) hmin = l;
390
391 l = (x20-x10)*(x20-x10) + (y20-y10)*(y20-y10);
392 if(l>hmax) hmax = l;
393 else if(l<hmin) hmin = l;
394
395 hmax = std::sqrt(hmax);
396 hmin = std::sqrt(hmin);
397 }

432 template<std::size_t TSize>
433 static void CalculateTetrahedraDistances(
434 const GeometryType& rGeometry, array_1d<double, TSize>& rDistances)
435 {
436 // Calculating the intersection points
437 array_1d<Point, 4> intersection_points;
438 int number_of_intersection_points = CalculateTetrahedraIntersectionPoints(rGeometry, rDistances, intersection_points);
439
440 if(number_of_intersection_points == 0) {
441 KRATOS_WARNING("CalculateTetrahedraDistances") << "WARNING:: The intersection with interface hasn't found!" << std::endl << "The distances are: " << rDistances << std::endl;
442 } else if(number_of_intersection_points == 1) {
443 // There is one point with zero distance. The distance of the nodes are their distance to this point
444 array_1d<double,3> temp;
445 // Loop over nodes to calculate their distance to the zero distance node.
446 for(unsigned int i_node = 0; i_node < rGeometry.size() ; ++i_node) {
447 noalias(temp) = intersection_points[0] - rGeometry[i_node];
448 rDistances[i_node] = norm_2(temp);
449 }
450 } else if(number_of_intersection_points == 2) {
451 // Loop over nodes to calculate their distance to the zero distance line.
452 for(unsigned int i_node = 0; i_node < rGeometry.size() ; ++i_node) {
453 rDistances[i_node] = PointDistanceToLineSegment3D(intersection_points[0], intersection_points[1], rGeometry[i_node]);
454 }
455 } else if(number_of_intersection_points == 3) {
456 // Loop over nodes to calculate their distance to the zero distance triangle.
457 for(unsigned int i_node = 0; i_node < rGeometry.size() ; ++i_node) {
458 rDistances[i_node] = PointDistanceToTriangle3D(intersection_points[0], intersection_points[1], intersection_points[2], rGeometry[i_node]);
459// rDistances[i_node] = std::abs(rGeometry[i_node].Z()); // TODO: To be removed. Pooyan.
460 }
461
462 } else if(number_of_intersection_points == 4) {
463 // Loop over nodes to calculate their distance to the each zero distance triangle.
464 for(unsigned int i_node = 0; i_node < rGeometry.size() ; ++i_node) {
465 // Here I'm taking in account the order of edges where I'm looking for intersection
466 double d1 = PointDistanceToTriangle3D(intersection_points[0], intersection_points[1], intersection_points[3], rGeometry[i_node]);
467 double d2 = PointDistanceToTriangle3D(intersection_points[0], intersection_points[3], intersection_points[2], rGeometry[i_node]);
468
469 rDistances[i_node] = (d1 > d2) ? d2 : d1;
470 }
471 }
472 }

514 template<std::size_t TSize>
515 static void CalculateExactDistancesToPlane(
516 const GeometryType& rThisGeometry,
517 array_1d<double, TSize>& rDistances
518 )
519 {
520 array_1d<Point, TSize> intersection_points;
521 int number_of_intersection_points = CalculateTetrahedraIntersectionPoints(rThisGeometry, rDistances, intersection_points);
522
523 if(number_of_intersection_points == 0) {
524 KRATOS_WARNING("GeometryUtilities") << "Warning: The intersection with interface hasn't found! The distances are" << rDistances << std::endl;
525 } else {
526 BoundedMatrix<double,TSize,TSize-1> DN_DX;
527 array_1d<double, TSize> N;
528 double volume;
529 GeometryUtils::CalculateGeometryData(rThisGeometry, DN_DX, N, volume);
530 array_1d<double, TSize-1> distance_gradient = prod(trans(DN_DX), rDistances);
531 double distance_gradient_norm = norm_2(distance_gradient);
532 if (distance_gradient_norm < 1e-15) distance_gradient_norm = 1e-15; //avoid division by zero
533 distance_gradient /= distance_gradient_norm;
534 // We use the first intersection point as reference
535 const auto &ref_point = intersection_points[0].Coordinates();
536 for (unsigned int i = 0; i < TSize; i++) {
537 double d = 0.0;
538 const auto &i_coords = rThisGeometry[i].Coordinates();
539 for (unsigned int Dim = 0; Dim < TSize - 1; Dim++) {
540 d += (i_coords[Dim] - ref_point[Dim]) * distance_gradient[Dim];
541 }
542 d = std::abs(d);
543 rDistances[i] = std::min(std::abs(rDistances[i]), d);
544 }
545 }
546 }

555 template<std::size_t TSize1, std::size_t TSize2>
556 static int CalculateTetrahedraIntersectionPoints(
557 const GeometryType& rGeometry,
558 array_1d<double, TSize1>& rDistances,
559 array_1d<Point, TSize2>& rIntersectionPoints
560 )
561 {
562 const double epsilon = 1e-15; //1.00e-9;
563
564 int number_of_intersection_points = 0;
565 for(unsigned int i = 0 ; i < TSize1 ; i++) {
566 if(std::abs(rDistances[i]) < epsilon) {
567 noalias(rIntersectionPoints[number_of_intersection_points].Coordinates()) = rGeometry[i].Coordinates();
568
569 number_of_intersection_points++;
570 continue;
571 }
572 for(unsigned int j = i + 1 ; j < TSize1 ; j++) {
573 if(std::abs(rDistances[j]) < epsilon)
574 continue; // we will add it to the intersections by the i index to be unique
575
576 if(rDistances[i] * rDistances[j] < 0.00) { // The interface passes through the edge
577 const double delta_d = std::abs(rDistances[i]) + std::abs(rDistances[j]); // we know that both distances are greater than epsilon.
578
579 const double di = std::abs(rDistances[i]) / delta_d;
580 const double dj = std::abs(rDistances[j]) / delta_d;
581
582 noalias(rIntersectionPoints[number_of_intersection_points].Coordinates()) = dj * rGeometry[i].Coordinates();
583 noalias(rIntersectionPoints[number_of_intersection_points].Coordinates()) += di * rGeometry[j].Coordinates();
584
585 number_of_intersection_points++;
586 }
587 }
588 }
589
590 return number_of_intersection_points;
591 }
