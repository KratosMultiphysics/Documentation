78 static void Triangulate_OPT(const Clipper2Lib::Path64& polygon, std::vector<Matrix>& triangles, const double factor)
79 {
80 if (polygon.size() == 4)
81 {
82 Matrix triangle(3, 2);
83 triangle(0, 0) = BrepTrimmingUtilities::IntPointToDoublePoint(polygon[0], factor)[0];
84 triangle(0, 1) = BrepTrimmingUtilities::IntPointToDoublePoint(polygon[0], factor)[1];
85 triangle(1, 0) = BrepTrimmingUtilities::IntPointToDoublePoint(polygon[1], factor)[0];
86 triangle(1, 1) = BrepTrimmingUtilities::IntPointToDoublePoint(polygon[1], factor)[1];
87 triangle(2, 0) = BrepTrimmingUtilities::IntPointToDoublePoint(polygon[2], factor)[0];
88 triangle(2, 1) = BrepTrimmingUtilities::IntPointToDoublePoint(polygon[2], factor)[1];
89 triangles.push_back(triangle);
90 return;
91 }
92
93 array_1d<double, 2> p1, p2, p3, p4;
94 int bestvertex;
95 double weight = 0;
96 double d1, d2 = 0.0;
97 double minweight = std::numeric_limits<double>::max();
98 Diagonal diagonal, newdiagonal;
99
100 std::list<Diagonal> diagonals;
101
102 IndexType n = polygon.size();
103 std::vector< Clipper2Lib::Point64 > const& points = polygon;
104 matrix<DPState> dpstates(n, n);
105
106 //init states and visibility
107 for (IndexType i = 0; i < (n - 1); i++) {
108 p1 = BrepTrimmingUtilities::IntPointToDoublePoint(points[i], factor);
109 for (IndexType j = i + 1; j < n; j++) {
110 dpstates(j, i).visible = true;
111 dpstates(j, i).weight = 0;
112 dpstates(j, i).bestvertex = -1;
113 if (j != (i + 1)) {
114 p2 = BrepTrimmingUtilities::IntPointToDoublePoint(points[j], factor);
115
116 //visibility check
117 if (i == 0) p3 = BrepTrimmingUtilities::IntPointToDoublePoint(points[n - 1], factor);
118 else p3 = BrepTrimmingUtilities::IntPointToDoublePoint(points[i - 1], factor);
119 if (i == (n - 1)) p4 = BrepTrimmingUtilities::IntPointToDoublePoint(points[0], factor);
120 else p4 = BrepTrimmingUtilities::IntPointToDoublePoint(points[i + 1], factor);
121 if (!BrepTrimmingUtilities::InCone(p3, p1, p4, p2)) {
122 dpstates(j, i).visible = false;
123 continue;
124 }
125
126 if (j == 0) p3 = BrepTrimmingUtilities::IntPointToDoublePoint(points[n - 1], factor);
127 else p3 = BrepTrimmingUtilities::IntPointToDoublePoint(points[j - 1], factor);
128 if (j == (n - 1)) p4 = BrepTrimmingUtilities::IntPointToDoublePoint(points[0], factor);
129 else p4 = BrepTrimmingUtilities::IntPointToDoublePoint(points[j + 1], factor);
130 if (!BrepTrimmingUtilities::InCone(p3, p2, p4, p1)) {
131 dpstates(j, i).visible = false;
132 continue;
133 }
134
135 for (IndexType k = 0; k < n; k++) {
136 p3 = BrepTrimmingUtilities::IntPointToDoublePoint(points[k], factor);
137 if (k == (n - 1)) p4 = BrepTrimmingUtilities::IntPointToDoublePoint(points[0], factor);
138 else p4 = BrepTrimmingUtilities::IntPointToDoublePoint(points[k + 1], factor);
139 if (BrepTrimmingUtilities::Intersects(p1, p2, p3, p4)) {
140 dpstates(j, i).visible = false;
141 break;
142 }
143 }
144 }
145 }
146 }
147
148 dpstates(n - 1, 0).visible = true;
149 dpstates(n - 1, 0).weight = 0;
150 dpstates(n - 1, 0).bestvertex = -1;
151
152 for (IndexType gap = 2; gap < n; gap++) {
153 for (IndexType i = 0; i < (n - gap); i++) {
154 IndexType j = i + gap;
155 if (!dpstates(j, i).visible) continue;
156 bestvertex = -1;
157 for (IndexType k = (i + 1); k < j; k++) {
158 if (!dpstates(k, i).visible) continue;
159 if (!dpstates(j, k).visible) continue;
160
161 if (k <= (i + 1)) d1 = 0;
162 else d1 = BrepTrimmingUtilities::Distance(BrepTrimmingUtilities::IntPointToDoublePoint(points[i], factor), BrepTrimmingUtilities::IntPointToDoublePoint(points[k], factor));
163 if (j <= (k + 1)) d2 = 0;
164 else d2 = BrepTrimmingUtilities::Distance(BrepTrimmingUtilities::IntPointToDoublePoint(points[k], factor), BrepTrimmingUtilities::IntPointToDoublePoint(points[j], factor));
165
166 weight = dpstates(k, i).weight + dpstates(j, k).weight + d1 + d2;
167
168 if ((bestvertex == -1) || (weight < minweight)) {
169 bestvertex = k;
170 minweight = weight;
171 }
172 }
173 if (bestvertex == -1) {
174 return;
175 }
176
177 dpstates(j, i).bestvertex = bestvertex;
178 dpstates(j, i).weight = minweight;
179 }
180 }
181
182 newdiagonal.index1 = 0;
183 newdiagonal.index2 = n - 1;
184 diagonals.push_back(newdiagonal);
185
186 while (!diagonals.empty()) {
187 diagonal = *(diagonals.begin());
188 diagonals.pop_front();
189
190 bestvertex = dpstates(diagonal.index2, diagonal.index1).bestvertex;
191 if (bestvertex == -1) {
192 break;
193 }
194 Matrix triangle(3, 2);
195 triangle(0, 0) = BrepTrimmingUtilities::IntPointToDoublePoint(points[diagonal.index1], factor)[0];
196 triangle(0, 1) = BrepTrimmingUtilities::IntPointToDoublePoint(points[diagonal.index1], factor)[1];
197 triangle(1, 0) = BrepTrimmingUtilities::IntPointToDoublePoint(points[bestvertex], factor)[0];
198 triangle(1, 1) = BrepTrimmingUtilities::IntPointToDoublePoint(points[bestvertex], factor)[1];
199 triangle(2, 0) = BrepTrimmingUtilities::IntPointToDoublePoint(points[diagonal.index2], factor)[0];
200 triangle(2, 1) = BrepTrimmingUtilities::IntPointToDoublePoint(points[diagonal.index2], factor)[1];
201
202 if (BrepTrimmingUtilities::GetAreaOfTriangle(triangle) > 1e-5)
203 triangles.push_back(triangle);
204 else
205 {
206 std::cout << "triangle with zero area" << BrepTrimmingUtilities::GetAreaOfTriangle(triangle) << std::endl;
207 }
208 if (bestvertex > (diagonal.index1 + 1)) {
209 newdiagonal.index1 = diagonal.index1;
210 newdiagonal.index2 = bestvertex;
211 diagonals.push_back(newdiagonal);
212 }
213 if (diagonal.index2 > (bestvertex + 1)) {
214 newdiagonal.index1 = bestvertex;
215 newdiagonal.index2 = diagonal.index2;
216 diagonals.push_back(newdiagonal);
217 }
218 }
219 }

221 static bool InCone(array_1d<double, 2>& p1, array_1d<double, 2>& p2,
222 array_1d<double, 2>& p3, array_1d<double, 2>& p)
223 {
224 if (BrepTrimmingUtilities::IsConvex(p1, p2, p3)) {
225 if (!BrepTrimmingUtilities::IsConvex(p1, p2, p)) return false;
226 if (!BrepTrimmingUtilities::IsConvex(p2, p3, p)) return false;
227 return true;
228 }
229 else {
230 if (BrepTrimmingUtilities::IsConvex(p1, p2, p)) return true;
231 if (BrepTrimmingUtilities::IsConvex(p2, p3, p)) return true;
232 return false;
233 }
234 }

236 static bool IsConvex(
237 const array_1d<double, 2>& p1,
238 const array_1d<double, 2>& p2,
239 const array_1d<double, 2>& p3)
240 {
241 double tmp;
242 tmp = (p3[1] - p1[1]) * (p2[0] - p1[0]) - (p3[0] - p1[0]) * (p2[1] - p1[1]);
243 if (tmp > 0) return true;
244 else return false;
245 }

247 static double Distance(array_1d<double, 2> point_1, array_1d<double, 2> point_2)
248 {
249 return sqrt(point_1[0] * point_2[0] + point_1[1] * point_2[1]);

252 static double GetAreaOfTriangle(const Matrix& triangle)
253 {
254 double area = (triangle(0, 0) * (triangle(1, 1) - triangle(2, 1))
255 + triangle(1, 0) * (triangle(2, 1) - triangle(0, 1))
256 + triangle(2, 0) * (triangle(0, 1) - triangle(1, 1))) / 2;
257
258 return area;
259 }

261 //checks if two lines intersect
262 static bool Intersects(array_1d<double, 2>& p11, array_1d<double, 2>& p12,
263 array_1d<double, 2>& p21, array_1d<double, 2>& p22)
264 {
265 if ((p11[0] == p21[0]) && (p11[1] == p21[1])) return false;
266 if ((p11[0] == p22[0]) && (p11[1] == p22[1])) return false;
267 if ((p12[0] == p21[0]) && (p12[1] == p21[1])) return false;
268 if ((p12[0] == p22[0]) && (p12[1] == p22[1])) return false;
269
270 array_1d<double, 2> v1ort, v2ort, v;
271 double dot11, dot12, dot21, dot22;
272
273 v1ort[0] = p12[1] - p11[1];
274 v1ort[1] = p11[0] - p12[0];
275
276 v2ort[0] = p22[1] - p21[1];
277 v2ort[1] = p21[0] - p22[0];
278
279 v[0] = p21[0] - p11[0];
280 v[1] = p21[1] - p11[1];
281 dot21 = v[0] * v1ort[0] + v[1] * v1ort[1];
282 v[0] = p22[0] - p11[0];
283 v[1] = p22[1] - p11[1];
284 dot22 = v[0] * v1ort[0] + v[1] * v1ort[1];
285
286 v[0] = p11[0] - p21[0];
287 v[1] = p11[1] - p21[1];
288 dot11 = v[0] * v2ort[0] + v[1] * v2ort[1];
289 v[0] = p12[0] - p21[0];
290 v[1] = p12[1] - p21[1];
291 dot12 = v[0] * v2ort[0] + v[1] * v2ort[1];
292
293 if (dot11 * dot12 > 0) return false;
294 if (dot21 * dot22 > 0) return false;
295
296 return true;
297 }
