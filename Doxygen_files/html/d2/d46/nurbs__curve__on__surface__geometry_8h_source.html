331 void ComputeAxisIntersectionSBM(
332 std::vector<double>& rSpans,
333 double Start,
334 double End,
335 const std::vector<double>& surface_spans_u,
336 const std::vector<double>& surface_spans_v) const
337 {
338 CoordinatesArrayType physical_coord_1 = ZeroVector(3);
339 CoordinatesArrayType local_coord_1 = ZeroVector(3);
340 local_coord_1[0] = Start;
341 mpNurbsCurve->GlobalCoordinates(physical_coord_1, local_coord_1);
342
343 CoordinatesArrayType physical_coord_2 = ZeroVector(3);
344 CoordinatesArrayType local_coord_2 = ZeroVector(3);
345 local_coord_2[0] = End;
346 mpNurbsCurve->GlobalCoordinates(physical_coord_2, local_coord_2);
347
348 // 2D need to re-order
349 bool reverse_order = false;
350 if ((physical_coord_2[0] - physical_coord_1[0] < 0 ) ||
351 (physical_coord_2[1] - physical_coord_1[1] < 0 )) {
352 reverse_order = true;
353 }
354 std::vector<double> tempSpans;
355 const double tolerance_orientation = 1e-10;
356 const double tolerance_intersection = 1e-15;
357
358 // Scale factor between surface coordinate and curve one
359 double physical_length_segment = norm_2(physical_coord_1-physical_coord_2);
360 double parameter_length_segment = norm_2(local_coord_1-local_coord_2);
361 double scale_factor = parameter_length_segment/physical_length_segment;
362
363 std::vector<double> knot_interval(2);
364
365 // Compute constant coordinate -> understand segment orientation (vertical/horizontal)
366 if (std::abs(physical_coord_1[0]-physical_coord_2[0]) > tolerance_orientation) {
367 // horizontal case
368 knot_interval[0] = physical_coord_1[0];
369 knot_interval[1] = physical_coord_2[0];
370 std::sort(knot_interval.begin(), knot_interval.end());
371
372 knot_interval[0] -= tolerance_intersection;
373 knot_interval[1] += tolerance_intersection;
374 // Compare with surface_spans_u
375 for (IndexType i = 0; i < surface_spans_u.size(); i++) {
376 double curr_knot_value = surface_spans_u[i];
377 if (curr_knot_value < knot_interval[0]) {continue;}
378 if (std::abs(curr_knot_value - knot_interval[0]) < tolerance_intersection*10) knot_interval[0] = curr_knot_value;
379 if (curr_knot_value > knot_interval[1]) {break;}
380 if (std::abs(curr_knot_value - knot_interval[1]) < tolerance_intersection*10) knot_interval[1] = curr_knot_value;
381 double knot_value_in_curve_parameter = Start + (curr_knot_value-knot_interval[0]) * scale_factor;
382
383 tempSpans.push_back(knot_value_in_curve_parameter);
384 }
385
386 } else if (std::abs(physical_coord_1[1]-physical_coord_2[1]) > tolerance_orientation) {
387 // vertical case
388 knot_interval[0] = physical_coord_1[1];
389 knot_interval[1] = physical_coord_2[1];
390 std::sort(knot_interval.begin(), knot_interval.end());
391
392 knot_interval[0] -= tolerance_intersection;
393 knot_interval[1] += tolerance_intersection;
394 // Compare with surface_spans_v
395 for (IndexType i = 0; i < surface_spans_v.size(); i++) {
396 double curr_knot_value = surface_spans_v[i];
397 if (curr_knot_value < knot_interval[0]) {continue;}
398 if (std::abs(curr_knot_value - knot_interval[0]) < tolerance_intersection*10) knot_interval[0] = curr_knot_value;
399 if (curr_knot_value > knot_interval[1]) {break;}
400 if (std::abs(curr_knot_value - knot_interval[1]) < tolerance_intersection*10) knot_interval[1] = curr_knot_value;
401 double knot_value_in_curve_parameter = Start + (curr_knot_value-knot_interval[0]) * scale_factor;
402
403 tempSpans.push_back(knot_value_in_curve_parameter);
404 }
405 } else {
406 KRATOS_ERROR << "ComputeAxisIntersectionSBM :: brep not parallel to any surface knot vector";
407 }
408
409 rSpans.resize(tempSpans.size());
410 if (tempSpans.size() > 2){
411 if (reverse_order) {
412 for (IndexType i = 0; i < tempSpans.size(); i++) {
413 rSpans[i] = End - tempSpans[tempSpans.size()-1-i];
414 }
415 } else {
416 rSpans = tempSpans;
417 }
418 } else rSpans = tempSpans;
419
420 KRATOS_ERROR_IF(rSpans.size()<2) << "ComputeAxisIntersectionSBM :: Wrong number of intersection found (<2)" << std::endl;
421 }

509 double Length() const override
510 {
511 IntegrationPointsArrayType integration_points;
512 IntegrationInfo integration_info = GetDefaultIntegrationInfo();
513 CreateIntegrationPoints(integration_points, integration_info);
514
515 double length = 0.0;
516 for (IndexType i = 0; i < integration_points.size(); ++i) {
517 const double determinant_jacobian = DeterminantOfJacobian(integration_points[i]);
518 length += integration_points[i].Weight() * determinant_jacobian;
519 }
520 return length;
521 }

581 void CreateQuadraturePointGeometries(
582 GeometriesArrayType& rResultGeometries,
583 IndexType NumberOfShapeFunctionDerivatives,
584 const IntegrationPointsArrayType& rIntegrationPoints,
585 IntegrationInfo& rIntegrationInfo) override
586 {
587 // shape function container.
588 NurbsSurfaceShapeFunction shape_function_container(
589 mpNurbsSurface->PolynomialDegreeU(), mpNurbsSurface->PolynomialDegreeV(),
590 NumberOfShapeFunctionDerivatives);
591
592 // Resize containers.
593 if (rResultGeometries.size() != rIntegrationPoints.size())
594 rResultGeometries.resize(rIntegrationPoints.size());
595
596 auto default_method = this->GetDefaultIntegrationMethod();
597 SizeType num_nonzero_cps = shape_function_container.NumberOfNonzeroControlPoints();
598
599 Matrix N(1, num_nonzero_cps);
600 DenseVector<Matrix> shape_function_derivatives(NumberOfShapeFunctionDerivatives - 1);
601 for (IndexType i = 0; i < NumberOfShapeFunctionDerivatives - 1; i++) {
602 shape_function_derivatives[i].resize(num_nonzero_cps, i + 2);
603 }
604
605 for (IndexType i = 0; i < rIntegrationPoints.size(); ++i)
606 {
607 std::vector<CoordinatesArrayType> global_space_derivatives(2);
608 mpNurbsCurve->GlobalSpaceDerivatives(
609 global_space_derivatives,
610 rIntegrationPoints[i],
611 1);
612
613 if (mpNurbsSurface->IsRational()) {
614 shape_function_container.ComputeNurbsShapeFunctionValues(
615 mpNurbsSurface->KnotsU(), mpNurbsSurface->KnotsV(), mpNurbsSurface->Weights(),
616 global_space_derivatives[0][0], global_space_derivatives[0][1]);
617 }
618 else {
619 shape_function_container.ComputeBSplineShapeFunctionValues(
620 mpNurbsSurface->KnotsU(), mpNurbsSurface->KnotsV(),
621 global_space_derivatives[0][0], global_space_derivatives[0][1]);
622 }
623
625 PointsArrayType nonzero_control_points(num_nonzero_cps);
626 auto cp_indices = shape_function_container.ControlPointIndices(
627 mpNurbsSurface->NumberOfControlPointsU(), mpNurbsSurface->NumberOfControlPointsV());
628 for (IndexType j = 0; j < num_nonzero_cps; j++) {
629 nonzero_control_points(j) = mpNurbsSurface->pGetPoint(cp_indices[j]);
630 }
632 for (IndexType j = 0; j < num_nonzero_cps; j++) {
633 N(0, j) = shape_function_container(j, 0);
634 }
635
637 if (NumberOfShapeFunctionDerivatives > 0) {
638 IndexType shape_derivative_index = 1;
639 for (IndexType n = 0; n < NumberOfShapeFunctionDerivatives - 1; n++) {
640 for (IndexType k = 0; k < n + 2; k++) {
641 for (IndexType j = 0; j < num_nonzero_cps; j++) {
642 shape_function_derivatives[n](j, k) = shape_function_container(j, shape_derivative_index + k);
643 }
644 }
645 shape_derivative_index += n + 2;
646 }
647 }
648
649 GeometryShapeFunctionContainer<GeometryData::IntegrationMethod> data_container(
650 default_method, rIntegrationPoints[i],
651 N, shape_function_derivatives);
652
653 rResultGeometries(i) = CreateQuadraturePointsUtility<NodeType>::CreateQuadraturePointCurveOnSurface(
654 data_container, nonzero_control_points,
655 global_space_derivatives[1][0], global_space_derivatives[1][1], this);
656 }
657 }

680 void CreateQuadraturePointGeometriesSBM(
681 GeometriesArrayType& rResultGeometries,
682 IndexType NumberOfShapeFunctionDerivatives,
683 const IntegrationPointsArrayType& rIntegrationPoints,
684 IntegrationInfo& rIntegrationInfo)
685 {
686 // shape function container.
687 NurbsSurfaceShapeFunction shape_function_container(
688 mpNurbsSurface->PolynomialDegreeU(), mpNurbsSurface->PolynomialDegreeV(),
689 NumberOfShapeFunctionDerivatives);
690
691 // Resize containers.
692 if (rResultGeometries.size() != rIntegrationPoints.size())
693 rResultGeometries.resize(rIntegrationPoints.size());
694
695 auto default_method = this->GetDefaultIntegrationMethod();
696 SizeType num_nonzero_cps = shape_function_container.NumberOfNonzeroControlPoints();
697
698 Matrix N(1, num_nonzero_cps);
699 DenseVector<Matrix> shape_function_derivatives(NumberOfShapeFunctionDerivatives - 1);
700 for (IndexType i = 0; i < NumberOfShapeFunctionDerivatives - 1; i++) {
701 shape_function_derivatives[i].resize(num_nonzero_cps, i + 2);
702 }
703
704 // the brep surface in which the brep curve is defined has been marked as sbm in the nurbs_modeler_sbm
705 bool is_brep_internal = false;
706 IndexType InternalBrepSpanU;
707 IndexType InternalBrepSpanV;
708
709 // SBM -> check to which axis the brep is alligned
710 std::vector<CoordinatesArrayType> first_integration_point(2); // first integration point of the brep in the parameter space
711 std::vector<CoordinatesArrayType> last_integration_point(2); // last integration point of the brep in the parameter space
712 mpNurbsCurve->GlobalSpaceDerivatives(first_integration_point,rIntegrationPoints[0],1);
713 mpNurbsCurve->GlobalSpaceDerivatives(last_integration_point,rIntegrationPoints[rIntegrationPoints.size()-1],1);
714
715 // check if the brep is internal (external breps do not need to be computed in a different knot span)
716 is_brep_internal = true;
717 const double tolerance = 1e-14;
718 // At this point all the true are boundary GPs
719 if (std::abs(first_integration_point[0][0]-mpNurbsSurface->KnotsU()[0]) < tolerance)
720 is_brep_internal = false;
721 if (std::abs(first_integration_point[0][0]-mpNurbsSurface->KnotsU()[mpNurbsSurface->KnotsU().size()-1]) < tolerance)
722 is_brep_internal = false;
723 if (std::abs(first_integration_point[0][1]-mpNurbsSurface->KnotsV()[0]) < tolerance)
724 is_brep_internal = false;
725 if (std::abs(first_integration_point[0][1]-mpNurbsSurface->KnotsV()[mpNurbsSurface->KnotsV().size()-1]) < tolerance)
726 is_brep_internal = false;
727
728 // If is_brep_internal -> the current GP is a surrogate GP otherwise it is an external GP
729 if (is_brep_internal) {
730 // brep = knot spans edge -> same SpanU and SpanV for all its integration points
731 InternalBrepSpanU = NurbsUtilities::GetLowerSpan(mpNurbsSurface->PolynomialDegreeU(), mpNurbsSurface->KnotsU(), first_integration_point[0][0]);
732 InternalBrepSpanV = NurbsUtilities::GetLowerSpan(mpNurbsSurface->PolynomialDegreeV(), mpNurbsSurface->KnotsV(), first_integration_point[0][1]);
733
734 // Understand if the knot-boundary is along U or V
735 bool gp_is_along_V;
736 if (std::abs(first_integration_point[0][0] - last_integration_point[0][0]) < tolerance) {
737 gp_is_along_V = true;
738 }
739 else {gp_is_along_V = false;}
740
741 if (gp_is_along_V && first_integration_point[0][1] > last_integration_point[0][1]) {
742 // Is decreasing along y -> Add 1 at InternalBrepSpanU
743 InternalBrepSpanU++;
744 }
745 if (!gp_is_along_V && first_integration_point[0][0] < last_integration_point[0][0]) {
746 // Is increasing along x -> Add 1 at InternalBrepSpanV
747 InternalBrepSpanV++;
748 }
749 }
750
751 // Loop over the integration points of the brep
752 for (IndexType i = 0; i < rIntegrationPoints.size(); ++i)
753 {
754 std::vector<CoordinatesArrayType> global_space_derivatives(2);
755 mpNurbsCurve->GlobalSpaceDerivatives(
756 global_space_derivatives,
757 rIntegrationPoints[i],
758 1);
759
760 if (mpNurbsSurface->IsRational()) {
761 shape_function_container.ComputeNurbsShapeFunctionValues(
762 mpNurbsSurface->KnotsU(), mpNurbsSurface->KnotsV(), mpNurbsSurface->Weights(),
763 global_space_derivatives[0][0], global_space_derivatives[0][1]);
764 }
765 else {
766 if (is_brep_internal) {
767 // SBM case on internal brep
768 shape_function_container.ComputeBSplineShapeFunctionValuesAtSpan(
769 mpNurbsSurface->KnotsU(),
770 mpNurbsSurface->KnotsV(),
771 InternalBrepSpanU,
772 InternalBrepSpanV,
773 global_space_derivatives[0][0],
774 global_space_derivatives[0][1]);
775 }
776 else {
777 // SBM case on external brep
778 shape_function_container.ComputeBSplineShapeFunctionValues(
779 mpNurbsSurface->KnotsU(), mpNurbsSurface->KnotsV(),
780 global_space_derivatives[0][0], global_space_derivatives[0][1]);
781 }
782 }
783
785 PointsArrayType nonzero_control_points(num_nonzero_cps);
786 auto cp_indices = shape_function_container.ControlPointIndices(
787 mpNurbsSurface->NumberOfControlPointsU(), mpNurbsSurface->NumberOfControlPointsV());
788 for (IndexType j = 0; j < num_nonzero_cps; j++) {
789 nonzero_control_points(j) = mpNurbsSurface->pGetPoint(cp_indices[j]);
790 }
792 for (IndexType j = 0; j < num_nonzero_cps; j++) {
793 N(0, j) = shape_function_container(j, 0);
794 }
795
797 if (NumberOfShapeFunctionDerivatives > 0) {
798 IndexType shape_derivative_index = 1;
799 for (IndexType n = 0; n < NumberOfShapeFunctionDerivatives - 1; n++) {
800 for (IndexType k = 0; k < n + 2; k++) {
801 for (IndexType j = 0; j < num_nonzero_cps; j++) {
802 shape_function_derivatives[n](j, k) = shape_function_container(j, shape_derivative_index + k);
803 }
804 }
805 shape_derivative_index += n + 2;
806 }
807 }
808
809 GeometryShapeFunctionContainer<GeometryData::IntegrationMethod> data_container(
810 default_method, rIntegrationPoints[i],
811 N, shape_function_derivatives);
812
813 rResultGeometries(i) = CreateQuadraturePointsUtility<NodeType>::CreateQuadraturePointCurveOnSurface(
814 data_container, nonzero_control_points,
815 global_space_derivatives[1][0], global_space_derivatives[1][1], this);
816 }
817 }

823 /*
824 * @brief This method maps from dimension space to working space.
825 * From Piegl and Tiller, The NURBS Book, Algorithm A3.1/ A4.1
826 * @param rResult array_1d<double, 3> with the coordinates in working space
827 * @param LocalCoordinates The local coordinates in dimension space
828 * @return array_1d<double, 3> with the coordinates in working space
829 * @see PointLocalCoordinates
830 */
831 CoordinatesArrayType& GlobalCoordinates(
832 CoordinatesArrayType& rResult,
833 const CoordinatesArrayType& rLocalCoordinates
834 ) const override
835 {
836 // Compute the coordinates of the embedded curve in the parametric space of the surface
837 CoordinatesArrayType result_local = mpNurbsCurve->GlobalCoordinates(rResult, rLocalCoordinates);
838
839 // Compute and return the coordinates of the surface in the geometric space
840 return mpNurbsSurface->GlobalCoordinates(rResult, result_local);
841 }
842
852 void GlobalSpaceDerivatives(
853 std::vector<CoordinatesArrayType>& rGlobalSpaceDerivatives,
854 const CoordinatesArrayType& rCoordinates,
855 const SizeType DerivativeOrder) const override
856 {
857 // Check size of output
858 if (rGlobalSpaceDerivatives.size() != DerivativeOrder + 1) {
859 rGlobalSpaceDerivatives.resize(DerivativeOrder + 1);
860 }
861
862 // Compute the gradients of the embedded curve in the parametric space of the surface
863 std::vector<array_1d<double, 3>> curve_derivatives;
864 mpNurbsCurve->GlobalSpaceDerivatives(curve_derivatives, rCoordinates, DerivativeOrder);
865
866 // Compute the gradients of the surface in the geometric space
867 array_1d<double, 3> surface_coordinates = ZeroVector(3);
868 surface_coordinates[0] = curve_derivatives[0][0];
869 surface_coordinates[1] = curve_derivatives[0][1];
870 std::vector<array_1d<double, 3>> surface_derivatives;
871 mpNurbsSurface->GlobalSpaceDerivatives(surface_derivatives, surface_coordinates, DerivativeOrder);
872
873 std::function<array_1d<double, 3>(int, int, int)> c;
874 c = [&](int DerivativeOrder, int i, int j) -> array_1d<double, 3> {
875 if (DerivativeOrder > 0) {
876 array_1d<double, 3> result = ZeroVector(3);
877
878 for (int a = 1; a <= DerivativeOrder; a++) {
879 result += (
880 c(DerivativeOrder - a, i + 1, j) * curve_derivatives[a][0] +
881 c(DerivativeOrder - a, i, j + 1) * curve_derivatives[a][1]
882 ) * NurbsUtilities::GetBinomCoefficient(DerivativeOrder - 1, a - 1);
883 }
884
885 return result;
886 }
887 else {
888 const int index = NurbsSurfaceShapeFunction::IndexOfShapeFunctionRow(i, j);
889 return surface_derivatives[index];
890 }
891 };
892 for (SizeType i = 0; i <= DerivativeOrder; i++) {
893 rGlobalSpaceDerivatives[i] = c(i, 0, 0);
894 }
895 }
