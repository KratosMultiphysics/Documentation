256
257 // The first iterator of the array of nodes

259
261 const Double3DArray zero_array = ZeroVector(3);
262 #pragma omp parallel for schedule(guided,512)
263 for (int i = 0; i < static_cast<int>(r_nodes.size()); ++i) {
264 auto it_node = (it_node_begin + i);
265 it_node->SetValue(NODAL_MASS, 0.0);

271 #pragma omp parallel for schedule(guided,512)
272 for (int i = 0; i < static_cast<int>(r_nodes.size()); ++i) {
273 auto it_node = (it_node_begin + i);
274 it_node->SetValue(NODAL_INERTIA, zero_array);
275 Double3DArray& r_fractional_acceleration = it_node->FastGetSolutionStepValue(FRACTIONAL_ANGULAR_ACCELERATION);
276 r_fractional_acceleration = ZeroVector(3);
277 }
278 }
279
280 #pragma omp parallel for schedule(guided,512)
281 for (int i = 0; i < static_cast<int>(r_nodes.size()); ++i) {
282 auto it_node = (it_node_begin + i);
283

287 r_current_residual[j] = 0.0;}

290 Double3DArray& r_current_residual_moment = it_node->FastGetSolutionStepValue(MOMENT_RESIDUAL);

383 template <typename TObjectType>
384 void UpdateAccelerationStage(
385 NodeIterator itCurrentNode,
386 const IndexType FieldPosition,
387 const ArrayVarType& rAccelerationVariable,
388 const ArrayVarType& rVelocityVariable,
389 const TObjectType& rDampingVariable,
390 const ArrayVarType& rResidualVariable,
391 const TObjectType& rIntertiaVariable,
392 const DoubleVarType& rFixVariable1,
393 const DoubleVarType& rFixVariable2,
394 const DoubleVarType& rFixVariable3,
395 const SizeType DomainSize = 3
396 )
397 {
398 Double3DArray& r_acceleration = itCurrentNode->FastGetSolutionStepValue(rAccelerationVariable);
399 Double3DArray& r_current_velocity = itCurrentNode->FastGetSolutionStepValue(rVelocityVariable);
400 const Double3DArray& r_current_residual = itCurrentNode->FastGetSolutionStepValue(rResidualVariable);
401
402 if (std::is_same<TObjectType,DoubleVarType>::value) {
403 const double& r_nodal_inertia = itCurrentNode->GetValue(NODAL_MASS);
404 const double& r_nodal_damping = itCurrentNode->GetValue(NODAL_DISPLACEMENT_DAMPING);
405
406 // Solution of the explicit equation:
407 if (r_nodal_inertia > numerical_limit)
408 {
409 noalias(r_acceleration) = (r_current_residual - r_nodal_damping * r_current_velocity) / r_nodal_inertia;
410 }
411 else
412 noalias(r_acceleration) = ZeroVector(3);
413 }
414 else if (std::is_same<TObjectType,ArrayVarType>::value) {
415 const Double3DArray& r_nodal_inertia = itCurrentNode->GetValue(NODAL_INERTIA);
416 const Double3DArray& r_nodal_rotational_damping = itCurrentNode->GetValue(NODAL_ROTATION_DAMPING);
417
418 const IndexType initial_k = DomainSize == 3 ? 0 : 2; // We do this because in 2D only the rotation Z is needed, then we start with 2, instead of 0
419 for (IndexType kk = initial_k; kk < 3; ++kk) {
420 if (r_nodal_inertia[kk] > numerical_limit)
421 r_acceleration[kk] = (r_current_residual[kk] - r_nodal_rotational_damping[kk] * r_current_velocity[kk]) / r_nodal_inertia[kk];
422 else
423 r_acceleration[kk] = 0.0;
424 }
425 }
426 else KRATOS_ERROR << "cannot handle input variable in \"explicit_multi_stage_kim_scheme.hpp\"" << std::endl;
427
428 std::array<bool, 3> fix_field = {false, false, false};
429 fix_field[0] = (itCurrentNode->GetDof(rFixVariable1, FieldPosition).IsFixed());
430 fix_field[1] = (itCurrentNode->GetDof(rFixVariable2, FieldPosition + 1).IsFixed());
431 if (DomainSize == 3)
432 fix_field[2] = (itCurrentNode->GetDof(rFixVariable3, FieldPosition + 2).IsFixed());
433
434 for (IndexType j = 0; j < DomainSize; j++) {
435 if (fix_field[j]) {
436 r_acceleration[j] = 0.0;
437 }
438 }
439 }

447 void UpdateDegreesOfFreedomStage1(
448 NodeIterator itCurrentNode,
449 const ArrayVarType& rVelocityVariable,
450 const ArrayVarType& rDisplacementVariable,
451 const ArrayVarType& rAccelerationVariable,
452 const SizeType DomainSize = 3
453 )
454 {
455 Double3DArray& r_current_velocity = itCurrentNode->FastGetSolutionStepValue(rVelocityVariable);
456 Double3DArray& r_current_displacement = itCurrentNode->FastGetSolutionStepValue(rDisplacementVariable);
457
458 const Double3DArray& r_previous_displacement = itCurrentNode->FastGetSolutionStepValue(rDisplacementVariable, 1);
459 const Double3DArray& r_previous_velocity = itCurrentNode->FastGetSolutionStepValue(rVelocityVariable, 1);
460 const Double3DArray& r_previous_acceleration = itCurrentNode->FastGetSolutionStepValue(rAccelerationVariable, 1);
461
462 for (IndexType j = 0; j < DomainSize; j++) {
463
464 r_current_displacement[j] = r_previous_displacement[j] + mTime.MidStep * r_previous_velocity[j];
465 r_current_displacement[j] += 0.50 * mTime.MidStep * mTime.MidStep * r_previous_acceleration[j];
466
467 r_current_velocity[j] = r_previous_velocity[j] + mTime.MidStep * r_previous_acceleration[j];
468 } // for DomainSize
469 }

471 void UpdateDegreesOfFreedomStage2(
472 NodeIterator itCurrentNode,
473 const ArrayVarType& rVelocityVariable,
474 const ArrayVarType& rDisplacementVariable,
475 const ArrayVarType& rAccelerationVariable,
476 const ArrayVarType& rFractionalAccelerationVariable,
477 const SizeType DomainSize = 3
478 )
479 {
480 Double3DArray& r_current_velocity = itCurrentNode->FastGetSolutionStepValue(rVelocityVariable);
481 Double3DArray& r_current_displacement = itCurrentNode->FastGetSolutionStepValue(rDisplacementVariable);
482 Double3DArray& r_fractional_acceleration = itCurrentNode->FastGetSolutionStepValue(rFractionalAccelerationVariable);
483
484 const Double3DArray& r_previous_displacement = itCurrentNode->FastGetSolutionStepValue(rDisplacementVariable, 1);
485 const Double3DArray& r_previous_velocity = itCurrentNode->FastGetSolutionStepValue(rVelocityVariable, 1);
486 const Double3DArray& r_previous_acceleration = itCurrentNode->FastGetSolutionStepValue(rAccelerationVariable, 1);
487
488
489 for (IndexType j = 0; j < DomainSize; j++) {
490
491 r_current_displacement[j] = r_previous_displacement[j] + mTime.Delta * r_previous_velocity[j];
492 r_current_displacement[j] += 0.50 * mTime.Delta * mTime.Delta * r_previous_acceleration[j] * ((3.0*mDeltaTime.Fraction-1.0)/(3.0*mDeltaTime.Fraction));
493 r_current_displacement[j] += 0.50 * mTime.Delta * mTime.Delta * r_fractional_acceleration[j] * (1.0/(3.0*mDeltaTime.Fraction));
494
495
496 r_current_velocity[j] = r_previous_velocity[j];
497 r_current_velocity[j] += mTime.Delta * r_previous_acceleration[j] * ((2.0*mDeltaTime.Fraction-1.0)/(2.0*mDeltaTime.Fraction));
498 r_current_velocity[j] += mTime.Delta * r_fractional_acceleration[j] * ((1.0)/(2.0*mDeltaTime.Fraction));
499 } // for DomainSize

503 NodeIterator itCurrentNode,
504 const ArrayVarType& rVelocityVariable,
505 const ArrayVarType& rDisplacementVariable,
506 const ArrayVarType& rAccelerationVariable,
507 const ArrayVarType& rFractionalAccelerationVariable,

512 Double3DArray& r_current_displacement = itCurrentNode->FastGetSolutionStepValue(rDisplacementVariable);
513 Double3DArray& r_current_acceleration = itCurrentNode->FastGetSolutionStepValue(rAccelerationVariable);

515
516 const Double3DArray& r_previous_displacement = itCurrentNode->FastGetSolutionStepValue(rDisplacementVariable, 1);
517 const Double3DArray& r_previous_velocity = itCurrentNode->FastGetSolutionStepValue(rVelocityVariable, 1);
518 const Double3DArray& r_previous_acceleration = itCurrentNode->FastGetSolutionStepValue(rAccelerationVariable, 1);
519
520
521 for (IndexType j = 0; j < DomainSize; j++) {
522
523 r_current_displacement[j] = r_previous_displacement[j] + mTime.Delta * r_previous_velocity[j];
524 r_current_displacement[j] += 0.50 * mTime.Delta * mTime.Delta * r_previous_acceleration[j] * ((4.0*mDeltaTime.Fraction-1.0)/(6.0*mDeltaTime.Fraction));
525 r_current_displacement[j] -= 0.50 * mTime.Delta * mTime.Delta * r_fractional_acceleration[j] * (1.0/(6.0*mDeltaTime.Fraction*(mDeltaTime.Fraction-1.0)));
526 r_current_displacement[j] += 0.50 * mTime.Delta * mTime.Delta * r_current_acceleration[j] * ((2.0*mDeltaTime.Fraction-1.0)/(6.0*(mDeltaTime.Fraction-1.0)));
527
528
529 r_current_velocity[j] = r_previous_velocity[j];
530 r_current_velocity[j] += mTime.Delta * r_previous_acceleration[j] * ((3.0*mDeltaTime.Fraction-1.0)/(6.0*mDeltaTime.Fraction));
531 r_current_velocity[j] -= mTime.Delta * r_fractional_acceleration[j] * ((1.0)/(6.0*mDeltaTime.Fraction*(mDeltaTime.Fraction-1.0)));
532 r_current_velocity[j] += mTime.Delta * r_current_acceleration[j] * ((3.0*mDeltaTime.Fraction-2.0)/(6.0*(mDeltaTime.Fraction-1.0)));
533 } // for DomainSize

541 virtual void SchemeCustomInitialization(
542 ModelPart& rModelPart,
543 const SizeType DomainSize = 3
544 )
545 {
546 KRATOS_TRY
547
548 // The array containing the nodes
549 NodesArrayType& r_nodes = rModelPart.Nodes();
550
551 // The fisrt node interator
552 const auto it_node_begin = rModelPart.NodesBegin();
553
554 // Auxiliary zero array
555 const Double3DArray zero_array = ZeroVector(3);
556
557 // If we consider the rotation DoF
558 const bool has_dof_for_rot_z = it_node_begin->HasDofFor(ROTATION_Z);
559
560 // Getting dof position
561 const IndexType disppos = it_node_begin->GetDofPosition(DISPLACEMENT_X);
562 const IndexType rotppos = has_dof_for_rot_z ? it_node_begin->GetDofPosition(ROTATION_X) : 0;
563
564 #pragma omp parallel for schedule(guided,512)
565 for (int i = 0; i < static_cast<int>(r_nodes.size()); ++i) {
566 // Current step information "N+1" (before step update).
567 auto it_node = it_node_begin + i;
568
569 const double nodal_mass = it_node->GetValue(NODAL_MASS);
570 const Double3DArray& r_current_residual = it_node->FastGetSolutionStepValue(FORCE_RESIDUAL);
571 Double3DArray& r_current_acceleration = it_node->FastGetSolutionStepValue(ACCELERATION);
572
573 // Solution of the explicit equation:
574 if (nodal_mass > numerical_limit) {
575 r_current_acceleration = r_current_residual / nodal_mass;
576 } else {
577 r_current_acceleration = zero_array;
578 }
579
580 std::array<bool, 3> fix_displacements = {false, false, false};
581
582 fix_displacements[0] = (it_node->GetDof(DISPLACEMENT_X, disppos).IsFixed());
583 fix_displacements[1] = (it_node->GetDof(DISPLACEMENT_Y, disppos + 1).IsFixed());
584 if (DomainSize == 3)
585 fix_displacements[2] = (it_node->GetDof(DISPLACEMENT_Z, disppos + 2).IsFixed());
586
587 for (IndexType j = 0; j < DomainSize; j++) {
588 if (fix_displacements[j]) {
589 r_current_acceleration[j] = 0.0;
590 }
591 } // for DomainSize
592
593
595 if (has_dof_for_rot_z) {
596 const Double3DArray& nodal_inertia = it_node->GetValue(NODAL_INERTIA);
597 const Double3DArray& r_current_residual_moment = it_node->FastGetSolutionStepValue(MOMENT_RESIDUAL);
598 Double3DArray& r_current_angular_acceleration = it_node->FastGetSolutionStepValue(ANGULAR_ACCELERATION);
599
600 const IndexType initial_k = DomainSize == 3 ? 0 : 2; // We do this because in 2D only the rotation Z is needed, then we start with 2, instead of 0
601 for (IndexType kk = initial_k; kk < 3; ++kk) {
602 if (nodal_inertia[kk] > numerical_limit) {
603 r_current_angular_acceleration[kk] = r_current_residual_moment[kk] / nodal_inertia[kk];
604 } else {
605 r_current_angular_acceleration[kk] = 0.0;
606 }
607 }
608
609 std::array<bool, 3> fix_rotation = {false, false, false};
610 if (DomainSize == 3) {
611 fix_rotation[0] = (it_node->GetDof(ROTATION_X, rotppos).IsFixed());
612 fix_rotation[1] = (it_node->GetDof(ROTATION_Y, rotppos + 1).IsFixed());
613 }
614 fix_rotation[2] = (it_node->GetDof(ROTATION_Z, rotppos + 2).IsFixed());
615
616 for (IndexType j = initial_k; j < 3; j++) {
617 if (fix_rotation[j]) {
618 r_current_angular_acceleration[j] = 0.0;
619 }
620 } // trans DoF
621 } // Rot DoF
622 } // for node parallel
623 KRATOS_CATCH("")
624 }

697 void Predict(
698 ModelPart& rModelPart,
699 DofsArrayType& rDofSet,
700 TSystemMatrixType& A,
701 TSystemVectorType& Dx,
702 TSystemVectorType& b
703 ) override
704 {
705 KRATOS_TRY;
706 // The current process info
707 const ProcessInfo& r_current_process_info = rModelPart.GetProcessInfo();
708
709 // The array of nodes
710 NodesArrayType& r_nodes = rModelPart.Nodes();
711
713 const SizeType dim = r_current_process_info[DOMAIN_SIZE];
714
715 // Step Update
716 // The first step is time = initial_time ( 0.0) + delta time
717 mTime.Delta = r_current_process_info[DELTA_TIME];
718 mTime.MidStep = mTime.Delta * mDeltaTime.Fraction;
719
720 // The iterator of the first node
721 const auto it_node_begin = rModelPart.NodesBegin();
722 const bool has_dof_for_rot_z = it_node_begin->HasDofFor(ROTATION_Z);
723
724
725 // Getting dof position
726 const IndexType disppos = it_node_begin->GetDofPosition(DISPLACEMENT_X);
727 const IndexType rotppos = has_dof_for_rot_z ? it_node_begin->GetDofPosition(ROTATION_X) : 0;
728
729 // _____________________________________________________________________
730 // ______________________________ STAGE 1 ______________________________
731 // _____________________________________________________________________
732
733 #pragma omp parallel for schedule(guided,512)
734 for (int i = 0; i < static_cast<int>(r_nodes.size()); ++i) {
735 // Current step information "N+1" (before step update).
736 this->UpdateDegreesOfFreedomStage1(it_node_begin + i,
737 VELOCITY,DISPLACEMENT,ACCELERATION,dim);
738 } // for Node parallel
739
740
741 if (has_dof_for_rot_z){
742 #pragma omp parallel for schedule(guided,512)
743 for (int i = 0; i < static_cast<int>(r_nodes.size()); ++i) {
744 // Current step information "N+1" (before step update).
745 this->UpdateDegreesOfFreedomStage1(it_node_begin + i,
746 ANGULAR_VELOCITY,ROTATION,ANGULAR_ACCELERATION,dim);
747 } // for Node parallel
748 }
749
750
751 InitializeResidual(rModelPart);
752 CalculateAndAddRHS(rModelPart);
753
754 #pragma omp parallel for schedule(guided,512)
755 for (int i = 0; i < static_cast<int>(r_nodes.size()); ++i) {
756 // Current step information "N+1" (before step update).
757 this->UpdateAccelerationStage(
758 it_node_begin + i, disppos,FRACTIONAL_ACCELERATION,
759 VELOCITY,NODAL_DISPLACEMENT_DAMPING,FORCE_RESIDUAL,NODAL_MASS,
760 DISPLACEMENT_X,DISPLACEMENT_Y,DISPLACEMENT_Z,dim);
761 } // for Node parallel
762
763 if (has_dof_for_rot_z){
764 #pragma omp parallel for schedule(guided,512)
765 for (int i = 0; i < static_cast<int>(r_nodes.size()); ++i) {
766 // Current step information "N+1" (before step update).
767 this->UpdateAccelerationStage(
768 it_node_begin + i, rotppos,FRACTIONAL_ANGULAR_ACCELERATION,
769 ANGULAR_VELOCITY,NODAL_ROTATION_DAMPING,MOMENT_RESIDUAL,
770 NODAL_INERTIA,ROTATION_X,ROTATION_Y,ROTATION_Z,dim);
771 } // for Node parallel
772 }
773
774 // _____________________________________________________________________
775 // ______________________________ STAGE 2 ______________________________
776 // _____________________________________________________________________
777
778 #pragma omp parallel for schedule(guided,512)
779 for (int i = 0; i < static_cast<int>(r_nodes.size()); ++i) {
780 // Current step information "N+1" (before step update).
781 this->UpdateDegreesOfFreedomStage2(it_node_begin + i,
782 VELOCITY,DISPLACEMENT,ACCELERATION,FRACTIONAL_ACCELERATION,dim);
783 } // for Node parallel
784
785 if (has_dof_for_rot_z){
786 #pragma omp parallel for schedule(guided,512)
787 for (int i = 0; i < static_cast<int>(r_nodes.size()); ++i) {
788 // Current step information "N+1" (before step update).
789 this->UpdateDegreesOfFreedomStage2(it_node_begin + i,
790 ANGULAR_VELOCITY,ROTATION,ANGULAR_ACCELERATION,
791 FRACTIONAL_ANGULAR_ACCELERATION,dim);
792 } // for Node parallel
793 }
794
795 InitializeResidual(rModelPart);
796 CalculateAndAddRHS(rModelPart);
797
798 KRATOS_CATCH("")
799 }
