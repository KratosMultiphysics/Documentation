314 {
315 TDataType aux_sum = TDataType();
316 #pragma omp parallel for reduction(+:aux_sum)
317 for (int i=0; i<static_cast<int>(rA.size1()); ++i) {
318 for (int j=0; j<static_cast<int>(rA.size2()); ++j) {

327 static TDataType JacobiNorm(const compressed_matrix<TDataType>& rA)
328 {
329 TDataType aux_sum = TDataType();
330
331 typedef typename compressed_matrix<TDataType>::const_iterator1 t_it_1;
332 typedef typename compressed_matrix<TDataType>::const_iterator2 t_it_2;
333
334 for (t_it_1 it_1 = rA.begin1(); it_1 != rA.end1(); ++it_1) {
335 for (t_it_2 it_2 = it_1.begin(); it_2 != it_1.end(); ++it_2) {
336 if (it_2.index1() != it_2.index2()) {
337 aux_sum += std::abs(*it_2);
338 }
339 }
340 }

447#ifndef _OPENMP

456 if (rX.size() != static_cast<unsigned int>(size) )
457 rX.resize(size, false);

459 if (A == 1.00)

461 #pragma omp parallel for

464 }
465 else if (A == -1.00)

473 #pragma omp parallel for

670 template<class TOtherMatrixType, class TEquationIdVectorType>
671 inline static void AssembleLHS(
672 MatrixType& A,
673 TOtherMatrixType& LHS_Contribution,
674 TEquationIdVectorType& EquationId
675 )
676 {
677 unsigned int system_size = A.size1();
678 unsigned int local_size = LHS_Contribution.size1();
679
680 for (unsigned int i_local = 0; i_local < local_size; i_local++)
681 {
682 unsigned int i_global = EquationId[i_local];
683 if (i_global < system_size)
684 {
685 for (unsigned int j_local = 0; j_local < local_size; j_local++)
686 {
687 unsigned int j_global = EquationId[j_local];
688 if (j_global < system_size)
689 A(i_global, j_global) += LHS_Contribution(i_local, j_local);
690 }
691 }
692 }

704 static double CheckAndCorrectZeroDiagonalValues(
705 const ProcessInfo& rProcessInfo,
706 MatrixType& rA,
707 VectorType& rb,
708 const SCALING_DIAGONAL ScalingDiagonal = SCALING_DIAGONAL::NO_SCALING
709 )
710 {
711 // The system size
712 const std::size_t system_size = rA.size1();
713
714 // The matrix data
715 auto& r_Avalues = rA.value_data();
716 const auto& r_Arow_indices = rA.index1_data();
717 const auto& r_Acol_indices = rA.index2_data();
718
719 // Define the iterators
720 const auto it_Acol_indices_begin = r_Acol_indices.begin();
721
722 // Define zero value tolerance
723 const double zero_tolerance = std::numeric_limits<double>::epsilon();
724
725 // The diagonal considered
726 const double scale_factor = GetScaleNorm(rProcessInfo, rA, ScalingDiagonal);
727
728 // Detect if there is a line of all zeros and set the diagonal to a 1 if this happens
729 IndexPartition(system_size).for_each([&](std::size_t Index){
730 bool empty = true;
731
732 const std::size_t col_begin = r_Arow_indices[Index];
733 const std::size_t col_end = r_Arow_indices[Index + 1];
734
735 for (std::size_t j = col_begin; j < col_end; ++j) {
736 if(std::abs(r_Avalues[j]) > zero_tolerance) {
737 empty = false;
738 break;
739 }
740 }
741
742 if(empty) {
743 const auto it_Acol_indices_row_begin = it_Acol_indices_begin + col_begin;
744 const auto it_Acol_indices_row_end = it_Acol_indices_begin + col_end;
745
746 const auto lower = std::lower_bound(it_Acol_indices_row_begin, it_Acol_indices_row_end, Index);
747 const auto upper = std::upper_bound(it_Acol_indices_row_begin, it_Acol_indices_row_end, Index);
748
749 if (lower != upper) { // Index was found
750 r_Avalues[std::distance(it_Acol_indices_begin, lower)] = scale_factor;
751 } else {
752 KRATOS_DEBUG_ERROR << "Diagonal term (" << Index << ", " << Index << ") is not defined in the system matrix" << std::endl;
753 KRATOS_WARNING("UblasSpace") << "Diagonal term (" << Index << ", " << Index << ") is not defined in the system matrix" << std::endl;
754 }
755 rb[Index] = 0.0;
756 }
757 });
758
759 return scale_factor;
760 }

769 static double GetScaleNorm(
770 const ProcessInfo& rProcessInfo,
771 const MatrixType& rA,
772 const SCALING_DIAGONAL ScalingDiagonal = SCALING_DIAGONAL::NO_SCALING
773 )
774 {
775 switch (ScalingDiagonal) {
776 case SCALING_DIAGONAL::NO_SCALING:
777 return 1.0;
778 case SCALING_DIAGONAL::CONSIDER_PRESCRIBED_DIAGONAL: {
779 KRATOS_ERROR_IF_NOT(rProcessInfo.Has(BUILD_SCALE_FACTOR)) << "Scale factor not defined at process info" << std::endl;
780 return rProcessInfo.GetValue(BUILD_SCALE_FACTOR);
781 }
782 case SCALING_DIAGONAL::CONSIDER_NORM_DIAGONAL:
783 return GetDiagonalNorm(rA)/static_cast<double>(rA.size1());
784 case SCALING_DIAGONAL::CONSIDER_MAX_DIAGONAL:
785 return GetMaxDiagonal(rA);
786 default:
787 return GetMaxDiagonal(rA);
788 }
789 }

796 static double GetDiagonalNorm(const MatrixType& rA)
797 {
798 const auto& Avalues = rA.value_data();
799 const auto& Arow_indices = rA.index1_data();
800 const auto& Acol_indices = rA.index2_data();
801
802 const double diagonal_norm = IndexPartition<std::size_t>(Size1(rA)).for_each<SumReduction<double>>([&](std::size_t Index){
803 const std::size_t col_begin = Arow_indices[Index];
804 const std::size_t col_end = Arow_indices[Index+1];
805 for (std::size_t j = col_begin; j < col_end; ++j) {
806 if (Acol_indices[j] == Index ) {
807 return std::pow(Avalues[j], 2);
808 }
809 }
810 return 0.0;
811 });
812
813 return std::sqrt(diagonal_norm);
814 }

831 static double GetMaxDiagonal(const MatrixType& rA)
832 {
833 const auto& Avalues = rA.value_data();
834 const auto& Arow_indices = rA.index1_data();
835 const auto& Acol_indices = rA.index2_data();
836
837 return IndexPartition<std::size_t>(Size1(rA)).for_each<MaxReduction<double>>([&](std::size_t Index){
838 const std::size_t col_begin = Arow_indices[Index];
839 const std::size_t col_end = Arow_indices[Index+1];
840 for (std::size_t j = col_begin; j < col_end; ++j) {
841 if (Acol_indices[j] == Index ) {
