193void CalculateAirMaterialResponse() {
194 const unsigned int strain_size = 3 * (TDim - 1);
195
196 if(this->C.size1() != strain_size)
197 this->C.resize(strain_size,strain_size,false);
198 if(this->ShearStress.size() != strain_size)
199 this->ShearStress.resize(strain_size,false);
200
201 ComputeStrain();
202
203 CalculateEffectiveViscosityAtGaussPoint();
204
205 const double mu = this->EffectiveViscosity;
206 const double c1 = 2.0*mu;
207 const double c2 = mu;
208 this->C.clear();
209 BoundedMatrix<double, strain_size, strain_size> c_mat = this->C;
210 Vector& stress = this->ShearStress;
211 Vector& strain = this->StrainRate;
212
213 FluidElementUtilities<TNumNodes>::GetNewtonianConstitutiveMatrix(mu, c_mat);
214 this->C = c_mat;
215
216 if constexpr (TDim == 2)
217 {
218 const double trace = strain[0] + strain[1];
219 const double volumetric_part = trace/2.0; // Note: this should be small for an incompressible fluid (it is basically the incompressibility error)
220
221 stress[0] = c1 * (strain[0] - volumetric_part);
222 stress[1] = c1 * (strain[1] - volumetric_part);
223 stress[2] = c2 * strain[2];
224 }
225
226 else if constexpr (TDim == 3)
227 {
228 const double trace = strain[0] + strain[1] + strain[2];
229 const double volumetric_part = trace/3.0; // Note: this should be small for an incompressible fluid (it is basically the incompressibility error)
230
231 stress[0] = c1*(strain[0] - volumetric_part);
232 stress[1] = c1*(strain[1] - volumetric_part);
233 stress[2] = c1*(strain[2] - volumetric_part);
234 stress[3] = c2*strain[3];
235 stress[4] = c2*strain[4];
236 stress[5] = c2*strain[5];
237 }
238}

240void ComputeStrain()
241{
242 const BoundedMatrix<double, TNumNodes, TDim>& v = Velocity;
243 const BoundedMatrix<double, TNumNodes, TDim>& DN = this->DN_DX;
244
245 // Compute strain (B*v)
246 // 3D strain computation
247 if constexpr (TDim == 3)
248 {
249 this->StrainRate[0] = DN(0,0)*v(0,0) + DN(1,0)*v(1,0) + DN(2,0)*v(2,0) + DN(3,0)*v(3,0);
250 this->StrainRate[1] = DN(0,1)*v(0,1) + DN(1,1)*v(1,1) + DN(2,1)*v(2,1) + DN(3,1)*v(3,1);
251 this->StrainRate[2] = DN(0,2)*v(0,2) + DN(1,2)*v(1,2) + DN(2,2)*v(2,2) + DN(3,2)*v(3,2);
252 this->StrainRate[3] = DN(0,0)*v(0,1) + DN(0,1)*v(0,0) + DN(1,0)*v(1,1) + DN(1,1)*v(1,0) + DN(2,0)*v(2,1) + DN(2,1)*v(2,0) + DN(3,0)*v(3,1) + DN(3,1)*v(3,0);
253 this->StrainRate[4] = DN(0,1)*v(0,2) + DN(0,2)*v(0,1) + DN(1,1)*v(1,2) + DN(1,2)*v(1,1) + DN(2,1)*v(2,2) + DN(2,2)*v(2,1) + DN(3,1)*v(3,2) + DN(3,2)*v(3,1);
254 this->StrainRate[5] = DN(0,0)*v(0,2) + DN(0,2)*v(0,0) + DN(1,0)*v(1,2) + DN(1,2)*v(1,0) + DN(2,0)*v(2,2) + DN(2,2)*v(2,0) + DN(3,0)*v(3,2) + DN(3,2)*v(3,0);
255 }
256 // 2D strain computation
257 else if constexpr (TDim == 2)
258 {
259 this->StrainRate[0] = DN(0,0)*v(0,0) + DN(1,0)*v(1,0) + DN(2,0)*v(2,0);
260 this->StrainRate[1] = DN(0,1)*v(0,1) + DN(1,1)*v(1,1) + DN(2,1)*v(2,1);
261 this->StrainRate[2] = DN(0,1)*v(0,0) + DN(0,0)*v(0,1) + DN(1,1)*v(1,0) + DN(1,0)*v(1,1) + DN(2,1)*v(2,0) + DN(2,0)*v(2,1);
262 }
263}

265double ComputeStrainNorm()
266{
267 double strain_rate_norm;
268 Vector& S = this->StrainRate;
269 if constexpr (TDim == 3)
270 {
271 strain_rate_norm = std::sqrt(2.*S[0] * S[0] + 2.*S[1] * S[1] + 2.*S[2] * S[2] +
272 S[3] * S[3] + S[4] * S[4] + S[5] * S[5]);
273 }
274
275 else if constexpr (TDim == 2)
276 {
277 strain_rate_norm = std::sqrt(2.*S[0] * S[0] + 2.*S[1] * S[1] + S[2] * S[2]);
278 }
279 return strain_rate_norm;
280}

282void CalculateDensityAtGaussPoint()
283{
284 double dist = 0.0;
285 for (unsigned int i = 0; i < TNumNodes; i++)
286 dist += this->N[i] * Distance[i];
287
288 int navg = 0;
289 double density = 0.0;
290 for (unsigned int i = 0; i < TNumNodes; i++)
291 {
292 if (dist * Distance[i] > 0.0)
293 {
294 navg += 1;
295 density += NodalDensity[i];
296 }
297 }
298
299 Density = density / navg;
300}
301
302void CalculateEffectiveViscosityAtGaussPoint()
303{
304 double dist = 0.0;
305 for (unsigned int i = 0; i < TNumNodes; i++)
306 dist += this->N[i] * Distance[i];
307
308 int navg = 0;
309 double dynamic_viscosity = 0.0;
310 for (unsigned int i = 0; i < TNumNodes; i++)
311 {
312 if (dist * Distance[i] > 0.0)
313 {
314 navg += 1;
315 dynamic_viscosity += NodalDynamicViscosity[i];
316 }
317 }
318 DynamicViscosity = dynamic_viscosity / navg;
319
320 if (SmagorinskyConstant > 0.0)
321 {
322 const double strain_rate_norm = ComputeStrainNorm();
323
324 double length_scale = SmagorinskyConstant*ElementSize;
325 length_scale *= length_scale; // square
326 this->EffectiveViscosity = DynamicViscosity + 2.0*length_scale*strain_rate_norm;
327 }
328 else this->EffectiveViscosity = DynamicViscosity;
329}
