215 const GeometryPointer pGetGeometryPart(const IndexType Index) const override
216 {
217 if (Index == GeometryType::BACKGROUND_GEOMETRY_INDEX)
218 return mpNurbsSurface;
219
220 for (IndexType i = 0; i < mOuterLoopArray.size(); ++i)
221 {
222 for (IndexType j = 0; j < mOuterLoopArray[i].size(); ++j)
223 {
224 if (mOuterLoopArray[i][j]->Id() == Index)
225 return mOuterLoopArray[i][j];
226 }
227 }
228
229 for (IndexType i = 0; i < mInnerLoopArray.size(); ++i)
230 {
231 for (IndexType j = 0; j < mInnerLoopArray[i].size(); ++j)
232 {
233 if (mInnerLoopArray[i][j]->Id() == Index)
234 return mInnerLoopArray[i][j];
235 }
236 }
237
238 for (IndexType i = 0; i < mEmbeddedEdgesArray.size(); ++i)
239 {
240 if (mEmbeddedEdgesArray[i]->Id() == Index)
241 return mEmbeddedEdgesArray[i];
242 }
243
244 KRATOS_ERROR << "Index " << Index << " not existing in BrepSurface: "
245 << this->Id() << std::endl;
246 }

432 /* Creates integration points on the nurbs surface of this geometry.
433 * Accounting for whether the surface is trimmed or untrimmed, and whether shifted
434 * boundary conditions are used
435 *
436 * - **Untrimmed Surface**: -> Non-cutting case
437 * - If `TShiftedBoundary` is true, the method prepares for the shifted boundary method (SBM)
438 * - Otherwise, it directly uses `CreateIntegrationPoints` from the underlying NURBS surface.
439 * - **Trimmed Surface**: -> Cutting case
440 * - It calls `BrepTrimmingUtilities::CreateBrepSurfaceTrimmingIntegrationPoints`
441 * to generate integration points that conform to the trimming curve.
442 *
443 * @param return integration points.
444 */
445 void CreateIntegrationPoints(
446 IntegrationPointsArrayType& rIntegrationPoints,
447 IntegrationInfo& rIntegrationInfo) const override
448 {
449 if (!mIsTrimmed) {
450 // sbm case
451 if constexpr (TShiftedBoundary) {
452 // TODO: Next PR -> Call "BrepSBMUtilities::CreateBrepSurfaceSBMIntegrationPoints"
453 mpNurbsSurface->CreateIntegrationPoints(
454 rIntegrationPoints, rIntegrationInfo);
455 }
456 // body-fitted case
457 else {
458 mpNurbsSurface->CreateIntegrationPoints(
459 rIntegrationPoints, rIntegrationInfo);
460 }
461 }
462 // trimmed case
463 else
464 {
465 std::vector<double> spans_u;
466 std::vector<double> spans_v;
467 mpNurbsSurface->SpansLocalSpace(spans_u, 0);
468 mpNurbsSurface->SpansLocalSpace(spans_v, 1);
469
470 BrepTrimmingUtilities::CreateBrepSurfaceTrimmingIntegrationPoints(
471 rIntegrationPoints,
472 mOuterLoopArray, mInnerLoopArray,
473 spans_u, spans_v,
474 rIntegrationInfo);
475 }
476 }
