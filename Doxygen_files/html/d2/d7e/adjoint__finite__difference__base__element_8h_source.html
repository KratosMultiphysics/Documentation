413 template <typename TDataType>
414 void CalculateAdjointFieldOnIntegrationPoints(const Variable<TDataType>& rVariable, std::vector< TDataType >& rOutput, const ProcessInfo& rCurrentProcessInfo)
415 {
416 KRATOS_TRY;
417
418 KRATOS_WARNING_IF("CalculateAdjointFieldOnIntegrationPoints", OpenMPUtils::IsInParallel() != 0)
419 << "The call of this non shared-memory-parallelized function within a parallel section should be avoided for efficiency reasons!" << std::endl;
420
421 const SizeType num_nodes = mpPrimalElement->GetGeometry().PointsNumber();
422 const SizeType dimension = mpPrimalElement->GetGeometry().WorkingSpaceDimension();
423 const SizeType num_dofs_per_node = (mHasRotationDofs) ? 2 * dimension : dimension;
424 const SizeType num_dofs = num_nodes * num_dofs_per_node;
425 Vector initial_state_variables;
426 initial_state_variables.resize(num_dofs);
427
428 Vector particular_solution = ZeroVector(num_dofs);
429 if(this->Has(ADJOINT_PARTICULAR_DISPLACEMENT)) {
430 particular_solution = this->GetValue(ADJOINT_PARTICULAR_DISPLACEMENT);
431 }
432
433 // Build vector of variables containing the DOF-variables of the primal problem
434 std::vector<Variable<double>*> primal_solution_variable_list;
435 (mHasRotationDofs) ? primal_solution_variable_list = {&DISPLACEMENT_X, &DISPLACEMENT_Y, &DISPLACEMENT_Z, &ROTATION_X, &ROTATION_Y, &ROTATION_Z} :
436 primal_solution_variable_list = {&DISPLACEMENT_X, &DISPLACEMENT_Y, &DISPLACEMENT_Z};
437
438 // Build vector of variables containing the DOF-variables of the adjoint problem
439 std::vector<Variable<double>*> adjoint_solution_variable_list;
440 (mHasRotationDofs) ? adjoint_solution_variable_list = {&ADJOINT_DISPLACEMENT_X, &ADJOINT_DISPLACEMENT_Y, &ADJOINT_DISPLACEMENT_Z, &ADJOINT_ROTATION_X, &ADJOINT_ROTATION_Y, &ADJOINT_ROTATION_Z} :
441 adjoint_solution_variable_list = {&ADJOINT_DISPLACEMENT_X, &ADJOINT_DISPLACEMENT_Y, &ADJOINT_DISPLACEMENT_Z};
442
443 for (IndexType i = 0; i < num_nodes; ++i) {
444 const IndexType index = i * num_dofs_per_node;
445 for(IndexType j = 0; j < primal_solution_variable_list.size(); ++j) {
446 initial_state_variables[index + j] = mpPrimalElement->GetGeometry()[i].FastGetSolutionStepValue(*primal_solution_variable_list[j]);
447 mpPrimalElement->GetGeometry()[i].FastGetSolutionStepValue(*primal_solution_variable_list[j]) =
448 this->GetGeometry()[i].FastGetSolutionStepValue(*adjoint_solution_variable_list[j]) +
449 particular_solution[index + j];
450 }
451 }
452
453 mpPrimalElement->CalculateOnIntegrationPoints(rVariable, rOutput, rCurrentProcessInfo);
454
455 for (IndexType i = 0; i < num_nodes; ++i) {
456 const IndexType index = i * num_dofs_per_node;
457 for(IndexType j = 0; j < primal_solution_variable_list.size(); ++j) {
458 mpPrimalElement->GetGeometry()[i].FastGetSolutionStepValue(*primal_solution_variable_list[j]) = initial_state_variables[index + j];
459 }
460 }
461
462 KRATOS_CATCH("")
463 }
