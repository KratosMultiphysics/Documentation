90
91 if (mEchoLevel > 1)
92 {
93 std::cout << " [ SELECT MESH ELEMENTS FOR CUT PFEM ANALYSIS in PfemFluid: (" << mrRemesh.OutMesh.GetNumberOfElements() << ") " << std::endl;
94 std::cout << "MODEL PART InNumberOfElements " << mrRemesh.InMesh.GetNumberOfElements() << std::endl;
95 std::cout << "MODEL PART InNumberOfPoints " << mrRemesh.InMesh.GetNumberOfPoints() << std::endl;

97 std::cout << "MODEL PART OutNumberOfPoints " << mrRemesh.OutMesh.GetNumberOfPoints() << std::endl;
98 }
99 const int &OutNumberOfElements = mrRemesh.OutMesh.GetNumberOfElements();
100 mrRemesh.PreservedElements.clear();
101 mrRemesh.PreservedElements.resize(OutNumberOfElements, false);
102 std::fill(mrRemesh.PreservedElements.begin(), mrRemesh.PreservedElements.end(), 0);
103 mrRemesh.MeshElementsSelectedFlag = true;
104
105 MesherUtilities MesherUtils;
106 double ModelPartVolume = MesherUtils.ComputeModelPartVolume(mrModelPart);
107 double CriticalVolume = 0.05 * ModelPartVolume / double(mrModelPart.Elements().size());
108
109 mrRemesh.Info->NumberOfElements = 0;
110
111 const ProcessInfo &rCurrentProcessInfo = mrModelPart.GetProcessInfo();
112 double currentTime = rCurrentProcessInfo[TIME];
113 double deltaTime = rCurrentProcessInfo[DELTA_TIME];
114 int number_of_slivers = 0;
115
116 bool refiningBox = false;
117 for (SizeType index = 0; index < mrRemesh.UseRefiningBox.size(); index++)
118 {
119 if (mrRemesh.UseRefiningBox[index] == true && currentTime > mrRemesh.RefiningBoxInitialTime[index] && currentTime < mrRemesh.RefiningBoxFinalTime[index])
120 {
121 refiningBox = true;
122 }
123 }
124
125 if (mrRemesh.ExecutionOptions.IsNot(MesherUtilities::SELECT_TESSELLATION_ELEMENTS))
126 {
127 for (int el = 0; el < OutNumberOfElements; el++)
128 {
129 mrRemesh.PreservedElements[el] = 1;
130 mrRemesh.Info->NumberOfElements += 1;
131 }
132 }
133 else
134 {
135 if (mEchoLevel > 1)
136 std::cout << " Start Element Selection " << OutNumberOfElements << std::endl;
137
138 ModelPart::ElementsContainerType::iterator element_begin = mrModelPart.ElementsBegin();
139 const SizeType nds = element_begin->GetGeometry().size();
140 const SizeType dimension = element_begin->GetGeometry().WorkingSpaceDimension();
141 int *OutElementList = mrRemesh.OutMesh.GetElementList();
142 ModelPart::NodesContainerType &rNodes = mrModelPart.Nodes();
143
144 int el = 0;
145 int number = 0;
146
147 // #pragma omp parallel for reduction(+:number) private(el)
148 for (el = 0; el < OutNumberOfElements; el++)
149 {
150 Geometry<Node> vertices;
151 double meanMeshSize = mrRemesh.Refine->CriticalRadius; // this must be inside because if there is a refined zone, each element has a different critical radius
152 double distance_tolerance = 0.05 * meanMeshSize;
153 SizeType numfreesurf = 0;
154 SizeType numboundary = 0;
155 SizeType numrigid = 0;
156 SizeType numInletNodes = 0;
157 SizeType numisolated = 0;
158 bool noremesh = false;
159 std::vector<double> normVelocityP;
160 normVelocityP.resize(nds, false);
161 SizeType checkedNodes = 0;
162 bool increaseAlfa = false;
163 SizeType previouslyFreeSurfaceNodes = 0;
164 SizeType previouslyIsolatedNodes = 0;
165 SizeType sumPreviouslyIsolatedFreeSurf = 0;
166 SizeType sumIsolatedFreeSurf = 0;
167 std::vector<array_1d<double, 3>> nodesCoordinates;
168 nodesCoordinates.resize(nds);
169 std::vector<array_1d<double, 3>> nodesVelocities;
170 nodesVelocities.resize(nds);
171 SizeType isolatedNodesInTheElement = 0;
172 double rigidNodeLocalMeshSize = 0;
173 double rigidNodeMeshCounter = 0;
174
175 for (SizeType pn = 0; pn < nds; pn++)
176 {
177 if (OutElementList[el * nds + pn] <= 0)
178 std::cout << " ERROR: something is wrong: nodal id < 0 " << el << std::endl;
179
180 if ((SizeType)OutElementList[el * nds + pn] > mrRemesh.NodalPreIds.size())
181 {
182 std::cout << " ERROR: something is wrong: node out of bounds " << std::endl;
183 break;
184 }
185 vertices.push_back(rNodes(OutElementList[el * nds + pn]));
186
187 if (vertices.back().IsNot(RIGID) && vertices.back().IsNot(SOLID) && vertices.back().GetSolutionStepValue(DISTANCE) > distance_tolerance)
188 {
189 isolatedNodesInTheElement += vertices.back().FastGetSolutionStepValue(ISOLATED_NODE);
190 }
191 // check flags on nodes
192 if (vertices.back().Is(ISOLATED))
193 {
194 numisolated++;
195 }
196 if (vertices.back().Is(PFEMFlags::PREVIOUS_FREESURFACE))
197 {
198 previouslyFreeSurfaceNodes++;
199 }
200 if (vertices.back().Is(PFEMFlags::PREVIOUS_ISOLATED))
201 {
202 previouslyIsolatedNodes++;
203 }
204 if (vertices.back().Is(BOUNDARY))
205 {
206 numboundary++;
207 }
208 if (vertices.back().GetValue(NO_MESH))
209 {
210 noremesh = true;
211 }
212
213 if (vertices.back().Is(RIGID) || vertices.back().Is(SOLID) || (vertices.back().GetSolutionStepValue(DISTANCE) < 0.0 && dimension == 3))
214 // if (vertices.back().Is(RIGID) || vertices.back().Is(SOLID))
215 {
216 if (vertices.back().Is(RIGID) || (vertices.back().GetSolutionStepValue(DISTANCE) < 0.0 && dimension == 3))
217 {
218 rigidNodeLocalMeshSize += vertices.back().FastGetSolutionStepValue(NODAL_H_WALL);
219 rigidNodeMeshCounter += 1.0;
220 }
221
222 numrigid++;
223 }
224
225 if (vertices.back().IsNot(RIGID) && vertices.back().Is(BOUNDARY) && vertices.back().GetSolutionStepValue(DISTANCE) > distance_tolerance)
226 {
227 numfreesurf++;
228 const array_1d<double, 3> &velocityP0 = vertices.back().FastGetSolutionStepValue(VELOCITY, 0);
229 normVelocityP[pn] = norm_2(velocityP0);
230 nodesVelocities[pn] = velocityP0;
231 checkedNodes++;
232 }
233 else if (vertices.back().Is(ISOLATED))
234 {
235 const array_1d<double, 3> &velocityP0 = vertices.back().FastGetSolutionStepValue(VELOCITY, 0);
236 normVelocityP[pn] = norm_2(velocityP0);
237 nodesVelocities[pn] = velocityP0;
238 checkedNodes++;
239 }
240 if (vertices.back().Is(INLET))
241 {
242 numInletNodes++;
243 }
244
245 if (refiningBox == true && vertices.back().IsNot(RIGID) && vertices.back().GetSolutionStepValue(DISTANCE) > distance_tolerance)
246 {
247 if (dimension == 2)
248 {
249 MesherUtils.DefineMeshSizeInTransitionZones2D(mrRemesh, currentTime, vertices.back().Coordinates(), meanMeshSize, increaseAlfa);
250 }
251 else if (dimension == 3)
252 {
253 MesherUtils.DefineMeshSizeInTransitionZones3D(mrRemesh, currentTime, vertices.back().Coordinates(), meanMeshSize, increaseAlfa);
254 }
255 CriticalVolume = 0.05 * (std::pow(meanMeshSize, 3) / (6.0 * std::sqrt(2)));
256 }
257
258 if (dimension == 3)
259 {
260 nodesCoordinates[pn] = vertices.back().Coordinates();
261 }
262 }
263
264 double Alpha = mrRemesh.AlphaParameter; //*nds;
265
266 if (rigidNodeMeshCounter > 0)
267 {
268 this->SetMeshSizeInBoundaryZones(meanMeshSize, previouslyFreeSurfaceNodes, numfreesurf, currentTime, deltaTime, rigidNodeLocalMeshSize, rigidNodeMeshCounter);
269 }
270
271 if (refiningBox == true)
272 {
273 this->IncreaseAlphaForRefininedZones(Alpha, increaseAlfa, nds, numfreesurf, numrigid, numisolated);
274 }
275
276 sumIsolatedFreeSurf = numisolated + numfreesurf;
277 sumPreviouslyIsolatedFreeSurf = previouslyFreeSurfaceNodes + previouslyIsolatedNodes;
278
279 this->ModifyAlpha(Alpha, dimension, nds, numfreesurf, numrigid, numisolated, numInletNodes, previouslyIsolatedNodes, previouslyFreeSurfaceNodes);
280
281 bool accepted = false;
282
283 accepted = MesherUtils.AlphaShape(Alpha, vertices, dimension, meanMeshSize);
284
285 if (numrigid == nds || noremesh == true)
286 {
287 accepted = false;
288 }
289
290 if (accepted == true && (numfreesurf == nds || sumIsolatedFreeSurf == nds || sumPreviouslyIsolatedFreeSurf == nds))
291 {
292 if (dimension == 2)
293 {
294 // this is to avoid the formation of isolated elements with different velocity fields. They can give convergence problems
295 if ((numfreesurf == nds || sumIsolatedFreeSurf == nds) && numrigid == 0)
296 {
297 if (checkedNodes == nds)
298 {
299 this->ControlSkewedElements2D(accepted, normVelocityP, nodesVelocities);
300 }
301 }
302 Geometry<Node> *triangle = new Triangle2D3<Node>(vertices);
303 double elementArea = triangle->Area();
304 if (elementArea < CriticalVolume)
305 {
306 accepted = false;
307 // std::cout << "ATTENTION!!! ELEMENT WITH AREA = " << elementArea << " versus critical area " << CriticalVolume << std::endl;
308 }
309 delete triangle;
310 }
311 else if (dimension == 3)
312 {
313 if ((numfreesurf == nds || sumIsolatedFreeSurf == nds || previouslyIsolatedNodes == nds || previouslyFreeSurfaceNodes == nds) && numrigid == 0 && isolatedNodesInTheElement > 0)
314 {
315 if (checkedNodes == nds)
316 {
317 this->ControlSkewedElements3D(accepted, normVelocityP, nodesVelocities);
318 }
319 }
320 }
321 }
322
323 // // to control that the element has a good shape
324 if (dimension == 3 && accepted && numrigid < 3 &&
325 (previouslyIsolatedNodes == 4 || previouslyFreeSurfaceNodes == 4 || sumIsolatedFreeSurf == 4 || numfreesurf == 4 || numisolated == 4 || (numrigid == 2 && isolatedNodesInTheElement > 1)))
326 {
327 this->ControlSliverElements(accepted, number_of_slivers, vertices, nodesCoordinates, CriticalVolume);
328 }
329
330 if (accepted)
331 {
332 number += 1;
333 mrRemesh.PreservedElements[el] = number;
334 }
335 }
336 mrRemesh.Info->NumberOfElements = number;
337 }
338
339 if (mEchoLevel > 1)
340 {
341 std::cout << "Number of Preserved Fluid Elements " << mrRemesh.Info->NumberOfElements << " (slivers detected: " << number_of_slivers << ") " << std::endl;
342 std::cout << "TOTAL removed nodes " << mrRemesh.Info->RemovedNodes << std::endl;
343 }
344 if (mrRemesh.ExecutionOptions.IsNot(MesherUtilities::KEEP_ISOLATED_NODES))
345 {
346 ModelPart::ElementsContainerType::iterator element_begin = mrModelPart.ElementsBegin();
347 const SizeType nds = (*element_begin).GetGeometry().size();
348
349 int *OutElementList = mrRemesh.OutMesh.GetElementList();
350
351 ModelPart::NodesContainerType &rNodes = mrModelPart.Nodes();
352
353 // check engaged nodes
354 for (int el = 0; el < OutNumberOfElements; el++)
355 {
356 if (mrRemesh.PreservedElements[el])
357 {
358 for (SizeType pn = 0; pn < nds; pn++)
359 {
360 // set vertices
361 rNodes[OutElementList[el * nds + pn]].Set(BLOCKED);
362 }
363 }
364 }
365
366 int count_release = 0;
367 for (ModelPart::NodesContainerType::iterator i_node = rNodes.begin(); i_node != rNodes.end(); i_node++)
368 {
369 if (i_node->IsNot(BLOCKED))
370 {
371 if (!(i_node->Is(FREE_SURFACE) || i_node->Is(RIGID) || i_node->GetSolutionStepValue(DISTANCE) < 0.0))
372 {
373 i_node->Set(TO_ERASE);
374 if (mEchoLevel > 0)
375 std::cout << " NODE " << i_node->Id() << " RELEASE " << std::endl;
376 if (i_node->Is(BOUNDARY))
377 std::cout << " ERROR: node " << i_node->Id() << " IS BOUNDARY RELEASE " << std::endl;
378 }
379 }
380 if (i_node->Is(TO_ERASE))
381 {
382 count_release++;
383 }
384
385 i_node->Reset(BLOCKED);
386 }
387
388 if (mEchoLevel > 0)
389 std::cout << " fluid NUMBER OF RELEASED NODES " << count_release << std::endl;
390 }
391 else
392 {
393 ModelPart::NodesContainerType &rNodes = mrModelPart.Nodes();
394 for (ModelPart::NodesContainerType::iterator i_node = rNodes.begin(); i_node != rNodes.end(); i_node++)
395 {
396 i_node->Reset(BLOCKED);
397 }
398 }
399
400 mrRemesh.InputInitializedFlag = false;
401 mMesherUtilities.SetNodes(mrModelPart, mrRemesh);
402 mrRemesh.InputInitializedFlag = true;
403
404 if (mEchoLevel > 1)
405 {
406 std::cout << " Generated_Elements :" << OutNumberOfElements << std::endl;
407 std::cout << " Passed_AlphaShape :" << mrRemesh.Info->NumberOfElements << std::endl;
408 std::cout << " SELECT MESH ELEMENTS ]; " << std::endl;
409 }
410
411 KRATOS_CATCH("")
412 }
