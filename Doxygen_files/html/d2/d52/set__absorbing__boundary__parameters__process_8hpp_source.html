29 SetAbsorbingBoundaryParametersProcess(ModelPart& model_part, Parameters rParameters)
30 : Process(Flags()), mrModelPart(model_part)
31 {
32 KRATOS_TRY
33
34 // only include validation with c++11 since raw_literals do not exist in c++03
35 Parameters default_parameters(R"(
36 {
37 "model_part_name":"PLEASE_CHOOSE_MODEL_PART_NAME",
38 "absorbing_factors": [1.0,1.0],
39 "virtual_thickness": 1e10
40 } )");
41
42 // Some values need to be mandatory prescribed since no meaningful default value exist. For
43 // this reason try accessing to them So that an error is thrown if they don't exist
44 rParameters["model_part_name"];
45
46 // Now validate against defaults -- this also ensures no type mismatch
47 rParameters.ValidateAndAssignDefaults(default_parameters);
48
49 // get absorbing factors
50 mAbsorbingFactors.resize(2, false);
51 mAbsorbingFactors(0) = rParameters["absorbing_factors"][0].GetDouble();
52 mAbsorbingFactors(1) = rParameters["absorbing_factors"][1].GetDouble();
53
54 // get virtual thickness
55 mVirtualThickness = rParameters["virtual_thickness"].GetDouble();
56
57 KRATOS_CATCH("")
58 }
