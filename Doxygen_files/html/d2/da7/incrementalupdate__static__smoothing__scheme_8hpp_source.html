65 void FinalizeSolutionStep(
66 ModelPart& rModelPart,
67 TSystemMatrixType& A,
68 TSystemVectorType& Dx,
69 TSystemVectorType& b) override
70 {
71 KRATOS_TRY
72
73 unsigned int Dim = rModelPart.GetProcessInfo()[DOMAIN_SIZE];
74
75 // Clear nodal variables
76 #pragma omp parallel
77 {
78 ModelPart::NodeIterator NodesBegin;
79 ModelPart::NodeIterator NodesEnd;
80 OpenMPUtils::PartitionedIterators(rModelPart.Nodes(),NodesBegin,NodesEnd);
81
82 for (ModelPart::NodeIterator itNode = NodesBegin; itNode != NodesEnd; ++itNode)
83 {
84 itNode->FastGetSolutionStepValue(NODAL_AREA) = 0.0;
85 Matrix& rNodalStress = itNode->FastGetSolutionStepValue(NODAL_CAUCHY_STRESS_TENSOR);
86 if(rNodalStress.size1() != Dim)
87 rNodalStress.resize(Dim,Dim,false);
88 noalias(rNodalStress) = ZeroMatrix(Dim,Dim);
89 itNode->FastGetSolutionStepValue(NODAL_JOINT_AREA) = 0.0;
90 itNode->FastGetSolutionStepValue(NODAL_JOINT_WIDTH) = 0.0;
91 }
92 }
93
94 BaseType::FinalizeSolutionStep(rModelPart,A,Dx,b);
95
96 // Compute smoothed nodal variables
97 #pragma omp parallel
98 {
99 ModelPart::NodeIterator NodesBegin;
100 ModelPart::NodeIterator NodesEnd;
101 OpenMPUtils::PartitionedIterators(rModelPart.Nodes(),NodesBegin,NodesEnd);
102
103 for (ModelPart::NodeIterator itNode = NodesBegin; itNode != NodesEnd; ++itNode)
104 {
105 const double& NodalArea = itNode->FastGetSolutionStepValue(NODAL_AREA);
106 if (NodalArea>1.0e-15)
107 {
108 const double InvNodalArea = 1.0/(NodalArea);
109 Matrix& rNodalStress = itNode->FastGetSolutionStepValue(NODAL_CAUCHY_STRESS_TENSOR);
110 for(unsigned int i = 0; i<Dim; i++)
111 {
112 for(unsigned int j = 0; j<Dim; j++)
113 {
114 rNodalStress(i,j) *= InvNodalArea;
115 }
116 }
117 }
118
119 const double& NodalJointArea = itNode->FastGetSolutionStepValue(NODAL_JOINT_AREA);
120 if (NodalJointArea>1.0e-15)
121 {
122 double& NodalJointWidth = itNode->FastGetSolutionStepValue(NODAL_JOINT_WIDTH);
123 NodalJointWidth = NodalJointWidth/NodalJointArea;
124 }
125 }
126 }
127
128 KRATOS_CATCH("")
129 }
