57template <typename DisplacementNodeRange, typename WaterPressureNodeRange>
58std::vector<Dof<double>*> ExtractUPwDofsFromNodes(const DisplacementNodeRange& rDisplacementNodes,
59 const WaterPressureNodeRange& rWaterPressureNodes,
60 std::size_t ModelDimension)
61{
62 auto result = std::vector<Dof<double>*>{};
63 auto displacement_variables =
64 Geo::ConstVariableRefs{std::cref(DISPLACEMENT_X), std::cref(DISPLACEMENT_Y)};
65 if (ModelDimension == 3) displacement_variables.push_back(std::cref(DISPLACEMENT_Z));
66
67 for (const auto& r_node : rDisplacementNodes) {
68 std::transform(std::begin(displacement_variables), std::end(displacement_variables),
69 std::back_inserter(result),
70 [&r_node](const auto& r_variable) { return r_node.pGetDof(r_variable.get()); });
71 }
72
73 ExtractDofsFromNodes(std::begin(rWaterPressureNodes), std::end(rWaterPressureNodes),
74 std::back_inserter(result), WATER_PRESSURE);
75
76 return result;
77}
