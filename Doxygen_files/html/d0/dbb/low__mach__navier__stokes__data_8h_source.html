101 void Initialize(const Element& rElement, const ProcessInfo& rProcessInfo) override
102 {
103 // Base class Initialize manages constitutive law parameters
104 FluidElementData<TDim,TNumNodes, true>::Initialize(rElement,rProcessInfo);
105
106 // Fill historical nodal data
107 const auto& r_geometry = rElement.GetGeometry();
108 this->FillFromHistoricalNodalData(Velocity, VELOCITY, r_geometry);
109 this->FillFromHistoricalNodalData(VelocityOldStep1, VELOCITY, r_geometry, 1);
110 this->FillFromHistoricalNodalData(VelocityOldStep2, VELOCITY, r_geometry, 2);
111 this->FillFromHistoricalNodalData(MeshVelocity, MESH_VELOCITY, r_geometry);
112 this->FillFromHistoricalNodalData(HeatFlux, HEAT_FLUX, r_geometry);
113 this->FillFromHistoricalNodalData(BodyForce, BODY_FORCE, r_geometry);
114 this->FillFromHistoricalNodalData(Pressure, PRESSURE, r_geometry);
115 this->FillFromHistoricalNodalData(PressureOldStep1, PRESSURE, r_geometry, 1);
116 this->FillFromHistoricalNodalData(PressureOldStep2, PRESSURE, r_geometry, 2);
117 this->FillFromHistoricalNodalData(Temperature, TEMPERATURE, r_geometry);
118 this->FillFromHistoricalNodalData(TemperatureOldStep1, TEMPERATURE, r_geometry, 1);
119 this->FillFromHistoricalNodalData(TemperatureOldStep2, TEMPERATURE, r_geometry, 2);
120 this->FillFromHistoricalNodalData(SolidFractionVelocity, SOLID_FRACTION_VELOCITY, r_geometry);
121
122 // Fill data from properties
123 const auto& r_properties = rElement.GetProperties();
124 this->FillFromProperties(Conductivity, CONDUCTIVITY, r_properties);
125 this->FillFromProperties(SpecificHeat, SPECIFIC_HEAT, r_properties);
126 this->FillFromProperties(HeatCapacityRatio, HEAT_CAPACITY_RATIO, r_properties);
127
128 // Fill data from ProcessInfo container
129 this->FillFromProcessInfo(DeltaTime, DELTA_TIME, rProcessInfo);
130 this->FillFromProcessInfo(DynamicTau, DYNAMIC_TAU, rProcessInfo);
131 if (rProcessInfo.Has(THERMODYNAMIC_PRESSURE)) {
132 this->FillFromProcessInfo(ThermodynamicPressure, THERMODYNAMIC_PRESSURE, rProcessInfo);
133 } else {
134 ThermodynamicPressure = 0.0;
135 }
136 if (rProcessInfo.Has(THERMODYNAMIC_PRESSURE_DERIVATIVE)) {
137 this->FillFromProcessInfo(ThermodynamicPressureDerivative, THERMODYNAMIC_PRESSURE_DERIVATIVE, rProcessInfo);
138 } else {
139 ThermodynamicPressureDerivative = 0.0;
140 }
141
142 // Get data from element database
143 this->FillFromElementData(Resistance, RESISTANCE, rElement);
144
145 // Set the BDF2 (time integration) coefficients
146 const auto& r_BDF_vector = rProcessInfo[BDF_COEFFICIENTS];
147 bdf0 = r_BDF_vector[0];
148 bdf1 = r_BDF_vector[1];
149 bdf2 = r_BDF_vector[2];
150
151 // Calculate element characteristic size
152 ElementSize = ElementSizeCalculator<TDim,TNumNodes>::MinimumElementSize(r_geometry);
153 }
