424 double Area() const override
425 {
426 array_1d<double, 3> p0 = 0.5 * (BaseType::GetPoint( 0 ) + BaseType::GetPoint( 3 ));
427 array_1d<double, 3> p1 = 0.5 * (BaseType::GetPoint( 1 ) + BaseType::GetPoint( 4 ));
428 array_1d<double, 3> p2 = 0.5 * (BaseType::GetPoint( 2 ) + BaseType::GetPoint( 5 ));
429
430 //heron formula
431 Vector side_a = ( p0 - p1 );
432 double a = MathUtils<double>::Norm3( side_a );
433 Vector side_b = ( p1 - p2 );
434 double b = MathUtils<double>::Norm3( side_b );
435 Vector side_c = ( p2 - p0 );
436 double c = MathUtils<double>::Norm3( side_c );
437 double s = ( a + b + c ) / 2;
438 return( sqrt( s*( s - a )*( s - b )*( s - c ) ) );
439 }

497 CoordinatesArrayType& PointLocalCoordinates(
498 CoordinatesArrayType& rResult,
499 const CoordinatesArrayType& rPoint
500 ) const override
501 {
502 BoundedMatrix<double,3,3> X;
503 BoundedMatrix<double,3,2> DN;
504 for(unsigned int i=0; i<3;i++) {
505 X(0,i ) = 0.5*(this->GetPoint( i ).X() + this->GetPoint( i+3 ).X());
506 X(1,i ) = 0.5*(this->GetPoint( i ).Y() + this->GetPoint( i+3 ).Y());
507 X(2,i ) = 0.5*(this->GetPoint( i ).Z() + this->GetPoint( i+3 ).Z());
508 }
509
510 double tol = 1.0e-8;
511 int maxiter = 1000;
512
513 Matrix J = ZeroMatrix( 2, 2 );
514 Matrix invJ = ZeroMatrix( 2, 2 );
515
516 //starting with xi = 0
517 rResult = ZeroVector( 3 );
518 Vector DeltaXi = ZeroVector( 2 );
519 array_1d<double,3> CurrentGlobalCoords;
520
521
522 //Newton iteration:
523 for ( int k = 0; k < maxiter; k++ )
524 {
525 noalias(CurrentGlobalCoords) = ZeroVector( 3 );
526 this->GlobalCoordinates( CurrentGlobalCoords, rResult );
527
528 noalias( CurrentGlobalCoords ) = rPoint - CurrentGlobalCoords;
529
530
531 //derivatives of shape functions
532 Matrix shape_functions_gradients;
533 shape_functions_gradients = ShapeFunctionsLocalGradients(shape_functions_gradients, rResult );
534 noalias(DN) = prod(X,shape_functions_gradients);
535
536 noalias(J) = prod(trans(DN),DN);
537 Vector res = prod(trans(DN),CurrentGlobalCoords);
538
539 //deteminant of Jacobian
540 const double det_j = J( 0, 0 ) * J( 1, 1 ) - J( 0, 1 ) * J( 1, 0 );
541
542 //filling matrix
543 invJ( 0, 0 ) = ( J( 1, 1 ) ) / ( det_j );
544 invJ( 1, 0 ) = -( J( 1, 0 ) ) / ( det_j );
545 invJ( 0, 1 ) = -( J( 0, 1 ) ) / ( det_j );
546 invJ( 1, 1 ) = ( J( 0, 0 ) ) / ( det_j );
547
548
549 DeltaXi( 0 ) = invJ( 0, 0 ) * res[0] + invJ( 0, 1 ) * res[1];
550 DeltaXi( 1 ) = invJ( 1, 0 ) * res[0] + invJ( 1, 1 ) * res[1];
551
552 rResult[0] += DeltaXi[0];
553 rResult[1] += DeltaXi[1];
554 rResult[2] = 0.0;
555
556 if ( k>0 && norm_2( DeltaXi ) > 30 )
557 {
558 KRATOS_ERROR << "Computation of local coordinates failed at iteration " << k<< std::endl;
559 }
560
561 if ( norm_2( DeltaXi ) < tol )
562 {
563 break;
564 }
565 }
566
567 return( rResult );
568 }

592 IndexType IntegrationPointIndex,
593 IntegrationMethod ThisMethod ) const override

629 Matrix& Jacobian( Matrix& rResult,
630 IndexType IntegrationPointIndex,
631 IntegrationMethod ThisMethod,
632 const Matrix& DeltaPosition ) const override
633 {
634 array_1d<double, 3> p0 = 0.5 * (BaseType::GetPoint( 0 ) + BaseType::GetPoint( 3 ));
635 array_1d<double, 3> p1 = 0.5 * (BaseType::GetPoint( 1 ) + BaseType::GetPoint( 4 ));
636 array_1d<double, 3> p2 = 0.5 * (BaseType::GetPoint( 2 ) + BaseType::GetPoint( 5 ));
637
638 Matrix deltaPosMid(3, 3);
639 for(unsigned int i = 0; i < 3; i++)
640 for(unsigned int j = 0; j < 3; j++)
641 deltaPosMid(i, j) = 0.5*( DeltaPosition(i, j) + DeltaPosition(i+3, j) );
642
643 if(rResult.size1() != 3 || rResult.size2() != 2)
644 rResult.resize(3, 2, false);
645
646 rResult( 0, 0 ) = -( p0[0] - deltaPosMid(0,0) ) + ( p1[0] - deltaPosMid(1,0) ); //on the Gauss points (J is constant at each element)
647 rResult( 1, 0 ) = -( p0[1] - deltaPosMid(0,1) ) + ( p1[1] - deltaPosMid(1,1) );
648 rResult( 2, 0 ) = -( p0[2] - deltaPosMid(0,2) ) + ( p1[2] - deltaPosMid(1,2) );
649 rResult( 0, 1 ) = -( p0[0] - deltaPosMid(0,0) ) + ( p2[0] - deltaPosMid(2,0) );
650 rResult( 1, 1 ) = -( p0[1] - deltaPosMid(0,1) ) + ( p2[1] - deltaPosMid(2,1) );
651 rResult( 2, 1 ) = -( p0[2] - deltaPosMid(0,2) ) + ( p2[2] - deltaPosMid(2,2) );
652
653 return rResult;

702 Vector& DeterminantOfJacobian( Vector& rResult,
703 IntegrationMethod ThisMethod ) const override
704 {
705 array_1d<double, 3> p0 = 0.5 * (BaseType::GetPoint( 0 ) + BaseType::GetPoint( 3 ));
706 array_1d<double, 3> p1 = 0.5 * (BaseType::GetPoint( 1 ) + BaseType::GetPoint( 4 ));
707 array_1d<double, 3> p2 = 0.5 * (BaseType::GetPoint( 2 ) + BaseType::GetPoint( 5 ));
708
709 array_1d<double, 3> Tangent1( p1 - p0 );
710 array_1d<double, 3> Tangent2( p2 - p0 );
711
712 array_1d<double, 3> Normal;
713 MathUtils<double>::CrossProduct( Normal, Tangent1, Tangent2);
714
715 double detJ = norm_2(Normal);
716
717 //for all integration points
718 if ( rResult.size() != this->IntegrationPointsNumber( ThisMethod ) )
719 rResult.resize( this->IntegrationPointsNumber( ThisMethod ), false );
720
721 for ( unsigned int pnt = 0; pnt < this->IntegrationPointsNumber( ThisMethod ); pnt++ )
722 rResult[pnt] = detJ;
723
724 return rResult;
725 }
