561 Vector& ShapeFunctionsValues(Vector& rResult, const CoordinatesArrayType& rCoordinates) const override
562 {
563 if (rResult.size() != 10) rResult.resize(10, false);
564 const double xi = rCoordinates[0];
565 const double et = rCoordinates[1];
566 const double zt = 1.0 - xi - et;
567 //
568 rResult[0] = zt * (3.0 * zt - 1.0) * (3.0 * zt - 2.0) * 0.5;
569 rResult[1] = xi * (3.0 * xi - 1.0) * (3.0 * xi - 2.0) * 0.5;
570 rResult[2] = et * (3.0 * et - 1.0) * (3.0 * et - 2.0) * 0.5;
571 rResult[3] = xi * zt * (3.0 * zt - 1.0) * 4.5;
572 rResult[4] = xi * zt * (3.0 * xi - 1.0) * 4.5;
573 rResult[5] = xi * et * (3.0 * xi - 1.0) * 4.5;
574 rResult[6] = xi * et * (3.0 * et - 1.0) * 4.5;
575 rResult[7] = et * zt * (3.0 * et - 1.0) * 4.5;
576 rResult[8] = et * zt * (3.0 * zt - 1.0) * 4.5;
577 rResult[9] = xi * et * zt * 27.0;
578 //
579 return rResult;
580 }

591 double ShapeFunctionValue(IndexType ShapeFunctionIndex, const CoordinatesArrayType& rPoint) const override
592 {
593 const double xi = rPoint[0];
594 const double et = rPoint[1];
595 const double zt = 1.0 - xi - et;
596 double shape = 0.0;
597 //
598 switch (ShapeFunctionIndex)
599 {
600 case 0:
601 shape = zt * (3.0 * zt - 1.0) * (3.0 * zt - 2.0) * 0.5;
602 break;
603 case 1:
604 shape = xi * (3.0 * xi - 1.0) * (3.0 * xi - 2.0) * 0.5;
605 break;
606 case 2:
607 shape = et * (3.0 * et - 1.0) * (3.0 * et - 2.0) * 0.5;
608 break;
609 case 3:
610 shape = xi * zt * (3.0 * zt - 1.0) * 4.5;
611 break;
612 case 4:
613 shape = xi * zt * (3.0 * xi - 1.0) * 4.5;
614 break;
615 case 5:
616 shape = xi * et * (3.0 * xi - 1.0) * 4.5;
617 break;
618 case 6:
619 shape = xi * et * (3.0 * et - 1.0) * 4.5;
620 break;
621 case 7:
622 shape = et * zt * (3.0 * et - 1.0) * 4.5;
623 break;
624 case 8:
625 shape = et * zt * (3.0 * zt - 1.0) * 4.5;
626 break;
627 case 9:
628 shape = xi * et * zt * 27.0;
629 break;
630 default:
631 KRATOS_ERROR << "Wrong index of shape function!" << *this << std::endl;
632 break;
633 }
634 //
635 return shape;
636 }

810 Matrix& ShapeFunctionsLocalGradients(Matrix& rResult, const CoordinatesArrayType& rPoint) const override
811 {
812 rResult.resize(10, 2, false);
813 const double xi = rPoint[0];
814 const double et = rPoint[1];
815 const double zt = 1.0 - xi - et;
816 //
817 noalias(rResult) = ZeroMatrix(10, 2);
818 //
819 rResult(0, 0) = -4.5 * zt * (3.0 * zt - 2.0) - 1.0;
820 rResult(0, 1) = -4.5 * zt * (3.0 * zt - 2.0) - 1.0;
821 //
822 rResult(1, 0) = 4.5 * xi * (3.0 * xi - 2.0) + 1.0;
823 rResult(1, 1) = 0.0;
824 //
825 rResult(2, 0) = 0.0;

827 //

829 rResult(3, 1) = -4.5 * xi * (6.0 * zt - 1.0);
830 //
831 rResult(4, 0) = 4.5 * (zt * (6.0 * xi - 1.0) - xi * (3.0 * xi - 1.0));
832 rResult(4, 1) = -4.5 * xi * (3.0 * xi - 1.0);
833 //
834 rResult(5, 0) = 4.5 * et * (6.0 * xi - 1.0);

910 ShapeFunctionsSecondDerivativesType& ShapeFunctionsSecondDerivatives(
911 ShapeFunctionsSecondDerivativesType& rResult, const CoordinatesArrayType& rPoint) const override
912 {
913 if (rResult.size() != this->PointsNumber())

915 ShapeFunctionsGradientsType temp(this->PointsNumber());

936 rResult[0](1, 0) = 9.0 * (3.0 * zt - 1.0);

940 rResult[1](0, 1) = 0.0;
941 rResult[1](1, 0) = 0.0;
942 rResult[1](1, 1) = 0.0;
943 //

945 rResult[2](0, 1) = 0.0;
946 rResult[2](1, 0) = 0.0;
947 rResult[2](1, 1) = 9.0 * (3.0 * et - 1.0);
948 //
949 rResult[3](0, 0) = 9.0 * (3.0 * xi - 6.0 * zt + 1.0);
950 rResult[3](0, 1) = 4.5 * (6.0 * xi - 6.0 * zt + 1.0);
951 rResult[3](1, 0) = 4.5 * (6.0 * xi - 6.0 * zt + 1.0);
952 rResult[3](1, 1) = 27.0 * xi;
953 //
954 rResult[4](0, 0) = 9.0 * (3.0 * zt - 6.0 * xi + 1.0);
955 rResult[4](0, 1) = -4.5 * (6.0 * xi - 1.0);
956 rResult[4](1, 0) = -4.5 * (6.0 * xi - 1.0);
957 rResult[4](1, 1) = 0.0;
958 //
959 rResult[5](0, 0) = 27.0 * et;
960 rResult[5](0, 1) = 4.5 * (6.0 * xi - 1.0);
961 rResult[5](1, 0) = 4.5 * (6.0 * xi - 1.0);
