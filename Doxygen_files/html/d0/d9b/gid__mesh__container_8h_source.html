99 KRATOS_TRY
100

102 if ( mMeshElements.size() != 0 )
103 {
104 //compute number of layers
105 int max_id = 0;
106 for ( ModelPart::ElementsContainerType::iterator it = mMeshElements.begin();
107 it != mMeshElements.end(); ++it )
108 {
109 const int prop_id = it->HasProperties() ? it->GetProperties().Id() : 0;
110 if (max_id < prop_id) max_id = prop_id;
111 }
112 if (max_id > 10000)
113 std::cout<< "a property Id > 10000 found. Are u sure you need so many properties?" << std::endl;
114 std::vector<int> elements_per_layer (max_id+1,0);
115 //KRATOS_WATCH(max_id);
116
117 //fill layer list
118 for ( ModelPart::ElementsContainerType::iterator it = mMeshElements.begin();
119 it != mMeshElements.end(); ++it )
120 {
121 const int prop_id = it->HasProperties() ? (it)->GetProperties().Id() : 0;
122 elements_per_layer[prop_id] += 1;
123 }
124 //std::cout << "start printing elements" <<std::endl;
125 for (unsigned int current_layer = 0; current_layer < elements_per_layer.size(); current_layer++)
126 {
127 if (elements_per_layer[current_layer] > 0)
128 {
129 //create an appropriate name

138 {
139 //std::cout << " -print element 3D mesh: layer ["<<current_layer<<"]-"<<std::endl;
140 GiD_fBeginMesh ( MeshFile, (char *) (current_layer_name.str() ).c_str(), GiD_3D, mGidElementType,mMeshElements.begin()->GetGeometry().size() );
141 }
142 else
143 KRATOS_THROW_ERROR (std::logic_error,"check working space dimension of model","");
144 //printing nodes

146 {
147 GiD_fBeginCoordinates(MeshFile);
148 for ( ModelPart::NodesContainerType::iterator it = mMeshNodes.begin();
149 it != mMeshNodes.end(); ++it )

151 if ( deformed )

162 //printing elements

166 it != mMeshElements.end(); ++it )
167 {
168 for ( unsigned int i=0; i< (it)->GetGeometry().size(); i++ )
169 nodes_id[i] = (it)->GetGeometry() [i].Id();
170
171 if (mGeometryType == GeometryData::KratosGeometryType::Kratos_Line2D3 || mGeometryType == GeometryData::KratosGeometryType::Kratos_Line3D3)
172 {
173 nodes_id[0] = (it)->GetGeometry() [0].Id();
174 nodes_id[1] = (it)->GetGeometry() [1].Id();
175 nodes_id[2] = (it)->GetGeometry() [2].Id();
176 }
177 const unsigned int elem_layer = it->HasProperties() ? it->GetProperties().Id() : 0;
178 nodes_id[ (it)->GetGeometry().size()]= elem_layer + 1;
179
180 if (it->IsActive())
181 if (elem_layer == current_layer)
182 GiD_fWriteElementMat ( MeshFile, (it)->Id(), nodes_id);
183
184 }
185 delete [] nodes_id;
186 GiD_fEndElements(MeshFile);
187 GiD_fEndMesh(MeshFile);
188 }
189 }
