338 template<class TDataType>
339 void CallFunction(const Condition::Pointer& pCondition, const double& time, TDataType& rValue)
340 {
341
342 Condition::GeometryType& rConditionGeometry = pCondition->GetGeometry();
343 unsigned int size = rConditionGeometry.size();
344 double value = 0;
345 unsigned int counter = 0;
346
347 rValue.resize(size*3,false);
348
349 if( mIsSpatialField ){
350
351 double x = 0.0, y = 0.0, z = 0.0;
352
353 for(unsigned int i=0; i<size; i++)
354 {
355 this->LocalAxesTransform(rConditionGeometry[i].X(), rConditionGeometry[i].Y(), rConditionGeometry[i].Z(), x, y, z);
356
357 value = mPyObject.attr(this->mPyMethodName.c_str())(x,y,z,time).cast<double>();
358
359 for(unsigned int j=0; j<3; j++)
360 {
361 rValue[counter] = value * mvector_value[j];
362 counter++;
363 }
364 }
365
366 }
367 else{
368
369 value = mPyObject.attr(this->mPyMethodName.c_str())(0.0,0.0,0.0,time).cast<double>();
370 for(unsigned int i=0; i<size; i++)
371 {
372 for(unsigned int j=0; j<3; j++)
373 {
374 rValue[counter] = value * mvector_value[j];
375 counter++;
376 }
377 }
378
379
380 }

385 template< class TVarType, class TDataType >
386 void AssignValueToConditions(TVarType& rVariable, TDataType& Value, const double& rTime )
387 {
388
389 typedef void (BaseType::*AssignmentMethodPointer) (ModelPart::ConditionType&, const TVarType&, const TDataType&);
390
391 AssignmentMethodPointer AssignmentMethod = this->GetAssignmentMethod<AssignmentMethodPointer>();
392
393 const int nconditions = mrModelPart.GetMesh().Conditions().size();
394
395 if(nconditions != 0)
396 {
397 ModelPart::ConditionsContainerType::iterator it_begin = mrModelPart.GetMesh().ConditionsBegin();
398
399 //#pragma omp parallel for //it does not work in parallel
400 for(int i = 0; i<nconditions; i++)
401 {
402 ModelPart::ConditionsContainerType::iterator it = it_begin + i;
403
404 this->CallFunction<TDataType>(*(it.base()), rTime, Value);
405
406 (this->*AssignmentMethod)(*it, rVariable, Value);
407 }
408 }
409
410 }
