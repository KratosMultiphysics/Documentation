118 /* Evaluates the corresponding IntegrationMethod to

120 */
121 static IntegrationMethod GetIntegrationMethod(
122 SizeType NumberOfIntegrationPointsPerSpan,
123 QuadratureMethod ThisQuadratureMethod)
124 {
125 switch (NumberOfIntegrationPointsPerSpan) {
126 case 1:
127 if (ThisQuadratureMethod == QuadratureMethod::GAUSS) {
128 return IntegrationMethod::GI_GAUSS_1;
129 }
130 else {
131 return IntegrationMethod::GI_EXTENDED_GAUSS_1;
132 }
133 break;
134 case 2:
135 if (ThisQuadratureMethod == QuadratureMethod::GAUSS) {
136 return IntegrationMethod::GI_GAUSS_2;
137 }
138 else {
139 return IntegrationMethod::GI_EXTENDED_GAUSS_2;
140 }
141 break;
142 case 3:
143 if (ThisQuadratureMethod == QuadratureMethod::GAUSS) {
144 return IntegrationMethod::GI_GAUSS_3;
145 }
146 else {
147 return IntegrationMethod::GI_EXTENDED_GAUSS_3;
148 }
149 break;
150 case 4:
151 if (ThisQuadratureMethod == QuadratureMethod::GAUSS) {
152 return IntegrationMethod::GI_GAUSS_4;
153 }
154 else {
155 return IntegrationMethod::GI_EXTENDED_GAUSS_4;
156 }
157 break;
158 case 5:
159 if (ThisQuadratureMethod == QuadratureMethod::GAUSS) {
160 return IntegrationMethod::GI_GAUSS_5;
161 }
162 else {
163 return IntegrationMethod::GI_EXTENDED_GAUSS_5;
164 }
165 break;
166 case 0:
167 return IntegrationMethod::NumberOfIntegrationMethods;
168 break;
169 }
170 KRATOS_WARNING("Evaluation of Integration Method")
171 << "Chosen combination of number of points per span and quadrature method does not has a corresponding IntegrationMethod in the KRATOS core."
172 << "NumberOfIntegrationPointsPerSpan: " << NumberOfIntegrationPointsPerSpan << std::endl;
173 return IntegrationMethod::NumberOfIntegrationMethods;
174 }
