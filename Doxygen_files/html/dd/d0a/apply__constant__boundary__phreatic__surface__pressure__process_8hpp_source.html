82 void ExecuteInitialize() override
83 {
84 KRATOS_TRY
85
86 const Variable<double>& var = KratosComponents<Variable<double>>::Get(mVariableName);
87
88 Vector3 direction = ZeroVector(3);
89 direction[mGravityDirection] = 1.0;
90
91 block_for_each(mrModelPart.Nodes(), [&var, &direction, this](Node& rNode) {
92 if (mIsFixed) rNode.Fix(var);
93 else if (mIsFixedProvided) rNode.Free(var);
94
95 double distance = inner_prod(mNormalVector, rNode.Coordinates());
96 const double d = inner_prod(mNormalVector, direction);
97 distance = -(distance - mEqRHS) / d;
98 const double pressure = mSpecificWeight * distance;
99 rNode.FastGetSolutionStepValue(var) = std::max(pressure, 0.0);
100 });
101
102 KRATOS_CATCH("")
103 }

125private:
126 void calculateEquationParameters()
127 {
128 const Vector3 v1 = mSecondReferenceCoordinate - mFirstReferenceCoordinate;
129 const Vector3 v2 = mThirdReferenceCoordinate - mFirstReferenceCoordinate;
130 mNormalVector = MathUtils<>::CrossProduct(v1, v2);
131 if (norm_2(mNormalVector) == 0.0)
132 KRATOS_ERROR << "Normal vector to phreatic surface has zero size!" << std::endl;
133
134 mEqRHS = inner_prod(mNormalVector, mFirstReferenceCoordinate);
135 }
