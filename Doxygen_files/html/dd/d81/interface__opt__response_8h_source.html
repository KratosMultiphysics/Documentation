104 double CalculateValue() override {
105 double total_mass = 0.0;
106 for(auto& eval_obj : mrResponseSettings["evaluated_objects"]){
107 ModelPart& r_eval_object = mrModel.GetModelPart(eval_obj.GetString());
108 const std::size_t domain_size = r_eval_object.GetProcessInfo()[DOMAIN_SIZE];
109 for (auto& elem_i : r_eval_object.Elements()){
110 const bool element_is_active = elem_i.IsDefined(ACTIVE) ? elem_i.Is(ACTIVE) : true;
111 if(element_is_active)
112 total_mass += CalculateElementValue(elem_i,domain_size);
113 }
114 }

118 double CalculateElementValue(Element& elem_i, const std::size_t DomainSize){
119 // We get the element geometry
120 auto& r_this_geometry = elem_i.GetGeometry();
121 const std::size_t number_of_nodes = r_this_geometry.size();
122
123 // We copy the current coordinates and move the coordinates to the initial configuration
124 std::vector<array_1d<double, 3>> current_coordinates(number_of_nodes);
125 for (std::size_t i_node = 0; i_node < number_of_nodes; ++i_node) {
126 noalias(current_coordinates[i_node]) = r_this_geometry[i_node].Coordinates();
127 noalias(r_this_geometry[i_node].Coordinates()) = r_this_geometry[i_node].GetInitialPosition().Coordinates();
128 }
129
130 double element_density = elem_i.GetProperties().GetValue(DENSITY);
131 double element_val = elem_i.GetGeometry().Volume() * std::pow((element_density) * (1.0-element_density),2);
132
133 // We restore the current configuration
134 for (std::size_t i_node = 0; i_node < number_of_nodes; ++i_node) {
135 noalias(r_this_geometry[i_node].Coordinates()) = current_coordinates[i_node];
136 }
137
138 return element_val;
139 }

169 void CalculateElementMaterialGradients(Element& elem_i, const std::size_t DomainSize){
170
171 // We get the element geometry
172 auto& r_this_geometry = elem_i.GetGeometry();
173 const std::size_t number_of_nodes = r_this_geometry.size();
174
175 double curr_density = elem_i.GetProperties().GetValue(DENSITY);
176 double elem_dens_grad = elem_i.GetGeometry().Volume() * 2 * std::pow((curr_density) * (1.0-curr_density),1) * (1.0 - 2.0 * curr_density);
177
178
179 for (SizeType i_node = 0; i_node < number_of_nodes; ++i_node){
180 const auto& d_pd_d_fd = r_this_geometry[i_node].FastGetSolutionStepValue(D_PD_D_FD);
181 r_this_geometry[i_node].FastGetSolutionStepValue(D_INTERFACE_D_FD) += d_pd_d_fd * elem_dens_grad / number_of_nodes;
182 }
183 };
