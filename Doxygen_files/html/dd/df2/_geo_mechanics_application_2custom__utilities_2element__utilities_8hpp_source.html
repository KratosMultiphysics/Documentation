220 template <typename VectorType1, typename VectorType2>
221 static void AddVectorAtPosition(const VectorType1& rSourceVector, VectorType2& rDestinationVector, std::size_t Offset)
222 {
223 auto pos = std::begin(rDestinationVector) + Offset;
224 std::transform(std::begin(rSourceVector), std::end(rSourceVector), pos, pos, std::plus<double>{});
225 }

227 template <typename MatrixType1, typename MatrixType2>
228 static void AddMatrixAtPosition(const MatrixType1& rSourceMatrix,
229 MatrixType2& rDestinationMatrix,
230 std::size_t RowOffset,
231 std::size_t ColumnOffset)
232 {
233 for (auto i = std::size_t{0}; i < rSourceMatrix.size1(); ++i) {
234 for (auto j = std::size_t{0}; j < rSourceMatrix.size2(); ++j) {
235 rDestinationMatrix(i + RowOffset, j + ColumnOffset) += rSourceMatrix(i, j);
236 }
237 }
