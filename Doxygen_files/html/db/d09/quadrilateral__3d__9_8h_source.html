514 CoordinatesArrayType& rResult,
515 const CoordinatesArrayType& rPoint
516 ) const override

518 const double tol = 1.0e-8;
519 const int maxiter = 1000;
520 //check orientation of surface
521 std::vector< unsigned int> orientation( 3 );
522
523 double dummy = this->GetPoint( 0 ).X();
524
525 if ( std::abs( this->GetPoint( 1 ).X() - dummy ) <= tol && std::abs( this->GetPoint( 2 ).X() - dummy ) <= tol && std::abs( this->GetPoint( 3 ).X() - dummy ) <= tol )
526 orientation[0] = 0;
527
528 dummy = this->GetPoint( 0 ).Y();
529
530 if ( std::abs( this->GetPoint( 1 ).Y() - dummy ) <= tol && std::abs( this->GetPoint( 2 ).Y() - dummy ) <= tol && std::abs( this->GetPoint( 3 ).Y() - dummy ) <= tol )
531 orientation[0] = 1;
532
533 dummy = this->GetPoint( 0 ).Z();
534
535 if ( std::abs( this->GetPoint( 1 ).Z() - dummy ) <= tol && std::abs( this->GetPoint( 2 ).Z() - dummy ) <= tol && std::abs( this->GetPoint( 3 ).Z() - dummy ) <= tol )
536 orientation[0] = 2;
537
538 switch ( orientation[0] )
539 {
540 case 0:
541 orientation[0] = 1;
542 orientation[1] = 2;
543 orientation[2] = 0;
544 break;
545 case 1:
546 orientation[0] = 0;
547 orientation[1] = 2;
548 orientation[2] = 1;
549 break;
550 case 2:
551 orientation[0] = 0;
552 orientation[1] = 1;
553 orientation[2] = 2;
554 break;
555 default:
556 orientation[0] = 0;
557 orientation[1] = 1;
558 orientation[2] = 2;
559 }
560
561 Matrix J = ZeroMatrix( 2, 2 );
562
563 Matrix invJ = ZeroMatrix( 2, 2 );
564
565 if ( rResult.size() != 2 )
566 rResult.resize( 2, false );
567
568 //starting with xi = 0
569 rResult = ZeroVector( 3 );
570
571 Vector DeltaXi = ZeroVector( 3 );
572
573 CoordinatesArrayType CurrentGlobalCoords( ZeroVector( 3 ) );
574
575 //Newton iteration:
576 for ( int k = 0; k < maxiter; k++ )
577 {
578 CurrentGlobalCoords = ZeroVector( 3 );
579 this->GlobalCoordinates( CurrentGlobalCoords, rResult );
580 noalias( CurrentGlobalCoords ) = rPoint - CurrentGlobalCoords;
581
582 //Caluclate Inverse of Jacobian
583 noalias(J) = ZeroMatrix( 2, 2 );
584
585 //derivatives of shape functions
586 Matrix shape_functions_gradients;
587 shape_functions_gradients = ShapeFunctionsLocalGradients(
588 shape_functions_gradients, rResult );
589
590 //Elements of jacobian matrix (e.g. J(1,1) = dX1/dXi1)
591 //loop over all nodes
592 for ( unsigned int i = 0; i < this->PointsNumber(); i++ )
593 {
594 Point dummyPoint = this->GetPoint( i );
595 J( 0, 0 ) += ( dummyPoint[orientation[0]] ) * ( shape_functions_gradients( i, 0 ) );
596 J( 0, 1 ) += ( dummyPoint[orientation[0]] ) * ( shape_functions_gradients( i, 1 ) );
597 J( 1, 0 ) += ( dummyPoint[orientation[1]] ) * ( shape_functions_gradients( i, 0 ) );
598 J( 1, 1 ) += ( dummyPoint[orientation[1]] ) * ( shape_functions_gradients( i, 1 ) );
599 }
600
601 //deteminant of Jacobian
602 double det_j = J( 0, 0 ) * J( 1, 1 ) - J( 0, 1 ) * J( 1, 0 );
603
604 //filling matrix
605 invJ( 0, 0 ) = ( J( 1, 1 ) ) / ( det_j );
606
607 invJ( 1, 0 ) = -( J( 1, 0 ) ) / ( det_j );
608
609 invJ( 0, 1 ) = -( J( 0, 1 ) ) / ( det_j );
610
611 invJ( 1, 1 ) = ( J( 0, 0 ) ) / ( det_j );
612
613 DeltaXi( 0 ) = invJ( 0, 0 ) * CurrentGlobalCoords( orientation[0] ) + invJ( 0, 1 ) * CurrentGlobalCoords( orientation[1] );
614
615 DeltaXi( 1 ) = invJ( 1, 0 ) * CurrentGlobalCoords( orientation[0] ) + invJ( 1, 1 ) * CurrentGlobalCoords( orientation[1] );
616
617 noalias( rResult ) += DeltaXi;
618
619 if ( MathUtils<double>::Norm3( DeltaXi ) > 30 )
620 {
621 break;
622 }
623
624 if ( MathUtils<double>::Norm3( DeltaXi ) < tol )
625 {
626 if ( !( std::abs( CurrentGlobalCoords( orientation[2] ) ) <= tol ) )
627 rResult( 0 ) = 2.0;
628
629 break;
630 }
631 }
632
633 return( rResult );
