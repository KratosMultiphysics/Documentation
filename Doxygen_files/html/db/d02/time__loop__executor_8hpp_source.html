61 std::vector<TimeStepEndState> Run(const TimeStepEndState& EndState) override
62 {
63 mStrategyWrapper->Initialize();
64 mStrategyWrapper->SaveTotalDisplacementFieldAtStartOfTimeLoop();
65 std::vector<TimeStepEndState> result;
66 TimeStepEndState NewEndState = EndState;
67 ScopedOutputFileAccess limit_output_file_access_to_this_scope{*mStrategyWrapper};
68 while (mTimeIncrementor->WantNextStep(NewEndState) && !IsCancelled()) {
69 mStrategyWrapper->IncrementStepNumber();
70 // clone without end time, the end time is overwritten anyway
71 mStrategyWrapper->CloneTimeStep();
72 NewEndState = RunCycleLoop(NewEndState);
73 mStrategyWrapper->AccumulateTotalDisplacementField();
74 mStrategyWrapper->ComputeIncrementalDisplacementField();
75 mStrategyWrapper->FinalizeSolutionStep();
76 mStrategyWrapper->OutputProcess();
77 result.emplace_back(NewEndState);
78 }

83private:
84 TimeStepEndState RunCycle(double PreviousTime)
85 {
86 // Setting the time and time increment may be needed for the processes
87 const auto time_increment = mTimeIncrementor->GetIncrement();
88 mStrategyWrapper->SetTimeIncrement(time_increment);
89 const auto end_time = PreviousTime + time_increment;
90 mStrategyWrapper->SetEndTime(end_time);
91
92 auto end_state = mTimeStepExecutor->Run(end_time);
93 mTimeIncrementor->PostTimeStepExecution(end_state);
94 UpdateProgress(end_time);
95 return end_state;
96 }
97
98 TimeStepEndState RunCycleLoop(const TimeStepEndState& previous_state)
99 {
100 auto cycle_number = 0;
101 auto end_state = previous_state;
102 while (mTimeIncrementor->WantRetryStep(cycle_number, end_state) && !IsCancelled()) {
103 if (cycle_number > 0) mStrategyWrapper->RestorePositionsAndDOFVectorToStartOfStep();
104 end_state = RunCycle(previous_state.time);
105 ++cycle_number;
106 }
107
108 end_state.num_of_cycles = cycle_number;
109 return end_state;
110 }
