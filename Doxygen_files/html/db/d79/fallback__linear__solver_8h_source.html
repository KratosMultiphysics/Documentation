635  LinearSolverPointer ConstructLinearSolverFromSettings(const Parameters Settings)
636  {
637  // Implementation that delegates to a factory method.
638  // The factory method 'Create' is responsible for parsing 'Settings'
639  // and constructing the appropriate solver.
640  const auto linear_solver_factory = LinearSolverFactoryType();
641  Parameters linear_solver_settings(Settings);
642  const std::string linear_solver_type = linear_solver_settings["solver_type"].GetString();
643  // If the solver type is a faster direct solver, try to use it
644  if (linear_solver_type == "faster_direct_solver") {
645  const std::vector<std::string> faster_direct_solvers = TSparseSpaceType::FastestDirectSolverList();
646  for (const std::string& r_solver_name : faster_direct_solvers) {
647  if (linear_solver_factory.Has(r_solver_name)) {
648  linear_solver_settings["solver_type"].SetString(r_solver_name);
649  }
650  }

728  void UpdateSolverIndex()
729  {
730  // Safety check to ensure we have solvers to work with
731  KRATOS_ERROR_IF(mSolvers.empty()) << "No solvers are configured." << std::endl;
732 
733  // Log the settings of the current (failing) solver, if applicable
734  if (mCurrentSolverIndex < mSolvers.size()) {
735  KRATOS_INFO("FallbackLinearSolver") << "Current solver " << GetCurrentSolver()->Info() << " failed with the following settings: " << mParameters["solvers"][mCurrentSolverIndex].PrettyPrintJsonString() << std::endl;
736  } else {
737  KRATOS_WARNING("FallbackLinearSolver") << "Solver index cannot be updated as all potential linear solvers has been already tried. Sticking to current solver " << GetCurrentSolver()->Info() << std::endl;
738  return;
739  }
740 
741  // Increment the counter
742  mCurrentSolverIndex++;
743 
744  // Print the new solver settings
745  if (mCurrentSolverIndex < mSolvers.size()) {
746  KRATOS_INFO("FallbackLinearSolver") << "Switching to new solver " << GetCurrentSolver()->Info() << " with the following settings: " << mParameters["solvers"][mCurrentSolverIndex].PrettyPrintJsonString() << std::endl;
747  } else {
748  KRATOS_WARNING("FallbackLinearSolver") << "Solver index cannot be updated as all potential linear solvers has been already tried. Sticking to current solver " << GetCurrentSolver()->Info() << std::endl;
749  return;
750  }
