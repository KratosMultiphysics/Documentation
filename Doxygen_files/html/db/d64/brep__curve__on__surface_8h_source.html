346 /* @brief checks and returns if local coordinate rPointLocalCoordinates[0]
347 * is inside the local/parameter space.
348 * @return on boundary -> 2 - meaning that it is equal to start or end point.
349 * inside -> 1
350 * outside -> 0
351 */
352 int IsInsideLocalSpace(
353 const CoordinatesArrayType& rPointLocalCoordinates,
354 const double Tolerance = std::numeric_limits<double>::epsilon()
355 ) const override
356 {
357 const double min_parameter = mCurveNurbsInterval.MinParameter();
358 if (rPointLocalCoordinates[0] < min_parameter) {
359 return 0;
360 } else if (std::abs(rPointLocalCoordinates[0] - min_parameter) < Tolerance) {
361 return 2;
362 }
363
364 const double max_parameter = mCurveNurbsInterval.MaxParameter();
365 if (rPointLocalCoordinates[0] > max_parameter) {
366 return 0;
367 } else if (std::abs(rPointLocalCoordinates[0] - max_parameter) < Tolerance) {
368 return 2;
369 }
370
371 return 1;
372 }
373
377
378 /* @brief Makes a check if the provided paramater rPointLocalCoordinates[0]
379 * is inside the curve, or on the boundary or if it lays outside.
380 * If it is outside, it is set to the boundary which is closer to it.
381 * @return if rPointLocalCoordinates[0] was before the projection:
382 * inside -> 1
383 * outside -> 0
384 * on boundary -> 2 - meaning that it is equal to start or end point.
385 */
386 virtual int ClosestPointLocalToLocalSpace(
387 const CoordinatesArrayType& rPointLocalCoordinates,
388 CoordinatesArrayType& rClosestPointLocalCoordinates,
389 const double Tolerance = std::numeric_limits<double>::epsilon()
390 ) const override
391 {
392 const double min_parameter = mCurveNurbsInterval.MinParameter();
393 if (rPointLocalCoordinates[0] < min_parameter) {
394 rClosestPointLocalCoordinates[0] = min_parameter;
395 return 0;
396 } else if (std::abs(rPointLocalCoordinates[0] - min_parameter) < Tolerance) {
397 rClosestPointLocalCoordinates[0] = rPointLocalCoordinates[0];
398 return 2;
399 }
400
401 const double max_parameter = mCurveNurbsInterval.MaxParameter();
402 if (rPointLocalCoordinates[0] > max_parameter) {
403 rClosestPointLocalCoordinates[0] = min_parameter;
404 return 0;
405 } else if (std::abs(rPointLocalCoordinates[0] - max_parameter) < Tolerance) {
406 rClosestPointLocalCoordinates[0] = rPointLocalCoordinates[0];
407 return 2;
408 }
409
410 rClosestPointLocalCoordinates[0] = rPointLocalCoordinates[0];
411 return 1;
412 }

510 double Length() const override
511 {
512 IntegrationPointsArrayType integration_points;
513 IntegrationInfo integration_info = GetDefaultIntegrationInfo();
514 CreateIntegrationPoints(integration_points, integration_info);
515
516 double length = 0.0;
517 for (IndexType i = 0; i < integration_points.size(); ++i) {
518 const double determinant_jacobian = mpCurveOnSurface->DeterminantOfJacobian(integration_points[i]);
519 length += integration_points[i].Weight() * determinant_jacobian;
520 }
521 return length;
522 }

584 rResultGeometries(i)->SetGeometryParent(this);
