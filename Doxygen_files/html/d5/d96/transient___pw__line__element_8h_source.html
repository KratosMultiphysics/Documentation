220 Vector CalculateIntegrationCoefficients(const Vector& rDetJContainer) const
221 {
222 const auto& r_properties = GetProperties();
223 const auto& r_integration_points = GetGeometry().IntegrationPoints(GetIntegrationMethod());
224
225 auto result = Vector{r_integration_points.size()};
226 std::transform(r_integration_points.begin(), r_integration_points.end(), rDetJContainer.begin(),
227 result.begin(), [&r_properties](const auto& rIntegrationPoint, const auto& rDetJ) {
228 return rIntegrationPoint.Weight() * rDetJ * r_properties[CROSS_AREA];
229 });
230 return result;
231 }

243 array_1d<double, TNumNodes> CalculateFluidBodyVector() const
244 {
245 Vector det_J_container;
246 GetGeometry().DeterminantOfJacobian(det_J_container, this->GetIntegrationMethod());
247 GeometryType::ShapeFunctionsGradientsType shape_function_gradients =
248 GetGeometry().ShapeFunctionsLocalGradients(this->GetIntegrationMethod());
249 std::transform(shape_function_gradients.begin(), shape_function_gradients.end(),
250 det_J_container.begin(), shape_function_gradients.begin(), std::divides<>());
251 const Matrix& N_container = GetGeometry().ShapeFunctionsValues(GetIntegrationMethod());
252
253 const auto integration_coefficients = CalculateIntegrationCoefficients(det_J_container);
254
255 const std::size_t number_integration_points =
256 GetGeometry().IntegrationPointsNumber(GetIntegrationMethod());
257 GeometryType::JacobiansType J_container;
258 J_container.resize(number_integration_points, false);
259 for (std::size_t i = 0; i < number_integration_points; ++i) {
260 J_container[i].resize(GetGeometry().WorkingSpaceDimension(),
261 GetGeometry().LocalSpaceDimension(), false);
262 }
263 GetGeometry().Jacobian(J_container, this->GetIntegrationMethod());
264
265 const auto& r_properties = GetProperties();
266 BoundedMatrix<double, 1, 1> constitutive_matrix;
267 GeoElementUtilities::FillPermeabilityMatrix(constitutive_matrix, r_properties);
268
269 RetentionLaw::Parameters RetentionParameters(GetProperties());
270
271 array_1d<double, TNumNodes * TDim> volume_acceleration;
272 GeoElementUtilities::GetNodalVariableVector<TDim, TNumNodes>(
273 volume_acceleration, GetGeometry(), VOLUME_ACCELERATION);
274
275 array_1d<double, TNumNodes> fluid_body_vector = ZeroVector(TNumNodes);
276 for (unsigned int integration_point_index = 0;
277 integration_point_index < GetGeometry().IntegrationPointsNumber(GetIntegrationMethod());
278 ++integration_point_index) {
279 array_1d<double, TDim> body_acceleration;
280 GeoElementUtilities::InterpolateVariableWithComponents<TDim, TNumNodes>(
281 body_acceleration, N_container, volume_acceleration, integration_point_index);
282
283 array_1d<double, TDim> tangent_vector = column(J_container[integration_point_index], 0);
284 tangent_vector /= norm_2(tangent_vector);
285
286 array_1d<double, 1> projected_gravity = ZeroVector(1);
287 projected_gravity(0) = MathUtils<double>::Dot(tangent_vector, body_acceleration);
288 const auto N = Vector{row(N_container, integration_point_index)};
289 double RelativePermeability =
290 mRetentionLawVector[integration_point_index]->CalculateRelativePermeability(RetentionParameters);
291 fluid_body_vector +=
292 r_properties[DENSITY_WATER] * RelativePermeability *
293 prod(prod(shape_function_gradients[integration_point_index], constitutive_matrix), projected_gravity) *
294 integration_coefficients[integration_point_index] / r_properties[DYNAMIC_VISCOSITY];
295 }
296 return fluid_body_vector;
297 }
