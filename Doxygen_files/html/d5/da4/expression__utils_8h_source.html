652 template <class TContainerType>
653 static ContainerExpression<TContainerType> Comb(const std::vector<typename ContainerExpression<TContainerType>::Pointer>& rpContainerExpressions)
654 {
655 std::vector<Expression::ConstPointer> exps;
656 std::transform(rpContainerExpressions.begin(), rpContainerExpressions.end(), std::back_inserter(exps), [](const auto& V) { return V->pGetExpression(); });
657 KRATOS_ERROR_IF(rpContainerExpressions.empty())
658 << "Empty container list provided for combing.";
659 auto copy = *rpContainerExpressions.front();
660 copy.SetExpression(UnaryCombineExpression::Create(exps.begin(), exps.end()));
661 return copy;
662 }
