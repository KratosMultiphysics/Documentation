635 IntegrationMethod GetIntegrationMethod() const override
636 {
637 // Setting the auxiliary integration points
638 const IndexType integration_order = GetProperties().Has(INTEGRATION_ORDER_CONTACT) ? GetProperties().GetValue(INTEGRATION_ORDER_CONTACT) : 2;
639 switch (integration_order) {
640 case 1: return GeometryData::IntegrationMethod::GI_GAUSS_1;
641 case 2: return GeometryData::IntegrationMethod::GI_GAUSS_2;
642 case 3: return GeometryData::IntegrationMethod::GI_GAUSS_3;
643 case 4: return GeometryData::IntegrationMethod::GI_GAUSS_4;
644 case 5: return GeometryData::IntegrationMethod::GI_GAUSS_5;
645 default: return GeometryData::IntegrationMethod::GI_GAUSS_2;
646 }
