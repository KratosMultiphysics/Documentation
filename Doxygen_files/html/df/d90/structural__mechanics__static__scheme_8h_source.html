112 void Initialize(ModelPart& rModelPart) override
113 {
114 // Call the base Initialize method
115 StaticBaseType::Initialize(rModelPart);
116
117 // Allocate the OSS projection variables
118 const auto &r_process_info = rModelPart.GetProcessInfo();
119 const bool oss_switch = r_process_info.Has(OSS_SWITCH) ? r_process_info[OSS_SWITCH] : false;
120 if (oss_switch) {
121 const array_1d<double,3> aux_zero = ZeroVector(3);
122 block_for_each(rModelPart.Nodes(), aux_zero, [this](Node& rNode, array_1d<double,3>& rAuxZero){
123 for (auto& rp_var : mProjectionScalarVariablesList) {
124 rNode.SetValue(*rp_var, 0.0);

126 for (auto& rp_var : mProjectionVectorVariablesList) {
127 rNode.SetValue(*rp_var, rAuxZero);
128 }
129 });
130 }
131
132 // Calculate the NODAL_AREA

138 void FinalizeNonLinIteration(
139 ModelPart& rModelPart,
140 TSystemMatrixType& rA,
141 TSystemVectorType& rDx,
142 TSystemVectorType& rb) override
143 {
144 KRATOS_TRY
145
146 // Call base class FinalizeNonLinIteration
147 StaticBaseType::FinalizeNonLinIteration(rModelPart, rA, rDx, rb);
148
149 // Check if the Orthogonal SubScales (OSS) are active
150 const auto& r_process_info = rModelPart.GetProcessInfo();
151 const bool oss_switch = r_process_info.Has(OSS_SWITCH) ? r_process_info[OSS_SWITCH] : false;
152
153 // Calculate the OSS projections
154 if (oss_switch) {
155 // Initialize the projection values
156 const array_1d<double,3> aux_zero = ZeroVector(3);
157 block_for_each(rModelPart.Nodes(), aux_zero, [this](Node& rNode, array_1d<double,3>& rAuxZero){
158 for (auto& rp_var : mProjectionScalarVariablesList) {
159 rNode.FastGetSolutionStepValue(*rp_var) = 0.0;
160 }
161 for (auto& rp_var : mProjectionVectorVariablesList) {
162 rNode.FastGetSolutionStepValue(*rp_var) = rAuxZero;
163 }
164 });
165
166 // Calculate the element residuals projection
167 std::tuple<double, array_1d<double,3>> oss_proj_tls;
168 block_for_each(rModelPart.Elements(), oss_proj_tls, [&, this](Element& rElement, std::tuple<double, array_1d<double,3>>& rOssProjTLS){
169 double& r_scalar_proj = std::get<0>(rOssProjTLS);
170 array_1d<double,3>& r_vector_proj = std::get<1>(rOssProjTLS);
171 for (auto& rp_var : mProjectionScalarVariablesList) {
172 rElement.Calculate(*rp_var, r_scalar_proj, r_process_info);
173 }
174 for (auto& rp_var : mProjectionVectorVariablesList) {
175 rElement.Calculate(*rp_var, r_vector_proj, r_process_info);
176 }
177 });
178
179 // Do the nodal weighting
180 //TODO: We the weighted L2 projection with the density should be done in the multimaterial case
181 //TODO: Note that this would require customizing the weights calculation in here (standard NODAL_AREA cannot be used)
182 block_for_each(rModelPart.Nodes(), [this](Node& rNode){
183 const double nodal_area = rNode.GetValue(NODAL_AREA);
184 for (auto& rp_var : mProjectionScalarVariablesList) {
185 rNode.FastGetSolutionStepValue(*rp_var) /= nodal_area;
186 }
187 for (auto& rp_var : mProjectionVectorVariablesList) {
188 rNode.FastGetSolutionStepValue(*rp_var) /= nodal_area;
189 }
190 });
191 }
