569 iterator insert(const TPointerType& value)
570 {
571 auto itr_pos = std::lower_bound(mData.begin(), mData.end(), KeyOf(*value), CompareKey());
572 if (itr_pos == mData.end()) {
573 // the position to insert is at the end.
574 mData.push_back(value);
575 mSortedPartSize = mData.size();
576 return iterator(mData.end() - 1);
577 } else if (EqualKeyTo(KeyOf(*value))(*itr_pos)) {
578 // already found existing element with the same key, hence returning the existing element.
579 return iterator(itr_pos);
580 } else {
581 // insert the new value before the itr_pos.
582 mSortedPartSize = mData.size() + 1;
583 return mData.insert(itr_pos, value);
584 }
585 }

597 iterator insert(const_iterator position_hint, const TPointerType& value)
598 {
599 if (empty()) {
600 // the dataset is empty. So use push back.
601 mData.push_back(value);
602 mSortedPartSize = mData.size();
603 return iterator(mData.end() - 1);
604 } else if (position_hint == cend()) {
605 // trying to insert at the end.
606 if (KeyOf(*(position_hint - 1)) < KeyOf(*value)) {
607 // key at the position hint is less than the value of key. Hence position hint
608 // is valid. So using the push back.
609 mData.push_back(value);
610 mSortedPartSize = mData.size();
611 return iterator(mData.end() - 1);
612 } else {
613 // given position is invalid. Hence, discarding the hint.
614 return insert(value);
615 }
616 } else if (position_hint == cbegin()) {
617 // trying to insert at the front.
618 if (KeyOf(*value) < KeyOf(*position_hint)) {
619 // key at the position hint is greater than the value of key. Hence position hint
620 // is valid. So using insertion at the beginning.
621 mSortedPartSize = mData.size() + 1;
622 return mData.insert(mData.begin(), value);
623 } else {
624 // given position is invalid. Hence, discarding the hint.
625 return insert(value);
626 }
627 } else {
628 // trying to insert at an arbitrary position.
629 if (KeyOf(*value) < KeyOf(*position_hint) && (KeyOf(*(position_hint - 1)) < KeyOf(*value))) {
630 mSortedPartSize = mData.size() + 1;
631 return mData.insert(mData.begin() + (position_hint - cbegin()), value);
632 } else {
633 // given position is invalid. Hence, discarding the hint.
634 return insert(value);
635 }
636 }

1073 template<class TIteratorType>
1074 void SortedInsert(TIteratorType first, TIteratorType last)
1075 {
1076 if (std::distance(first, last) == 0) {
1077 return;
1078 }
1079
1080 if (empty()) {
1081 mData.reserve(std::distance(first, last));
1082 for (auto it = first; it != last; ++it) {
1083 mData.push_back(GetPointer(it));
1084 }
1085 } else {
1086 if (KeyOf(GetReference(first)) > KeyOf(*(mData.back()))) {
1087 // all are pointing to the end of the vector, hence pushing back.
1088 mData.reserve(mData.size() + std::distance(first, last));
1089 for (auto it = first; it != last; ++it) {
1090 mData.push_back(GetPointer(it));
1091 }
1092 } else {
1093 TContainerType temp;
1094 temp.reserve(mData.size() + std::distance(first, last));
1095 auto p_existing_data_itr = mData.begin();
1096
1097 CompareKey key_comparator;
1098
1099 while (p_existing_data_itr != mData.end() && first != last) {
1100 if (key_comparator(*p_existing_data_itr, KeyOf(GetReference(first)))) {
1101 temp.push_back(GetPointer(p_existing_data_itr++));
1102 } else if (EqualKeyTo(KeyOf(GetReference(p_existing_data_itr)))(*first)) {
1103 // here we keep the old entity, and discard the new entity.
1104 temp.push_back(GetPointer(p_existing_data_itr++));
1105 ++first;
1106 } else {
1107 temp.push_back(GetPointer(first++));
1108 }
1109 }
1110
1111 // now either p_existing_data_itr reached the end or first reached the end,
1112 // hence we add the remaining without checking
1113 for (; p_existing_data_itr != mData.end(); ++p_existing_data_itr) {
1114 temp.push_back(GetPointer(p_existing_data_itr));
1115 }
1116 for (; first != last; ++first) {
1117 temp.push_back(GetPointer(first));
1118 }
1119
1120 mData.swap(temp);
1121 }
1122 }
1123
1124 // TODO: To be removed once push back is removed.
1125 // insert assumes the PointerVectorSet is already sorted,
1126 // hence mSortedPartSize should be mData.size()
1127 mSortedPartSize = mData.size();
1128 }

1171 template<class TIteratorType>
1172 inline auto& GetReference(TIteratorType Iterator) const
1173 {
1174 // It is difficult to use std::iterator_traits to get the value
1175 // type of the iterator because, boost::indirect has a value type
1176 // which is harder to guess, and cryptic. Hence, using the decltype.
1177 using iterator_value_type = std::decay_t<decltype(*Iterator)>;
1178
1179 if constexpr(std::is_same_v<iterator_value_type, std::remove_cv_t<value_type>>) {
1180 // in here, std::remove_cv is only used for the value_type because,
1181 // the PointerVectorSet can be with TDataType which is const, but the passed pointers
1182 // must be always TDataType::Pointer which is defined for non cost TDataType. This is
1183 // a valid use case. Other way is not possible, hence std::remove_cv is not used on
1184 // iterator_value_type.
1185 return *Iterator;
1186 } else if constexpr(std::is_same_v<iterator_value_type, pointer>) {
1187 return **Iterator;
1188 } else {
1189 static_assert(!std::is_same_v<TIteratorType, TIteratorType>, "Unsupported iterator type.");
1190 return 0;
1191 }
1192 }

1194 template<class TIteratorType>
1195 inline TPointerType GetPointer(TIteratorType Iterator) const
1196 {
1197 // It is difficult to use std::iterator_traits to get the value
1198 // type of the iterator because, boost::indirect has a value type
1199 // which is harder to guess, and cryptic. Hence, using the decltype.
1200 using iterator_value_type = std::decay_t<decltype(*Iterator)>;
1201
1202 if constexpr(std::is_same_v<iterator_value_type, std::remove_cv_t<TPointerType>>) {
1203 // this supports any type of pointers
1204 return *Iterator;
1205 } else if constexpr(std::is_same_v<iterator_value_type, std::remove_cv_t<value_type>> && std::is_same_v<TPointerType, Kratos::intrusive_ptr<std::decay_t<decltype(*Iterator)>>>) {
1206 // now the *iterator points to a value type. Then we can only return the pointer if
1207 // the PointerVectorSet is of the type with intrusive_ptrs. Then we can safely construct the intrusive ptrs
1208 return TPointerType(&*Iterator);
1209 } else {
1210 static_assert(!std::is_same_v<TIteratorType, TIteratorType>, "Unsupported iterator type.");
1211 return 0;
1212 }
