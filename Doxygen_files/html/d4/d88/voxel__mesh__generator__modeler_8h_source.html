134 public:

136
137 void SetNumberOfDivisions(std::size_t XDivisions, std::size_t YDivisions, std::size_t ZDivisions)
138 {
139 mNumberOfDivisions[0] = XDivisions;
140 mNumberOfDivisions[1] = YDivisions;
141 mNumberOfDivisions[2] = ZDivisions;
142
143 KRATOS_ERROR_IF((mNumberOfDivisions[0] < 1) || (mNumberOfDivisions[1] < 1) || (mNumberOfDivisions[2] < 1))
144 << "The number of division cannot be zero" << std::endl;
145
146 std::size_t size = mNumberOfDivisions[0] * mNumberOfDivisions[1] * mNumberOfDivisions[2];
147 mNodalData.assign(size, CartesianNodalData());
148 }
149
150 std::size_t GetNodeIndex(std::size_t I, std::size_t J, std::size_t K) const
151 {
152 const std::size_t index = I + J * mNumberOfDivisions[0] + K * mNumberOfDivisions[1] * mNumberOfDivisions[0];
153 return index;
154 }
155
156 CartesianNodalData &GetNodalData(std::size_t I, std::size_t J, std::size_t K)
157 {
158 const std::size_t index = GetNodeIndex(I,J,K);
159 return mNodalData[index];
160 }
161
162 const array_1d<std::size_t, 3> &GetNumberOfDivisions()
163 {
164 return mNumberOfDivisions;
165 }
166
167 bool IsInitialized() const
168 {
169 return mNumberOfDivisions[0] && mNumberOfDivisions[1] && mNumberOfDivisions[2];
170 }
