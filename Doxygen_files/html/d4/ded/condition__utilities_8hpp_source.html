116 template <unsigned int TDim, unsigned int TNumNodes>
117 static double CalculateIntegrationCoefficient(const Matrix& rJacobian, double Weight)
118 {
119 auto normal_vector = Vector{TDim, 0.0};
120
121 if constexpr (TDim == 2) {
122 normal_vector = column(rJacobian, 0);
123 } else if constexpr (TDim == 3) {
124 MathUtils<>::CrossProduct(normal_vector, column(rJacobian, 0), column(rJacobian, 1));
125 }
126 return Weight * MathUtils<>::Norm(normal_vector);
127 }
