1201 virtual Vector& LumpingFactors(
1202 Vector& rResult,
1203 const LumpingMethods LumpingMethod = LumpingMethods::ROW_SUM
1204 ) const
1205 {
1206 const SizeType number_of_nodes = this->size();
1207 const SizeType local_space_dimension = this->LocalSpaceDimension();
1208
1209 // Clear lumping factors
1210 if (rResult.size() != number_of_nodes)
1211 rResult.resize(number_of_nodes, false);
1212 noalias(rResult) = ZeroVector(number_of_nodes);
1213
1214 if (LumpingMethod == LumpingMethods::ROW_SUM) {
1215 const IntegrationMethod integration_method = GetDefaultIntegrationMethod();
1216 const GeometryType::IntegrationPointsArrayType& r_integrations_points = this->IntegrationPoints( integration_method );
1217 const Matrix& r_Ncontainer = this->ShapeFunctionsValues(integration_method);
1218
1219 // Vector fo jacobians
1220 Vector detJ_vector(r_integrations_points.size());
1221 DeterminantOfJacobian(detJ_vector, integration_method);
1222
1223 // Iterate over the integration points
1224 double domain_size = 0.0;
1225 for ( IndexType point_number = 0; point_number < r_integrations_points.size(); ++point_number ) {
1226 const double integration_weight = r_integrations_points[point_number].Weight() * detJ_vector[point_number];
1227 const Vector& rN = row(r_Ncontainer,point_number);
1228
1229 // Computing domain size
1230 domain_size += integration_weight;
1231
1232 for ( IndexType i = 0; i < number_of_nodes; ++i ) {
1233 rResult[i] += rN[i] * integration_weight;
1234 }
1235 }
1236
1237 // Divide by the domain size
1238 for ( IndexType i = 0; i < number_of_nodes; ++i ) {
1239 rResult[i] /= domain_size;
1240 }
1241 } else if (LumpingMethod == LumpingMethods::DIAGONAL_SCALING) {
1242 IntegrationMethod integration_method = GetDefaultIntegrationMethod();
1243 int j = std::min(static_cast<int>(integration_method) + 1, 4);
1244 integration_method = static_cast<IntegrationMethod>(j);
1245 const GeometryType::IntegrationPointsArrayType& r_integrations_points = this->IntegrationPoints( integration_method );
1246 const Matrix& r_Ncontainer = this->ShapeFunctionsValues(integration_method);
1247
1248 // Vector fo jacobians
1249 Vector detJ_vector(r_integrations_points.size());
1250 DeterminantOfJacobian(detJ_vector, integration_method);
1251
1252 // Iterate over the integration points
1253 for ( IndexType point_number = 0; point_number < r_integrations_points.size(); ++point_number ) {
1254 const double detJ = detJ_vector[point_number];
1255 const double integration_weight = r_integrations_points[point_number].Weight() * detJ;
1256 const Vector& rN = row(r_Ncontainer,point_number);
1257
1258 for ( IndexType i = 0; i < number_of_nodes; ++i ) {
1259 rResult[i] += std::pow(rN[i], 2) * integration_weight;
1260 }
1261 }
1262
1263 // Computing diagonal scaling coefficient
1264 double total_value = 0.0;
1265 for ( IndexType i = 0; i < number_of_nodes; ++i ) {
1266 total_value += rResult[i];
1267 }
1268 for ( IndexType i = 0; i < number_of_nodes; ++i ) {
1269 rResult[i] /= total_value;
1270 }
1271 } else if (LumpingMethod == LumpingMethods::QUADRATURE_ON_NODES) {
1272 // Divide by the domain size
1273 const double domain_size = DomainSize();
1274
1275 // Getting local coordinates
1276 Matrix local_coordinates(number_of_nodes, local_space_dimension);
1277 PointsLocalCoordinates(local_coordinates);
1278 Point local_point(ZeroVector(3));

1874 virtual CoordinatesArrayType& PointLocalCoordinates(
1875 CoordinatesArrayType& rResult,
1876 const CoordinatesArrayType& rPoint
1877 ) const
1878 {
1879 KRATOS_ERROR_IF(WorkingSpaceDimension() != LocalSpaceDimension()) << "ERROR:: Attention, the Point Local Coordinates must be specialized for the current geometry" << std::endl;
1880
1881 Matrix J = ZeroMatrix( WorkingSpaceDimension(), LocalSpaceDimension() );
1882
1883 rResult.clear();
1884
1885 Vector DeltaXi = ZeroVector( LocalSpaceDimension() );
1886
1887 CoordinatesArrayType CurrentGlobalCoords( ZeroVector( 3 ) );
1888
1889 static constexpr double MaxNormPointLocalCoordinates = 30.0;
1890 static constexpr std::size_t MaxIteratioNumberPointLocalCoordinates = 1000;
1891 static constexpr double MaxTolerancePointLocalCoordinates = 1.0e-8;
1892
1893 //Newton iteration:
1894 for(std::size_t k = 0; k < MaxIteratioNumberPointLocalCoordinates; k++) {
1895 CurrentGlobalCoords.clear();
1896 DeltaXi.clear();
1897
1898 GlobalCoordinates( CurrentGlobalCoords, rResult );
1899 noalias( CurrentGlobalCoords ) = rPoint - CurrentGlobalCoords;
1900 InverseOfJacobian( J, rResult );
1901 for(unsigned int i = 0; i < WorkingSpaceDimension(); i++) {
1902 for(unsigned int j = 0; j < WorkingSpaceDimension(); j++) {
1903 DeltaXi[i] += J(i,j)*CurrentGlobalCoords[j];
1904 }
1905 rResult[i] += DeltaXi[i];
1906 }
1907
1908 const double norm2DXi = norm_2(DeltaXi);
1909
1910 if(norm2DXi > MaxNormPointLocalCoordinates) {
1911 KRATOS_WARNING("Geometry") << "Computation of local coordinates failed at iteration " << k << std::endl;
1912 break;
1913 }
1914
1915 if(norm2DXi < MaxTolerancePointLocalCoordinates) {
1916 break;
1917 }
1918 }
1919
1920 return rResult;
1921 }
