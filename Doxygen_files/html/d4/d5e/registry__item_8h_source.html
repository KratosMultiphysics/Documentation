243 template<typename TItemType, class... TArgumentsList>
244 RegistryItem& AddItem(
245 std::string const& ItemName,
246 TArgumentsList&&... Arguments)
247 {
248 KRATOS_ERROR_IF(this->HasItem(ItemName))
249 << "The RegistryItem '" << this->Name() << "' already has an item with name "
250 << ItemName << "." << std::endl;
251
252 using ValueType = typename std::conditional<std::is_same<TItemType, RegistryItem>::value, SubRegistryItemFunctor, SubValueItemFunctor<TItemType>>::type;
253
254 auto insert_result = GetSubRegistryItemMap().emplace(
255 std::make_pair(
256 ItemName,
257 ValueType::Create(ItemName, std::forward<TArgumentsList>(Arguments)...)
258 )
259 );
260
261 KRATOS_ERROR_IF_NOT(insert_result.second)
262 << "Error in inserting '" << ItemName
263 << "' in registry item with name '" << this->Name() << "'." << std::endl;
264
265 return *insert_result.first->second;
266 }
