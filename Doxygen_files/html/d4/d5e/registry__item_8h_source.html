198 // Adding sub value item
199 template<typename TItemType, class... TArgumentsList>
200 RegistryItem& AddItem(
201 std::string const& ItemName,
202 TArgumentsList&&... Arguments)
203 {
204 KRATOS_ERROR_IF(this->HasItem(ItemName))
205 << "The RegistryItem '" << this->Name() << "' already has an item with name "
206 << ItemName << "." << std::endl;
207
208 using ValueType = typename std::conditional<std::is_same<TItemType, RegistryItem>::value, SubRegistryItemFunctor, SubValueItemFunctor<TItemType>>::type;
209
210 auto insert_result = GetSubRegistryItemMap().emplace(
211 std::make_pair(
212 ItemName,
213 ValueType::Create(ItemName, std::forward<TArgumentsList>(Arguments)...)
214 )
215 );
216
217 KRATOS_ERROR_IF_NOT(insert_result.second)
218 << "Error in inserting '" << ItemName
219 << "' in registry item with name '" << this->Name() << "'." << std::endl;
220
221 return *insert_result.first->second;
222 }
