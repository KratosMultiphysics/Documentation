44 {
45 Parameters parameters = this->GetParameters();

47
48 auto dx = parameters["voxel_sizes"].GetVector();
49 // Now we find the min and max value for the node coordinates
50 Vector min_v(3,std::numeric_limits<double>::max()), max_v(3, -std::numeric_limits<double>::max());
51 for(auto& node: GetInputModelPart().Nodes()){
52 double x = node.X(), y = node.Y(), z = node.Z();
53 min_v[0] = std::min(min_v[0], x); max_v[0] = std::max(max_v[0], x);
54 min_v[1] = std::min(min_v[1], y); max_v[1] = std::max(max_v[1], y);
55 min_v[2] = std::min(min_v[2], z); max_v[2] = std::max(max_v[2], z);
56 }
57 // Now we fill the PP and complete it
58 double epsilon = parameters["margin"].GetDouble();
59 bounding_box_parameters.AddVector("min_point", min_v - ((1.0 + epsilon)*dx));
60 bounding_box_parameters.AddVector("max_point", max_v + ((1.0 + epsilon)*dx));
61
62 KeyPlaneGenerationByBoundingBox::Generate(bounding_box_parameters);
