48 void ExecuteInitializeSolutionStep() override
49 {
50 KRATOS_TRY
51
52 const auto& r_variable = KratosComponents<Variable<double>>::Get(GetVariableName());
53 const auto time = GetModelPart().GetProcessInfo()[TIME] / mTimeUnitConverter;
54 const auto delta_h = mpTable->GetValue(time);
55
56 block_for_each(GetModelPart().Nodes(), [&delta_h, &r_variable, this](Node& rNode) {
57 auto xcoord = rNode.Coordinates()[GetHorizontalDirection()];
58 xcoord = std::max(xcoord, GetMinHorizontalCoordinate());
59 xcoord = std::min(xcoord, GetMaxHorizontalCoordinate());
60 const auto height =
61 GetSlope() * (xcoord - GetFirstReferenceCoordinate()[GetHorizontalDirection()]) +
62 GetFirstReferenceCoordinate()[GetGravityDirection()];
63 const auto distance = height - rNode.Coordinates()[GetGravityDirection()];
64 const auto pressure = GetSpecificWeight() * (distance + delta_h);
65 rNode.FastGetSolutionStepValue(r_variable) = std::max(pressure, 0.0);
66 });
67
68 KRATOS_CATCH("")
69 }
