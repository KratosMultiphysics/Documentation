184 KRATOS_DEPRECATED_MESSAGE("This is legacy version (using array instead of vector for shape function)") bool FindPointOnMesh(
185 const array_1d<double, 3 >& rCoordinates,
186 array_1d<double, TDim + 1 >& rNShapeFunction,
187 typename EntityType::Pointer& pEntity,
188 ResultIteratorType ItResultBegin,
189 const SizeType MaxNumberOfResults = 1000,
190 const double Tolerance = 1.0e-5
191 )
192 {
193 // Ask to the container for the list of candidate entities
194 SizeType results_found = mpBinsObjectDynamic->SearchObjectsInCell(BinsPointType{rCoordinates}, ItResultBegin, MaxNumberOfResults);
195
196 if (results_found > 0) {
197 // Loop over the candidate entities and check if the particle falls within
198 for (IndexType i = 0; i < results_found; i++) {
199 GeometryType& r_geom = (*(ItResultBegin + i))->GetGeometry();
200
201 // Find local position
202 array_1d<double, 3> point_local_coordinates;
203 Vector shape_function;
204 const bool is_found = LocalIsInside(r_geom, rCoordinates, point_local_coordinates, Tolerance);
205 r_geom.ShapeFunctionsValues(shape_function, point_local_coordinates);
206 noalias(rNShapeFunction) = shape_function;
207
208 if (is_found) {
209 pEntity = (*(ItResultBegin + i));
210 return true;
211 }
212 }
213 }
214
215 // Not found case
216 pEntity = nullptr;
217 return false;
218 }
