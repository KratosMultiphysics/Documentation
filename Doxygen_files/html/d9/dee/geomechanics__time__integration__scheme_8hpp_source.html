127 template <typename MemFuncPtr>
128 void BlockForEachActiveElement(ModelPart& rModelPart, MemFuncPtr pMemberFunction)
129 {
130 const auto& r_current_process_info = rModelPart.GetProcessInfo();
131 block_for_each(rModelPart.Elements(), [&r_current_process_info, pMemberFunction](auto& rElement) {
132 if (IsActive(rElement)) {
133 (rElement.*pMemberFunction)(r_current_process_info);
134 }
135 });
136 }
137
138 template <typename MemFuncPtr>
139 void BlockForEachActiveCondition(ModelPart& rModelPart, MemFuncPtr pMemberFunction)
140 {
141 const auto& r_current_process_info = rModelPart.GetProcessInfo();
142 block_for_each(rModelPart.Conditions(), [&r_current_process_info, pMemberFunction](Condition& rCondition) {
143 if (IsActive(rCondition)) (rCondition.*pMemberFunction)(r_current_process_info);
144 });
145 }

290 void CheckAllocatedVariables(const ModelPart& rModelPart) const
291 {
292 for (const auto& r_node : rModelPart.Nodes()) {
293 for (const auto& r_first_order_scalar_variable : mFirstOrderScalarVariables) {
294 this->CheckSolutionStepsData(r_node, r_first_order_scalar_variable.instance);
295 this->CheckSolutionStepsData(r_node, r_first_order_scalar_variable.first_time_derivative);
296 this->CheckDof(r_node, r_first_order_scalar_variable.instance);
297 }
298 }
299
300 for (const auto& r_node : rModelPart.Nodes()) {
301 for (const auto& r_second_order_vector_variable : this->mSecondOrderVectorVariables) {
302 if (!rModelPart.HasNodalSolutionStepVariable(r_second_order_vector_variable.instance))
303 continue;
304
305 this->CheckSolutionStepsData(r_node, r_second_order_vector_variable.instance);
306 this->CheckSolutionStepsData(r_node, r_second_order_vector_variable.first_time_derivative);
307 this->CheckSolutionStepsData(r_node, r_second_order_vector_variable.second_time_derivative);
308
309 // We don't check for "Z", since it is optional (in case of a 2D problem)
310 std::vector<std::string> components{"X", "Y"};
311 for (const auto& component : components) {
312 const auto& variable_component = VariablesUtilities::GetComponentFromVectorVariable(
313 r_second_order_vector_variable.instance.Name(), component);
314 this->CheckDof(r_node, variable_component);
315 }
316 }
317 }
318 }
