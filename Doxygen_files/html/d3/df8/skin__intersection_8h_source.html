53 void ComputeCutInsideCell(
54 SplitGeometryPointerType pGeometry,
55 const Point& rLowPoint,
56 const Point& rHighPoint,
57 std::size_t I,
58 std::size_t J,
59 std::size_t K)
60 {

62 geometries.push_back(pGeometry);
63
64 for (std::size_t i = 0; i < 3; i++) {
65 geometries = TrimLower<SplitGeometryType>(geometries, rLowPoint, i);

79 double GetIntegrationPointData(
80 std::vector<Vector>& rIntegrationPointData,
81 const Point& rLowPoint,
82 const Point& rHighPoint,
83 std::size_t I,
84 std::size_t J,
85 std::size_t K,
86 const GeometryData::IntegrationMethod Method,
87 double Thickness) const
88 {
89 rIntegrationPointData.resize(0);
90 auto& r_cut_geometries = CutGeometriesInCell(I, J, K);
91
92 for (auto& p_geom: r_cut_geometries) {
93 AddTriangleIntegrationPointData(rIntegrationPointData, *p_geom, Method);
94 }
95
96 // Assuming that the element is a hexahedra with plane faces and local coordinates in [-1,1]^3
97 array_1d<double,3> len = rHighPoint - rLowPoint;
98 double vol = len[0]*len[1]*len[2];
99 double weight_factor = Thickness*8.0/vol; // Integration point weights add to 8 for hexas
100 double effective_area = 0.0;
101
102 for (auto& r_data: rIntegrationPointData) {
103 for (std::size_t d = 0; d < 3; d++) {
104 r_data[d] = 2*(r_data[d] - rLowPoint[d])/len[d] - 1;
105 }
106 effective_area += r_data[3];
107 r_data[3] *= weight_factor;
108 //TODO: PointIsOnSharedFace should account for same-color neighbors only
109 //if (PointIsOnSharedFace(r_data, I, J, K)) {
110 // r_data[3] *= 0.5*weight_factor;
111 //}
112 //else {
113 // r_data[3] *= weight_factor;
114 //}
115 }
116
117 const double full_fraction = Thickness*effective_area / vol;
118 return full_fraction;
119 }

130 template<class GeometryType>
131 static std::size_t SignedDistanceUnder(
132 Vector& rNodalDistances,
133 const GeometryType& rGeometry,
134 const Point& rReferencePoint,
135 const std::size_t Component)
136 {
137 auto num_nodes = rGeometry.PointsNumber();
138 if (rNodalDistances.size() != num_nodes) {
139 rNodalDistances.resize(num_nodes);
140 noalias(rNodalDistances) = ZeroVector(num_nodes);
141 }
142
143 auto idistance = rNodalDistances.begin();
144 const double reference_coord = rReferencePoint[Component];
145 std::size_t points_under = 0;
146
147 for (const auto& point: rGeometry) {
148 const double d = reference_coord - point[Component];
149 if (d >= 0) points_under++;
150 *(idistance++) = d;
151 }
152
153 return points_under;
154 }

184 template<class GeometryType, class GeometryPointerType = typename GeometryType::Pointer>
185 static std::vector<GeometryPointerType> TrimHigher(
186 const std::vector<GeometryPointerType>& rGeometries,
187 const Point& rLowPoint,
188 std::size_t Component)
189 {
190 using PointType = typename GeometryType::PointType;
191 std::vector<GeometryPointerType> cut, trimmed_geometries;
192 Vector nodal_distances;
193
194 for (auto iter = rGeometries.begin(); iter < rGeometries.end(); ++iter) {
195 auto& r_geom = **iter;
196 cut.resize(0);
197
198 std::size_t points_under = SignedDistanceUnder(nodal_distances, r_geom, rLowPoint, Component);
199 if (points_under == 3) {
200 trimmed_geometries.push_back(*iter);
201 }
202 else if (points_under > 0) {
203 auto splitter = DivideTriangle3D3<PointType>(r_geom, nodal_distances);
204 splitter.GenerateDivision();
205 cut = splitter.GetPositiveSubdivisions();
206 trimmed_geometries.insert(trimmed_geometries.end(), cut.begin(), cut.end());
207 }
208 }
209
210 return trimmed_geometries;
211 }

242 // Each component of rIntegrationPointData is a 4-component vector
243 // Components 0-2 are the GLOBAL coordinates of the integration point
244 // Component 3 is the integration point weight in terms of area (DetJ*Gauss_W)
245 void AddTriangleIntegrationPointData(
246 std::vector<Vector>& rIntegrationPointData,
247 const SplitGeometryType& rGeom,
248 const GeometryData::IntegrationMethod Method) const
249 {
250 const std::size_t num_nodes = rGeom.PointsNumber();
251 const auto& r_integration_points = rGeom.IntegrationPoints(Method);
252 const std::size_t num_gauss = r_integration_points.size();
253 std::size_t new_point_offset = rIntegrationPointData.size();
254
255 Matrix shape_functions = rGeom.ShapeFunctionsValues(Method);
256 Vector det_j;
257 rGeom.DeterminantOfJacobian(det_j, Method);
258
259 for (std::size_t g = 0; g < num_gauss; g++) {
260 Vector data = ZeroVector(4);
261 data[3] = r_integration_points[g].Weight() * det_j[g];
262 rIntegrationPointData.push_back(data);
263 }
264
265 for (std::size_t n = 0; n < num_nodes; n++) {
266 const auto& r_coordinates = rGeom[n].Coordinates();
267 for (std::size_t g = 0; g < num_gauss; g++) {
268 for (std::size_t d = 0; d < 3; d++) {
269 rIntegrationPointData[new_point_offset+g][d] += shape_functions(g, n)*r_coordinates[d];
270 }
271 }
272 }

275 bool PointIsOnSharedFace(Vector& rPointData, std::size_t i, std::size_t j, std::size_t k) const
276 {
277 constexpr double coordinate_tol = 1e-9;
278 if (std::abs(rPointData[0] + 1.0) < coordinate_tol && i != 0) return true;
279 if (std::abs(rPointData[0] - 1.0) < coordinate_tol && i != mNumDivisions[0]) return true;
280 if (std::abs(rPointData[1] + 1.0) < coordinate_tol && j != 0) return true;
281 if (std::abs(rPointData[1] - 1.0) < coordinate_tol && j != mNumDivisions[1]) return true;
282 if (std::abs(rPointData[2] + 1.0) < coordinate_tol && k != 0) return true;
283 if (std::abs(rPointData[2] - 1.0) < coordinate_tol && k != mNumDivisions[1]) return true;
284 return false;
285 }
