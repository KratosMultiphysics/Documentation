25class ApplyConstantBoundaryPhreaticLinePressureProcess : public Process

27public:

31 ApplyConstantBoundaryPhreaticLinePressureProcess(ModelPart& model_part, Parameters rParameters)
32 : Process(Flags()), mrModelPart(model_part)
33 {
34 KRATOS_TRY
35
36 // only include validation with c++11 since raw_literals do not exist in c++03
37 Parameters default_parameters(R"(
38 {
39 "model_part_name":"PLEASE_CHOOSE_MODEL_PART_NAME",
40 "variable_name": "PLEASE_PRESCRIBE_VARIABLE_NAME",
41 "is_fixed": false,
42 "gravity_direction": 1,
43 "out_of_plane_direction": 2,
44 "first_reference_coordinate": [0.0,1.0,0.0],
45 "second_reference_coordinate": [1.0,0.5,0.0],
46 "specific_weight" : 10000.0,
47 "table" : 1
48 } )");
49
50 // Some values need to be mandatory prescribed since no meaningful default value exist. For
51 // this reason try accessing to them So that an error is thrown if they don't exist
52 rParameters["first_reference_coordinate"];
53 rParameters["second_reference_coordinate"];
54 rParameters["variable_name"];
55 rParameters["model_part_name"];
56
57 mIsFixedProvided = rParameters.Has("is_fixed");
58
59 // Now validate against defaults -- this also ensures no type mismatch
60 rParameters.ValidateAndAssignDefaults(default_parameters);
61
62 mVariableName = rParameters["variable_name"].GetString();
63 mIsFixed = rParameters["is_fixed"].GetBool();
64 mGravityDirection = rParameters["gravity_direction"].GetInt();
65 mOutOfPlaneDirection = rParameters["out_of_plane_direction"].GetInt();
66 KRATOS_ERROR_IF(mGravityDirection == mOutOfPlaneDirection)
67 << "Gravity direction cannot be the same as Out-of-Plane directions " << rParameters
68 << std::endl;
69
70 for (unsigned int i = 0; i < N_DIM_3D; ++i)
71 if (i != mGravityDirection && i != mOutOfPlaneDirection) mHorizontalDirection = i;
72
73 mFirstReferenceCoordinate = rParameters["first_reference_coordinate"].GetVector();
74 mSecondReferenceCoordinate = rParameters["second_reference_coordinate"].GetVector();
75
76 mMinHorizontalCoordinate = std::min(mFirstReferenceCoordinate[mHorizontalDirection],
77 mSecondReferenceCoordinate[mHorizontalDirection]);
78 mMaxHorizontalCoordinate = std::max(mFirstReferenceCoordinate[mHorizontalDirection],
79 mSecondReferenceCoordinate[mHorizontalDirection]);
80
81 KRATOS_ERROR_IF_NOT(mMaxHorizontalCoordinate > mMinHorizontalCoordinate)
82 << "First and second point on the phreatic line have the same horizontal coordinate"
83 << rParameters << std::endl;
84
85 mSlope =
86 (mSecondReferenceCoordinate[mGravityDirection] - mFirstReferenceCoordinate[mGravityDirection]) /
87 (mSecondReferenceCoordinate[mHorizontalDirection] - mFirstReferenceCoordinate[mHorizontalDirection]);
88
89 mSpecificWeight = rParameters["specific_weight"].GetDouble();
90
91 KRATOS_CATCH("")
92 }

100 void ExecuteInitialize() override
101 {
102 KRATOS_TRY
103
104 const Variable<double>& var = KratosComponents<Variable<double>>::Get(mVariableName);
105
106 block_for_each(mrModelPart.Nodes(), [&var, this](Node& rNode) {
107 if (mIsFixed) rNode.Fix(var);
108 else if (mIsFixedProvided) rNode.Free(var);
109
110 double horCoord = rNode.Coordinates()[mHorizontalDirection];
111 horCoord = std::max(horCoord, mMinHorizontalCoordinate);
112 horCoord = std::min(horCoord, mMaxHorizontalCoordinate);
113 const double height = mSlope * (horCoord - mFirstReferenceCoordinate[mHorizontalDirection]) +
114 mFirstReferenceCoordinate[mGravityDirection];
115 const double distance = height - rNode.Coordinates()[mGravityDirection];
116 const double pressure = mSpecificWeight * distance;
117 rNode.FastGetSolutionStepValue(var) = std::max(pressure, 0.0);
118 });
119
120 KRATOS_CATCH("")
121 }
