49 void ExecuteInitializeSolutionStep() override

52
53 const Variable<double>& var = KratosComponents<Variable<double>>::Get(mVariableName);
54 const double Time = mrModelPart.GetProcessInfo()[TIME] / mTimeUnitConverter;
55 const double deltaH = mpTable->GetValue(Time);
56
57 block_for_each(mrModelPart.Nodes(), [&var, &deltaH, this](Node& rNode) {
58 const double distance = mReferenceCoordinate - rNode.Coordinates()[mGravityDirection];
59 const double pressure = -PORE_PRESSURE_SIGN_FACTOR * mSpecificWeight * (distance + deltaH);
60 if (mIsSeepage) {
61 if (pressure < PORE_PRESSURE_SIGN_FACTOR * mPressureTensionCutOff) { // Before 0. was used i.s.o. the tension cut off value -> no effect in any test.
62 rNode.FastGetSolutionStepValue(var) = pressure;
63 if (mIsFixed) rNode.Fix(var);
64 } else {
65 if (mIsFixedProvided) rNode.Free(var);
66 }
67 } else {
68 rNode.FastGetSolutionStepValue(var) =
69 std::min(pressure, PORE_PRESSURE_SIGN_FACTOR * mPressureTensionCutOff);
70 }
71 });
72
73 KRATOS_CATCH("")
74 }
