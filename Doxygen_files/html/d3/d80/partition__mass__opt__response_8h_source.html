113
114 double element_density = elem_i.GetProperties().GetValue(DENSITY);

117
118 if((element_density>=(element_density_c-0.01))&&(element_density<=element_density_c)){

144 double CalculateElementValue(Element& elem_i, const std::size_t DomainSize){
145 // We get the element geometry
146 auto& r_this_geometry = elem_i.GetGeometry();
147 const std::size_t number_of_nodes = r_this_geometry.size();
148
149 // We copy the current coordinates and move the coordinates to the initial configuration
150 std::vector<array_1d<double, 3>> current_coordinates(number_of_nodes);
151 for (std::size_t i_node = 0; i_node < number_of_nodes; ++i_node) {
152 noalias(current_coordinates[i_node]) = r_this_geometry[i_node].Coordinates();
153 noalias(r_this_geometry[i_node].Coordinates()) = r_this_geometry[i_node].GetInitialPosition().Coordinates();
154 }
155
156 double element_density = elem_i.GetProperties().GetValue(DENSITY);
157 double element_density_f = std::floor(element_density);
158 double element_val = elem_i.GetGeometry().Volume() * std::pow(element_density-element_density_f,3);
159
160 // We restore the current configuration
161 for (std::size_t i_node = 0; i_node < number_of_nodes; ++i_node) {
162 noalias(r_this_geometry[i_node].Coordinates()) = current_coordinates[i_node];
163 }
164
165 return element_val;
166 }

196 void CalculateElementMaterialGradients(Element& elem_i, const std::size_t DomainSize){
197
198 // We get the element geometry
199 auto& r_this_geometry = elem_i.GetGeometry();
200 const std::size_t number_of_nodes = r_this_geometry.size();
201
202 double elem_dens_grad = 0.0;
203 double element_density = elem_i.GetProperties().GetValue(DENSITY);
204 int element_density_f = 8;
205 int element_density_c = 8;
206
207 if((element_density>=(element_density_c-0.01))&&(element_density<=element_density_c)){
208 elem_dens_grad = elem_i.GetGeometry().Volume();
209 }else if((element_density>=(element_density_f))&&(element_density<=element_density_f+0.01)){
210 elem_dens_grad = elem_i.GetGeometry().Volume();
211 }
212
213
214 for (SizeType i_node = 0; i_node < number_of_nodes; ++i_node){
215 const auto& d_pd_d_fd = r_this_geometry[i_node].FastGetSolutionStepValue(D_PD_D_FD);
216 r_this_geometry[i_node].FastGetSolutionStepValue(D_PARTITION_MASS_D_FD) += d_pd_d_fd * elem_dens_grad / number_of_nodes;
217 }
218 };
