109 template<class TIteratorType >
110 static void AddNodesFromOrderedContainer(ModelPart& rTargetModelPart, TIteratorType iNodesBegin, TIteratorType iNodesEnd)
111 {
112 KRATOS_TRY
113 ModelPart::NodesContainerType aux;
114 ModelPart* root_model_part = &rTargetModelPart.GetRootModelPart();
115
116 for(TIteratorType it = iNodesBegin; it!=iNodesEnd; it++) {
117 aux.push_back( *(it.base()) );
118 }
119 // Add the nodes to the root modelpart
120 root_model_part->Nodes() = JoinOrderedNodesContainerType(root_model_part->Nodes().begin(), root_model_part->Nodes().end(), aux.begin(), aux.end());
121
122 // Add to all of the leaves
123 ModelPart* current_part = &rTargetModelPart;
124 while(current_part->IsSubModelPart()) {
125 current_part->Nodes() = JoinOrderedNodesContainerType(current_part->Nodes().begin(), current_part->Nodes().end(), aux.begin(), aux.end());
126 current_part = &(current_part->GetParentModelPart());
127 }
128
129 KRATOS_CATCH("")
130 }

143 template<class TEntity>
144 static void GenerateEntitiesFromConnectivities(
145 const TEntity& rPrototypeEntity,
146 const std::vector<std::vector<std::size_t>>& rEntitiesConnectivities,
147 ModelPart::NodesContainerType& rThisNodes,
148 PointerVectorSet<TEntity, IndexedObject, std::less<typename IndexedObject::result_type>, std::equal_to<typename IndexedObject::result_type>, typename TEntity::Pointer, std::vector<typename TEntity::Pointer>>& rThisEntities,
149 const Properties::Pointer pProperties = nullptr
150 )
151 {
152 // Some definitions
153 const std::size_t number_of_entities = rEntitiesConnectivities.size();
154
155 // Number of nodes per entity
156 const std::size_t number_of_nodes_per_entity = rPrototypeEntity.GetGeometry().size();
157
158 const std::size_t current_size = rThisEntities.size();
159 rThisEntities.reserve(current_size + number_of_entities);
160
161 // Define a TLS struct to store the entity nodes
162 struct TLS {
163 TLS (const std::size_t NumberEntities) : entity_nodes(NumberEntities) {}
164 typename TEntity::NodesArrayType entity_nodes;
165 };
166
167 // From connectivities generate entities
168 const auto entities_vector = IndexPartition<IndexType>(number_of_entities).for_each<AccumReduction<typename TEntity::Pointer>>(TLS(number_of_nodes_per_entity), [&](const IndexType i, TLS& rTLS) {
169 const auto& r_nodes_ids = rEntitiesConnectivities[i];
170 for(std::size_t j = 0; j < number_of_nodes_per_entity; ++j) {
171 const std::size_t node_id = r_nodes_ids[j];
172 auto it_node = rThisNodes.find(node_id);
173 rTLS.entity_nodes(j) = *(it_node.base());
174 }
175
176 // Add the entity to the list
177 return rPrototypeEntity.Create(i + 1, rTLS.entity_nodes, pProperties);
178 });
179
180 // Add the entities to the list
181 for (std::size_t i = 0; i < number_of_entities; ++i) {
182 rThisEntities.push_back(entities_vector[i]);
183 }
184 }

316 template<class TIteratorType >
317 static typename ModelPart::NodesContainerType JoinOrderedNodesContainerType(
318 TIteratorType iC1Begin,
319 TIteratorType iC1End,
320 TIteratorType iC2Begin,
321 TIteratorType iC2End
322 )
323 {
324 std::size_t c1length = std::distance(iC1Begin,iC1End);
325 std::size_t c2length = std::distance(iC2Begin,iC2End);
326 TIteratorType blong, elong, bshort, eshort;
327 ModelPart::NodesContainerType aux;
328 aux.reserve(c1length + c2length);
329
330 // We order c1 and c2 to long and short
331 if(c1length>c2length){
332 blong = iC1Begin; elong = iC1End;
333 bshort = iC2Begin; eshort = iC2End;
334 } else {
335 blong = iC2Begin; elong = iC2End;
336 bshort = iC1Begin; eshort = iC1End;
337 }
338
339 // If short is empty we return long. If both empty it returns empty aux
340 if(c2length == 0 || c1length == 0){
341 for(TIteratorType it1=blong; it1!=elong; it1++){
342 aux.push_back(*(it1.base()) );
343 }
344 return aux;
345 }
346 TIteratorType it2 = blong;
347 for(TIteratorType it1=bshort; it1!=eshort; it1++) {
348 while(it2!=elong && it2->Id()<it1->Id()) {
349 aux.push_back(*(it2.base()));
350 it2++;
351 }
352 aux.push_back(*(it1.base()) );
353 if(it2!=elong && (it1->Id() == it2->Id())) { //If both are the same, then we need to skip
354 it2++;
355 }
356 }
357 while(it2 != elong) {
358 aux.push_back(*(it2.base()));
359 it2++;
360 }
361
362 return aux;
363 }

374 template<class TEntity>
375 static const TEntity& CheckEntity(const std::string& rEntityName)
376 {
377 // Check if the entity is registered in Kratos
378 if(!KratosComponents<TEntity>::Has(rEntityName)) {
379 std::stringstream buffer;
380 if constexpr(std::is_same_v<TEntity, Element>) {
381 buffer << "Element " << rEntityName << " is not registered in Kratos.";
382 buffer << " Please check the spelling of the element name and see if the application which containing it, is registered correctly.";
383 } else if constexpr(std::is_same_v<TEntity, Condition>) {
384 buffer << "Condition " << rEntityName << " is not registered in Kratos.";
385 buffer << " Please check the spelling of the condition name and see if the application which containing it, is registered correctly.";
386 } else {
387 static_assert(!std::is_same_v<TEntity, TEntity>, "Unsupported entity type.");
388 }
389 KRATOS_ERROR << buffer.str() << std::endl;
390 }
391 return KratosComponents<TEntity>::Get(rEntityName);
392 }
