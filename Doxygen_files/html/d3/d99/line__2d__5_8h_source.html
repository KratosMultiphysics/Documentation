427 CoordinatesArrayType& PointLocalCoordinates(CoordinatesArrayType& rResult,
428 const CoordinatesArrayType& rPoint) const override
429 {
430 BoundedMatrix<double, 3, 5> X;
431 BoundedMatrix<double, 3, 1> DN;
432 for (IndexType i = 0; i < this->size(); ++i)
433 {
434 const auto& r_node = this->GetPoint(i);
435 X(0, i) = r_node.X();
436 X(1, i) = r_node.Y();
437 X(2, i) = r_node.Z();
438 }
439
440 static constexpr double MaxNormPointLocalCoordinates = 300.0;
441 static constexpr std::size_t MaxIteratioNumberPointLocalCoordinates = 500;
442 static constexpr double MaxTolerancePointLocalCoordinates = 1.0e-8;
443
444 Matrix J = ZeroMatrix(1, 1);
445 Matrix invJ = ZeroMatrix(1, 1);
446
447 // Starting with xi = 0
448 if (rResult.size() != 3) rResult.resize(3, false);
449 noalias(rResult) = ZeroVector(3);
450 double delta_xi = 0.0;
451 const array_1d<double, 3> zero_array = ZeroVector(3);
452 array_1d<double, 3> current_global_coords;
453 array_1d<double, 1> res;
454
455 //Newton iteration:
456 for (IndexType k = 0; k < MaxIteratioNumberPointLocalCoordinates; ++k) {
457 noalias(current_global_coords) = zero_array;
458 this->GlobalCoordinates(current_global_coords, rResult);
459
460 noalias(current_global_coords) = rPoint - current_global_coords;
461
462 // Derivatives of shape functions
463 Matrix shape_functions_gradients;
464 shape_functions_gradients = ShapeFunctionsLocalGradients(shape_functions_gradients, rResult);
465 noalias(DN) = prod(X, shape_functions_gradients);
466 noalias(J) = prod(trans(DN), DN);
467 noalias(res) = prod(trans(DN), current_global_coords);
468
469 // The inverted jacobian matrix
470 invJ(0, 0) = 1.0 / J(0, 0);
471
472 delta_xi = invJ(0, 0) * res[0];
473
474 rResult[0] += delta_xi;
475
476 if (delta_xi > MaxNormPointLocalCoordinates) {
477 KRATOS_WARNING_IF("Line2D5", k > 0) << "detJ =\t" << J(0, 0) << " DeltaX =\t" << delta_xi << " stopping calculation. Iteration:\t" << k << std::endl;
478 break;
479 }
480
481 if (delta_xi < MaxTolerancePointLocalCoordinates)
482 break;
483 }
484 return rResult;
485 }

505 JacobiansType& Jacobian(JacobiansType& rResult, IntegrationMethod ThisMethod) const override
506 {
507 // Getting derivatives of shape functions
508 const ShapeFunctionsGradientsType shape_functions_gradients
509 = CalculateShapeFunctionsIntegrationPointsLocalGradients(ThisMethod);
510 const std::size_t number_of_integration_points = this->IntegrationPointsNumber(ThisMethod);
511 if (rResult.size() != number_of_integration_points) {
512 JacobiansType temp(number_of_integration_points);
513 rResult.swap(temp);
514 }
515 // Loop over all integration points
516 for (std::size_t pnt = 0; pnt < number_of_integration_points; ++pnt) {
517 // Initializing jacobian matrix
518 noalias(rResult[pnt]) = ZeroMatrix(2, 1);
519 // Loop over all nodes
520 for (std::size_t i = 0; i < this->PointsNumber(); ++i) {
521 const auto& r_node = this->GetPoint(i);
522 rResult[pnt](0, 0) += r_node.X() * shape_functions_gradients[pnt](i, 0);
523 rResult[pnt](1, 0) += r_node.Y() * shape_functions_gradients[pnt](i, 0);
524 }
525 } // End of loop over all integration points
526 return rResult;
527 }

715 Vector& ShapeFunctionsValues(Vector& rResult, const CoordinatesArrayType& rCoordinates) const override
716 {
717 if (rResult.size() != 5) rResult.resize(5, false);
718 //
719 const double xi = rCoordinates[0];
720 const double fx1 = xi - 1.0;
721 const double fx2 = xi + 1.0;
722 const double fx3 = fx1 * fx2;
723 const double gx1 = 2.0 * xi - 1.0;
724 const double gx2 = 2.0 * xi + 1.0;
725 const double gx3 = gx1 * gx2;
726 const double cof1 = 1.0 / 6.0;
727 const double cof2 = 8.0 * cof1;
728 //
729 rResult[0] = xi * fx1 * gx3 * cof1;
730 rResult[1] = xi * fx2 * gx3 * cof1;
731 rResult[2] = -xi * fx3 * gx1 * cof2;
732 rResult[3] = fx3 * gx3;
733 rResult[4] = -xi * fx3 * gx2 * cof2;
734 //
735 return rResult;
736 }

747 double ShapeFunctionValue(const IndexType ShapeFunctionIndex, const CoordinatesArrayType& rPoint) const override
748 {
749 const double xi = rPoint[0];
750 const double fx1 = xi - 1.0;
751 const double fx2 = xi + 1.0;
752 const double fx3 = fx1 * fx2;
753 const double gx1 = 2.0 * xi - 1.0;
754 const double gx2 = 2.0 * xi + 1.0;
755 const double gx3 = gx1 * gx2;
756 const double cof1 = 1.0 / 6.0;
757 const double cof2 = 8.0 * cof1;
758 double shape = 0.0;
759 //
760 switch (ShapeFunctionIndex)
761 {
762 case 0:
763 shape = xi * fx1 * gx3 * cof1;
764 break;
765 case 1:
766 shape = xi * fx2 * gx3 * cof1;
767 break;
768 case 2:
769 shape = -xi * fx3 * gx1 * cof2;
770 break;
771 case 3:
772 shape = fx3 * gx3;
773 break;
774 case 4:
775 shape = -xi * fx3 * gx2 * cof2;
776 break;
777 default:
778 KRATOS_ERROR << "Wrong index of shape function!" << *this << std::endl;
779 break;
780 }
781 //
782 return shape;
783 }

888 Matrix& ShapeFunctionsLocalGradients(Matrix& rResult, const CoordinatesArrayType& rPoint) const override
889 {
890 // Setting up result matrix
891 if (rResult.size1() != 5 || rResult.size2() != 1) rResult.resize(5, 1, false);
892 noalias(rResult) = ZeroMatrix(5, 1);
893 //
894 const double xi = rPoint[0];
895 const double fx1 = xi - 1.0;
896 const double fx2 = xi + 1.0;
897 const double fx3 = fx1 * fx2;
898 const double gx1 = 2.0 * xi - 1.0;
899 const double gx2 = 2.0 * xi + 1.0;
900 const double gx3 = gx1 * gx2;
901 const double cof1 = 1.0 / 6.0;
902 const double cof2 = 8.0 * cof1;
903 //
904 rResult(0, 0) = gx3 * gx1 * cof1 + xi * xi * fx1 * cof2;
905 rResult(1, 0) = gx3 * gx2 * cof1 + xi * xi * fx2 * cof2;
906 rResult(2, 0) = -(gx1 * (gx3 - fx3 - 1.0) + 2.0 * xi * fx3) * cof2;
907 rResult(3, 0) = 2.0 * xi * (gx3 + 4.0 * fx3);
908 rResult(4, 0) = -(gx2 * (gx3 - fx3 - 1.0) + 2.0 * xi * fx3) * cof2;
909 return rResult;

1070 const IntegrationPointsArrayType& IntegrationPoints = all_integration_points[static_cast<int>(ThisMethod)];
1071 ShapeFunctionsGradientsType DN_De(IntegrationPoints.size());

1073 //
1074 const double cof1 = 1.0 / 6.0;
1075 const double cof2 = 8.0 * cof1;
1076 //
1077 for (unsigned int pnt = 0; pnt < IntegrationPoints.size(); ++pnt)
1078 {
1079 const double xi = IntegrationPoints[pnt].X();
1080 const double fx1 = xi - 1.0;
1081 const double fx2 = xi + 1.0;
1082 const double fx3 = fx1 * fx2;
1083 const double gx1 = 2.0 * xi - 1.0;
1084 const double gx2 = 2.0 * xi + 1.0;
1085 const double gx3 = gx1 * gx2;

1087 DN_De[pnt](0, 0) = gx3 * gx1 * cof1 + xi * xi * fx1 * cof2;
1088 DN_De[pnt](1, 0) = gx3 * gx2 * cof1 + xi * xi * fx2 * cof2;
1089 DN_De[pnt](2, 0) = -(gx1 * (gx3 - fx3 - 1.0) + 2.0 * xi * fx3) * cof2;
1090 DN_De[pnt](3, 0) = 2.0 * xi * (gx3 + 4.0 * fx3);
1091 DN_De[pnt](4, 0) = -(gx2 * (gx3 - fx3 - 1.0) + 2.0 * xi * fx3) * cof2;
1092 }
1093 return DN_De;
