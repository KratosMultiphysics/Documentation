413 static void BtDBProductOperation(
414 MatrixType& rA,
415 const MatrixType& rD,
416 const MatrixType& rB,
417 const bool CallFillCompleteOnResult = true,
418 const bool KeepAllHardZeros = false,
419 const bool EnforceInitialGraph = false
420 )
421 {
422 // Define first auxiliary matrix
423 std::vector<int> NumNz;
424 MatrixType aux_1(::Copy, rA.RowMap(), NumNz.data());
425
426 // First multiplication
427 TransposeMult(rB, rD, aux_1, {true, false}, CallFillCompleteOnResult, KeepAllHardZeros);
428
429 // If we enforce the initial connectivity
430 if (EnforceInitialGraph) {
431 // Second multiplication
432 MatrixType aux_2(::Copy, rA.RowMap(), NumNz.data());
433 Mult(aux_1, rB, aux_2, CallFillCompleteOnResult, KeepAllHardZeros);
434
435 // Create an Epetra_Matrix
436 MatrixType* aux_3 = new MatrixType(::Copy, CombineMatricesGraphs(rA, aux_2));
437
438 // Copy values
439 CopyMatrixValues(*aux_3, aux_2);
440
441 // Doing a swap
442 std::swap(rA, *aux_3);
443
444 // Delete the new matrix
445 delete aux_3;
446 } else { // A new matrix
447 // Already existing matrix
448 if (rA.NumGlobalNonzeros() > 0) {
449 // Create an Epetra_Matrix
450 MatrixType* aux_2 = new MatrixType(::Copy, rB.RowMap(), NumNz.data());
451
452 // Second multiplication
453 Mult(aux_1, rB, *aux_2, CallFillCompleteOnResult, KeepAllHardZeros);
454
455 // Doing a swap
456 std::swap(rA, *aux_2);
457
458 // Delete the new matrix
459 delete aux_2;
460 } else { // Empty matrix
461 // Second multiplication
462 Mult(aux_1, rB, rA, CallFillCompleteOnResult, KeepAllHardZeros);
463 }
464 }
465 }

771 inline static void AssembleLHS(
772 MatrixType& rA,
773 const Matrix& rLHSContribution,
774 const std::vector<std::size_t>& rEquationId
775 )
776 {
777 const unsigned int system_size = Size1(rA);
778
779 // Count active indices
780 unsigned int active_indices = 0;
781 for (unsigned int i = 0; i < rEquationId.size(); i++)
782 if (rEquationId[i] < system_size)
783 ++active_indices;
784
785 if (active_indices > 0) {
786 // Size Epetra vectors
787 Epetra_IntSerialDenseVector indices(active_indices);
788 Epetra_SerialDenseMatrix values(active_indices, active_indices);
789
790 // Fill epetra vectors
791 unsigned int loc_i = 0;
792 for (unsigned int i = 0; i < rEquationId.size(); i++) {
793 if (rEquationId[i] < system_size) {
794 indices[loc_i] = rEquationId[i];
795
796 unsigned int loc_j = 0;
797 for (unsigned int j = 0; j < rEquationId.size(); j++) {
798 if (rEquationId[j] < system_size) {
799 values(loc_i, loc_j) = rLHSContribution(i, j);
800 ++loc_j;
801 }
802 }
803 ++loc_i;
804 }
805 }
806
807 const int ierr = rA.SumIntoGlobalValues(indices, values);
808 KRATOS_ERROR_IF(ierr != 0) << "Epetra failure found" << std::endl;
809 }
810 }

945 MatrixPointerType ReadMatrixMarket(
946 const std::string FileName,
947 Epetra_MpiComm& rComm
948 )
949 {
950 KRATOS_TRY
951
952 Epetra_CrsMatrix* pp = nullptr;
953
954 int error_code = EpetraExt::MatrixMarketFileToCrsMatrix(FileName.c_str(), rComm, pp);
955
956 KRATOS_ERROR_IF(error_code != 0) << "Eerror thrown while reading Matrix Market file "<<FileName<< " error code is : " << error_code;
957
958 rComm.Barrier();
959
960 const Epetra_CrsGraph& rGraph = pp->Graph();
961 MatrixPointerType paux = Kratos::make_shared<Epetra_FECrsMatrix>( ::Copy, rGraph, false );
962
963 IndexType NumMyRows = rGraph.RowMap().NumMyElements();
964
965 int* MyGlobalElements = new int[NumMyRows];
966 rGraph.RowMap().MyGlobalElements(MyGlobalElements);
967
968 for(IndexType i = 0; i < NumMyRows; ++i) {
969// std::cout << pA->Comm().MyPID() << " : I=" << i << std::endl;
970 IndexType GlobalRow = MyGlobalElements[i];
971
972 int NumEntries;
973 std::size_t Length = pp->NumGlobalEntries(GlobalRow); // length of Values and Indices
974
975 double* Values = new double[Length]; // extracted values for this row
976 int* Indices = new int[Length]; // extracted global column indices for the corresponding values
977
978 error_code = pp->ExtractGlobalRowCopy(GlobalRow, Length, NumEntries, Values, Indices);
979
980 KRATOS_ERROR_IF(error_code != 0) << "Error thrown in ExtractGlobalRowCopy : " << error_code;
981
982 error_code = paux->ReplaceGlobalValues(GlobalRow, Length, Values, Indices);
983
984 KRATOS_ERROR_IF(error_code != 0) << "Error thrown in ReplaceGlobalValues : " << error_code;
985
986 delete [] Values;
987 delete [] Indices;
988 }
989
990 paux->GlobalAssemble();
991
992 delete [] MyGlobalElements;
993 delete pp;
994
995 return paux;
996 KRATOS_CATCH("");
997 }

1186 static double CheckAndCorrectZeroDiagonalValues(
1187 const ProcessInfo& rProcessInfo,
1188 MatrixType& rA,
1189 VectorType& rb,
1190 const SCALING_DIAGONAL ScalingDiagonal = SCALING_DIAGONAL::NO_SCALING
1191 )
1192 {
1193 KRATOS_TRY
1194
1195 // Define zero value tolerance
1196 const double zero_tolerance = std::numeric_limits<double>::epsilon();
1197
1198 // The diagonal considered
1199 const double scale_factor = GetScaleNorm(rProcessInfo, rA, ScalingDiagonal);
1200
1201 for (int i = 0; i < rA.NumMyRows(); i++) {
1202 int numEntries; // Number of non-zero entries
1203 double* vals; // Row non-zero values

1205 rA.ExtractMyRowView(i, numEntries, vals, cols);
1206 const int row_gid = rA.RowMap().GID(i);
1207 bool empty = true;
1208 int j;
1209 for (j = 0; j < numEntries; j++) {
1210 const int col_gid = rA.ColMap().GID(cols[j]);
1211 // Check diagonal value
1212 if (col_gid == row_gid) {
1213 if(std::abs(vals[j]) > zero_tolerance) {
1214 empty = false;
1215 }
1216 break;
1217 }
1218 }
1219
1220 // If diagonal empty assign scale factor
1221 if (empty) {
1222 vals[j] = scale_factor;
1223 rb[0][i] = 0.0;
1224 }
1225 }
1226
1227 // Global assembly
1228 rb.GlobalAssemble();
1229 rA.GlobalAssemble();
1230
1231 return scale_factor;
1232
1233 KRATOS_CATCH("");

1243 static double GetScaleNorm(
1244 const ProcessInfo& rProcessInfo,
1245 const MatrixType& rA,
1246 const SCALING_DIAGONAL ScalingDiagonal = SCALING_DIAGONAL::NO_SCALING
1247 )
1248 {

1250
1251 switch (ScalingDiagonal) {
1252 case SCALING_DIAGONAL::NO_SCALING:
1253 return 1.0;
1254 case SCALING_DIAGONAL::CONSIDER_PRESCRIBED_DIAGONAL: {
