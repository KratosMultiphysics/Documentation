464 IntegrationMethod GetIntegrationMethod() const override
465 {
466 // Setting the auxiliary integration points
467 const IndexType integration_order = GetProperties().Has(INTEGRATION_ORDER_CONTACT) ? GetProperties().GetValue(INTEGRATION_ORDER_CONTACT) : 2;
468 switch (integration_order) {
469 case 1: return GeometryData::IntegrationMethod::GI_GAUSS_1;
470 case 2: return GeometryData::IntegrationMethod::GI_GAUSS_2;
471 case 3: return GeometryData::IntegrationMethod::GI_GAUSS_3;
472 case 4: return GeometryData::IntegrationMethod::GI_GAUSS_4;
473 case 5: return GeometryData::IntegrationMethod::GI_GAUSS_5;
474 default: return GeometryData::IntegrationMethod::GI_GAUSS_2;
475 }
