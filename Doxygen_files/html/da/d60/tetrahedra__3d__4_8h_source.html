573 double Circumradius() const override {
574 // | s10 y10 z10 | | s10 x10 z10 | | s10 x10 y10 | | x10 y10 z10 |
575 // MX | s20 y20 z20 | MY | s20 x20 z20 | MZ | s20 x20 y20 | AL | x20 y20 z20 |
576 // | s30 y30 z30 | | s30 x30 z30 | | s30 x30 y30 | | x30 y30 z30 |
577
578 const CoordinatesArrayType& rP0 = this->Points()[0].Coordinates();
579 const CoordinatesArrayType& rP1 = this->Points()[1].Coordinates();
580 const CoordinatesArrayType& rP2 = this->Points()[2].Coordinates();
581 const CoordinatesArrayType& rP3 = this->Points()[3].Coordinates();
582
583 const double aDot = rP0[0] * rP0[0] + rP0[1] * rP0[1] + rP0[2] * rP0[2];
584 const double bDot = rP1[0] * rP1[0] + rP1[1] * rP1[1] + rP1[2] * rP1[2];
585 const double cDot = rP2[0] * rP2[0] + rP2[1] * rP2[1] + rP2[2] * rP2[2];
586 const double dDot = rP3[0] * rP3[0] + rP3[1] * rP3[1] + rP3[2] * rP3[2];
587
588 // Build the simplified matrices
589 const double s10 = aDot - dDot;
590 const double x10 = rP0[0] - rP3[0];
591 const double y10 = rP0[1] - rP3[1];
592 const double z10 = rP0[2] - rP3[2];
593
594 const double s20 = bDot - dDot;
595 const double x20 = rP1[0] - rP3[0];
596 const double y20 = rP1[1] - rP3[1];
597 const double z20 = rP1[2] - rP3[2];
598
599 const double s30 = cDot - dDot;
600 const double x30 = rP2[0] - rP3[0];
601 const double y30 = rP2[1] - rP3[1];
602 const double z30 = rP2[2] - rP3[2];
603
604 const double detJMX = s10 * y20 * z30 + y10 * z20 * s30 + z10 * s20 * y30 - s30 * y20 * z10 - y30 * z20 * s10 - z30 * s20 * y10;
605 const double detJMY = s10 * x20 * z30 + x10 * z20 * s30 + z10 * s20 * x30 - s30 * x20 * z10 - x30 * z20 * s10 - z30 * s20 * x10;
606 const double detJMZ = s10 * x20 * y30 + x10 * y20 * s30 + y10 * s20 * x30 - s30 * x20 * y10 - x30 * y20 * s10 - y30 * s20 * x10;
607 const double detJAL = x10 * y20 * z30 + y10 * z20 * x30 + z10 * x20 * y30 - x30 * y20 * z10 - y30 * z20 * x10 - z30 * x20 * y10;
608
609 return std::sqrt( (detJMX*detJMX + detJMY*detJMY + detJMZ*detJMZ)) / (2*std::abs(detJAL));
610 }

617 double Inradius() const override {
618 // | x10 y10 z10 |
619 // AL | x20 y20 z20 |
620 // | x30 y30 z30 |
621
622 const CoordinatesArrayType& rP0 = this->Points()[0].Coordinates();
623 const CoordinatesArrayType& rP1 = this->Points()[1].Coordinates();
624 const CoordinatesArrayType& rP2 = this->Points()[2].Coordinates();
625 const CoordinatesArrayType& rP3 = this->Points()[3].Coordinates();
626
627 array_1d<double, 3> c012, c013, c023, c123;
628
629 MathUtils<double>::CrossProduct(c012, rP2-rP0, rP1-rP0);
630 MathUtils<double>::CrossProduct(c013, rP3-rP0, rP1-rP0);
631 MathUtils<double>::CrossProduct(c023, rP3-rP0, rP2-rP0);
632 MathUtils<double>::CrossProduct(c123, rP3-rP1, rP2-rP1);
633
634 const double n012 = std::sqrt(c012[0]*c012[0] + c012[1]*c012[1] + c012[2]*c012[2]);
635 const double n013 = std::sqrt(c013[0]*c013[0] + c013[1]*c013[1] + c013[2]*c013[2]);
636 const double n023 = std::sqrt(c023[0]*c023[0] + c023[1]*c023[1] + c023[2]*c023[2]);
637 const double n123 = std::sqrt(c123[0]*c123[0] + c123[1]*c123[1] + c123[2]*c123[2]);
638
639 // Build the simplified matrices
640 const double x10 = rP0[0] - rP3[0];
641 const double y10 = rP0[1] - rP3[1];
642 const double z10 = rP0[2] - rP3[2];
643
644 const double x20 = rP1[0] - rP3[0];
645 const double y20 = rP1[1] - rP3[1];
646 const double z20 = rP1[2] - rP3[2];
647
648 const double x30 = rP2[0] - rP3[0];
649 const double y30 = rP2[1] - rP3[1];
650 const double z30 = rP2[2] - rP3[2];
651
652 const double detJAL = x10 * y20 * z30 + y10 * z20 * x30 + z10 * x20 * y30 - x30 * y20 * z10 - y30 * z20 * x10 - z30 * x20 * y10;
653
654 return std::abs(detJAL) / (n012 + n013 + n023 + n123);
655 }

925 coords(i, j) = xyz[j];
926 }
927 //we have to check 6 different angles (one per edge)
928 //to get an easy-to-read algorithm, we find the angles by creating the normal planes to each face and then find the angles between them
929 //to do so, we create a list of 3 nodes per angle. face1 is defined with nodes 0,1,2a , and face2 is defined with nodes 0,1,2b
930 const int node0[] = {0, 0, 0, 1, 1, 2};
931 const int node1[] = {1, 3, 2, 3, 2, 3};
932 const int node2a[] = {2, 1, 1, 0, 0, 0};
933 const int node2b[] = {3, 2, 3, 2, 3, 1};
934
935 array_1d<double,3> edge1,edge2a,edge2b,normal1,normal2;
936
937 //now we only loop through the six edges to see which one has the lowest angle
938 for (unsigned int i = 0; i < 6; i++) {
939 //first we find the edges
940 for (unsigned int j = 0; j < 3; ++j) {
941 edge1[j] = coords(node1[i],j) - coords(node0[i],j) ;
942 edge2a[j] = coords(node2a[i],j) - coords(node0[i],j) ;
943 edge2b[j] = coords(node2b[i],j) - coords(node0[i],j) ;
944 }
945 //now we find the normals to the planes
946 MathUtils<double>::CrossProduct(normal1,edge1,edge2a);
947 MathUtils<double>::CrossProduct(normal2,edge1,edge2b);
948 normal1 /= std::sqrt(normal1[0]*normal1[0] + normal1[1]*normal1[1] + normal1[2]*normal1[2]);
949 normal2 /= std::sqrt(normal2[0]*normal2[0] + normal2[1]*normal2[1] + normal2[2]*normal2[2]);
950
951 //and finally the cos of the angle:
952 const double angle_cos = ( normal1[0]*normal2[0] + normal1[1]*normal2[1] + normal1[2]*normal2[2] );
953 rDihedralAngles[i] = std::acos(angle_cos);
954 }

1462 void SplitAndDecompose(

1465 {
1466
1467 // Determine on which side of the plane the points of the tetrahedron lie.
1468 int i = 0;
1469 int positive = 0, negative = 0, zero = 0;
1470 array_1d<double,4> C;
1471 array_1d<int,4> pos, neg, zer;
1472
1473 noalias(C) = ZeroVector(4);
1474
1475 pos[0] = 0;
1476 neg[0] = 0;
1477 zer[0] = 0;
1478 pos[1] = 0;
1479 neg[1] = 0;
1480 zer[1] = 0;
1481 pos[2] = 0;
1482 neg[2] = 0;
1483 zer[2] = 0;
1484 pos[3] = 0;
1485 neg[3] = 0;
1486 zer[3] = 0;
1487
1488 for (i = 0; i < 4; ++i) {
1489 const array_1d<double,3>& p = tetra[i].Coordinates();
1490 C[i] = plane.DistanceTo(p);
1491 if (C[i] > 0.00)
1492 pos[positive++] = i;
1493 else if (C[i] < 0.00)
1494 neg[negative++] = i;
1495 else
1496 zer[zero++] = i;
1497 }
1498
1499 // For a split to occur, one of the c_i must be positive and one must
1500 // be negative.
1501 if (negative == 0) {
1502 // Tetrahedron is completely on the positive side of plane, full clip.
1503 return;
1504 }
1505
1506 if (positive == 0) {
1507 // Tetrahedron is completely on the negative side of plane.
1508 inside.push_back(tetra);
1509 return;
1510 }
1511
1512 double w0, w1, invCDiff;
1513 array_1d<array_1d<double, 3 >, 4> intp;
1514 array_1d<array_1d<double, 3 >, 4> V;
1515
1516 if (positive == 3) {
1517 // +++-
1518 for (i = 0; i < positive; ++i) {
1519 invCDiff = 1.00/(C[pos[i]] - C[neg[0]]);
1520 w0 = -C[neg[0]]*invCDiff;
1521 w1 = +C[pos[i]]*invCDiff;
1522 V[pos[i]] = w0*tetra[pos[i]].Coordinates() + w1*tetra[neg[0]].Coordinates();
1523 }

1525 } else if (positive == 2) {

1527 // ++--
1528 for (i = 0; i < positive; ++i) {
1529 invCDiff = (1.00)/(C[pos[i]] - C[neg[0]]);
1530 w0 = -C[neg[0]]*invCDiff;
1531 w1 = +C[pos[i]]*invCDiff;
1532 intp[i] = w0*tetra[pos[i]].Coordinates() + w1*tetra[neg[0]].Coordinates();
1533 }
1534 for (i = 0; i < negative; ++i) {
1535 invCDiff = (1.00)/(C[pos[i]] - C[neg[1]]);
1536 w0 = -C[neg[1]]*invCDiff;
1537 w1 = +C[pos[i]]*invCDiff;
1538 intp[i+2] = w0*tetra[pos[i]].Coordinates() + w1*tetra[neg[1]].Coordinates();
1539 }
1540
1541 V[pos[0]] = intp[2];
1542 V[pos[1]] = intp[1];
1543 inside.push_back(tetra);
1544 } else {
1545 // ++-0
1546 for (i = 0; i < positive; ++i) {
1547 invCDiff = (1.00)/(C[pos[i]] - C[neg[0]]);
1548 w0 = -C[neg[0]]*invCDiff;

1555 if (negative == 3) {
1556 // +---

1558 invCDiff = (1.00)/(C[pos[0]] - C[neg[i]]);
1559 w0 = -C[neg[i]]*invCDiff;
