146 void AddDampingToLHS(LocalSystemMatrixType& LHS_Contribution,LocalSystemMatrixType& C,const ProcessInfo& rCurrentProcessInfo)
147 {
148 // adding damping contribution
149 if (C.size1() != 0)
150 noalias(LHS_Contribution) += 1.0/(mTheta_u*mDeltaTime)*C;
151 }
152
153//----------------------------------------------------------------------------------------------------------------------------------------------------------------------------------
154
155 void AddDampingToRHS(Element& rCurrentElement,LocalSystemVectorType& RHS_Contribution,
156 LocalSystemMatrixType& C,const ProcessInfo& rCurrentProcessInfo)
157 {
158 int thread = OpenMPUtils::ThisThread();
159
160 //adding damping contribution
161 if (C.size1() != 0)
162 {
163 rCurrentElement.GetFirstDerivativesVector(mVelocityVector[thread], 0);
164
165 noalias(RHS_Contribution) -= prod(C, mVelocityVector[thread]);
166 }
167 }
