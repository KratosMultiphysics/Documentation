1567 CoordinatesArrayType& rResult,

1570 ) const
1571 {
1572 BoundedMatrix<double,3,4> X;
1573 BoundedMatrix<double,3,2> DN;
1574 for(IndexType i=0; i<this->size();i++) {
1575 X(0, i) = this->GetPoint( i ).X();

1580 static constexpr double MaxNormPointLocalCoordinates = 300.0;
1581 static constexpr std::size_t MaxIteratioNumberPointLocalCoordinates = 500;
1582 static constexpr double MaxTolerancePointLocalCoordinates = 1.0e-8;
1583
1584 Matrix J = ZeroMatrix( 2, 2 );
1585 Matrix invJ = ZeroMatrix( 2, 2 );
1586
1587 // Starting with xi = 0
1588 rResult = ZeroVector( 3 );
1589 array_1d<double, 2> DeltaXi = ZeroVector( 2 );
1590 const array_1d<double, 3> zero_array = ZeroVector(3);
1591 array_1d<double, 3> CurrentGlobalCoords;
1592
1593 //Newton iteration:
1594 for ( IndexType k = 0; k < MaxIteratioNumberPointLocalCoordinates; k++ ) {
1595 noalias(CurrentGlobalCoords) = zero_array;
1596 this->GlobalCoordinates( CurrentGlobalCoords, rResult );
1597
1598 noalias( CurrentGlobalCoords ) = rPoint - CurrentGlobalCoords;
1599
1600 // Derivatives of shape functions
1601 Matrix shape_functions_gradients;
1602 shape_functions_gradients = ShapeFunctionsLocalGradients(shape_functions_gradients, rResult );
1603 noalias(DN) = prod(X,shape_functions_gradients);
1604
1605 noalias(J) = prod(trans(DN),DN);
1606 const array_1d<double, 2> res = prod(trans(DN), CurrentGlobalCoords);
1607
1608 // Deteminant of Jacobian
1609 const double det_j = J( 0, 0 ) * J( 1, 1 ) - J( 0, 1 ) * J( 1, 0 );
1610
1611 // Filling matrix
1612 invJ( 0, 0 ) = ( J( 1, 1 ) ) / ( det_j );
1613 invJ( 1, 0 ) = -( J( 1, 0 ) ) / ( det_j );
1614 invJ( 0, 1 ) = -( J( 0, 1 ) ) / ( det_j );
1615 invJ( 1, 1 ) = ( J( 0, 0 ) ) / ( det_j );
1616
1617 DeltaXi( 0 ) = invJ( 0, 0 ) * res[0] + invJ( 0, 1 ) * res[1];
1618 DeltaXi( 1 ) = invJ( 1, 0 ) * res[0] + invJ( 1, 1 ) * res[1];
1619
1620 rResult[0] += DeltaXi[0];
1621 rResult[1] += DeltaXi[1];
1622
1623 if ( norm_2( DeltaXi ) > MaxNormPointLocalCoordinates ) {
1624 KRATOS_WARNING_IF("Quadrilateral3D4", IsInside == false && k > 0) << "detJ =\t" << det_j << " DeltaX =\t" << DeltaXi << " stopping calculation. Iteration:\t" << k << std::endl;
1625 break;
1626 }
1627
1628 if ( norm_2( DeltaXi ) < MaxTolerancePointLocalCoordinates )
1629 break;
1630 }
1631
1632 return rResult;
