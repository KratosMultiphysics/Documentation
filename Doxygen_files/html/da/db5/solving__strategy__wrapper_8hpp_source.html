108 void AccumulateTotalDisplacementField() override
109 {
110 if (mResetDisplacements) {
111 KRATOS_ERROR_IF_NOT(mrModelPart.Nodes().size() == mOldTotalDisplacements.size())
112 << "The number of old displacements (" << mOldTotalDisplacements.size()
113 << ") does not match the current number of nodes (" << mrModelPart.Nodes().size() << ").";
114 std::size_t count = 0;
115 for (auto& node : mrModelPart.Nodes()) {
116 node.GetSolutionStepValue(TOTAL_DISPLACEMENT) =
117 mOldTotalDisplacements[count] + node.GetSolutionStepValue(DISPLACEMENT);
118 ++count;
119 }
120 }
121 }

123 void ComputeIncrementalDisplacementField() override

126 node.GetSolutionStepValue(INCREMENTAL_DISPLACEMENT) =
127 node.GetSolutionStepValue(DISPLACEMENT, 0) - node.GetSolutionStepValue(DISPLACEMENT, 1);
128 }
129 }

157 {
158 if (!mrModelPart.HasNodalSolutionStepVariable(rVariable)) return;
159
160 for (auto& node : mrModelPart.Nodes()) {
