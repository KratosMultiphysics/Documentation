66 template<class NeighbourClassType>
67 void CalculateTangentialForceWithNeighbour(const double normal_contact_force,
68 const double OldLocalElasticContactForce[3],
69 double LocalElasticContactForce[3],
70 const double LocalDeltDisp[3],
71 bool& sliding,
72 SphericParticle* const element,
73 NeighbourClassType* const neighbour) {
74 // Compute shear force
75 LocalElasticContactForce[0] = OldLocalElasticContactForce[0] - mKt * LocalDeltDisp[0];
76 LocalElasticContactForce[1] = OldLocalElasticContactForce[1] - mKt * LocalDeltDisp[1];
77 const double tangent_contact_force = sqrt(LocalElasticContactForce[0] * LocalElasticContactForce[0] + LocalElasticContactForce[1] * LocalElasticContactForce[1]);
78
79 // Compute maximum admisible shear force
80 Properties& properties_of_this_contact = element->GetProperties().GetSubProperties(neighbour->GetProperties().Id());
81 const double friction_angle_tg = std::tan(properties_of_this_contact[STATIC_FRICTION]);
82 const double MaximumAdmisibleShearForce = normal_contact_force * friction_angle_tg;
83
84 // Check for sliding: apply Coulomb friction condition
85 if (tangent_contact_force > MaximumAdmisibleShearForce) {
86 sliding = true;
87 const double fraction = MaximumAdmisibleShearForce / tangent_contact_force;
88 LocalElasticContactForce[0] *= fraction;
89 LocalElasticContactForce[1] *= fraction;
90 }
91 }
