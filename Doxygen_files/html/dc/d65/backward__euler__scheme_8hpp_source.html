33 inline void SetTimeFactors(ModelPart& rModelPart) override
34 {
35 KRATOS_TRY
36
37 GeoMechanicsTimeIntegrationScheme<TSparseSpace, TDenseSpace>::SetTimeFactors(rModelPart);
38
39 for (const auto& r_first_order_scalar_variable : this->GetFirstOrderScalarVariables()) {
40 rModelPart.GetProcessInfo()[r_first_order_scalar_variable.delta_time_coefficient] =
41 1.0 / this->GetDeltaTime();

51 inline void UpdateVariablesDerivatives(ModelPart& rModelPart) override
52 {
53 KRATOS_TRY
54
55 block_for_each(rModelPart.Nodes(), [this](Node& rNode) {
56 UpdateVectorTimeDerivatives(rNode);
57
58 for (const auto& r_first_order_scalar_variable : this->GetFirstOrderScalarVariables()) {
59 if (rNode.IsFixed(r_first_order_scalar_variable.first_time_derivative)) continue;
60
61 rNode.FastGetSolutionStepValue(r_first_order_scalar_variable.first_time_derivative) =
62 CalculateDerivative(r_first_order_scalar_variable.instance, rNode);
63 }
64 });
65
66 KRATOS_CATCH("")
67 }
68
69 void UpdateVectorTimeDerivatives(Node& rNode) const
70 {
71 for (const auto& r_second_order_vector_variable : this->GetSecondOrderVectorVariables()) {
72 if (!rNode.SolutionStepsDataHas(r_second_order_vector_variable.instance)) continue;
73
74 const auto updated_first_time_derivative =
75 CalculateDerivative(r_second_order_vector_variable.instance, rNode);
76
77 NodeUtilities::AssignUpdatedVectorVariableToNonFixedComponents(
78 rNode, r_second_order_vector_variable.first_time_derivative, updated_first_time_derivative);
79
80 // Make sure that setting the second_time_derivative is done
81 // after setting the first_time_derivative.
82 const auto updated_second_time_derivative =
83 CalculateDerivative(r_second_order_vector_variable.first_time_derivative, rNode);
84
85 NodeUtilities::AssignUpdatedVectorVariableToNonFixedComponents(
86 rNode, r_second_order_vector_variable.second_time_derivative, updated_second_time_derivative);
87 }
88 }

90 template <class T>
91 T CalculateDerivative(const Variable<T>& rInstanceVariable, Node& rNode) const
92 {
93 return (rNode.FastGetSolutionStepValue(rInstanceVariable, 0) -
94 rNode.FastGetSolutionStepValue(rInstanceVariable, 1)) /
95 this->GetDeltaTime();
96 }
97};
