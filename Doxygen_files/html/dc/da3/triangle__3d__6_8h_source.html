527 const CoordinatesArrayType& rPoint,

529 const double Tolerance = std::numeric_limits<double>::epsilon()
530 )const override
531 {
532 this->PointLocalCoordinates( rResult, rPoint );
533
534 if ( (rResult[0] >= (0.0-Tolerance)) && (rResult[0] <= (1.0+Tolerance)) )
535 {
536 if ( (rResult[1] >= (0.0-Tolerance)) && (rResult[1] <= (1.0+Tolerance)) )
537 {
538 if ( (rResult[0] + rResult[1]) <= (1.0+Tolerance) )
539 {
540 return true;

564 double ShapeFunctionValue( IndexType ShapeFunctionIndex,
565 const CoordinatesArrayType& rPoint ) const override
566 {
567
568 double thirdCoord = 1 - rPoint[0] - rPoint[1];
569
570 switch ( ShapeFunctionIndex )
571 {
572 case 0:
573 return( thirdCoord*( 2*thirdCoord - 1 ) );
574 case 1:
575 return( rPoint[0]*( 2*rPoint[0] - 1 ) );
576 case 2:
577 return( rPoint[1]*( 2*rPoint[1] - 1 ) );
578 case 3:
579 return( 4*thirdCoord*rPoint[0] );
580 case 4:
581 return( 4*rPoint[0]*rPoint[1] );
582 case 5:
583 return( 4*rPoint[1]*thirdCoord );
584
585 default:
586 KRATOS_ERROR << "Wrong index of shape function!" << *this << std::endl;
587 }
588
589 return 0;
590 }

761 Matrix& ShapeFunctionsLocalGradients( Matrix& rResult,
762 const CoordinatesArrayType& rPoint ) const override
763 {
764 rResult.resize( 6, 2,false );
765 double thirdCoord = 1 - rPoint[0] - rPoint[1];
766 double thirdCoord_DX = -1;
767 double thirdCoord_DY = -1;
768
769 noalias( rResult ) = ZeroMatrix( 6, 2 );
770 rResult( 0, 0 ) = ( 4 * thirdCoord - 1 ) * thirdCoord_DX;
771 rResult( 0, 1 ) = ( 4 * thirdCoord - 1 ) * thirdCoord_DY;
772 rResult( 1, 0 ) = 4 * rPoint[0] - 1;
773 rResult( 1, 1 ) = 0;
774 rResult( 2, 0 ) = 0;
775 rResult( 2, 1 ) = 4 * rPoint[1] - 1;
776 rResult( 3, 0 ) = 4 * thirdCoord_DX * rPoint[0] + 4 * thirdCoord;
777 rResult( 3, 1 ) = 4 * thirdCoord_DY * rPoint[0];
778 rResult( 4, 0 ) = 4 * rPoint[1];
779 rResult( 4, 1 ) = 4 * rPoint[0];
780 rResult( 5, 0 ) = 4 * rPoint[1] * thirdCoord_DX;
781 rResult( 5, 1 ) = 4 * rPoint[1] * thirdCoord_DY + 4 * thirdCoord;
782 return rResult;
783 }

997 static Matrix CalculateShapeFunctionsIntegrationPointsValues(
998 typename BaseType::IntegrationMethod ThisMethod )
999 {
1000 IntegrationPointsContainerType all_integration_points =
1001 AllIntegrationPoints();
1002 IntegrationPointsArrayType integration_points = all_integration_points[static_cast<int>(ThisMethod)];
1003 //number of integration points
1004 const int integration_points_number = integration_points.size();
1005 //number of nodes in current geometry
1006 const int points_number = 6;
1007 //setting up return matrix
1008 Matrix shape_function_values( integration_points_number, points_number );
1009 //loop over all integration points
1010
1011 for ( int pnt = 0; pnt < integration_points_number; pnt++ )
1012 {
1013 double thirdCoord = 1 - integration_points[pnt].X() - integration_points[pnt].Y();
1014
1015 shape_function_values( pnt, 0 ) = thirdCoord * ( 2 * thirdCoord - 1 ) ;
1016 shape_function_values( pnt, 1 ) = integration_points[pnt].X() * ( 2 * integration_points[pnt].X() - 1 ) ;
1017 shape_function_values( pnt, 2 ) = integration_points[pnt].Y() * ( 2 * integration_points[pnt].Y() - 1 ) ;
1018 shape_function_values( pnt, 3 ) = 4 * thirdCoord * integration_points[pnt].X();
1019 shape_function_values( pnt, 4 ) = 4 * integration_points[pnt].X() * integration_points[pnt].Y();
1020 shape_function_values( pnt, 5 ) = 4 * integration_points[pnt].Y() * thirdCoord;
1021
1022 }
1023
1024 return shape_function_values;
