77 ModelPart& rModelPart,

79 TSystemVectorType& Dx,
80 TSystemVectorType& b) override
81 {
82 KRATOS_TRY
83
84 unsigned int Dim = rModelPart.GetProcessInfo()[DOMAIN_SIZE];
85
86 // Clear nodal variables
87 #pragma omp parallel
88 {
89 ModelPart::NodeIterator NodesBegin;
90 ModelPart::NodeIterator NodesEnd;
91 OpenMPUtils::PartitionedIterators(rModelPart.Nodes(),NodesBegin,NodesEnd);
92
93 for (ModelPart::NodeIterator itNode = NodesBegin; itNode != NodesEnd; ++itNode)
94 {
95 itNode->FastGetSolutionStepValue(NODAL_AREA) = 0.0;
96 Matrix& rNodalStress = itNode->FastGetSolutionStepValue(NODAL_CAUCHY_STRESS_TENSOR);
97 if(rNodalStress.size1() != Dim)
98 rNodalStress.resize(Dim,Dim,false);
99 noalias(rNodalStress) = ZeroMatrix(Dim,Dim);
100 itNode->FastGetSolutionStepValue(NODAL_JOINT_AREA) = 0.0;
101 itNode->FastGetSolutionStepValue(NODAL_JOINT_WIDTH) = 0.0;
102 }
103 }
104
105 BaseType::FinalizeSolutionStep(rModelPart,A,Dx,b);
106
107 // Compute smoothed nodal variables
108 #pragma omp parallel
109 {
110 ModelPart::NodeIterator NodesBegin;
111 ModelPart::NodeIterator NodesEnd;
112 OpenMPUtils::PartitionedIterators(rModelPart.Nodes(),NodesBegin,NodesEnd);
113
114 for (ModelPart::NodeIterator itNode = NodesBegin; itNode != NodesEnd; ++itNode)
115 {
116 const double& NodalArea = itNode->FastGetSolutionStepValue(NODAL_AREA);
117 if (NodalArea>1.0e-15)
118 {
119 const double InvNodalArea = 1.0/(NodalArea);
120 Matrix& rNodalStress = itNode->FastGetSolutionStepValue(NODAL_CAUCHY_STRESS_TENSOR);
121 for(unsigned int i = 0; i<Dim; i++)
122 {
123 for(unsigned int j = 0; j<Dim; j++)
124 {
125 rNodalStress(i,j) *= InvNodalArea;
126 }
127 }
128 }
129
130 const double& NodalJointArea = itNode->FastGetSolutionStepValue(NODAL_JOINT_AREA);
131 if (NodalJointArea>1.0e-15)
132 {
133 double& NodalJointWidth = itNode->FastGetSolutionStepValue(NODAL_JOINT_WIDTH);
134 NodalJointWidth = NodalJointWidth/NodalJointArea;
135 }
136 }
137 }
138
139 KRATOS_CATCH("")
140 }
