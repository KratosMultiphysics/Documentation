174 {

190
191 // Preparing the time values for the first step (where time = initial_time +
192 // dt)
193 mTime.Current = r_current_process_info[TIME] + r_current_process_info[DELTA_TIME];
194 mTime.Delta = r_current_process_info[DELTA_TIME];
195 mTime.Middle = mTime.Current - 0.5 * mTime.Delta;
196 mTime.Previous = mTime.Current - mTime.Delta;

204 InitializeExplicitScheme(rModelPart, dim);

273 void InitializeExplicitScheme(
274 ModelPart& rModelPart,
275 const SizeType DomainSize = 3
276 )
277 {
278 KRATOS_TRY
279
281 NodesArrayType& r_nodes = rModelPart.Nodes();
282
283 if (!r_nodes.empty()) {
284 // The first iterator of the array of nodes
285 const auto it_node_begin = rModelPart.NodesBegin();
286
287 // Construct the node initializer lambda based on whether rot_z exists
288 std::function<void(Node&)> initializer_base, initializer;
289 const array_1d<double, 3> zero_array = ZeroVector(3);
290
291 initializer_base = [&zero_array, DomainSize](Node& rNode){
292 rNode.SetValue(NODAL_MASS, 0.0);
293 rNode.FastGetSolutionStepValue(MIDDLE_VELOCITY) = zero_array;
294
295 array_1d<double, 3>& r_middle_velocity = rNode.FastGetSolutionStepValue(MIDDLE_VELOCITY);
296 const array_1d<double, 3>& r_current_velocity = rNode.FastGetSolutionStepValue(VELOCITY);
297 array_1d<double, 3>& r_current_residual = rNode.FastGetSolutionStepValue(FORCE_RESIDUAL);
298 //array_1d<double,3>& r_current_displacement = rNode.FastGetSolutionStepValue(DISPLACEMENT);
299
300 for (IndexType j = 0; j < DomainSize; j++) {
301 r_middle_velocity[j] = r_current_velocity[j];
302 r_current_residual[j] = 0.0;
303 //r_current_displacement[j] = 0.0; // this might be wrong for presribed displacement // NOTE: then you should check if the dof is fixed
304 }
305 };
306
307 const bool has_dof_for_rot_z = it_node_begin->HasDofFor(ROTATION_Z);
308 if (has_dof_for_rot_z) {
309 initializer = [&zero_array, DomainSize, &initializer_base](Node& rNode){
310 initializer_base(rNode);
311 rNode.SetValue(NODAL_INERTIA, zero_array);
312 rNode.FastGetSolutionStepValue(MIDDLE_ANGULAR_VELOCITY) = zero_array;
313
314 array_1d<double, 3>& r_middle_angular_velocity = rNode.FastGetSolutionStepValue(MIDDLE_ANGULAR_VELOCITY);
315 const array_1d<double, 3>& r_current_angular_velocity = rNode.FastGetSolutionStepValue(ANGULAR_VELOCITY);
316 array_1d<double, 3>& r_current_residual_moment = rNode.FastGetSolutionStepValue(MOMENT_RESIDUAL);
317 //array_1d<double,3>& current_rotation = rNode.FastGetSolutionStepValue(ROTATION);
318
319 const IndexType initial_j = DomainSize == 3 ? 0 : 2; // We do this because in 2D only the rotation Z is needed, then we start with 2, instead of 0
320 for (IndexType j = initial_j; j < 3; j++) {
321 r_middle_angular_velocity[j] = r_current_angular_velocity[j];
322 r_current_residual_moment[j] = 0.0;
323 // current_rotation[j] = 0.0; // this might be wrong for presribed rotations // NOTE: then you should check if the dof is fixed
324 }
325 };
326 } else {
327 initializer = initializer_base;
328 }
329
331 block_for_each(r_nodes, initializer);
332 } // if not nodes.empty()
333
334 KRATOS_CATCH("")
335 }

345 void Update(
346 ModelPart& rModelPart,
347 DofsArrayType& rDofSet,
348 TSystemMatrixType& rA,
349 TSystemVectorType& rDx,
350 TSystemVectorType& rb
351 ) override
352 {
353 KRATOS_TRY
354 // The array of nodes
355 NodesArrayType& r_nodes = rModelPart.Nodes();
356
357 if (!r_nodes.empty()) {
358 // The current process info
359 const ProcessInfo& r_current_process_info = rModelPart.GetProcessInfo();
360
362 const SizeType dim = r_current_process_info[DOMAIN_SIZE];
363
364 // Step Update
365 // The first step is time = initial_time ( 0.0) + delta time
366 mTime.Current = r_current_process_info[TIME];
367 mTime.Delta = r_current_process_info[DELTA_TIME];
368
369 mTime.Middle = mTime.Current - 0.50*mTime.Delta;
370 mTime.Previous = mTime.Current - 1.00*mTime.Delta;
371 mTime.PreviousMiddle = mTime.Middle - 1.00*mTime.Delta;
372
373 if (mTime.Previous<0.0) mTime.Previous=0.00;
374 if (mTime.PreviousMiddle<0.0) mTime.PreviousMiddle=0.00;
375 // The iterator of the first node
376 const auto it_node_begin = rModelPart.NodesBegin();
377 const bool has_dof_for_rot_z = it_node_begin->HasDofFor(ROTATION_Z);
378
379 // Getting dof position
380 const IndexType disppos = it_node_begin->GetDofPosition(DISPLACEMENT_X);
381 const IndexType rotppos = has_dof_for_rot_z ? it_node_begin->GetDofPosition(ROTATION_X) : 0;
382
383 // Construct loop lambda
384 std::function<void(SizeType)> updater_base, updater;
385 updater_base = [it_node_begin, dim, &disppos, this](SizeType index){
386 this->UpdateTranslationalDegreesOfFreedom(it_node_begin + index, disppos, dim);
387 };
388
389 if (has_dof_for_rot_z) {
390 updater = [it_node_begin, dim, &rotppos, this, &updater_base](SizeType index){
391 updater_base(index);
392 this->UpdateRotationalDegreesOfFreedom(it_node_begin + index, rotppos, dim);
393 };
394 } else {
395 updater = updater_base;
396 }
397
398 // Update nodes
399 IndexPartition<SizeType>(r_nodes.size()).for_each(updater);
400 } // if not nodes.empty()
401
402 KRATOS_CATCH("")
403 }

411 void UpdateTranslationalDegreesOfFreedom(
412 NodeIterator itCurrentNode,
413 const IndexType DisplacementPosition,
414 const SizeType DomainSize = 3
415 )
416 {
417
418 const double nodal_mass = itCurrentNode->GetValue(NODAL_MASS);
419 const double nodal_displacement_damping = itCurrentNode->GetValue(NODAL_DISPLACEMENT_DAMPING);
420 const array_1d<double, 3>& r_current_residual = itCurrentNode->FastGetSolutionStepValue(FORCE_RESIDUAL);
421
422
423 array_1d<double, 3>& r_current_velocity = itCurrentNode->FastGetSolutionStepValue(VELOCITY);
424 array_1d<double, 3>& r_current_displacement = itCurrentNode->FastGetSolutionStepValue(DISPLACEMENT);
425 array_1d<double, 3>& r_middle_velocity = itCurrentNode->FastGetSolutionStepValue(MIDDLE_VELOCITY);
426
427 array_1d<double, 3>& r_current_acceleration = itCurrentNode->FastGetSolutionStepValue(ACCELERATION);
428
429 const array_1d<double, 3>& r_previous_displacement = itCurrentNode->FastGetSolutionStepValue(DISPLACEMENT, 1);
430 const array_1d<double, 3>& r_previous_middle_velocity = itCurrentNode->FastGetSolutionStepValue(MIDDLE_VELOCITY, 1);
431 // Solution of the explicit equation:
432 if (nodal_mass > numerical_limit)
433 noalias(r_current_acceleration) = (r_current_residual - nodal_displacement_damping * r_current_velocity) / nodal_mass;
434 else
435 noalias(r_current_acceleration) = ZeroVector(3);
436
437 std::array<bool, 3> fix_displacements = {false, false, false};
438
439 fix_displacements[0] = (itCurrentNode->GetDof(DISPLACEMENT_X, DisplacementPosition).IsFixed());
440 fix_displacements[1] = (itCurrentNode->GetDof(DISPLACEMENT_Y, DisplacementPosition + 1).IsFixed());
441 if (DomainSize == 3)
442 fix_displacements[2] = (itCurrentNode->GetDof(DISPLACEMENT_Z, DisplacementPosition + 2).IsFixed());
443
444 for (IndexType j = 0; j < DomainSize; j++) {
445 if (fix_displacements[j]) {
446 r_current_acceleration[j] = 0.0;
447 r_middle_velocity[j] = 0.0;
448 }
449
450 r_current_velocity[j] = r_previous_middle_velocity[j] + (mTime.Previous - mTime.PreviousMiddle) * r_current_acceleration[j]; //+ actual_velocity;
451 r_middle_velocity[j] = r_current_velocity[j] + (mTime.Middle - mTime.Previous) * r_current_acceleration[j];
452 r_current_displacement[j] = r_previous_displacement[j] + mTime.Delta * r_middle_velocity[j];
453 } // for DomainSize
454 }

462 void UpdateRotationalDegreesOfFreedom(
463 NodeIterator itCurrentNode,
464 const IndexType RotationPosition,
465 const SizeType DomainSize = 3
466 )
467 {
469 const array_1d<double, 3>& nodal_inertia = itCurrentNode->GetValue(NODAL_INERTIA);
470 const array_1d<double, 3>& nodal_rotational_damping = itCurrentNode->GetValue(NODAL_ROTATION_DAMPING);
471 const array_1d<double, 3>& r_current_residual_moment = itCurrentNode->FastGetSolutionStepValue(MOMENT_RESIDUAL);
472 array_1d<double, 3>& r_current_angular_velocity = itCurrentNode->FastGetSolutionStepValue(ANGULAR_VELOCITY);
473 array_1d<double, 3>& r_current_rotation = itCurrentNode->FastGetSolutionStepValue(ROTATION);
474 array_1d<double, 3>& r_middle_angular_velocity = itCurrentNode->FastGetSolutionStepValue(MIDDLE_ANGULAR_VELOCITY);
475 array_1d<double, 3>& r_current_angular_acceleration = itCurrentNode->FastGetSolutionStepValue(ANGULAR_ACCELERATION);
476
477
478 const array_1d<double, 3>& r_previous_rotation = itCurrentNode->FastGetSolutionStepValue(ROTATION, 1);
479 const array_1d<double, 3>& r_previous_middle_angular_velocity = itCurrentNode->FastGetSolutionStepValue(MIDDLE_ANGULAR_VELOCITY, 1);
480
481 const IndexType initial_k = DomainSize == 3 ? 0 : 2; // We do this because in 2D only the rotation Z is needed, then we start with 2, instead of 0
482 for (IndexType kk = initial_k; kk < 3; ++kk) {
483 if (nodal_inertia[kk] > numerical_limit)
484 r_current_angular_acceleration[kk] = (r_current_residual_moment[kk] - nodal_rotational_damping[kk] * r_current_angular_velocity[kk]) / nodal_inertia[kk];
485 else
486 r_current_angular_acceleration[kk] = 0.0;
487 }
488
489 std::array<bool, 3> fix_rotation = {false, false, false};
490 if (DomainSize == 3) {
491 fix_rotation[0] = (itCurrentNode->GetDof(ROTATION_X, RotationPosition).IsFixed());
492 fix_rotation[1] = (itCurrentNode->GetDof(ROTATION_Y, RotationPosition + 1).IsFixed());
493 }
494 fix_rotation[2] = (itCurrentNode->GetDof(ROTATION_Z, RotationPosition + 2).IsFixed());
495
496 for (IndexType j = initial_k; j < 3; j++) {
497 if (fix_rotation[j]) {
498 r_current_angular_acceleration[j] = 0.0;
499 r_middle_angular_velocity[j] = 0.0;
500 }
501 r_current_angular_velocity[j] = r_previous_middle_angular_velocity[j] + (mTime.Previous - mTime.PreviousMiddle) * r_current_angular_acceleration[j];
502 r_middle_angular_velocity[j] = r_current_angular_velocity[j] + (mTime.Middle - mTime.Previous) * r_current_angular_acceleration[j];
503 r_current_rotation[j] = r_previous_rotation[j] + mTime.Delta * r_middle_angular_velocity[j];
504 }

604 r_current_angular_acceleration[j] = 0.0;
605 r_middle_angular_velocity[j] = 0.0;
606 }
607

613 } else {
614 initializer = initializer_base;
615 }
616
617 // Initialize all nodes
618 block_for_each(r_nodes, initializer);

620

623 KRATOS_CATCH("")

668 void CalculateAndAddRHS(ModelPart& rModelPart)
669 {
670 KRATOS_TRY
671
672 const ProcessInfo& r_current_process_info = rModelPart.GetProcessInfo();
673 ConditionsArrayType& r_conditions = rModelPart.Conditions();

684 r_current_process_info);
