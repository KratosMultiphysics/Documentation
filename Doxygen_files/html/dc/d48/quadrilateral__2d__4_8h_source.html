606 double ShapeFunctionValue( IndexType ShapeFunctionIndex,
607 const CoordinatesArrayType& rPoint ) const override
608 {
609 switch ( ShapeFunctionIndex )
610 {
611 case 0:
612 return( 0.25*( 1.0 - rPoint[0] )*( 1.0 - rPoint[1] ) );
613 case 1:
614 return( 0.25*( 1.0 + rPoint[0] )*( 1.0 - rPoint[1] ) );
615 case 2:
616 return( 0.25*( 1.0 + rPoint[0] )*( 1.0 + rPoint[1] ) );
617 case 3:
618 return( 0.25*( 1.0 - rPoint[0] )*( 1.0 + rPoint[1] ) );
619 default:
620 KRATOS_ERROR << "Wrong index of shape function!" << *this << std::endl;
621 }
622
623 return 0;
624 }

956 static Matrix CalculateShapeFunctionsIntegrationPointsValues(
957 typename BaseType::IntegrationMethod ThisMethod )
958 {
959 IntegrationPointsContainerType all_integration_points =
960 AllIntegrationPoints();
961 IntegrationPointsArrayType integration_points = all_integration_points[static_cast<int>(ThisMethod)];
962 //number of integration points

964 //number of nodes in current geometry
965 const int points_number = 4;
966 //setting up return matrix
967 Matrix shape_function_values( integration_points_number, points_number );
968 //loop over all integration points
969
970 for ( int pnt = 0; pnt < integration_points_number; pnt++ )
971 {
972 shape_function_values( pnt, 0 ) =
973 0.25 * ( 1.0 - integration_points[pnt].X() )
974 * ( 1.0 - integration_points[pnt].Y() );
975 shape_function_values( pnt, 1 ) =
976 0.25 * ( 1.0 + integration_points[pnt].X() )
977 * ( 1.0 - integration_points[pnt].Y() );

979 0.25 * ( 1.0 + integration_points[pnt].X() )
980 * ( 1.0 + integration_points[pnt].Y() );

982 0.25 * ( 1.0 - integration_points[pnt].X() )
983 * ( 1.0 + integration_points[pnt].Y() );
