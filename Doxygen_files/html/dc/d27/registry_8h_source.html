88 template<typename TItemType, class... TArgumentsList >
89 static RegistryItem& AddItem(
90 std::string const& rItemFullName,
91 TArgumentsList&&... Arguments)
92 {
93 KRATOS_TRY
94
95 const std::lock_guard<LockObject> scope_lock(ParallelUtilities::GetGlobalLock());
96
97 auto item_path = StringUtilities::SplitStringByDelimiter(rItemFullName, '.');
98 KRATOS_ERROR_IF(item_path.empty()) << "The item full name is empty" << std::endl;
99
100 RegistryItem* p_current_item = &GetRootRegistryItem();
101
102 for(std::size_t i = 0 ; i < item_path.size() - 1 ; i++) {
103 auto& r_item_name = item_path[i];
104 if(p_current_item->HasItem(r_item_name)){
105 p_current_item = &p_current_item->GetItem(r_item_name);
106 } else {
107 p_current_item = &p_current_item->AddItem<RegistryItem>(r_item_name);
108 }
109 }
110
111 // I am doing the last one out of the loop to create it with the given type and argument
112 auto& r_item_name = item_path.back();
113 if (p_current_item->HasItem(r_item_name)) {
114 KRATOS_ERROR << "The item \"" << rItemFullName << "\" is already registered." << std::endl;
115 } else {
116 p_current_item = &p_current_item->AddItem<TItemType>(r_item_name, std::forward<TArgumentsList>(Arguments)...);
117 }
118
119 return *p_current_item;
120
121 KRATOS_CATCH("")
122 }
