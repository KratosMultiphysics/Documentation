555 double ShapeFunctionValue( IndexType ShapeFunctionIndex,
556 const CoordinatesArrayType& rPoint ) const override
557 {
558 double fx1 = 0.5 * ( rPoint[0] - 1 ) * rPoint[0];
559 double fx2 = 0.5 * ( rPoint[0] + 1 ) * rPoint[0];
560 double fx3 = 1 - rPoint[0] * rPoint[0];
561 double fy1 = 0.5 * ( rPoint[1] - 1 ) * rPoint[1];
562 double fy2 = 0.5 * ( rPoint[1] + 1 ) * rPoint[1];
563 double fy3 = 1 - rPoint[1] * rPoint[1];
564
565 switch ( ShapeFunctionIndex )
566 {
567 case 0:
568 return( fx1*fy1 );
569 case 1:
570 return( fx2*fy1 );
571 case 2:
572 return( fx2*fy2 );
573 case 3:
574 return( fx1*fy2 );
575 case 4:
576 return( fx3*fy1 );
577 case 5:
578 return( fx2*fy3 );
579 case 6:
580 return( fx3*fy2 );
581 case 7:
582 return( fx1*fy3 );
583 case 8:
584 return( fx3*fy3 );
585 default:
586 KRATOS_ERROR << "Wrong index of shape function!" << *this << std::endl;
587 }
588
589 return 0;
590 }
