187#define KRATOS_EXPECT_EXCEPTION_IS_THROWN(TheStatement, TheErrorMessage) { \
188 EXPECT_THROW({ \
189 try { \
190 TheStatement; \
191 KRATOS_ERROR << #TheStatement << " exited without throwing an error." << std::endl; \
192 } catch( const Kratos::Exception& e ) { \
193 EXPECT_THAT(e.what(), HasSubstr(TheErrorMessage)) \
194 << "Test Failed: " << #TheStatement \
195 << " did not throw the expected error." << std::endl \
196 << "Expected:" << std::endl << TheErrorMessage << std::endl \
197 << "Got:" << std::endl << e.what() << std::endl; \
198 throw; \
199 } \
200 }, Kratos::Exception); \
201}
