265 void EraseCriticalNodes2D(Element::GeometryType &eElement, SizeType &inside_nodes_removed, SizeType &nodes_removed_inlet_zone)
266 {
267
268 KRATOS_TRY
269
270 double safetyCoefficient2D = 0.5;
271 double elementVolume = eElement.Area();
272
273 const ProcessInfo &rCurrentProcessInfo = mrModelPart.GetProcessInfo();
274 const SizeType principalModelPartId = rCurrentProcessInfo[MAIN_MATERIAL_PROPERTY];
275
276 array_1d<double, 3> Edges(3, 0.0);
277 array_1d<SizeType, 3> FirstEdgeNode(3, 0);
278 array_1d<SizeType, 3> SecondEdgeNode(3, 0);
279 double wallLength = 0;
280 array_1d<double, 3> CoorDifference = eElement[1].Coordinates() - eElement[0].Coordinates();
281 double SquaredLength = CoorDifference[0] * CoorDifference[0] + CoorDifference[1] * CoorDifference[1];
282 Edges[0] = std::sqrt(SquaredLength);
283 FirstEdgeNode[0] = 0;
284 SecondEdgeNode[0] = 1;
285 if (eElement[0].Is(RIGID) && eElement[1].Is(RIGID))
286 {
287 wallLength = Edges[0];
288 }
289 bool inletElement = false;
290 if (eElement[0].Is(PFEMFlags::EULERIAN_INLET) || eElement[1].Is(PFEMFlags::EULERIAN_INLET) || eElement[2].Is(PFEMFlags::EULERIAN_INLET))
291 {
292 inletElement = true;
293 }
294 SizeType counter = 0;
295 for (SizeType i = 2; i < eElement.size(); i++)
296 {
297 for (SizeType j = 0; j < i; j++)
298 {
299 noalias(CoorDifference) = eElement[i].Coordinates() - eElement[j].Coordinates();
300 SquaredLength = CoorDifference[0] * CoorDifference[0] + CoorDifference[1] * CoorDifference[1];
301 counter += 1;
302 Edges[counter] = std::sqrt(SquaredLength);
303 FirstEdgeNode[counter] = j;
304 SecondEdgeNode[counter] = i;
305 if (eElement[i].Is(RIGID) && eElement[j].Is(RIGID) && Edges[counter] > wallLength)
306 {
307 wallLength = Edges[counter];
308 }
309 }
310 }
311
313 for (SizeType i = 0; i < eElement.size(); i++)
314 {
315 if (eElement[i].IsNot(RIGID) && eElement[i].IsNot(TO_ERASE) && eElement[i].IsNot(SOLID) && eElement[i].IsNot(ISOLATED))
316 {
317 double height = elementVolume * 2.0 / wallLength;
318
320 if (eElement[i].Is(FREE_SURFACE))
321 {
322 NodeWeakPtrVectorType &neighb_nodes = eElement[i].GetValue(NEIGHBOUR_NODES);
323 SizeType countRigid = 0;
324 SizeType countFreeSurface = 0;
325 for (NodeWeakPtrVectorType::iterator nn = neighb_nodes.begin(); nn != neighb_nodes.end(); nn++)
326 {
327 if ((nn)->Is(RIGID) || (nn)->Is(SOLID))
328 {
329 countRigid++;
330 }
331 if ((nn)->Is(FREE_SURFACE) && (nn)->IsNot(RIGID) && (nn)->IsNot(SOLID))
332 {
333 countFreeSurface++;
334 }
335 }
336 if ((countRigid + countFreeSurface) == neighb_nodes.size() && countRigid > 0)
337 {
338 safetyCoefficient2D = 0.25;
339 }
340 }
341
343 if (height < (0.5 * safetyCoefficient2D * wallLength))
344 {
345 eElement[i].Set(TO_ERASE);
346 inside_nodes_removed++;
347
348 if (inletElement)
349 {
350 nodes_removed_inlet_zone++;
351 }
352
353 const SizeType propertyIdNode = eElement[i].FastGetSolutionStepValue(PROPERTY_ID);
354 if (propertyIdNode != principalModelPartId) // this is to conserve the number of nodes of the smaller domain in case of a two-fluid analysis
355 {
356 mrRemesh.Info->BalancePrincipalSecondaryPartsNodes += -1;
357 }
358 }
359 }
360 }
361
362 bool longDamBreak = false; // to attivate in case of long dam breaks to avoid separeted elements in the water front
363 if (longDamBreak)
364 ControlForLongDamBreakProblems2D(eElement, inletElement, inside_nodes_removed, nodes_removed_inlet_zone);

366 KRATOS_CATCH("")

865 double finalTime = mrRemesh.RefiningBoxFinalTime[index];
866 bool refiningBox = mrRemesh.UseRefiningBox[index];
