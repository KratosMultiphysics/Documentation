61 const double d = inner_prod(mNormalVector, direction);
62 distance = -(distance - mEqRHS) / d;
63 const double pressure = mSpecificWeight * (distance + deltaH);
64 rNode.FastGetSolutionStepValue(var) = std::max(pressure, 0.0);
65 });
