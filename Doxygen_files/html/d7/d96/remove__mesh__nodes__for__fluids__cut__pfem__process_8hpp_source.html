193 bool RemoveNodesOnDistance(SizeType &inside_nodes_removed,
194 SizeType &boundary_nodes_removed,
195 SizeType &nodes_removed_inlet_zone)
196 {
197 KRATOS_TRY
198
199 const SizeType dimension = mrModelPart.ElementsBegin()->GetGeometry().WorkingSpaceDimension();
200
201 bool some_node_is_removed = false;
202
203 // bucket size definition:
204 SizeType bucket_size = 20;
205
206 // create the list of the nodes to be check during the search
207 std::vector<Node::Pointer> list_of_nodes;
208 list_of_nodes.reserve(mrModelPart.NumberOfNodes());
209 for (ModelPart::NodesContainerType::iterator i_node = mrModelPart.NodesBegin(); i_node != mrModelPart.NodesEnd(); i_node++)
210 {
211 (list_of_nodes).push_back(*(i_node.base()));
212 }
213
214 KdtreeType nodes_tree(list_of_nodes.begin(), list_of_nodes.end(), bucket_size);
216 // all of the nodes in this list will be preserved
217 SizeType num_neighbours = 100;
218 std::vector<Node::Pointer> neighbours(num_neighbours);
219 std::vector<double> neighbour_distances(num_neighbours);
220
221 // radius means the distance, if the distance between two nodes is closer to radius -> mark for removing
222 double radius = 0;
223 Node work_point(0, 0.0, 0.0, 0.0);
224 SizeType n_points_in_radius;
225
226 const ProcessInfo &rCurrentProcessInfo = mrModelPart.GetProcessInfo();
227 const double currentTime = rCurrentProcessInfo[TIME];
228 double meshSize = mrRemesh.Refine->CriticalRadius;
229
230 bool refiningBox = false;
231 for (SizeType index = 0; index < mrRemesh.UseRefiningBox.size(); index++)
232 {
233 if (mrRemesh.UseRefiningBox[index] == true && currentTime > mrRemesh.RefiningBoxInitialTime[index] && currentTime < mrRemesh.RefiningBoxFinalTime[index])
234 {
235 refiningBox = true;
236 }
237 }
238
239 const SizeType principalModelPartId = rCurrentProcessInfo[MAIN_MATERIAL_PROPERTY];
240 for (ModelPart::ElementsContainerType::const_iterator ie = mrModelPart.ElementsBegin();
241 ie != mrModelPart.ElementsEnd(); ie++)
242 {
243 SizeType rigidNodes = 0;
244 // coordinates
245 for (SizeType i = 0; i < ie->GetGeometry().size(); i++)
246 {
247 if ((ie->GetGeometry()[i].Is(RIGID) && ie->GetGeometry()[i].IsNot(PFEMFlags::LAGRANGIAN_INLET)) ||
248 ie->GetGeometry()[i].Is(SOLID) ||
249 ie->GetGeometry()[i].Is(PFEMFlags::EULERIAN_INLET) ||
250 ie->GetGeometry()[i].GetSolutionStepValue(DISTANCE) < 0)
251 {
252 rigidNodes++;
253 }
254 }
255 if (dimension == 2)
256 {
257 if (rigidNodes > 0)
258 this->EraseCriticalNodes2D(ie->GetGeometry(), inside_nodes_removed, nodes_removed_inlet_zone);
259 }
260 else if (dimension == 3)
261 {
262 if (rigidNodes > 1)
263 EraseCriticalNodes3D(ie->GetGeometry(), inside_nodes_removed, nodes_removed_inlet_zone, rigidNodes);
264 }
265 }
266
267 for (ModelPart::NodesContainerType::const_iterator in = mrModelPart.NodesBegin(); in != mrModelPart.NodesEnd(); in++)
268 {
269
270 const SizeType propertyIdNode = in->FastGetSolutionStepValue(PROPERTY_ID);
271 meshSize = mrRemesh.Refine->CriticalRadius;
272 double rigidNodeLocalMeshSize = 0;
273 double rigidNodeMeshCounter = 0;
274 NodeWeakPtrVectorType &rN = in->GetValue(NEIGHBOUR_NODES);
275
276 for (SizeType i = 0; i < rN.size(); i++)
277 {
278 if (rN[i].Is(RIGID))
279 {
280 rigidNodeLocalMeshSize += rN[i].FastGetSolutionStepValue(NODAL_H_WALL);
281 rigidNodeMeshCounter += 1.0;
282 }
283 }
284
285 if (refiningBox == true)
286 {
287 array_1d<double, 3> NodeCoordinates = in->Coordinates();
288 if (dimension == 2)
289 {
290 bool insideTransitionZone = false;
291 mMesherUtilities.DefineMeshSizeInTransitionZones2D(mrRemesh, currentTime, NodeCoordinates, meshSize, insideTransitionZone);
292 }
293 else if (dimension == 3)
294 {
295 bool insideTransitionZone = false;
296 mMesherUtilities.DefineMeshSizeInTransitionZones3D(mrRemesh, currentTime, NodeCoordinates, meshSize, insideTransitionZone);
297 }
298 }
299
300 const double distance_tolerance = 0.05 * meshSize;
301
302 this->SetMeshSizeNearBoundaries(meshSize, rigidNodeLocalMeshSize, rigidNodeMeshCounter);
303
304 const double size_for_distance_boundary = 0.6 * meshSize;
305 const double size_for_wall_tip_contact_side = 0.15 * mrRemesh.Refine->CriticalSide;
306
307 if (in->Is(TO_ERASE))
308 {
309 some_node_is_removed = true;
310 }
311 bool on_contact_tip = false;
312
313 if (in->Is(TO_SPLIT) || in->Is(CONTACT))
314 on_contact_tip = true;
315
316 if (in->IsNot(NEW_ENTITY) && in->IsNot(INLET) && in->IsNot(ISOLATED) && in->IsNot(RIGID) && in->IsNot(SOLID) && (dimension == 2 || (dimension == 3 && in->GetSolutionStepValue(DISTANCE) > 0)))
317 {
318 SizeType neighErasedNodes = 0;
319 SizeType freeSurfaceNeighNodes = 0;
320 bool inletElement = false;
321 bool interfaceElement = false;
322
323 radius = 0.6 * meshSize;
324 work_point[0] = in->X();
325 work_point[1] = in->Y();
326 work_point[2] = in->Z();
327
328 if (in->Is(FREE_SURFACE))
329 {
330 // it must be more difficult to erase a free_surface node, otherwise, lot of volume is lost
331 // this value has a strong effect on volume variation due to remeshing
332 radius = 0.475 * meshSize; // compared with element radius
333 NodeWeakPtrVectorType &neighb_nodes = in->GetValue(NEIGHBOUR_NODES);
334 SizeType countRigid = 0;
335
336 for (NodeWeakPtrVectorType::iterator nn = neighb_nodes.begin(); nn != neighb_nodes.end(); nn++)
337 {
338 if ((nn)->Is(RIGID) || (nn)->Is(SOLID))
339 {
340 countRigid++;
341 }
342 if ((nn)->Is(TO_ERASE))
343 {
344 neighErasedNodes++;
345 }
346 }
347 if (countRigid == neighb_nodes.size())
348 {
349 radius = 0.15 * meshSize;
350 }
351 }
352 else
353 {
354 NodeWeakPtrVectorType &neighb_nodes = in->GetValue(NEIGHBOUR_NODES);
355 for (NodeWeakPtrVectorType::iterator nn = neighb_nodes.begin(); nn != neighb_nodes.end(); nn++)
356 {
357 const SizeType propertyIdSecondNode = (nn)->FastGetSolutionStepValue(PROPERTY_ID);
358 if ((nn)->Is(FREE_SURFACE))
359 {
360 freeSurfaceNeighNodes++;
361 }
362 if ((nn)->Is(TO_ERASE))
363 {
364 neighErasedNodes++;
365 }
366 if ((nn)->Is(PFEMFlags::EULERIAN_INLET))
367 {
368 inletElement = true;
369 }
370 if (propertyIdNode != propertyIdSecondNode && (nn)->IsNot(RIGID))
371 {
372 interfaceElement = true;
373 }
374 }
375 }
376
377 if (freeSurfaceNeighNodes > 1)
378 {
379 radius = 0.5 * meshSize;
380 }
381 else if (interfaceElement == true)
382 {
383 if (dimension == 2)
384 radius = 0.54 * meshSize; // 10% less than normal nodes
385 if (dimension == 3)
386 radius = 0.48 * meshSize; // 20% less than normal nodes
387 }
388 if (in->GetSolutionStepValue(DISTANCE) < distance_tolerance)
389 {
390 if (dimension == 2)
391 radius = 0.2 * meshSize; // 1/3 of normal nodes
392 if (dimension == 3)
393 radius = 0.15 * meshSize; // 1/3 of normal nodes
394 }
395 else if (in->GetSolutionStepValue(DISTANCE) < distance_tolerance * 2.0)
396 {
397 if (dimension == 2)
398 radius = 0.4 * meshSize; // 1/3 of normal nodes
399 if (dimension == 3)
400 radius = 0.3 * meshSize; // 1/3 of normal nodes
401 }
402 n_points_in_radius = nodes_tree.SearchInRadius(work_point, radius, neighbours.begin(), neighbour_distances.begin(), num_neighbours);
403
404 if (n_points_in_radius > 1 && neighErasedNodes == 0 && in->IsNot(INLET))
405 {
406
407 if (in->IsNot(RIGID) && in->IsNot(SOLID) && in->IsNot(ISOLATED))
408 {
409 if (mrRemesh.Refine->RemovingOptions.Is(MesherUtilities::REMOVE_NODES_ON_DISTANCE))
410 {
411
412 if (in->IsNot(FREE_SURFACE) && in->IsNot(RIGID) && freeSurfaceNeighNodes == dimension)
413 {
414 this->SetVariablesToFreeSurfaceElements(in, dimension);
415 }
416 else
417 {
418 in->Set(TO_ERASE);
419 some_node_is_removed = true;
420 inside_nodes_removed++;
421 if (inletElement)
422 {
423 nodes_removed_inlet_zone++;
424 }
425 if (propertyIdNode != principalModelPartId) // this is to conserve the number of nodes of the smaller domain in case of a two-fluid analysis
426 {
427 mrRemesh.Info->BalancePrincipalSecondaryPartsNodes += -1;
428 }
429 }
430 }
431 }
432 else
433 {
434 SizeType k = 0;
435 SizeType counter = 0;
436 for (std::vector<Node::Pointer>::iterator nn = neighbours.begin(); nn != neighbours.begin() + n_points_in_radius; nn++)
437 {
438 bool nn_on_contact_tip = false;
439
440 if ((*nn)->Is(TO_SPLIT) || (*nn)->Is(CONTACT))
441 nn_on_contact_tip = true;
442
443 if ((*nn)->Is(BOUNDARY) && !nn_on_contact_tip && neighbour_distances[k] < size_for_distance_boundary && neighbour_distances[k] > 0.0)
444 {
445 if ((*nn)->IsNot(TO_ERASE))
446 {
447 counter += 1;
448 }
449 }
450
451 if ((*nn)->Is(BOUNDARY) && nn_on_contact_tip && neighbour_distances[k] < size_for_wall_tip_contact_side)
452 {
453 if ((*nn)->IsNot(TO_ERASE))
454 {
455 counter += 1;
456 }
457 }
458 k++;
459 }
460
461 if (counter > 1 && in->IsNot(RIGID) && in->IsNot(SOLID) && in->IsNot(NEW_ENTITY) && !on_contact_tip && (dimension == 2 || (dimension == 3 && in->GetSolutionStepValue(DISTANCE) > distance_tolerance)))
462 { // Can be inserted in the boundary refine
463 in->Set(TO_ERASE);
464 if (mEchoLevel > 1)
465 std::cout << " Removed Boundary Node [" << in->Id() << "] on Distance " << std::endl;
466 some_node_is_removed = true;
467 boundary_nodes_removed++;
468
469 if (propertyIdNode != principalModelPartId) // this is to conserve the number of nodes of the smaller domain in case of a two-fluid analysis
470 {
471 mrRemesh.Info->BalancePrincipalSecondaryPartsNodes += -1;
472 }
473 }
474 }
475 }
476 }
477
478 if (in->Is(ISOLATED) && in->GetSolutionStepValue(DISTANCE) < distance_tolerance && in->IsNot(TO_ERASE))
479 {
480 in->Set(TO_ERASE);
481 some_node_is_removed = true;
482 inside_nodes_removed++;
483 if (propertyIdNode != principalModelPartId) // this is to conserve the number of nodes of the smaller domain in case of a two-fluid analysis
484 {
485 mrRemesh.Info->BalancePrincipalSecondaryPartsNodes += -1;
486 }
487 }
488 }
489
490 if (boundary_nodes_removed > 0 || inside_nodes_removed > 0)
491 {
492 some_node_is_removed = true;
493 }
494 // Build boundary after removing boundary nodes due distance criterion
495 if (mEchoLevel > 1)
496 {
497 std::cout << "boundary_nodes_removed " << boundary_nodes_removed << std::endl;
498 std::cout << "inside_nodes_removed " << inside_nodes_removed << std::endl;
499 }
500 return some_node_is_removed;
501
502 KRATOS_CATCH(" ")
503 }

505 void EraseCriticalNodes3D(Element::GeometryType &eElement, SizeType &inside_nodes_removed, SizeType &nodes_removed_inlet_zone, SizeType rigidNodes)
506 {
507
508 KRATOS_TRY
509 double safetyCoefficient3D = 0.6;
510
511 const ProcessInfo &rCurrentProcessInfo = mrModelPart.GetProcessInfo();
512 SizeType principalModelPartId = rCurrentProcessInfo[MAIN_MATERIAL_PROPERTY];
513
514 SizeType freeSurfaceNodes = 0;
515 SizeType numNodes = eElement.size();
516 double elementVolume = eElement.Volume();
517 double criticalVolume = 0.1 * mrRemesh.Refine->MeanVolume;
518
519 std::vector<array_1d<double, 3>> rigidNodesCoordinates;
520 std::vector<array_1d<double, 3>> rigidNodesNormals;
521 array_1d<double, 3> notRigidNodesCoordinates(3, 0.0);
522 SizeType notRigidNodeId = 0;
523 rigidNodesCoordinates.resize(3);
524 rigidNodesNormals.resize(3);
525
526 this->CriticalVolumeForRefinedBoxes(eElement, criticalVolume, safetyCoefficient3D);
527
528 bool inletElement = false;
529 if (eElement[0].Is(PFEMFlags::EULERIAN_INLET) || eElement[1].Is(PFEMFlags::EULERIAN_INLET) || eElement[2].Is(PFEMFlags::EULERIAN_INLET) || eElement[3].Is(PFEMFlags::EULERIAN_INLET))
530 {
531 inletElement = true;
532 }
533
534 SizeType rigidNode = 0;
535 array_1d<double, 3> WallBaricenter = ZeroVector(3);
536
537 for (SizeType i = 0; i < numNodes; i++)
538 {
539 if (eElement[i].Is(FREE_SURFACE))
540 {
541 freeSurfaceNodes++;
542 }
543 if (rigidNodes == 3)
544 {
545 if (eElement[i].Is(RIGID))
546 {
547 WallBaricenter += eElement[i].Coordinates() / 3.0;
548 rigidNodesCoordinates[rigidNode] = eElement[i].Coordinates();
549 rigidNodesNormals[rigidNode] = eElement[i].FastGetSolutionStepValue(NORMAL);
550 rigidNode++;
551 }
552 else
553 {
554 notRigidNodesCoordinates = eElement[i].Coordinates();
555 notRigidNodeId = i;
556 }
557 }
558 if (elementVolume < criticalVolume)
559 {
560
561 if (eElement[i].IsNot(RIGID) && eElement[i].IsNot(SOLID) && eElement[i].IsNot(TO_ERASE) && eElement[i].GetSolutionStepValue(DISTANCE) > 0)
562 {
563 eElement[i].Set(TO_ERASE);
564 if (mEchoLevel > 1)
565 std::cout << "erase this layer node because it may be potentially dangerous and pass through the solid contour" << std::endl;
566 inside_nodes_removed++;
567
568 if (inletElement)
569 {
570 nodes_removed_inlet_zone++;
571 }
572
573 const SizeType propertyIdNode = eElement[i].FastGetSolutionStepValue(PROPERTY_ID);
574 if (propertyIdNode != principalModelPartId) // this is to conserve the number of nodes of the smaller domain in case of a two-fluid analysis
575 {
576 mrRemesh.Info->BalancePrincipalSecondaryPartsNodes += -1;
577 }
578 }
579 }
580 }
581 double wallNodeDistance = -1;
582 if (rigidNode == 3 && eElement[notRigidNodeId].IsNot(TO_ERASE))
583 {
584 this->CheckSliverElements(eElement,
585 inside_nodes_removed,
586 nodes_removed_inlet_zone,
587 inletElement,
588 notRigidNodeId,
589 principalModelPartId,
590 wallNodeDistance,
591 WallBaricenter,
592 rigidNodesCoordinates,
593 rigidNodesNormals,
594 notRigidNodesCoordinates);
595 }
596
597 array_1d<double, 6> Edges(6, 0.0);
598 array_1d<SizeType, 6> FirstEdgeNode(6, 0);
599 array_1d<SizeType, 6> SecondEdgeNode(6, 0);
600 int removableNode = -1;
601 double wallLength = 0;
602 double minimumLength = 0;
603 this->DetectRemovableNodes(eElement, Edges, FirstEdgeNode, SecondEdgeNode, removableNode, wallLength, minimumLength, safetyCoefficient3D);
604
605 if ((minimumLength < (0.35 * wallLength) || (wallNodeDistance < (0.25 * wallLength) && wallNodeDistance > 0)) && removableNode > -1)
606 {
607 if (eElement[removableNode].IsNot(RIGID) && eElement[removableNode].IsNot(TO_ERASE) && eElement[removableNode].IsNot(SOLID) && eElement[removableNode].IsNot(ISOLATED) && eElement[removableNode].IsNot(FREE_SURFACE))
608 {
609 eElement[removableNode].Set(TO_ERASE);
610 inside_nodes_removed++;
611
612 if (inletElement)
613 {
614 nodes_removed_inlet_zone++;
615 }
616 const SizeType propertyIdNode = eElement[removableNode].FastGetSolutionStepValue(PROPERTY_ID);
617 if (propertyIdNode != principalModelPartId) // this is to conserve the number of nodes of the smaller domain in case of a two-fluid analysis
618 {
619 mrRemesh.Info->BalancePrincipalSecondaryPartsNodes += -1;
620 }
621 }
622 }
623
624 // //////// to compare the non-wall length to wall edge length /////////
625 this->CompareEdgeLengthsNearWalls(eElement, Edges, FirstEdgeNode, SecondEdgeNode, freeSurfaceNodes, safetyCoefficient3D, wallLength, inletElement, inside_nodes_removed, nodes_removed_inlet_zone);
626
627 KRATOS_CATCH("")
628 }
