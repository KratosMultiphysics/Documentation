509 bool IsInside(
510 const CoordinatesArrayType& rPoint,
511 CoordinatesArrayType& rResult,
512 const double Tolerance = std::numeric_limits<double>::epsilon()
513 ) const override
514 {
515 this->PointLocalCoordinates( rResult, rPoint );
516
517 if ( (rResult[0] >= (0.0 - Tolerance)) && (rResult[0] <= (1.0 + Tolerance)) )
518 if ( (rResult[1] >= (0.0 - Tolerance)) && (rResult[1] <= (1.0 + Tolerance)) )
519 if ( (rResult[2] >= (-1.0 - Tolerance)) && (rResult[2] <= (1.0 + Tolerance)) )
520 if ((( 1.0 - ( rResult[0] + rResult[1] ) ) >= (0.0 - Tolerance) ) && (( 1.0 - ( rResult[0] + rResult[1] ) ) <= (1.0 + Tolerance) ) )
521 return true;
522
523 return false;
524 }

798 static double CalculateShapeFunctionValue(
799 const IndexType ShapeFunctionIndex,
800 const CoordinatesArrayType& rPoint
801 )
802 {
803 const double x = rPoint[0];
804 const double y = rPoint[1];
805 const double z = rPoint[2];
806 switch ( ShapeFunctionIndex )
807 {
808 case 0 : return (1.0/2.0)*(2*z - 2.0)*(2*z - 1)*(-2.0*x - 2.0*y + 1.0)*(-x - y + 1.0) ;
809 case 1 : return (1.0/2.0)*x*(2.0*x - 1.0)*(2*z - 2.0)*(2*z - 1) ;
810 case 2 : return (1.0/2.0)*y*(2.0*y - 1.0)*(2*z - 2.0)*(2*z - 1) ;
811 case 3 : return z*(2*z - 1)*(-2.0*x - 2.0*y + 1.0)*(-x - y + 1.0) ;
812 case 4 : return x*z*(2.0*x - 1.0)*(2*z - 1) ;
813 case 5 : return y*z*(2.0*y - 1.0)*(2*z - 1) ;
814 case 6 : return (1.0/2.0)*x*(2*z - 2.0)*(2*z - 1)*(-4.0*x - 4.0*y + 4.0) ;
815 case 7 : return 2.0*x*y*(2*z - 2.0)*(2*z - 1) ;
816 case 8 : return 2.0*y*(2*z - 2.0)*(2*z - 1)*(-x - y + 1.0) ;
817 case 9 : return (1.0 - std::pow(2*z - 1, 2))*(-x - y + 1.0) ;
818 case 10 : return x*(1.0 - std::pow(2*z - 1, 2)) ;
819 case 11 : return y*(1.0 - std::pow(2*z - 1, 2)) ;
820 case 12 : return x*z*(2*z - 1)*(-4.0*x - 4.0*y + 4.0) ;
821 case 13 : return 4.0*x*y*z*(2*z - 1) ;
822 case 14 : return 4.0*y*z*(2*z - 1)*(-x - y + 1.0) ;
823 default:
824 KRATOS_ERROR << "Wrong index of shape function!" << ShapeFunctionIndex << std::endl;
825 }
826 }

936 static Matrix CalculateShapeFunctionsIntegrationPointsValues(typename BaseType::IntegrationMethod ThisMethod )
937 {
938 IntegrationPointsContainerType all_integration_points = AllIntegrationPoints();
939 IntegrationPointsArrayType integration_points = all_integration_points[static_cast<int>(ThisMethod)];
940
941 // Number of integration points
942 const std::size_t integration_points_number = integration_points.size();
943
944 //Setting up return matrix
945 Matrix shape_function_values( integration_points_number, 15 );
946
947 // Loop over all integration points
948 double x, y, z;
949 for ( std::size_t pnt = 0; pnt < integration_points_number; pnt++ ) {
950 const auto& r_point = integration_points[pnt];
951 x = r_point[0];
952 y = r_point[1];
