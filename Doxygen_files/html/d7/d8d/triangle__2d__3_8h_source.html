847 CoordinatesArrayType& PointLocalCoordinates(
848 CoordinatesArrayType& rResult,
849 const CoordinatesArrayType& rPoint
850 ) const override {
851
852 rResult = ZeroVector(3);
853
854 const TPointType& point_0 = this->GetPoint(0);
855
856 // Compute the Jacobian matrix and its determinant
857 BoundedMatrix<double, 2, 2> J;
858 J(0,0) = this->GetPoint(1).X() - this->GetPoint(0).X();
859 J(0,1) = this->GetPoint(2).X() - this->GetPoint(0).X();
860 J(1,0) = this->GetPoint(1).Y() - this->GetPoint(0).Y();
861 J(1,1) = this->GetPoint(2).Y() - this->GetPoint(0).Y();
862 const double det_J = J(0,0)*J(1,1) - J(0,1)*J(1,0);
863
864 // Compute eta and xi
865 const double eta = (J(1,0)*(point_0.X() - rPoint(0)) +
866 J(0,0)*(rPoint(1) - point_0.Y())) / det_J;
867 const double xi = (J(1,1)*(rPoint(0) - point_0.X()) +
868 J(0,1)*(point_0.Y() - rPoint(1))) / det_J;
869
870 rResult(0) = xi;
871 rResult(1) = eta;
872 rResult(2) = 0.0;
873
874 return rResult;
875 }

1623 /* Triangle/triangle intersection test routine,
1624 * by Tomas Moller, 1997.
1625 * See article "A Fast Triangle-Triangle Intersection Test",
1626 * Journal of Graphics Tools, 2(2), 1997
1627 *
1628 * Updated June 1999: removed the divisions -- a little faster now!
1629 * Updated October 1999: added {} to CROSS and SUB macros
1630 *
1631 * int NoDivTriTriIsect(float V0[3],float V1[3],float V2[3],
1632 * float U0[3],float U1[3],float U2[3])
1633 *
1634 * parameters: vertices of triangle 1: V0,V1,V2
1635 * vertices of triangle 2: U0,U1,U2
1636 * result : returns 1 if the triangles intersect, otherwise 0
1637 *
1638 */
1639
1640 bool NoDivTriTriIsect( const Point& V0,
1641 const Point& V1,
1642 const Point& V2,
1643 const Point& U0,
1644 const Point& U1,
1645 const Point& U2) const
1646 {
1647 short index;
1648 double d1,d2;
1649 double du0,du1,du2,dv0,dv1,dv2;
1650 double du0du1,du0du2,dv0dv1,dv0dv2;
1651 double vp0,vp1,vp2;
1652 double up0,up1,up2;
1653 double bb,cc,max;
1654 array_1d<double,2> isect1, isect2;
1655 array_1d<double,3> D;
1656 array_1d<double,3> E1,E2;
1657 array_1d<double,3> N1,N2;
1658
1659
1660 const double epsilon = 1E-6;
1661
1662// compute plane equation of triangle(V0,V1,V2) //
1663 noalias(E1) = V1-V0;
1664 noalias(E2) = V2-V0;
1665 MathUtils<double>::CrossProduct(N1, E1, E2);
1666 d1=-inner_prod(N1,V0);
1667// plane equation 1: N1.X+d1=0 //
1668
1669// put U0,U1,U2 into plane equation 1 to compute signed distances to the plane//
1670 du0=inner_prod(N1,U0)+d1;
1671 du1=inner_prod(N1,U1)+d1;
1672 du2=inner_prod(N1,U2)+d1;
1673
1674// coplanarity robustness check //
1675 if(fabs(du0)<epsilon) du0=0.0;
1676 if(fabs(du1)<epsilon) du1=0.0;
1677 if(fabs(du2)<epsilon) du2=0.0;
1678
1679 du0du1=du0*du1;
1680 du0du2=du0*du2;
1681
1682 if(du0du1>0.00 && du0du2>0.00)// same sign on all of them + not equal 0 ? //
1683 return false; // no intersection occurs //
1684
1685// compute plane of triangle (U0,U1,U2) //
1686 noalias(E1) = U1 - U0;
1687 noalias(E2) = U2 - U0;
1688 MathUtils<double>::CrossProduct(N2, E1, E2);
1689 d2=-inner_prod(N2,U0);
1690// plane equation 2: N2.X+d2=0 //
1691
1692// put V0,V1,V2 into plane equation 2 //
1693 dv0=inner_prod(N2,V0)+d2;
1694 dv1=inner_prod(N2,V1)+d2;
1695 dv2=inner_prod(N2,V2)+d2;
1696
1697 if(fabs(dv0)<epsilon) dv0=0.0;
1698 if(fabs(dv1)<epsilon) dv1=0.0;
1699 if(fabs(dv2)<epsilon) dv2=0.0;
1700
1701 dv0dv1=dv0*dv1;
1702 dv0dv2=dv0*dv2;
1703
1704 if(dv0dv1>0.00 && dv0dv2>0.00)// same sign on all of them + not equal 0 ? //
1705 return false; // no intersection occurs //
1706
1707// compute direction of intersection line //
1708 MathUtils<double>::CrossProduct(D, N1, N2);
1709
1710// compute and index to the largest component of D //
1711 max=(double)fabs(D[0]);
1712 index=0;
1713 bb=(double)fabs(D[1]);
1714 cc=(double)fabs(D[2]);
1715 if(bb>max)
1716 {
1717 max=bb,index=1;
1718 }
1719 if(cc>max)
1720 {
1721 max=cc,index=2;
1722 }
1723
1724// this is the simplified projection onto L//
1725 vp0=V0[index];
1726 vp1=V1[index];
1727 vp2=V2[index];
1728
1729 up0=U0[index];
1730 up1=U1[index];
1731 up2=U2[index];
1732
1733
1734// compute interval for triangle 1 //
1735 double a,b,c,x0,x1;
1736 if( New_Compute_Intervals(vp0,vp1,vp2,dv0, dv1,dv2,dv0dv1,dv0dv2, a,b,c,x0,x1)==true )
1737 {
1738 return coplanar_tri_tri(N1,V0,V1,V2,U0,U1,U2);
1739 }
1740
1741// compute interval for triangle 2 //
1742 double d,e,f,y0,y1;
1743 if( New_Compute_Intervals(up0, up1, up2, du0, du1, du2, du0du1, du0du2, d, e, f, y0, y1)==true )
1744 {
1745 return coplanar_tri_tri(N1,V0,V1,V2,U0,U1,U2);
1746 }
1747
1748
1749 double xx,yy,xxyy,tmp;
1750 xx=x0*x1;
1751 yy=y0*y1;
1752 xxyy=xx*yy;
1753
1754 tmp=a*xxyy;
1755 isect1[0]=tmp+b*x1*yy;
1756 isect1[1]=tmp+c*x0*yy;
1757
1758 tmp=d*xxyy;
1759 isect2[0]=tmp+e*xx*y1;
1760 isect2[1]=tmp+f*xx*y0;
1761
1762
1763 Sort(isect1[0],isect1[1]);
1764 Sort(isect2[0],isect2[1]);
1765
1766 if(isect1[1]<isect2[0] || isect2[1]<isect1[0]) return false;
1767 return true;
1768 }

2044 inline bool TriBoxOverlap(Point& rBoxCenter, Point& rBoxHalfSize) const
2045 {
2046 double abs_ex, abs_ey;
2047 array_1d<double,3 > vert0, vert1, vert2;
2048 array_1d<double,3 > edge0, edge1, edge2;
2049 std::pair<double, double> min_max;
2050
2051 // move everything so that the boxcenter is in (0,0,0)
2052 noalias(vert0) = this->GetPoint(0) - rBoxCenter;
2053 noalias(vert1) = this->GetPoint(1) - rBoxCenter;
2054 noalias(vert2) = this->GetPoint(2) - rBoxCenter;
2055
2056 // compute triangle edges
2057 noalias(edge0) = vert1 - vert0;
2058 noalias(edge1) = vert2 - vert1;
2059 noalias(edge2) = vert0 - vert2;
2060
2061 // Bullet 3:
2062 // test the 9 tests first (this was faster)
2063 // We are only interested on z-axis test, which defines the {x,y} plane
2064 // Note that projections onto crossproduct tri-{x,y} are always 0:
2065 // that means there is no separating axis on X,Y-axis tests
2066 abs_ex = std::abs(edge0[0]);
2067 abs_ey = std::abs(edge0[1]);
2068 if (AxisTestZ(edge0[0],edge0[1],abs_ex,abs_ey,vert0,vert2,rBoxHalfSize)) return false;
2069
2070 abs_ex = std::abs(edge1[0]);
2071 abs_ey = std::abs(edge1[1]);
2072 if (AxisTestZ(edge1[0],edge1[1],abs_ex,abs_ey,vert1,vert0,rBoxHalfSize)) return false;
2073
2074 abs_ex = std::abs(edge2[0]);
2075 abs_ey = std::abs(edge2[1]);
2076 if (AxisTestZ(edge2[0],edge2[1],abs_ex,abs_ey,vert2,vert1,rBoxHalfSize)) return false;
2077
2078 // Bullet 1:
2079 // first test overlap in the {x,y,(z)}-directions
2080 // find min, max of the triangle each direction, and test for overlap in
2081 // that direction -- this is equivalent to testing a minimal AABB around
2082 // the triangle against the AABB
2083
2084 // test in X-direction
2085 min_max = std::minmax({vert0[0],vert1[0],vert2[0]});
2086 if(min_max.first>rBoxHalfSize[0] || min_max.second<-rBoxHalfSize[0]) return false;
2087
2088 // test in Y-direction
2089 min_max = std::minmax({vert0[1],vert1[1],vert2[1]});
2090 if(min_max.first>rBoxHalfSize[1] || min_max.second<-rBoxHalfSize[1]) return false;
2091
2092 // test in Z-direction
2093 // we do not consider rBoxHalfSize[2] since we are working in 2D
2094
2095 // Bullet 2:
2096 // test if the box intersects the plane of the triangle
2097 // compute plane equation of triangle: normal*x+d=0
2098 // Triangle and box are on the {x-y} plane, so this test won't return false
2099
2100 return true; // box and triangle overlaps
2101 }
