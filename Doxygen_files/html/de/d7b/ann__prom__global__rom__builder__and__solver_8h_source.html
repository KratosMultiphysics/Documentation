277 void MonotonicityPreserving(
278 TSystemMatrixType& rA,
279 TSystemVectorType& rB
280 )
281 {
282 const auto& r_dof_set = BaseType::GetDofSet();
283 Vector dofs_values = ZeroVector(r_dof_set.size());
284
285 block_for_each(r_dof_set, [&](Dof<double>& rDof){
286 const std::size_t id = rDof.EquationId();
287 dofs_values[id] = rDof.GetSolutionStepValue();
288 });
289 double *values_vector = rA.value_data().begin();
290 std::size_t *index1_vector = rA.index1_data().begin();
291 std::size_t *index2_vector = rA.index2_data().begin();
292
293 IndexPartition<std::size_t>(rA.size1()).for_each(
294 [&](std::size_t i)
295 {
296 for (std::size_t k = index1_vector[i]; k < index1_vector[i + 1]; k++) {
297 const double value = values_vector[k];
298 if (value > 0.0) {
299 const auto j = index2_vector[k];
300 if (j > i) {
301 // TODO: Partition in blocks to gain efficiency by avoiding thread locks.
302 // Values conflicting with other threads
303 auto& r_aij = rA(i,j).ref();
304 AtomicAdd(r_aij, -value);
305 auto& r_aji = rA(j,i).ref();
306 AtomicAdd(r_aji, -value);
307 auto& r_aii = rA(i,i).ref();
308 AtomicAdd(r_aii, value);
309 auto& r_ajj = rA(j,j).ref();
310 AtomicAdd(r_ajj, value);
311 auto& r_bi = rB[i];
312 AtomicAdd(r_bi, value*dofs_values[j] - value*dofs_values[i]);
313 auto& r_bj = rB[j];
314 AtomicAdd(r_bj, value*dofs_values[i] - value*dofs_values[j]);
315 }
316 }
317 }
318 }
319 );
320 }

494 void InitializeHROMWeights(ModelPart& rModelPart)
495 {
496 KRATOS_TRY
497
498 using ElementQueue = moodycamel::ConcurrentQueue<Element::Pointer>;
499 using ConditionQueue = moodycamel::ConcurrentQueue<Condition::Pointer>;
500
501 // Inspecting elements
502 ElementQueue element_queue;
503 block_for_each(rModelPart.Elements().GetContainer(),
504 [&](Element::Pointer p_element)
505 {
506 if (p_element->Has(HROM_WEIGHT)) {
507 element_queue.enqueue(std::move(p_element));
508 } else {
509 p_element->SetValue(HROM_WEIGHT, 1.0);
510 }
511 });
512
513 // Inspecting conditions
514 ConditionQueue condition_queue;
515 block_for_each(rModelPart.Conditions().GetContainer(),
516 [&](Condition::Pointer p_condition)
517 {
518 if (p_condition->Has(HROM_WEIGHT)) {
519 condition_queue.enqueue(std::move(p_condition));
520 } else {
521 p_condition->SetValue(HROM_WEIGHT, 1.0);
522 }
523 });
524
525 // Dequeueing elements
526 std::size_t err_id;
527 mSelectedElements.reserve(element_queue.size_approx());
528 Element::Pointer p_element;
529 while ( (err_id = element_queue.try_dequeue(p_element)) != 0) {
530 mSelectedElements.push_back(std::move(p_element));
531 }
532
533 // Dequeueing conditions
534 mSelectedConditions.reserve(condition_queue.size_approx());
535 Condition::Pointer p_condition;
536 while ( (err_id = condition_queue.try_dequeue(p_condition)) != 0) {
537 mSelectedConditions.push_back(std::move(p_condition));
538 }
539
540 // Wrap-up
541 mHromSimulation = !(mSelectedElements.empty() && mSelectedConditions.empty());
542 mHromWeightsInitialized = true;
543
544 KRATOS_CATCH("")
545 }

547 static DofQueue ExtractDofSet(
548 typename TSchemeType::Pointer pScheme,
549 ModelPart& rModelPart)
550 {
551 KRATOS_TRY
552
553 DofQueue dof_queue;
554
555 // Emulates ConcurrentQueue::enqueue_bulk adding move semantics to avoid atomic ops
556 const auto enqueue_bulk_move = [](DofQueue& r_queue, DofsVectorType& r_dof_list) {
557 for(auto& p_dof: r_dof_list) {
558 r_queue.enqueue(std::move(p_dof));
559 }
560 r_dof_list.clear();
561 };
562
563 // Inspecting elements
564 DofsVectorType tls_dof_list; // Preallocation
565 block_for_each(rModelPart.Elements(), tls_dof_list,
566 [&](const Element& r_element, DofsVectorType& r_dof_list)
567 {
568 pScheme->GetDofList(r_element, r_dof_list, rModelPart.GetProcessInfo());
569 enqueue_bulk_move(dof_queue, r_dof_list);
570 });
571
572 // Inspecting conditions
573 block_for_each(rModelPart.Conditions(), tls_dof_list,
574 [&](const Condition& r_condition, DofsVectorType& r_dof_list)
575 {
576 pScheme->GetDofList(r_condition, r_dof_list, rModelPart.GetProcessInfo());
577 enqueue_bulk_move(dof_queue, r_dof_list);
578 });
579
580 // Inspecting master-slave constraints
581 std::pair<DofsVectorType, DofsVectorType> tls_ms_dof_lists; // Preallocation
582 block_for_each(rModelPart.MasterSlaveConstraints(), tls_ms_dof_lists,
583 [&](const MasterSlaveConstraint& r_constraint, std::pair<DofsVectorType, DofsVectorType>& r_dof_lists)
584 {
585 r_constraint.GetDofList(r_dof_lists.first, r_dof_lists.second, rModelPart.GetProcessInfo());
586
587 enqueue_bulk_move(dof_queue, r_dof_lists.first);
588 enqueue_bulk_move(dof_queue, r_dof_lists.second);
589 });
590
591 return dof_queue;
592
593 KRATOS_CATCH("")
594 }

669 void Build(
670 typename TSchemeType::Pointer pScheme,
671 ModelPart& rModelPart,
672 TSystemMatrixType& A,
673 TSystemVectorType& b) override
674 {
675 KRATOS_TRY
676
677 KRATOS_ERROR_IF(!pScheme) << "No scheme provided!" << std::endl;
678
679 // // Getting the elements from the model
680 const int nelements = mHromSimulation ? mSelectedElements.size() : rModelPart.Elements().size();
681
682 // // Getting the array of the conditions
683 const int nconditions = mHromSimulation ? mSelectedConditions.size() : rModelPart.Conditions().size();
684
685 const ProcessInfo& CurrentProcessInfo = rModelPart.GetProcessInfo();
686 ModelPart::ElementsContainerType::iterator el_begin = mHromSimulation ? mSelectedElements.begin() : rModelPart.Elements().begin();
687 ModelPart::ConditionsContainerType::iterator cond_begin = mHromSimulation ? mSelectedConditions.begin() : rModelPart.Conditions().begin();
688
689 //contributions to the system
690 LocalSystemMatrixType LHS_Contribution = LocalSystemMatrixType(0, 0);
691 LocalSystemVectorType RHS_Contribution = LocalSystemVectorType(0);
692
693 //vector containing the localization in the system of the different
694 //terms
695 Element::EquationIdVectorType EquationId;
696
697 // Assemble all elements
698 const auto timer = BuiltinTimer();
699
700 #pragma omp parallel firstprivate(nelements,nconditions, LHS_Contribution, RHS_Contribution, EquationId )
701 {
702 # pragma omp for schedule(guided, 512) nowait
703 for (int k = 0; k < nelements; k++) {
704 auto it_elem = el_begin + k;
705
706 if (it_elem->IsActive()) {
707 // Calculate elemental contribution
708 pScheme->CalculateSystemContributions(*it_elem, LHS_Contribution, RHS_Contribution, EquationId, CurrentProcessInfo);
709
710 //Get HROM weight and multiply it by its contribution
711 const double h_rom_weight = mHromSimulation ? it_elem->GetValue(HROM_WEIGHT) : 1.0;
712 LHS_Contribution *= h_rom_weight;
713 RHS_Contribution *= h_rom_weight;
714
715 // Assemble the elemental contribution
716 BaseType::Assemble(A, b, LHS_Contribution, RHS_Contribution, EquationId);
717 }
718
719 }
720
721 #pragma omp for schedule(guided, 512)
722 for (int k = 0; k < nconditions; k++) {
723 auto it_cond = cond_begin + k;
724
725 if (it_cond->IsActive()) {
726 // Calculate elemental contribution
727 pScheme->CalculateSystemContributions(*it_cond, LHS_Contribution, RHS_Contribution, EquationId, CurrentProcessInfo);
728
729 //Get HROM weight and multiply it by its contribution
730 const double h_rom_weight = mHromSimulation ? it_cond->GetValue(HROM_WEIGHT) : 1.0;
731 LHS_Contribution *= h_rom_weight;
732 RHS_Contribution *= h_rom_weight;
733
734 // Assemble the elemental contribution
735 BaseType::Assemble(A, b, LHS_Contribution, RHS_Contribution, EquationId);
736 }
737 }
738 }
739
740 KRATOS_INFO_IF("GlobalROMResidualBasedBlockBuilderAndSolver", this->GetEchoLevel() >= 1) << "Build time: " << timer.ElapsedSeconds() << std::endl;
741
742 KRATOS_INFO_IF("GlobalROMResidualBasedBlockBuilderAndSolver", (this->GetEchoLevel() > 2 && rModelPart.GetCommunicator().MyPID() == 0)) << "Finished parallel building" << std::endl;

744 KRATOS_CATCH("")

750 virtual void ProjectROM(
751 ModelPart &rModelPart,
752 TSystemMatrixType &rA,
753 TSystemVectorType &rb)
754 {
755 KRATOS_TRY
756
757 auto& r_root_mp = rModelPart.GetRootModelPart();
758
759 if (mRightRomBasisInitialized==false){
760 mPhiGlobal = ZeroMatrix(BaseBuilderAndSolverType::GetEquationSystemSize(), GetNumberOfROMModes());

762 TSystemVectorType& r_xBase = r_root_mp.GetValue(SOLUTION_BASE);
763 vector<Matrix>& r_svdPhiMatrices = r_root_mp.GetValue(SVD_PHI_MATRICES);
764 vector<Matrix>& r_nnLayers = r_root_mp.GetValue(NN_LAYERS);
765 Vector& r_refSnapshot = r_root_mp.GetValue(SOLUTION_REFERENCE);
766
767 RomNNUtility<TSparseSpace,TDenseSpace>::GetXAndDecoderGradient(r_xRom, r_xBase, mPhiGlobal, r_svdPhiMatrices, r_nnLayers, r_refSnapshot);
768
769 mRightRomBasisInitialized = true;
770 }
771 else if (r_root_mp.GetValue(UPDATE_PHI_EFFECTIVE_BOOL)==true){
772 RomSystemVectorType& r_xRomTotal = r_root_mp.GetValue(ROM_SOLUTION_TOTAL);
773 vector<Matrix>& r_svdPhiMatrices = r_root_mp.GetValue(SVD_PHI_MATRICES);
774 vector<Matrix>& r_nnLayers = r_root_mp.GetValue(NN_LAYERS);
775 RomNNUtility<TSparseSpace,TDenseSpace>::GetDecoderGradient(r_xRomTotal, mPhiGlobal, r_svdPhiMatrices, r_nnLayers);
776 }
777
778 SetFixedDofsROMBasisToZero();
779
780 auto a_wrapper = UblasWrapper<double>(rA);
781 const auto& eigen_rA = a_wrapper.matrix();
782 Eigen::Map<EigenDynamicVector> eigen_rb(rb.data().begin(), rb.size());
783 Eigen::Map<EigenDynamicMatrix> eigen_mPhiGlobal(mPhiGlobal.data().begin(), mPhiGlobal.size1(), mPhiGlobal.size2());
784
785 EigenDynamicMatrix eigen_rA_times_mPhiGlobal = eigen_rA * eigen_mPhiGlobal; //TODO: Make it in parallel.
786
787 // Compute the matrix multiplication
788 mEigenRomA = eigen_mPhiGlobal.transpose() * eigen_rA_times_mPhiGlobal; //TODO: Make it in parallel.
789 mEigenRomB = eigen_mPhiGlobal.transpose() * eigen_rb; //TODO: Make it in parallel.
790
791 KRATOS_CATCH("")

797 virtual void SolveROM(
798 ModelPart &rModelPart,
799 EigenDynamicMatrix &rEigenRomA,
800 EigenDynamicVector &rEigenRomB,
801 TSystemVectorType &rDx)
802 {
803 KRATOS_TRY
804
805 // Save the current ROM solution and other variables from the root modelpart database
806 auto& r_root_mp = rModelPart.GetRootModelPart();
807
808 RomSystemVectorType& r_xRom = r_root_mp.GetValue(ROM_SOLUTION_BASE);
809 RomSystemVectorType& r_xRomTotal = r_root_mp.GetValue(ROM_SOLUTION_TOTAL);
810
811 r_xRom = r_xRomTotal;
812
813 RomSystemVectorType& r_dxRom = r_root_mp.GetValue(ROM_SOLUTION_INCREMENT);
814
815 Vector& r_xBase = r_root_mp.GetValue(SOLUTION_BASE);
816 Vector xNew (rDx.size());
817
818 vector<Matrix>& svdPhiMatrices = r_root_mp.GetValue(SVD_PHI_MATRICES);
819 vector<Matrix>& nnLayers = r_root_mp.GetValue(NN_LAYERS);
820 Vector& refSnapshot = r_root_mp.GetValue(SOLUTION_REFERENCE);
821
822 const auto solving_timer = BuiltinTimer();
823
824 Eigen::Map<EigenDynamicVector> dxrom_eigen(r_dxRom.data().begin(), r_dxRom.size());
825 if(rEigenRomA.colPivHouseholderQr().isInvertible() == false){
826 KRATOS_INFO("AnnPromGlobalROMBuilderAndSolver") << "Linear system to be solved by QR is not invertible" << std::endl;
827 }
828 dxrom_eigen = rEigenRomA.colPivHouseholderQr().solve(rEigenRomB);
829
830 double time = solving_timer.ElapsedSeconds();
831 KRATOS_INFO_IF("AnnPromGlobalROMBuilderAndSolver", (this->GetEchoLevel() > 0)) << "Solve reduced system time: " << time << std::endl;
832
833 // project reduced solution back to full order modela
834 const auto backward_projection_timer = BuiltinTimer();
835
836 RomNNUtility<TSparseSpace,TDenseSpace>::GetXAndDecoderGradient(r_xRom+r_dxRom,xNew, mPhiGlobal, svdPhiMatrices, nnLayers, refSnapshot);
837
838 GetDxAndUpdateXBase(xNew, r_xBase, rDx);
839 KRATOS_INFO_IF("AnnPromGlobalROMBuilderAndSolver", (this->GetEchoLevel() > 0)) << "Project to fine basis time: " << backward_projection_timer.ElapsedSeconds() << std::endl;
840
841 r_xRomTotal = r_xRom+r_dxRom;
842
843 r_root_mp.SetValue(UPDATE_PHI_EFFECTIVE_BOOL, false);
844
845 KRATOS_CATCH("")
846 }

865 noalias(row(mPhiGlobal, r_dof.EquationId())) = ZeroVector(mPhiGlobal.size2());
866 }
867 });
