55 void ExecuteInitializeSolutionStep() override
56 {
57 KRATOS_TRY
58
59 const Variable<double>& var = KratosComponents<Variable<double>>::Get(mVariableName);
60
61 const double Time = mrModelPart.GetProcessInfo()[TIME] / mTimeUnitConverter;
62 array_1d<double, 2> deltaH;
63 for (unsigned int i = 0; i < mpTable.size(); ++i) {
64 if (mpTable[i]) {
65 deltaH[i] = mpTable[i]->GetValue(Time);
66 } else {
67 deltaH[i] = 0.0;
68 }
69 }
70
71 array_1d<double, 2> y;
72 y[0] = mFirstReferenceCoordinate[mGravityDirection];
73 y[1] = mSecondReferenceCoordinate[mGravityDirection];
74 y += deltaH;
75
76 mSlope = (y[1] - y[0]) / (mSecondReferenceCoordinate[mHorizontalDirection] -
77 mFirstReferenceCoordinate[mHorizontalDirection]);
78
79 block_for_each(mrModelPart.Nodes(), [&var, &y, this](Node& rNode) {
80 const double pressure = PORE_PRESSURE_SIGN_FACTOR * CalculatePressurewithTable(rNode, y);
81 if (mIsSeepage) {
82 if (pressure < PORE_PRESSURE_SIGN_FACTOR * mPressureTensionCutOff) { // Before 0. was used i.s.o. the tension cut off value -> no effect in any test.
83 rNode.FastGetSolutionStepValue(var) = pressure;
84 if (mIsFixed) rNode.Fix(var);
85 } else {
86 if (mIsFixedProvided) rNode.Free(var);
87 }
88 } else {
89 rNode.FastGetSolutionStepValue(var) =
90 std::min(pressure, PORE_PRESSURE_SIGN_FACTOR * mPressureTensionCutOff);
91 }
92 });
93
94 KRATOS_CATCH("")
95 }
96
98 std::string Info() const override { return "ApplyPhreaticLinePressureTableProcess"; }
99
100protected:
101 double CalculatePressurewithTable(const Node& rNode, const array_1d<double, 2>& y) const
102 {
103 double horCoord = rNode.Coordinates()[mHorizontalDirection];
104 horCoord = std::max(horCoord, mMinHorizontalCoordinate);
105 horCoord = std::min(horCoord, mMaxHorizontalCoordinate);
106 const double height = mSlope * (horCoord - mFirstReferenceCoordinate[mHorizontalDirection]) + y[0];
107 const double distance = height - rNode.Coordinates()[mGravityDirection];
108 return -mSpecificWeight * distance;
109 }
110
111private:
