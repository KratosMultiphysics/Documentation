29 template<SizeType TDimension>
30 void GenerateMaterialPointElement( ModelPart& rBackgroundGridModelPart,
31 ModelPart& rInitialModelPart,
32 ModelPart& rMPMModelPart,
33 bool IsMixedFormulation) {
34 const bool IsAxisSymmetry = (rBackgroundGridModelPart.GetProcessInfo().Has(IS_AXISYMMETRIC))
35 ? rBackgroundGridModelPart.GetProcessInfo().GetValue(IS_AXISYMMETRIC)
36 : false;
37
38 // Initialize zero the variables needed
39 std::vector<array_1d<double, 3>> xg = { ZeroVector(3) };
40 std::vector<array_1d<double, 3>> mp_displacement = { ZeroVector(3) };
41 std::vector<array_1d<double, 3>> mp_velocity = { ZeroVector(3) };
42 std::vector<array_1d<double, 3>> mp_acceleration = { ZeroVector(3) };
43 std::vector<array_1d<double, 3>> mp_volume_acceleration = { ZeroVector(3) };
44
45 std::vector<Vector> mp_cauchy_stress_vector = { ZeroVector(6) };
46 std::vector<Vector> mp_almansi_strain_vector = { ZeroVector(6) };
47 std::vector<double> mp_pressure = { 0.0 };
48
49 std::vector<double> mp_mass(1);
50 std::vector<double> mp_volume(1);
51
52 // Determine element index: This convention is done in order for the purpose of visualization in GiD
53 const unsigned int number_elements = rBackgroundGridModelPart.NumberOfElements() + rInitialModelPart.NumberOfElements();
54 const unsigned int number_nodes = rBackgroundGridModelPart.NumberOfNodes();
55 unsigned int last_element_id = (number_nodes > number_elements) ? (number_nodes + 1) : (number_elements + 1);
56
57 BinBasedFastPointLocator<TDimension> SearchStructure(rBackgroundGridModelPart);
58 SearchStructure.UpdateSearchDatabase();
59 typename BinBasedFastPointLocator<TDimension>::ResultContainerType results(100);
60
61 // Loop over the submodelpart of rInitialModelPart
62 for (auto& submodelpart : rInitialModelPart.SubModelParts())
63 {
64 std::string submodelpart_name = submodelpart.Name();
65
66 rMPMModelPart.CreateSubModelPart(submodelpart_name);
67
68 // Loop over the element of submodelpart and generate mpm element to be appended to the rMPMModelPart
69 for (ModelPart::ElementIterator i = submodelpart.ElementsBegin();
70 i != submodelpart.ElementsEnd(); i++)
71 {
72 if (i->IsDefined(ACTIVE))
73 {
74 Properties::Pointer properties = i->pGetProperties();
75 const double density = i->GetProperties()[DENSITY];
76
77 // Check number of material point per element to be created
78 unsigned int material_points_per_element;
79 if (i->GetProperties().Has(MATERIAL_POINTS_PER_ELEMENT)) {
80 material_points_per_element = i->GetProperties()[MATERIAL_POINTS_PER_ELEMENT];
81 }
82 else {
83 std::string warning_msg = "MATERIAL_POINTS_PER_ELEMENT is not specified in Properties, ";
84 warning_msg += "1 material point per element is assumed.";
85 KRATOS_WARNING("MaterialPointGeneratorUtility") << warning_msg << std::endl;
86 material_points_per_element = 1;
87 }
88
89 // Get geometry and dimension of the background grid
90 const GeometryData::KratosGeometryType background_geo_type = rBackgroundGridModelPart.ElementsBegin()->GetGeometry().GetGeometryType();
91 const std::size_t domain_size = rBackgroundGridModelPart.GetProcessInfo()[DOMAIN_SIZE];
92 const Geometry< Node >& r_geometry = i->GetGeometry(); // current element's geometry
93
94 // Get integration method and shape function values
95 IntegrationMethod int_method = GeometryData::IntegrationMethod::GI_GAUSS_1;
96 Matrix shape_functions_values;
97 bool is_equal_int_volumes = false;
98 DetermineIntegrationMethodAndShapeFunctionValues(r_geometry, material_points_per_element,
99 int_method, shape_functions_values, is_equal_int_volumes);
100
101 // Get volumes of the material points
102 const unsigned int integration_point_per_elements = shape_functions_values.size1();
103 Vector int_volumes (integration_point_per_elements);
104 if (is_equal_int_volumes) {
105 for (size_t j = 0; j < integration_point_per_elements; ++j) int_volumes[j] = r_geometry.DomainSize() / integration_point_per_elements;
106 }
107 else GetIntegrationPointVolumes(r_geometry, int_method, int_volumes);
108 if (domain_size == 2 && i->GetProperties().Has(THICKNESS)) {
109 for (size_t j = 0; j < integration_point_per_elements; ++j) int_volumes[j] *= i->GetProperties()[THICKNESS];
110 }
111
112 // Set element type
113 std::string element_type_name = "MPMUpdatedLagrangian";
114 if (IsMixedFormulation) {
115 if (background_geo_type == GeometryData::KratosGeometryType::Kratos_Triangle2D3) element_type_name = "MPMUpdatedLagrangianUP";
116 else KRATOS_ERROR << "Element for mixed U-P formulation is only implemented for 2D Triangle Elements." << std::endl;
117 }
118 else if (IsAxisSymmetry && domain_size == 3) KRATOS_ERROR << "Axisymmetric elements must be used in a 2D domain. You specified a 3D domain." << std::endl;
119 else if (rBackgroundGridModelPart.GetProcessInfo().Has(IS_PQMPM)) {
120 if (rBackgroundGridModelPart.GetProcessInfo().GetValue(IS_PQMPM)) {
121 element_type_name = "MPMUpdatedLagrangianPQ";
122 KRATOS_ERROR_IF(IsAxisSymmetry) << "PQMPM is not implemented for axisymmetric elements yet." << std::endl;
123 }
124 }
125
126 // Get new element
127 const Element& new_element = KratosComponents<Element>::Get(element_type_name);
128
129 // Loop over the material points that fall in each grid element
130 unsigned int new_element_id = 0;
131 for (unsigned int PointNumber = 0; PointNumber < integration_point_per_elements; PointNumber++)
132 {
133 mp_volume[0] = int_volumes[PointNumber];
134 mp_mass[0] = int_volumes[PointNumber]*density;
135
136 std::vector<double> MP_density = { density };
137
138 xg[0].clear();
139
140 // Loop over the nodes of the grid element
141 for (unsigned int dimension = 0; dimension < r_geometry.WorkingSpaceDimension(); dimension++)
142 {
143 for (unsigned int j = 0; j < r_geometry.size(); j++)
144 {
145 xg[0][dimension] = xg[0][dimension] + shape_functions_values(PointNumber, j) * r_geometry[j].Coordinates()[dimension];
146 }
147 }
148 if (IsAxisSymmetry) {
149 mp_volume[0] *= 2.0 * Globals::Pi * xg[0][0];
150 mp_mass[0] = mp_volume[0] * density;
151 }
152
153 typename BinBasedFastPointLocator<TDimension>::ResultIteratorType result_begin = results.begin();
154 Element::Pointer pelem;
155 Vector N;
156
157 // FindPointOnMesh find the background element in which a given point falls and the relative shape functions
158 bool is_found = SearchStructure.FindPointOnMesh(xg[0], N, pelem, result_begin);
159 if (!is_found) KRATOS_WARNING("MaterialPointGeneratorUtility") << "::search failed." << std::endl;
160
161 pelem->Set(ACTIVE);
162 auto p_new_geometry = CreateQuadraturePointsUtility<Node>::CreateFromCoordinates(
163 pelem->pGetGeometry(), xg[0],
164 mp_volume[0]);
165
166 // Create new material point element
167 new_element_id = last_element_id + PointNumber;
168 Element::Pointer p_element = new_element.Create(
169 new_element_id, p_new_geometry, properties);
170
171 const ProcessInfo process_info = ProcessInfo();
172
173 // Setting material point element initial condition
174 p_element->SetValuesOnIntegrationPoints(MP_DENSITY, MP_density, process_info);
175 p_element->SetValuesOnIntegrationPoints(MP_MASS, mp_mass, process_info);
176 p_element->SetValuesOnIntegrationPoints(MP_VOLUME, mp_volume, process_info);
177 p_element->SetValuesOnIntegrationPoints(MP_COORD, xg, process_info);
178 p_element->SetValuesOnIntegrationPoints(MP_DISPLACEMENT, mp_displacement, process_info);
179 p_element->SetValuesOnIntegrationPoints(MP_VELOCITY, mp_velocity, process_info);
180 p_element->SetValuesOnIntegrationPoints(MP_ACCELERATION, mp_acceleration, process_info);
181 p_element->SetValuesOnIntegrationPoints(MP_VOLUME_ACCELERATION, mp_volume_acceleration, process_info);
182 p_element->SetValuesOnIntegrationPoints(MP_CAUCHY_STRESS_VECTOR, mp_cauchy_stress_vector, process_info);
183 p_element->SetValuesOnIntegrationPoints(MP_ALMANSI_STRAIN_VECTOR, mp_almansi_strain_vector, process_info);
184
185 if (IsMixedFormulation)
186 {
187 p_element->SetValuesOnIntegrationPoints(MP_PRESSURE, mp_pressure, process_info);
188 }
189
190 // Add the MP Element to the model part
191 rMPMModelPart.GetSubModelPart(submodelpart_name).AddElement(p_element);
192 }
193
194 last_element_id += integration_point_per_elements;
195
196 }
197 }
198 }
199 }
205 template <SizeType TDimension>
206 void GenerateMaterialPointCondition(ModelPart& rBackgroundGridModelPart,
207 ModelPart& rInitialModelPart,
208 ModelPart& rMPMModelPart){
209 // Initialize zero the variables needed
210 std::vector<array_1d<double, 3>> mpc_xg = { ZeroVector(3) };
211 array_1d<double,3> mpc_normal = ZeroVector(3);
212 std::vector<array_1d<double, 3>> mpc_displacement = { ZeroVector(3) };
213 std::vector<array_1d<double, 3>> mpc_delta_displacement = { ZeroVector(3) };
214 std::vector<array_1d<double, 3>> mpc_imposed_displacement = { ZeroVector(3) };
215 std::vector<array_1d<double, 3>> mpc_velocity = { ZeroVector(3) };
216 std::vector<array_1d<double, 3>> mpc_imposed_velocity = { ZeroVector(3) };
217 std::vector<array_1d<double, 3>> mpc_acceleration = { ZeroVector(3) };
218 std::vector<array_1d<double, 3>> mpc_imposed_acceleration = { ZeroVector(3) };
219 std::vector<array_1d<double, 3>> mpc_contact_force = { ZeroVector(3) };
220 std::vector<array_1d<double, 3>> point_load = { ZeroVector(3) };
221
222 std::vector<double> mpc_area(1);
223 std::vector<double> mpc_penalty_factor(1);
224 PointerVector<Condition> MaterialPointConditions;
225
226 // Determine condition index: This convention is done in order for the purpose of visualization in GiD
227 const unsigned int number_conditions = rBackgroundGridModelPart.NumberOfConditions();
228 const unsigned int number_elements = rBackgroundGridModelPart.NumberOfElements() + rInitialModelPart.NumberOfElements() + rMPMModelPart.NumberOfElements();
229 const unsigned int number_nodes = rBackgroundGridModelPart.NumberOfNodes();
230 unsigned int condition_id;
231 if (number_elements > number_nodes && number_elements > number_conditions)
232 condition_id = number_elements+1;
233 else if (number_nodes > number_elements && number_nodes > number_conditions)
234 condition_id = number_nodes+1;
235 else
236 condition_id = number_conditions+1;
237
238
239 BinBasedFastPointLocator<TDimension> SearchStructure(rBackgroundGridModelPart);
240 SearchStructure.UpdateSearchDatabase();
241 typename BinBasedFastPointLocator<TDimension>::ResultContainerType results(100);
242
243 const bool friction_active = rBackgroundGridModelPart.GetProcessInfo()[FRICTION_ACTIVE];
244
245 // initialize NODAL_AREA for SLIP nodes on NonHistorialVariable
246 // [ since NODAL_AREA may not be incl. as solution step variable for conforming conditions ]
247 if (friction_active) VariableUtils().SetNonHistoricalVariable<double>(NODAL_AREA, 0.0, rBackgroundGridModelPart.Nodes(), SLIP);
248
249
250 // Loop over the submodelpart of rBackgroundGridModelPart
251 for (auto& submodelpart : rBackgroundGridModelPart.SubModelParts())
252 {
253 // For submodelpart without condition, exit
254 if (submodelpart.NumberOfConditions() != 0){
255
256 std::string submodelpart_name = submodelpart.Name();
257
258 // For regular conditions: straight copy all conditions
259 if (!submodelpart.ConditionsBegin()->Is(BOUNDARY)){
260 if (submodelpart.NodesBegin()->Is(SLIP)){
261 // Do not copy conditions, this is a slip condition applied directly
262 // to the background grid nodes.
263 // Check 'apply_mpm_slip_boundary_process.py'
264
265 // Compute NODAL_AREA and perform checks for conforming friction
266 if (friction_active) {
267 for (auto& r_cond : submodelpart.Conditions()) {
268 const Geometry<Node>& r_geometry = r_cond.GetGeometry(); // current condition's geometry
269
270 const double condition_area = r_geometry.DomainSize();
271 const auto number_of_nodes = static_cast<double>(r_geometry.PointsNumber());
272
273 // assume equal contribution to all constituent nodes
274 for (Node& r_node : r_geometry.Points()) {
275 double& r_nodal_area = r_node.GetValue(NODAL_AREA);
276 r_nodal_area += condition_area / number_of_nodes;
277
278 // if required, flip normal direction
279 if (r_node.Is(MODIFIED)) {
280 array_1d<double, 3>& r_normal = r_node.FastGetSolutionStepValue(NORMAL);
281 r_normal *= -1.0;
282 r_node.Reset(MODIFIED);
283 }
284
285 // check required variables are present for friction
286 KRATOS_CHECK_VARIABLE_IN_NODAL_DATA(FRICTION_STATE, r_node);
287 KRATOS_CHECK_VARIABLE_IN_NODAL_DATA(STICK_FORCE, r_node);
288 }
289 }
290 }
291 }
292 else {
293 rMPMModelPart.CreateSubModelPart(submodelpart_name);
294 rMPMModelPart.SetConditions(submodelpart.pConditions());
295 }
296 }
297 // For boundary conditions: create material point conditions for all the necessary conditions
298 else{
299 MaterialPointConditions.clear();
300 // NOTE: To create material point Condition, we consider both the nodal position as well as the position of integration point
301 // Loop over the conditions of submodelpart and generate mpm condition to be appended to the rMPMModelPart
302 rMPMModelPart.CreateSubModelPart(submodelpart_name);
303 for (auto& r_cond : submodelpart.Conditions())
304 {
305 Properties::Pointer properties = r_cond.pGetProperties();
306 // Flag whether condition is Neumann or Dirichlet
307 const bool is_neumann_condition = r_cond.GetValue(MPC_IS_NEUMANN);
308 const int boundary_condition_type = r_cond.GetValue(MPC_BOUNDARY_CONDITION_TYPE);
309
310 // Check number of material points per condition to be created
311 unsigned int material_points_per_condition = 0; // Default zero
312 if (r_cond.Has( MATERIAL_POINTS_PER_CONDITION )){
313 material_points_per_condition = r_cond.GetValue(MATERIAL_POINTS_PER_CONDITION);
314 }
315 else{
316 KRATOS_WARNING("MaterialPointGeneratorUtility") << "MATERIAL_POINTS_PER_CONDITION is not specified. Only one material point is assumed." << std::endl;
317 }
318
319 // Get condition variables:
320 // Normal vector
321 if (r_cond.Has(NORMAL)) mpc_normal = r_cond.GetValue(NORMAL);
322 MPMMathUtilities<double>::Normalize(mpc_normal);
323
324 // Get shape_function_values from defined material_points_per_condition
325 const Geometry< Node >& r_geometry = r_cond.GetGeometry(); // current condition's geometry
326
327 // Check number of material points per condition to be created
328 bool is_equal_int_volumes = false; // default GAUSS
329 if (r_cond.Has( IS_EQUAL_DISTRIBUTED )){
330 is_equal_int_volumes = r_cond.GetValue(IS_EQUAL_DISTRIBUTED);
331 }
332
333 std::vector<IntegrationPoint<3>> integration_points;
334 IndexType number_of_points_per_span;
335 array_1d<double, 3> local_coordinates;
336
337 const GeometryData::KratosGeometryType geo_type = r_geometry.GetGeometryType();
338 IntegrationMethod integration_method = r_geometry.GetDefaultIntegrationMethod();
339
340 if (is_equal_int_volumes){
341 if (geo_type == GeometryData::KratosGeometryType::Kratos_Line2D2 || geo_type == GeometryData::KratosGeometryType::Kratos_Line3D2)
342 {
343 number_of_points_per_span = material_points_per_condition;
344 std::vector<double> spans = {-1, 1};
345
346 auto integration_info = IntegrationInfo(r_geometry.LocalSpaceDimension(), number_of_points_per_span, IntegrationInfo::QuadratureMethod::GRID);
347
348 IntegrationPointUtilities::CreateIntegrationPoints1D(
349 integration_points, spans, integration_info);
350
351 integration_method = integration_info.GetIntegrationMethod(0);
352 }
353 else{
354 KRATOS_WARNING("MaterialPointGeneratorUtility") << "Equal distribution of material point conditions only available for line segments: " << std::endl;
355 }
356 }
357 else{
358 if (geo_type != GeometryData::KratosGeometryType::Kratos_Point2D && geo_type != GeometryData::KratosGeometryType::Kratos_Point3D)
359 {
360 DetermineGeometryIntegrationMethod(r_geometry, material_points_per_condition,
361 number_of_points_per_span);
362
363 auto integration_info = IntegrationInfo(r_geometry.LocalSpaceDimension(), number_of_points_per_span, IntegrationInfo::QuadratureMethod::GAUSS);
364 r_geometry.CreateIntegrationPoints(integration_points,integration_info);
365 integration_method = integration_info.GetIntegrationMethod(0);
366 }
367 }
368
369 // Check condition variables
370 if (r_cond.Has(DISPLACEMENT))
371 mpc_imposed_displacement[0] = r_cond.GetValue(DISPLACEMENT);
372 if (r_cond.Has(VELOCITY))
373 mpc_imposed_velocity[0] = r_cond.GetValue(VELOCITY);
374 if (r_cond.Has(ACCELERATION))
375 mpc_imposed_acceleration[0] = r_cond.GetValue(ACCELERATION);
376 if (r_cond.Has(PENALTY_FACTOR))
377 mpc_penalty_factor[0] = r_cond.GetValue(PENALTY_FACTOR);
378
379 const bool is_slip = r_cond.Is(SLIP);
380 const bool is_contact = r_cond.Is(CONTACT);
381 const bool is_interface = r_cond.Is(INTERFACE);
382 const bool flip_normal_direction = r_cond.Is(MODIFIED);
383
384 std::string condition_type_name;
385
386 // If dirichlet boundary or coupling interface
387 if (!is_neumann_condition){
388 condition_type_name = "MPMParticlePenaltyDirichletCondition";
389 }
390 else{
391 if(r_cond.Has( POINT_LOAD ) ){
392 point_load[0] = r_cond.GetValue( POINT_LOAD );
393 condition_type_name = "MPMParticlePointLoadCondition";
394 }
395 else{
396 KRATOS_ERROR << "Material Point line load / surface load condition is not yet implemented." << std::endl;
397 }
398 }
399 // Get new condition
400 const Condition& new_condition = KratosComponents<Condition>::Get(condition_type_name);
401
402 // Check Normal direction
403 if (flip_normal_direction) mpc_normal *= -1.0;
404
405 // Create Material Point Point Load Condition
406 if (condition_type_name == "MPMParticlePointLoadCondition" ){
407 // create point load condition
408 mpc_area[0] = 1;
409
410
411 mpc_xg[0].clear();
412
413 for (unsigned int dimension = 0; dimension < r_geometry.WorkingSpaceDimension(); dimension++){
414 mpc_xg[0][dimension] = r_geometry[0].Coordinates()[dimension];
415 }
416
417 typename BinBasedFastPointLocator<TDimension>::ResultIteratorType result_begin = results.begin();
418 Element::Pointer pelem;
419 Vector N;
420
421 // FindPointOnMesh find the background element in which a given point falls and the relative shape functions
422 bool is_found = SearchStructure.FindPointOnMesh(mpc_xg[0], N, pelem, result_begin);
423 if (!is_found) KRATOS_WARNING("MaterialPointGeneratorUtility") << "::search failed." << std::endl;
424
425 auto p_new_geometry = CreateQuadraturePointsUtility<Node>::CreateFromCoordinates(
426 pelem->pGetGeometry(), mpc_xg[0],
427 mpc_area[0]);
428
429 Condition::Pointer p_condition = new_condition.Create(
430 condition_id, p_new_geometry, properties);
431
432
433 ProcessInfo process_info = ProcessInfo();
434 if (is_interface){
435 p_condition->Set(INTERFACE);
436 }
437
438 // Setting material point condition's initial condition
439 p_condition->SetValuesOnIntegrationPoints(MPC_COORD, mpc_xg , process_info);
440 p_condition->SetValuesOnIntegrationPoints(MPC_AREA, mpc_area, process_info);
441 p_condition->SetValuesOnIntegrationPoints(POINT_LOAD, { point_load }, process_info);
442 p_condition->SetValuesOnIntegrationPoints(MPC_DISPLACEMENT, { mpc_displacement }, process_info);
443 p_condition->SetValuesOnIntegrationPoints(MPC_DELTA_DISPLACEMENT, { mpc_delta_displacement }, process_info);
444 p_condition->SetValuesOnIntegrationPoints(MPC_VELOCITY, { mpc_velocity }, process_info);
445 p_condition->SetValuesOnIntegrationPoints(MPC_ACCELERATION, { mpc_acceleration }, process_info);
446 // Mark as boundary condition
447 p_condition->Set(BOUNDARY, true);
448
449 // Add the MP Condition to the model part
450 rMPMModelPart.GetSubModelPart(submodelpart_name).AddCondition(p_condition);
451 condition_id +=1;
452
453 }
454 // Loop over the conditions to create inner material point condition (except point load condition)
455 else{
456 std::vector<array_1d<double, 3>> xg_tmp;
457 std::vector<double> area_temp(1);
458 for ( IndexType i_integration_point = 0; i_integration_point < integration_points.size(); ++i_integration_point )
459 {
460 local_coordinates = integration_points[i_integration_point];
461
462 r_geometry.GlobalCoordinates(mpc_xg[0], local_coordinates);
463
464 mpc_area[0] = integration_points[i_integration_point].Weight() * r_geometry.DeterminantOfJacobian(i_integration_point, integration_method);
465
466 typename BinBasedFastPointLocator<TDimension>::ResultIteratorType result_begin = results.begin();
467 Element::Pointer pelem;
468 Vector N;
469 bool is_found = SearchStructure.FindPointOnMesh(mpc_xg[0], N, pelem, result_begin);
470 if (!is_found) KRATOS_WARNING("MaterialPointGeneratorUtility") << "::MPC search failed." << std::endl;
471
472 pelem->Set(ACTIVE);
473 auto p_quadrature_point_geometry = CreateQuadraturePointsUtility<Node>::CreateFromCoordinates(
474 pelem->pGetGeometry(), mpc_xg[0],
475 mpc_area[0]);
476
477 // Material point condition are not created twice
478 bool create_condition = true;
479 // loop only necessary for equal material points distribution to avoid doubled conditions
480 if (is_equal_int_volumes){
481 for(auto it=MaterialPointConditions.begin(); it!=MaterialPointConditions.end(); ++it)
482 {
483 it->CalculateOnIntegrationPoints(MPC_COORD, xg_tmp, rMPMModelPart.GetProcessInfo());
484
485 if (mpc_xg[0][0] == xg_tmp[0][0] && mpc_xg[0][1] == xg_tmp[0][1] && mpc_xg[0][2] == xg_tmp[0][2] )
486 {
487 create_condition = false;
488 it->CalculateOnIntegrationPoints(MPC_AREA, area_temp, rMPMModelPart.GetProcessInfo());
489 area_temp[0] += mpc_area[0];
490 it->SetValuesOnIntegrationPoints(MPC_AREA, area_temp, rMPMModelPart.GetProcessInfo());
491 }
492 }
493 }
494
495 if (create_condition){
496
497 Condition::Pointer p_condition = new_condition.Create(
498 condition_id, p_quadrature_point_geometry, properties);
499
500 MaterialPointConditions.push_back(p_condition);
501
502 ProcessInfo process_info = ProcessInfo();
503
504 // Setting material points condition's initial condition
505 //p_condition->SetValuesOnIntegrationPoints(MPC_CONDITION_ID, mpc_condition_id, process_info);
506 p_condition->SetValuesOnIntegrationPoints(MPC_COORD, mpc_xg , process_info);
507 p_condition->SetValuesOnIntegrationPoints(MPC_AREA, mpc_area , process_info);
508 p_condition->SetValuesOnIntegrationPoints(MPC_NORMAL, { mpc_normal }, process_info);
509 p_condition->SetValuesOnIntegrationPoints(MPC_DISPLACEMENT, { mpc_displacement }, process_info);
510 p_condition->SetValuesOnIntegrationPoints(MPC_IMPOSED_DISPLACEMENT, { mpc_imposed_displacement }, process_info);
511 p_condition->SetValuesOnIntegrationPoints(MPC_VELOCITY, { mpc_velocity }, process_info);
512 p_condition->SetValuesOnIntegrationPoints(MPC_IMPOSED_VELOCITY, { mpc_imposed_velocity }, process_info);
513 p_condition->SetValuesOnIntegrationPoints(MPC_ACCELERATION, { mpc_acceleration }, process_info);
514 p_condition->SetValuesOnIntegrationPoints(MPC_IMPOSED_ACCELERATION, { mpc_imposed_acceleration }, process_info);
515 p_condition->SetValuesOnIntegrationPoints(MPC_CONTACT_FORCE, mpc_contact_force , process_info);
516
517 // Mark as boundary condition
518 p_condition->Set(BOUNDARY, true);
519
520 if (boundary_condition_type == 1)
521 {
522 p_condition->SetValuesOnIntegrationPoints(PENALTY_FACTOR, mpc_penalty_factor , process_info);
523 }
524
525 if (is_slip)
526 p_condition->Set(SLIP);
527 if (is_contact)
528 p_condition->Set(CONTACT);
529 if (is_interface)
530 p_condition->Set(INTERFACE);
531
532 // Add the MP Condition to the model part
533 rMPMModelPart.GetSubModelPart(submodelpart_name).AddCondition(p_condition);
534 condition_id +=1;

826 {

837 void DetermineIntegrationMethodAndShapeFunctionValues(const GeometryType& rGeom, const SizeType MaterialPointsPerElement,
838 IntegrationMethod& rIntegrationMethod, Matrix& rN, bool& IsEqualVolumes)
839 {
840 const GeometryData::KratosGeometryType geo_type = rGeom.GetGeometryType();
841 const SizeType domain_size = rGeom.WorkingSpaceDimension();
842
843 if (geo_type == GeometryData::KratosGeometryType::Kratos_Tetrahedra3D4 || geo_type == GeometryData::KratosGeometryType::Kratos_Triangle2D3)
844 {
845 switch (MaterialPointsPerElement)
846 {
847 case 1:
848 rIntegrationMethod = GeometryData::IntegrationMethod::GI_GAUSS_1;
849 break;
850 case 3:
851 rIntegrationMethod = GeometryData::IntegrationMethod::GI_GAUSS_2;
852 break;
853 case 6:
854 rIntegrationMethod = GeometryData::IntegrationMethod::GI_GAUSS_4;
855 break;
856 case 12:
857 rIntegrationMethod = GeometryData::IntegrationMethod::GI_GAUSS_5;
858 break;
859 case 16:
860 if (domain_size == 2) {
861 IsEqualVolumes = true;
862
863 KRATOS_WARNING("MaterialPointGeneratorUtility") << "16 material points per triangle element is only valid for undistorted triangles." << std::endl;
864 rN = MP16ShapeFunctions();
865 break;
866 }
867 case 33:
868 if (domain_size == 2) {
869 IsEqualVolumes = true;
870 KRATOS_WARNING("MaterialPointGeneratorUtility") << "33 material points per triangle element is only valid for undistorted triangles." << std::endl;
871 rN = MP33ShapeFunctions();
872 break;
873 }
874 default:
875 rIntegrationMethod = GeometryData::IntegrationMethod::GI_GAUSS_2; // default to 3 material points per tri
876
877 std::string warning_msg = "The input number of MATERIAL_POINTS_PER_ELEMENT: " + std::to_string(MaterialPointsPerElement);
878 warning_msg += " is not available for Triangular" + std::to_string(domain_size) + "D.\n";
879 warning_msg += "Available options are: 1, 3, 6, 12, 16 (only 2D), and 33 (only 2D).\n";
880 warning_msg += "The default number of material points: 3 is currently assumed.";
881 KRATOS_WARNING("MaterialPointGeneratorUtility") << warning_msg << std::endl;
882 break;
883 }
884 }
885 else if (geo_type == GeometryData::KratosGeometryType::Kratos_Hexahedra3D8 || geo_type == GeometryData::KratosGeometryType::Kratos_Quadrilateral2D4)
886 {
887 switch (MaterialPointsPerElement)
888 {
889 case 1:
890 rIntegrationMethod = GeometryData::IntegrationMethod::GI_GAUSS_1;
891 break;
892 case 4:
893 rIntegrationMethod = GeometryData::IntegrationMethod::GI_GAUSS_2;
894 break;
895 case 9:
896 rIntegrationMethod = GeometryData::IntegrationMethod::GI_GAUSS_3;
897 break;
898 case 16:
899 rIntegrationMethod = GeometryData::IntegrationMethod::GI_GAUSS_4;
900 break;
901 default:
902 rIntegrationMethod = GeometryData::IntegrationMethod::GI_GAUSS_2; // default to 4 material points per quad
903
904 std::string warning_msg = "The input number of MATERIAL_POINTS_PER_ELEMENT: " + std::to_string(MaterialPointsPerElement);
905 warning_msg += " is not available for Quadrilateral" + std::to_string(domain_size) + "D.\n";
906 warning_msg += "Available options are: 1, 4, 9, 16.\n";
907 warning_msg += "The default number of material points: 4 is currently assumed.";
908 KRATOS_WARNING("MaterialPointGeneratorUtility") << warning_msg << std::endl;
909 break;
910 }
911 }
912
913 // Get shape function values
914 if (!IsEqualVolumes) rN = rGeom.ShapeFunctionsValues(rIntegrationMethod);
