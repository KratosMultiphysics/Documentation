30 ApplyConstantBoundaryHydrostaticPressureProcess(ModelPart& model_part, Parameters rParameters)

33 KRATOS_TRY

35 // only include validation with c++11 since raw_literals do not exist in c++03
36 Parameters default_parameters(R"(

38 "model_part_name":"PLEASE_CHOOSE_MODEL_PART_NAME",

40 "is_fixed": false,

45 } )");
46
47 // Some values have to be input by the user since no meaningful default value exist. For
48 // this reason, we try to access them, so that an error is thrown if they don't exist.
49 rParameters["reference_coordinate"];
50 rParameters["variable_name"];
51 rParameters["model_part_name"];

73 void ExecuteInitialize() override
74 {
75 KRATOS_TRY
76
77 const auto& r_variable = KratosComponents<Variable<double>>::Get(GetVariableName());
78
79 block_for_each(GetModelPart().Nodes(), [&r_variable, this](Node& rNode) {
80 if (mIsFixed) rNode.Fix(r_variable);
81 else if (mIsFixedProvided) rNode.Free(r_variable);
82
83 const auto pressure = GetSpecificWeight() * (GetReferenceCoordinate() -
84 rNode.Coordinates()[GetGravityDirection()]);
85 rNode.FastGetSolutionStepValue(r_variable) = std::max(pressure, 0.);
86 });
87
88 KRATOS_CATCH("")
89 }
