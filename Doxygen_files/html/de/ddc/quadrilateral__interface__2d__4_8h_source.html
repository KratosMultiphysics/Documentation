220 typename PointType::Pointer pSecondPoint,
221 typename PointType::Pointer pThirdPoint,
222 typename PointType::Pointer pFourthPoint )
223 : BaseType( PointsArrayType(), &msGeometryData )
224 {
225 double p2_p1_X = (pThirdPoint->X() + pFourthPoint->X() - pFirstPoint->X() - pSecondPoint->X())*0.5;
226 double p2_p1_Y = (pThirdPoint->Y() + pFourthPoint->Y() - pFirstPoint->Y() - pSecondPoint->Y())*0.5;
227 double p4_p3_X = (pThirdPoint->X() + pSecondPoint->X() - pFirstPoint->X() - pFourthPoint->X())*0.5;
228 double p4_p3_Y = (pThirdPoint->Y() + pSecondPoint->Y() - pFirstPoint->Y() - pFourthPoint->Y())*0.5;
229
230 double lx = std::sqrt(p4_p3_X*p4_p3_X + p4_p3_Y*p4_p3_Y);
231 double ly = std::sqrt(p2_p1_X*p2_p1_X + p2_p1_Y*p2_p1_Y);
232
233 if(ly < lx)
234 {
235 this->Points().push_back( pFirstPoint );
236 this->Points().push_back( pSecondPoint );
237 this->Points().push_back( pThirdPoint );
238 this->Points().push_back( pFourthPoint );
239 }

242 this->Points().push_back( pFourthPoint );
243 this->Points().push_back( pFirstPoint );

495 CoordinatesArrayType& PointLocalCoordinates(
496 CoordinatesArrayType& rResult,
497 const CoordinatesArrayType& rPoint
498 ) const override
499 {
500 rResult.clear();
501
502 array_1d<double, 3> FirstPoint;
503 noalias(FirstPoint) = 0.5 * (BaseType::GetPoint( 0 ) + BaseType::GetPoint( 3 ));
504 array_1d<double, 3> SecondPoint;
505 noalias(SecondPoint) = 0.5 * (BaseType::GetPoint( 1 ) + BaseType::GetPoint( 2 ));
506
507 // Project point
508 double tol = 1e-14; // Tolerance
509
510 // Normal
511 array_1d<double,2> Normal = ZeroVector(2);
512 Normal[0] = SecondPoint[1] - FirstPoint[1];
513 Normal[1] = FirstPoint[0] - SecondPoint[0];
514 double norm = std::sqrt(Normal[0] * Normal[0] + Normal[1] * Normal[1]);
515 Normal /= norm;
516
517 // Vector point and distance
518 array_1d<double,2> VectorPoint = ZeroVector(2);
519 VectorPoint[0] = rPoint[0] - FirstPoint[0];
520 VectorPoint[1] = rPoint[1] - FirstPoint[1];
521 double dist_proy = VectorPoint[0] * Normal[0] + VectorPoint[1] * Normal[1];
522
523 if (dist_proy < tol)
524 {
525 double L = Length();
526
527 double l1 = (rPoint[0] - FirstPoint[0]) * (rPoint[0] - FirstPoint[0])
528 + (rPoint[1] - FirstPoint[1]) * (rPoint[1] - FirstPoint[1]);
529 l1 = std::sqrt(l1);
530
531 double l2 = (rPoint[0] - SecondPoint[0]) * (rPoint[0] - SecondPoint[0])
532 + (rPoint[1] - SecondPoint[1]) * (rPoint[1] - SecondPoint[1]);
533 l2 = std::sqrt(l2);
534
535// std::cout << "L: " << L << " l1: " << l1 << " l2: " << l2 << std::endl;
536
537 if (l1 <= (L + tol) && l2 <= (L + tol))
538 {
539 rResult[0] = 2.0 * l1/(L + tol) - 1.0;
540 }
541 else
542 {
543 rResult[0] = 2.0; // Out of the line!!! TODO: Check if this value gives problems
544 }
545 }
546 else
547 {
548 rResult[0] = 2.0; // Out of the line!!!
549 }
550
551 return( rResult );
552 }

608 Matrix& Jacobian( Matrix& rResult,
609 IndexType IntegrationPointIndex,
610 IntegrationMethod ThisMethod,
611 const Matrix& DeltaPosition ) const override
612 {
613 array_1d<double, 3> p0 = 0.5 * (BaseType::GetPoint( 0 ) + BaseType::GetPoint( 3 ));
614 array_1d<double, 3> p1 = 0.5 * (BaseType::GetPoint( 1 ) + BaseType::GetPoint( 2 ));
615
616 array_1d<double, 2> dp0;
617 array_1d<double, 2> dp1;
618 dp0[0] = (DeltaPosition(0,0) + DeltaPosition(3,0))*0.5;
619 dp0[1] = (DeltaPosition(0,1) + DeltaPosition(3,1))*0.5;
620 dp1[0] = (DeltaPosition(1,0) + DeltaPosition(2,0))*0.5;
621 dp1[1] = (DeltaPosition(1,1) + DeltaPosition(2,1))*0.5;
622
623 if(rResult.size1() != 2 || rResult.size2() != 1)
624 rResult.resize(2, 1, false);
625 rResult( 0, 0 ) = ( (p1[0]-dp1[0]) - (p0[0]-dp0[0]) ) * 0.5;
626 rResult( 1, 0 ) = ( (p1[1]-dp1[1]) - (p0[1]-dp0[1]) ) * 0.5;
627
628 return rResult;
