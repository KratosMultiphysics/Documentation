422 CoordinatesArrayType& PointLocalCoordinates(CoordinatesArrayType& rResult,
423 const CoordinatesArrayType& rPoint) const override
424 {
425 BoundedMatrix<double, 3, 4> X;
426 BoundedMatrix<double, 3, 1> DN;
427 for (IndexType i = 0; i < this->size(); ++i)
428 {
429 const auto& r_node = this->GetPoint(i);
430 X(0, i) = r_node.X();
431 X(1, i) = r_node.Y();
432 X(2, i) = r_node.Z();
433 }
434
435 static constexpr double MaxNormPointLocalCoordinates = 300.0;
436 static constexpr std::size_t MaxIteratioNumberPointLocalCoordinates = 500;
437 static constexpr double MaxTolerancePointLocalCoordinates = 1.0e-8;
438
439 Matrix J = ZeroMatrix(1, 1);
440 Matrix invJ = ZeroMatrix(1, 1);
441
442 // Starting with xi = 0
443 if (rResult.size() != 3) rResult.resize(3, false);
444 noalias(rResult) = ZeroVector(3);
445 double delta_xi = 0.0;
446 const array_1d<double, 3> zero_array = ZeroVector(3);
447 array_1d<double, 3> current_global_coords;
448 array_1d<double, 1> res;
449
450 //Newton iteration:
451 for (IndexType k = 0; k < MaxIteratioNumberPointLocalCoordinates; ++k) {
452 noalias(current_global_coords) = zero_array;
453 this->GlobalCoordinates(current_global_coords, rResult);
454
455 noalias(current_global_coords) = rPoint - current_global_coords;
456
457 // Derivatives of shape functions
458 Matrix shape_functions_gradients;
459 shape_functions_gradients = ShapeFunctionsLocalGradients(shape_functions_gradients, rResult);
460 noalias(DN) = prod(X, shape_functions_gradients);
461 noalias(J) = prod(trans(DN), DN);
462 noalias(res) = prod(trans(DN), current_global_coords);
463
464 // The inverted jacobian matrix
465 invJ(0, 0) = 1.0 / J(0, 0);
466
467 delta_xi = invJ(0, 0) * res[0];
468
469 rResult[0] += delta_xi;
470
471 if (delta_xi > MaxNormPointLocalCoordinates) {
472 KRATOS_WARNING_IF("Line2D4", k > 0) << "detJ =\t" << J(0, 0) << " DeltaX =\t" << delta_xi << " stopping calculation. Iteration:\t" << k << std::endl;
473 break;
474 }
475
476 if (delta_xi < MaxTolerancePointLocalCoordinates)
477 break;
478 }
479
480 return rResult;
481 }

501 JacobiansType& Jacobian(JacobiansType& rResult, IntegrationMethod ThisMethod) const override
502 {
503 // Getting derivatives of shape functions
504 const ShapeFunctionsGradientsType shape_functions_gradients
505 = CalculateShapeFunctionsIntegrationPointsLocalGradients(ThisMethod);
506 const std::size_t number_of_integration_points = this->IntegrationPointsNumber(ThisMethod);
507 if (rResult.size() != number_of_integration_points) {
508 JacobiansType temp(number_of_integration_points);
509 rResult.swap(temp);
510 }
511 // Loop over all integration points
512 for (std::size_t pnt = 0; pnt < number_of_integration_points; ++pnt) {
513 // Initializing jacobian matrix
514 noalias(rResult[pnt]) = ZeroMatrix(2, 1);
515 // Loop over all nodes
516 for (std::size_t i = 0; i < this->PointsNumber(); ++i) {
517 const auto& r_node = this->GetPoint(i);
518 rResult[pnt](0, 0) += r_node.X() * shape_functions_gradients[pnt](i, 0);
519 rResult[pnt](1, 0) += r_node.Y() * shape_functions_gradients[pnt](i, 0);
520 }
521 } // End of loop over all integration points
522 return rResult;
523 }
