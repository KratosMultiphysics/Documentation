30template <class TSparseSpace, class TDenseSpace, class TLinearSolver>
31class SolvingStrategyFactory
32{
33public:
34 [[nodiscard]] static std::unique_ptr<SolvingStrategy<TSparseSpace, TDenseSpace>> Create(const Parameters& rSolverSettings,
35 ModelPart& rModelPart)
36 {
37 const std::string strategy_type = "strategy_type";
38
39 KRATOS_ERROR_IF_NOT(rSolverSettings.Has(strategy_type))
40 << "The parameter strategy_type is undefined, aborting.";
41
42 const auto echo_level = rSolverSettings["echo_level"].GetInt();
43
44 auto solver = LinearSolverFactory<TSparseSpace, TDenseSpace>().Create(
45 rSolverSettings["linear_solver_settings"]);
46 KRATOS_ERROR_IF_NOT(solver) << "Failed to create a linear solver" << std::endl;
47
48 auto scheme = SchemeFactory<TSparseSpace, TDenseSpace>::Create(rSolverSettings);
49 KRATOS_ERROR_IF_NOT(scheme) << "Failed to create a scheme" << std::endl;
50
51 auto builder_and_solver =
52 BuilderAndSolverFactory<TSparseSpace, TDenseSpace, TLinearSolver>::Create(rSolverSettings, solver);
53 KRATOS_ERROR_IF_NOT(builder_and_solver) << "Failed to create a builder-and-solver" << std::endl;
54 builder_and_solver->SetEchoLevel(echo_level);
55
56 auto criteria = ConvergenceCriteriaFactory<TSparseSpace, TDenseSpace>::Create(rSolverSettings);
57 KRATOS_ERROR_IF_NOT(criteria) << "Failed to create convergence criteria" << std::endl;
58 criteria->SetEchoLevel(echo_level);
59
60 if (rSolverSettings[strategy_type].GetString() == "newton_raphson") {
61 const auto max_iterations = rSolverSettings["max_iterations"].GetInt();
62 const auto calculate_reactions = rSolverSettings["calculate_reactions"].GetBool();
63 const auto reform_dof_set_at_each_step = rSolverSettings["reform_dofs_at_each_step"].GetBool();
64 const auto move_mesh_flag = rSolverSettings["move_mesh_flag"].GetBool();
65
66 const std::vector<std::string> strategy_entries = {"loads_sub_model_part_list",
67 "loads_variable_list"};
68 auto strategy_parameters =
69 ParametersUtilities::CopyOptionalParameters(rSolverSettings, strategy_entries);
70 auto result =
71 std::make_unique<GeoMechanicsNewtonRaphsonStrategy<TSparseSpace, TDenseSpace, TLinearSolver>>(
72 rModelPart, scheme, criteria, builder_and_solver, strategy_parameters,
73 max_iterations, calculate_reactions, reform_dof_set_at_each_step, move_mesh_flag);
74 result->SetEchoLevel(echo_level);
75 return result;
76 } else if (rSolverSettings[strategy_type].GetString() == "line_search") {
77 const std::vector<std::string> strategy_entries = {"compute_reactions",
78 "max_line_search_iterations",
79 "first_alpha_value",
80 "second_alpha_value",
81 "min_alpha",
82 "max_alpha",
83 "line_search_tolerance",
84 "move_mesh_flag",
85 "reform_dofs_at_each_step",
86 "echo_level"};
87
88 auto strategy_parameters =
89 ParametersUtilities::CopyOptionalParameters(rSolverSettings, strategy_entries);
90
91 // For residual-based Newton-Raphson strategies (and derived ones), the Kratos Core uses
92 // the value of "max_iteration". In the GeoMechanicsApplication, we use the value of
93 // "max_iterations" (note the extra 's' at the end). This is inconsistent, and to
94 // overcome it we use a similar approach as in our Python scripts:
95 if (rSolverSettings.Has("max_iterations")) {
96 strategy_parameters.AddValue("max_iteration", rSolverSettings["max_iterations"]);
97 }
98
99 auto result = std::make_unique<LineSearchStrategy<TSparseSpace, TDenseSpace, TLinearSolver>>(
100 rModelPart, scheme, solver, criteria, strategy_parameters);
101 return result;
102 }
103
104 return nullptr;
105 }
106};
