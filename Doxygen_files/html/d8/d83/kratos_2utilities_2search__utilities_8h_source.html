175 template<class TPointType>
176 static bool PointIsInsideBoundingBox(
177 const BoundingBox<TPointType>& rBoundingBox,
178 const array_1d<double, 3>& rCoords
179 )
180 {
181 // Get the bounding box points
182 const auto& r_max_point = rBoundingBox.GetMaxPoint();
183 const auto& r_min_point = rBoundingBox.GetMinPoint();
184
185 // The Bounding Box check
186 if (rCoords[0] < r_max_point[0] && rCoords[0] > r_min_point[0]) // check x-direction
187 if (rCoords[1] < r_max_point[1] && rCoords[1] > r_min_point[1]) // check y-direction
188 if (rCoords[2] < r_max_point[2] && rCoords[2] > r_min_point[2]) // check z-direction
189 return true;
190 return false;
191 }

221 template<class TPointType>
222 static bool PointIsInsideBoundingBox(
223 const BoundingBox<TPointType>& rBoundingBox,
224 const array_1d<double, 3>& rCoords,
225 const double Tolerance
226 )
227 {
228 // Check if the tolerance is negative, if negative we don't even check
229 if (Tolerance < 0) {
230 return false;
231 }
232
233 // Get the bounding box points
234 auto max_point = rBoundingBox.GetMaxPoint();
235 auto min_point = rBoundingBox.GetMinPoint();
236
237 // Apply Tolerances
238 for (unsigned int i = 0; i < 3; ++i) {
239 max_point[i] += Tolerance;
240 min_point[i] -= Tolerance;
241 }
242
243 // The Bounding Box check
244 if (rCoords[0] < max_point[0] && rCoords[0] > min_point[0]) // check x-direction
245 if (rCoords[1] < max_point[1] && rCoords[1] > min_point[1]) // check y-direction
246 if (rCoords[2] < max_point[2] && rCoords[2] > min_point[2]) // check z-direction
247 return true;
248 return false;
249 }

469 template<class TContainer>
470 static std::vector<typename PointObject<typename TContainer::value_type>::Pointer> PreparePointsSearch(
471 const TContainer& rStructure,
472 const int Rank = -1
473 )
474 {
475 // Some definitions
476 using ObjectType = typename TContainer::value_type;
477 using PointType = PointObject<ObjectType>;
478 using PointTypePointer = typename PointType::Pointer;
479 using PointVector = std::vector<PointTypePointer>;
480
481 // Creating the points
482 PointVector points;
483 const std::size_t structure_size = rStructure.size();
484 points.reserve(structure_size);
485 const auto it_begin = rStructure.begin();
486 if constexpr (std::is_same<TContainer, ModelPart::NodesContainerType>::value) { // For nodes
487 const bool serial_check = (Rank < 0 || !it_begin->SolutionStepsDataHas(PARTITION_INDEX)) ? true : false;
488 if (serial_check) { // Checking if Rank is defined
489 for (std::size_t i = 0; i < structure_size; ++i) {
490 auto it = it_begin + i;
491 points.push_back(PointTypePointer(new PointType(*(it.base()))));
492 }
493 } else { // When rank is defined nodes are added only if local
494 for (std::size_t i = 0; i < structure_size; ++i) {
495 auto it = it_begin + i;
496 if ( it->FastGetSolutionStepValue(PARTITION_INDEX) == Rank) {
497 points.push_back(PointTypePointer(new PointType(*(it.base()))));
498 }
499 }
500 // Reduce size
501 points.shrink_to_fit();
502 }
503 } else { // For other entities
504 for (std::size_t i = 0; i < structure_size; ++i) {
505 auto it = it_begin + i;
506 points.push_back(PointTypePointer(new PointType(*(it.base()))));
507 }
508 }
509
510 return points;
511 }

525 template<class TContainer, class TSpatialContainer, class TResultType>
526 static void ParallelSearch(
527 const TContainer& rInput,
528 const RadiusArrayType& rRadius,
529 TSpatialContainer& rSearch,
530 TResultType& rResults,
531 VectorDistanceType& rResultsDistance,
532 const int AllocationSize = 1000
533 )
534 {
535 // Some definitions
536 using ObjectType = typename TContainer::value_type;
537 using PointType = PointObject<ObjectType>;
538 using PointTypePointer = typename PointType::Pointer;
539 using PointVector = std::vector<PointTypePointer>;
540 using DistanceVector = std::vector<double>;
541 const std::size_t input_size = rInput.size();
542
543 // Performing search
544 IndexPartition<std::size_t>(input_size).for_each([&](std::size_t i) {
545 auto it = rInput.begin() + i;
546 PointType aux_point(*(it.base()));
547 PointVector results(AllocationSize);
548 DistanceVector results_distances(AllocationSize);
549 const std::size_t number_of_results = rSearch.SearchInRadius(aux_point, rRadius[i], results.begin(), results_distances.begin(), AllocationSize);
550 if (number_of_results > 0) {
551 auto& r_results = rResults[i];
552 auto& r_results_distance = rResultsDistance[i];
553 r_results.reserve(number_of_results);
554 r_results_distance.reserve(number_of_results);
555 for (std::size_t j = 0; j < number_of_results; ++j) {
556 auto p_point = results[j];
557 r_results.push_back(p_point->pGetObject());
558 r_results_distance.push_back(results_distances[j]);
559 }
560 }
561 });
562 }

582 template<typename TPointIteratorType>
583 static std::vector<int> SynchronizePoints(
584 TPointIteratorType itPointBegin,
585 TPointIteratorType itPointEnd,
586 std::vector<double>& rAllPointsCoordinates,
587 std::vector<SignedIndexType>& rAllPointsIds,
588 const DataCommunicator& rDataCommunicator,
589 const int NumberOfPoints,
590 const int TotalNumberOfPoints
591 )
592 {
593 // Get the World Size and rank in MPI
594 const int world_size = rDataCommunicator.Size();
595 const int rank = rDataCommunicator.Rank();
596
597 // Getting global number of points
598 std::vector<int> points_per_partition(world_size);
599 std::vector<int> send_points_per_partition(1, NumberOfPoints);
600 rDataCommunicator.AllGather(send_points_per_partition, points_per_partition);
601 std::size_t initial_id = 0;
602 if constexpr (!std::is_same<TPointIteratorType, ModelPart::NodeIterator>::value || std::is_same<TPointIteratorType, ModelPart::NodeConstantIterator>::value) {
603 initial_id = std::accumulate(points_per_partition.begin(), points_per_partition.begin() + rank, 0);
604 }
605
606 // Initialize and resize vectors
607 if (rAllPointsCoordinates.size() != static_cast<unsigned int>(TotalNumberOfPoints * 3)) {
608 rAllPointsCoordinates.resize(TotalNumberOfPoints * 3);
609 }
610 if (rAllPointsIds.size() != static_cast<unsigned int>(TotalNumberOfPoints)) {
611 rAllPointsIds.resize(TotalNumberOfPoints);
612 }
613 std::vector<int> recv_sizes(world_size, 0);
614
615 // Auxiliary variables
616 std::size_t counter = 0;
617 array_1d<double, 3> coordinates;
618 unsigned int i_coord;
619
620 // If distributed
621 if (rDataCommunicator.IsDistributed()) {
622 // Initialize local points coordinates
623 std::vector<double> send_points_coordinates(NumberOfPoints * 3);
624 std::vector<IndexType> send_points_ids(NumberOfPoints);
625 for (auto it_point = itPointBegin ; it_point != itPointEnd ; ++it_point) {
626 // In case of considering nodes
627 if constexpr (std::is_same<TPointIteratorType, ModelPart::NodeIterator>::value || std::is_same<TPointIteratorType, ModelPart::NodeConstantIterator>::value) {
628 if (it_point->FastGetSolutionStepValue(PARTITION_INDEX) != rank) {
629 continue; // Skip if not local
630 }
631 }
632 noalias(coordinates) = it_point->Coordinates();
633 if constexpr (std::is_same<TPointIteratorType, ModelPart::NodeIterator>::value || std::is_same<TPointIteratorType, ModelPart::NodeConstantIterator>::value) {
634 // Verify that Index is not bigger than the maximum positive values of SignedIndexType
635 KRATOS_DEBUG_ERROR_IF(it_point->Id() > static_cast<IndexType>(std::numeric_limits<SignedIndexType>::max())) << "Index is bigger than the maximum positive value of SignedIndexType" << std::endl;
636 send_points_ids[counter] = it_point->Id();
637 } else {
638 // Verify that Index is not bigger than the maximum positive values of SignedIndexType
639 KRATOS_DEBUG_ERROR_IF(initial_id + counter > static_cast<IndexType>(std::numeric_limits<SignedIndexType>::max())) << "Index is bigger than the maximum positive value of SignedIndexType" << std::endl;
640 send_points_ids[counter] = initial_id + counter;
641 }
642 for (i_coord = 0; i_coord < 3; ++i_coord) {
643 send_points_coordinates[3 * counter + i_coord] = coordinates[i_coord];
644 }
645 ++counter;
646 }
647
648 /* Sync coordinates */
649
650 // Generate vectors with sizes for AllGatherv
651 for (int i_rank = 0; i_rank < world_size; ++i_rank) {
652 recv_sizes[i_rank] = 3 * points_per_partition[i_rank];
653 }
654 std::vector<int> recv_offsets(world_size, 0);
655 for (int i_rank = 1; i_rank < world_size; ++i_rank) {
656 recv_offsets[i_rank] = recv_offsets[i_rank - 1] + recv_sizes[i_rank - 1];
657 }
658
659 // Invoke AllGatherv
660 rDataCommunicator.AllGatherv(send_points_coordinates, rAllPointsCoordinates, recv_sizes, recv_offsets);
661
662 /* Sync Ids */
663
664 // Generate vectors with sizes for AllGatherv
665 for (int i_rank = 0; i_rank < world_size; ++i_rank) {
666 recv_sizes[i_rank] = points_per_partition[i_rank];
667 }
668 for (int i_rank = 1; i_rank < world_size; ++i_rank) {
669 recv_offsets[i_rank] = recv_offsets[i_rank - 1] + recv_sizes[i_rank - 1];
670 }
671
672 // Invoke AllGatherv
673 std::vector<IndexType> all_points_pure_ids(TotalNumberOfPoints);
674 rDataCommunicator.AllGatherv(send_points_ids, all_points_pure_ids, recv_sizes, recv_offsets);
675
676 // Copy values
677 IndexPartition<std::size_t>(TotalNumberOfPoints).for_each([&all_points_pure_ids, &rAllPointsIds](std::size_t i) {
678 rAllPointsIds[i] = all_points_pure_ids[i];
679 });
680 } else { // Serial
681 // Assign values
682 for (auto it_point = itPointBegin ; it_point != itPointEnd ; ++it_point) {
683 noalias(coordinates) = it_point->Coordinates();
684 if constexpr (std::is_same<TPointIteratorType, ModelPart::NodeIterator>::value || std::is_same<TPointIteratorType, ModelPart::NodeConstantIterator>::value) {
685 // Verify that Index is not bigger than the maximum positive values of SignedIndexType
686 KRATOS_DEBUG_ERROR_IF(it_point->Id() > static_cast<IndexType>(std::numeric_limits<SignedIndexType>::max())) << "Index is bigger than the maximum positive value of SignedIndexType" << std::endl;
687 rAllPointsIds[counter] = it_point->Id();
688 } else {
689 // Verify that Index is not bigger than the maximum positive values of SignedIndexType
690 KRATOS_DEBUG_ERROR_IF(initial_id + counter > static_cast<IndexType>(std::numeric_limits<SignedIndexType>::max())) << "Index is bigger than the maximum positive value of SignedIndexType" << std::endl;
691 rAllPointsIds[counter] = initial_id + counter;
692 }
693 for (i_coord = 0; i_coord < 3; ++i_coord) {
694 rAllPointsCoordinates[3 * counter + i_coord] = coordinates[i_coord];
695 }
696 ++counter;
697 }
698 }
699
700 // Return the recv sizes
701 return recv_sizes;
702 }

772 if constexpr (std::is_same<TPointIteratorType, ModelPart::NodeIterator>::value || std::is_same<TPointIteratorType, ModelPart::NodeConstantIterator>::value) {

780
781 // In case of considering nodes
782 if constexpr (std::is_same<TPointIteratorType, ModelPart::NodeIterator>::value || std::is_same<TPointIteratorType, ModelPart::NodeConstantIterator>::value) {
783 if (it_point->FastGetSolutionStepValue(PARTITION_INDEX) != rank) {
784 continue; // Skip if not local
785 } else if (IndexItIsJustPosition) {
786 local_points_ids.push_back({it_point->Id(), i_position});

788 }
789 const auto& r_coordinates = it_point->Coordinates();
790 if constexpr (std::is_same<TPointIteratorType, ModelPart::NodeIterator>::value || std::is_same<TPointIteratorType, ModelPart::NodeConstantIterator>::value) {

792 KRATOS_DEBUG_ERROR_IF(it_point->Id() > static_cast<IndexType>(std::numeric_limits<SignedIndexType>::max())) << "Index is bigger than the maximum positive value of SignedIndexType" << std::endl;
793 send_points_ids[counter] = it_point->Id();
794 } else {
795 // Verify that Index is not bigger than the maximum positive values of SignedIndexType
796 KRATOS_DEBUG_ERROR_IF(initial_id + counter > static_cast<IndexType>(std::numeric_limits<SignedIndexType>::max())) << "Index is bigger than the maximum positive value of SignedIndexType" << std::endl;
797 send_points_ids[counter] = initial_id + counter;
798 }
799 send_points_ranks[counter] = rank;
800 for (i_coord = 0; i_coord < 3; ++i_coord) {
801 send_points_coordinates[3 * counter + i_coord] = r_coordinates[i_coord];
802 }
803 ++counter;
804 }
805
806 /* Sync coordinates */
807
808 // Generate vectors with sizes for AllGatherv
809 for (int i_rank = 0; i_rank < world_size; ++i_rank) {
810 recv_sizes[i_rank] = 3 * points_per_partition[i_rank];
811 }
812 std::vector<int> recv_offsets(world_size, 0);
813 for (int i_rank = 1; i_rank < world_size; ++i_rank) {
814 recv_offsets[i_rank] = recv_offsets[i_rank - 1] + recv_sizes[i_rank - 1];
815 }
816
817 // Invoke AllGatherv
818 rDataCommunicator.AllGatherv(send_points_coordinates, rAllPointsCoordinates, recv_sizes, recv_offsets);
819
820 /* Sync Ids */
821
822 // Generate vectors with sizes for AllGatherv
823 for (int i_rank = 0; i_rank < world_size; ++i_rank) {
824 recv_sizes[i_rank] = points_per_partition[i_rank];
825 }
826 for (int i_rank = 1; i_rank < world_size; ++i_rank) {
827 recv_offsets[i_rank] = recv_offsets[i_rank - 1] + recv_sizes[i_rank - 1];
828 }
829
830 // Invoke AllGatherv
831 rDataCommunicator.AllGatherv(send_points_ranks, all_points_ranks, recv_sizes, recv_offsets);
832 rDataCommunicator.AllGatherv(send_points_ids, rAllPointsGlobalIds, recv_sizes, recv_offsets);
833
834 // Some additional post-process
835 if constexpr (std::is_same<TPointIteratorType, ModelPart::NodeIterator>::value || std::is_same<TPointIteratorType, ModelPart::NodeConstantIterator>::value) {
836 if (IndexItIsJustPosition) {
837 // Set proper indices
838 std::fill(rAllPointsLocalIds.begin(), rAllPointsLocalIds.end(), -1);
839 const auto it_pure_ids_begin = rAllPointsGlobalIds.begin();
840 const auto it_pure_ids_end = rAllPointsGlobalIds.end();
841 for (auto& r_pair : local_points_ids) {
842 const auto& index = r_pair.first;
843 const auto it_points_pure_ids_found = std::find(rAllPointsGlobalIds.begin(), rAllPointsGlobalIds.end(), index);
844 if (it_points_pure_ids_found != it_pure_ids_end) {
845 const auto i_point = std::distance(it_pure_ids_begin, it_points_pure_ids_found);
846 const auto& position = r_pair.second;
847 rAllPointsLocalIds[i_point] = position;
848 }
849 }
850 } else {
851 // Copy values
852 IndexPartition<std::size_t>(TotalNumberOfPoints).for_each([&rAllPointsGlobalIds, &rAllPointsLocalIds](std::size_t i) {
853 rAllPointsLocalIds[i] = rAllPointsGlobalIds[i];
854 });
855 }
856 } else {
857 // Copy values
858 IndexPartition<std::size_t>(TotalNumberOfPoints).for_each([&rAllPointsGlobalIds, &rAllPointsLocalIds](std::size_t i) {
859 rAllPointsLocalIds[i] = rAllPointsGlobalIds[i];
860 });
861 }
862 } else { // Serial
863 // Assign values
864 for (auto it_point = itPointBegin ; it_point != itPointEnd ; ++it_point) {
865 const auto& r_coordinates = it_point->Coordinates();
866 if constexpr (std::is_same<TPointIteratorType, ModelPart::NodeIterator>::value || std::is_same<TPointIteratorType, ModelPart::NodeConstantIterator>::value) {
867 // Verify that Index is not bigger than the maximum positive values of SignedIndexType
868 KRATOS_DEBUG_ERROR_IF(it_point->Id() > static_cast<IndexType>(std::numeric_limits<SignedIndexType>::max())) << "Index is bigger than the maximum positive value of SignedIndexType" << std::endl;
869 rAllPointsGlobalIds[counter] = it_point->Id();
870 } else {
871 // Verify that Index is not bigger than the maximum positive values of SignedIndexType
872 KRATOS_DEBUG_ERROR_IF(initial_id + counter > static_cast<IndexType>(std::numeric_limits<SignedIndexType>::max())) << "Index is bigger than the maximum positive value of SignedIndexType" << std::endl;
873 rAllPointsGlobalIds[counter] = initial_id + counter;
874 }
875 for (i_coord = 0; i_coord < 3; ++i_coord) {
876 rAllPointsCoordinates[3 * counter + i_coord] = r_coordinates[i_coord];
877 }
878 ++counter;
879 }
880 // Copy values
881 IndexPartition<std::size_t>(TotalNumberOfPoints).for_each([&rAllPointsLocalIds](std::size_t i) {
882 rAllPointsLocalIds[i] = i;
883 });
884 }
885
886 // Return the ranks
887 return all_points_ranks;

906 template<typename TPointIteratorType, typename TBoundingBoxType>
907 static void SynchronizePointsWithBoundingBox(
908 TPointIteratorType itPointBegin,
909 TPointIteratorType itPointEnd,
910 DistributedSearchInformation& rSearchInfo,
911 const TBoundingBoxType& rBoundingBox,
912 const double ThresholdBoundingBox,
913 const DataCommunicator& rDataCommunicator,
914 const int NumberOfPoints,
915 const int TotalNumberOfPoints,
916 const bool IndexItIsJustPosition = false
917 )
918 {
919 // Initialize and resize vectors
920 rSearchInfo.Reserve(TotalNumberOfPoints);
921 std::vector<double> all_points_coordinates(TotalNumberOfPoints * 3);
922 std::vector<SignedIndexType> all_points_local_ids(TotalNumberOfPoints);
923 std::vector<IndexType> all_points_global_ids(TotalNumberOfPoints);
924
925 // Sync all points first
926 std::vector<int> all_points_ranks = SynchronizePointsWithRanks(itPointBegin, itPointEnd, all_points_coordinates, all_points_local_ids, all_points_global_ids, rDataCommunicator, NumberOfPoints, TotalNumberOfPoints, IndexItIsJustPosition);
927
928 // Some definitions
929 IndexType i_coord = 0;
930
931 // If distributed
932 if (rDataCommunicator.IsDistributed()) {
933 // Prepare MPI data
934 const int rank = rDataCommunicator.Rank();
935 const int world_size = rDataCommunicator.Size();
936 std::vector<int> ranks(1, rank);
937 std::vector<int> inside_ranks(world_size);
938
939 // Fill actual vectors
940 Point point_to_test;
941 for (IndexType i_point = 0; i_point < static_cast<IndexType>(TotalNumberOfPoints); ++i_point) {
942 point_to_test[0] = all_points_coordinates[3 * i_point + 0];
943 point_to_test[1] = all_points_coordinates[3 * i_point + 1];
944 point_to_test[2] = all_points_coordinates[3 * i_point + 2];
945 const bool is_inside = PointIsInsideBoundingBox(rBoundingBox, point_to_test, ThresholdBoundingBox);
946 const int search_rank = all_points_ranks[i_point];
947 const bool to_be_included = is_inside || search_rank == rank;
948 inside_ranks.resize(world_size);
949 if (to_be_included) {
950 for (i_coord = 0; i_coord < 3; ++i_coord) {
951 rSearchInfo.PointCoordinates.push_back(all_points_coordinates[3 * i_point + i_coord]);
952 }
953 rSearchInfo.LocalIndices.push_back(all_points_local_ids[i_point]);
954 rSearchInfo.GlobalIndices.push_back(all_points_global_ids[i_point]);
955 rSearchInfo.GlobalPosition.push_back(i_point);
956 ranks[0] = rank;
957 } else {
958 ranks[0] = -1;
959 }
960 rDataCommunicator.AllGather(ranks, inside_ranks);
961 // Use std::remove_if and vector::erase to remove elements less than 0
962 inside_ranks.erase(
963 std::remove_if(
964 inside_ranks.begin(),
965 inside_ranks.end(),
966 [](const int rank) { return rank < 0; }
967 ),
968 inside_ranks.end()
969 );
970
971 // Now adding
972 if (to_be_included) {
973 rSearchInfo.Ranks.push_back(inside_ranks);
974 }
975 }
976 } else { // Serial
977 // Assign values
978 const std::size_t total_number_of_points = itPointEnd - itPointBegin;
979 for (IndexType i_point = 0 ; i_point < total_number_of_points; ++i_point) {
980 auto it_point = itPointBegin + i_point;
981 const auto& r_coordinates = it_point->Coordinates();
982 for (i_coord = 0; i_coord < 3; ++i_coord) {
983 rSearchInfo.PointCoordinates.push_back(r_coordinates[i_coord]);
984 }
985 rSearchInfo.LocalIndices.push_back(all_points_local_ids[i_point]);
986 rSearchInfo.GlobalIndices.push_back(all_points_global_ids[i_point]);
987 rSearchInfo.GlobalPosition.push_back(i_point);
988 rSearchInfo.Ranks.push_back({0});
989 }
990 }
991
992 // Shrink to actual size
993 rSearchInfo.Shrink();
994
995 // Check that size is equal in all partitions
996 rSearchInfo.Check(rDataCommunicator);
997 }
