156 template<class TMatrixType>
157 static double Cofactor(const TMatrixType& rMat, IndexType i, IndexType j)
158 {
159 static_assert(std::is_same<typename TMatrixType::value_type, double>::value, "Bad value type.");
160
161 KRATOS_ERROR_IF(rMat.size1() != rMat.size2() || rMat.size1() == 0) << "Bad matrix dimensions." << std::endl;
162
163 if (rMat.size1() == 1)
164 return 1;
165
166 IndirectArrayType ia1(rMat.size1() - 1), ia2(rMat.size2() - 1);
167
168 // Construct the submatrix structure for the first minor.
169 unsigned i_sub = 0;
170 for (unsigned k = 0; k < rMat.size1(); ++k)
171 if (k != i)
172 ia1(i_sub++) = k;
173
174 unsigned j_sub = 0;
175 for (unsigned k = 0; k < rMat.size2(); ++k)
176 if (k != j)
177 ia2(j_sub++) = k;
178
179 boost::numeric::ublas::matrix_indirect<const TMatrixType, IndirectArrayType> sub_mat(rMat, ia1, ia2);
180 const double first_minor = Det(sub_mat);
181 return ((i + j) % 2) ? -first_minor : first_minor;
182 }

210 template<SizeType TDim>
211 KRATOS_DEPRECATED_MESSAGE("Please use InvertMatrix() instead")
212 static inline BoundedMatrix<double, TDim, TDim> InvertMatrix(
213 const BoundedMatrix<double, TDim, TDim>& rInputMatrix,
214 double& rInputMatrixDet,
215 const double Tolerance = ZeroTolerance
216 )
217 {
218 BoundedMatrix<double, TDim, TDim> inverted_matrix;
219
220 /* Compute Determinant of the matrix */
221 rInputMatrixDet = Det(rInputMatrix);
222
223 if constexpr (TDim == 1) {
224 inverted_matrix(0,0) = 1.0/rInputMatrix(0,0);
225 rInputMatrixDet = rInputMatrix(0,0);
226 } else if constexpr (TDim == 2) {
227 InvertMatrix2(rInputMatrix, inverted_matrix, rInputMatrixDet);
228 } else if constexpr (TDim == 3) {
229 InvertMatrix3(rInputMatrix, inverted_matrix, rInputMatrixDet);
230 } else if constexpr (TDim == 4) {
231 InvertMatrix4(rInputMatrix, inverted_matrix, rInputMatrixDet);
232 } else {
233 KRATOS_ERROR << "Size not implemented. Size: " << TDim << std::endl;
234 }
235
236 // Checking condition number
237 if (Tolerance > 0.0) { // Check is skipped for negative tolerances
238 CheckConditionNumber(rInputMatrix, inverted_matrix, Tolerance);
239 }
240
241 return inverted_matrix;
242 }

250 template<class TMatrix1, class TMatrix2>
251 static inline bool CheckConditionNumber(
252 const TMatrix1& rInputMatrix,
253 TMatrix2& rInvertedMatrix,
254 const double Tolerance = ZeroTolerance,
255 const bool ThrowError = true
256 )
257 {
258 // We want at least 4 significant digits
259 const double max_condition_number = (1.0/Tolerance) * 1.0e-4;
260
261 // Find the condition number to define is inverse is OK
262 const double input_matrix_norm = norm_frobenius(rInputMatrix);
263 const double inverted_matrix_norm = norm_frobenius(rInvertedMatrix);
264
265 // Now the condition number is the product of both norms
266 const double cond_number = input_matrix_norm * inverted_matrix_norm ;
267 // Finally check if the condition number is low enough
268 if (cond_number > max_condition_number) {
269 if (ThrowError) {
270 KRATOS_WATCH(rInputMatrix);
271 KRATOS_ERROR << " Condition number of the matrix is too high!, cond_number = " << cond_number << std::endl;
272 }
273 return false;
274 }
275
276 return true;
277 }

288 template<class TMatrix1, class TMatrix2>
289 static void GeneralizedInvertMatrix(
290 const TMatrix1& rInputMatrix,
291 TMatrix2& rInvertedMatrix,
292 double& rInputMatrixDet,
293 const double Tolerance = ZeroTolerance
294 )
295 {
296 const SizeType size_1 = rInputMatrix.size1();
297 const SizeType size_2 = rInputMatrix.size2();
298
299 if (size_1 == size_2) {
300 InvertMatrix(rInputMatrix, rInvertedMatrix, rInputMatrixDet, Tolerance);
301 } else if (size_1 < size_2) { // Right inverse
302 if (rInvertedMatrix.size1() != size_2 || rInvertedMatrix.size2() != size_1) {
303 rInvertedMatrix.resize(size_2, size_1, false);
304 }
305 const Matrix aux = prod(rInputMatrix, trans(rInputMatrix));
306 Matrix auxInv;
307 InvertMatrix(aux, auxInv, rInputMatrixDet, Tolerance);
308 rInputMatrixDet = std::sqrt(rInputMatrixDet);
309 noalias(rInvertedMatrix) = prod(trans(rInputMatrix), auxInv);
310 } else { // Left inverse
311 if (rInvertedMatrix.size1() != size_2 || rInvertedMatrix.size2() != size_1) {
312 rInvertedMatrix.resize(size_2, size_1, false);
313 }
314 const Matrix aux = prod(trans(rInputMatrix), rInputMatrix);
315 Matrix auxInv;
316 InvertMatrix(aux, auxInv, rInputMatrixDet, Tolerance);
317 rInputMatrixDet = std::sqrt(rInputMatrixDet);
318 noalias(rInvertedMatrix) = prod(auxInv, trans(rInputMatrix));
319 }

342 template<class TMatrix1, class TMatrix2>
343 static void InvertMatrix(
344 const TMatrix1& rInputMatrix,
345 TMatrix2& rInvertedMatrix,
346 double& rInputMatrixDet,
347 const double Tolerance = ZeroTolerance
348 )
349 {
350 KRATOS_DEBUG_ERROR_IF_NOT(rInputMatrix.size1() == rInputMatrix.size2()) << "Matrix provided is non-square" << std::endl;
351
352 const SizeType size = rInputMatrix.size2();
353
354 if(size == 1) {
355 if(rInvertedMatrix.size1() != 1 || rInvertedMatrix.size2() != 1) {
356 rInvertedMatrix.resize(1,1,false);
357 }
358 rInvertedMatrix(0,0) = 1.0/rInputMatrix(0,0);
359 rInputMatrixDet = rInputMatrix(0,0);
360 } else if (size == 2) {
361 InvertMatrix2(rInputMatrix, rInvertedMatrix, rInputMatrixDet);
362 } else if (size == 3) {
363 InvertMatrix3(rInputMatrix, rInvertedMatrix, rInputMatrixDet);
364 } else if (size == 4) {
365 InvertMatrix4(rInputMatrix, rInvertedMatrix, rInputMatrixDet);
366 } else if (std::is_same<TMatrix1, Matrix>::value) {
367
368 const SizeType size1 = rInputMatrix.size1();
369 const SizeType size2 = rInputMatrix.size2();
370 if(rInvertedMatrix.size1() != size1 || rInvertedMatrix.size2() != size2) {
371 rInvertedMatrix.resize(size1, size2,false);
372 }
373
374 Matrix A(rInputMatrix);
375 typedef permutation_matrix<SizeType> pmatrix;
376 pmatrix pm(A.size1());
377 const int singular = lu_factorize(A,pm);
378 rInvertedMatrix.assign( IdentityMatrix(A.size1()));
379 KRATOS_ERROR_IF(singular == 1) << "Matrix is singular: " << rInputMatrix << std::endl;
380 lu_substitute(A, pm, rInvertedMatrix);
381
382 // Calculating determinant
383 rInputMatrixDet = 1.0;
384
385 for (IndexType i = 0; i < size1;++i) {
386 IndexType ki = pm[i] == i ? 0 : 1;
387 rInputMatrixDet *= (ki == 0) ? A(i,i) : -A(i,i);
388 }
389 } else { // Bounded-matrix case
390 const SizeType size1 = rInputMatrix.size1();
391 const SizeType size2 = rInputMatrix.size2();
392
393 Matrix A(rInputMatrix);
394 Matrix invA(rInvertedMatrix);
395
396 typedef permutation_matrix<SizeType> pmatrix;
397 pmatrix pm(size1);
398 const int singular = lu_factorize(A,pm);
399 invA.assign( IdentityMatrix(size1));
400 KRATOS_ERROR_IF(singular == 1) << "Matrix is singular: " << rInputMatrix << std::endl;
401 lu_substitute(A, pm, invA);
402
403 // Calculating determinant
404 rInputMatrixDet = 1.0;
405
406 for (IndexType i = 0; i < size1;++i) {
407 IndexType ki = pm[i] == i ? 0 : 1;
408 rInputMatrixDet *= (ki == 0) ? A(i,i) : -A(i,i);
409 }
410
411 for (IndexType i = 0; i < size1;++i) {
412 for (IndexType j = 0; j < size2;++j) {
413 rInvertedMatrix(i,j) = invA(i,j);
414 }
415 }
416 }
417
418 // Checking condition number
419 if (Tolerance > 0.0) { // Check is skipped for negative tolerances
420 CheckConditionNumber(rInputMatrix, rInvertedMatrix, Tolerance);
421 }
422 }

431 static void InvertMatrix2(
432 const TMatrix1& rInputMatrix,
433 TMatrix2& rInvertedMatrix,
434 double& rInputMatrixDet
435 )
436 {
437 KRATOS_TRY;
438
439 KRATOS_DEBUG_ERROR_IF_NOT(rInputMatrix.size1() == rInputMatrix.size2()) << "Matrix provided is non-square" << std::endl;

463 template<class TMatrix1, class TMatrix2>
464 static void InvertMatrix3(
465 const TMatrix1& rInputMatrix,
466 TMatrix2& rInvertedMatrix,
467 double& rInputMatrixDet
468 )
469 {
470 KRATOS_TRY;
471
472 KRATOS_DEBUG_ERROR_IF_NOT(rInputMatrix.size1() == rInputMatrix.size2()) << "Matrix provided is non-square" << std::endl;
473
474 if(rInvertedMatrix.size1() != 3 || rInvertedMatrix.size2() != 3) {
475 rInvertedMatrix.resize(3,3,false);
476 }
477
478 // Filling the inverted matrix with the algebraic complements
479 // First column
480 rInvertedMatrix(0,0) = rInputMatrix(1,1)*rInputMatrix(2,2) - rInputMatrix(1,2)*rInputMatrix(2,1);
481 rInvertedMatrix(1,0) = -rInputMatrix(1,0)*rInputMatrix(2,2) + rInputMatrix(1,2)*rInputMatrix(2,0);
482 rInvertedMatrix(2,0) = rInputMatrix(1,0)*rInputMatrix(2,1) - rInputMatrix(1,1)*rInputMatrix(2,0);
483
484 // Second column
485 rInvertedMatrix(0,1) = -rInputMatrix(0,1)*rInputMatrix(2,2) + rInputMatrix(0,2)*rInputMatrix(2,1);
486 rInvertedMatrix(1,1) = rInputMatrix(0,0)*rInputMatrix(2,2) - rInputMatrix(0,2)*rInputMatrix(2,0);
487 rInvertedMatrix(2,1) = -rInputMatrix(0,0)*rInputMatrix(2,1) + rInputMatrix(0,1)*rInputMatrix(2,0);
488
489 // Third column
490 rInvertedMatrix(0,2) = rInputMatrix(0,1)*rInputMatrix(1,2) - rInputMatrix(0,2)*rInputMatrix(1,1);
491 rInvertedMatrix(1,2) = -rInputMatrix(0,0)*rInputMatrix(1,2) + rInputMatrix(0,2)*rInputMatrix(1,0);
492 rInvertedMatrix(2,2) = rInputMatrix(0,0)*rInputMatrix(1,1) - rInputMatrix(0,1)*rInputMatrix(1,0);
493
494 // Calculation of determinant (of the input matrix)
495 rInputMatrixDet = rInputMatrix(0,0)*rInvertedMatrix(0,0) + rInputMatrix(0,1)*rInvertedMatrix(1,0) + rInputMatrix(0,2)*rInvertedMatrix(2,0);
496
497 // Finalizing the calculation of the inverted matrix
498 rInvertedMatrix /= rInputMatrixDet;
499
500 KRATOS_CATCH("")
501 }

835

1682 template<class TMatrixType1, class TMatrixType2>
1683 static inline bool GaussSeidelEigenSystem(
1684 const TMatrixType1& rA,
1685 TMatrixType2& rEigenVectorsMatrix,
1686 TMatrixType2& rEigenValuesMatrix,
1687 const double Tolerance = 1.0e-18,
1688 const SizeType MaxIterations = 20
1689 )
1690 {
1691 bool is_converged = false;
1692
1693 const SizeType size = rA.size1();
1694
1695 if (rEigenVectorsMatrix.size1() != size || rEigenVectorsMatrix.size2() != size)
1696 rEigenVectorsMatrix.resize(size, size, false);
1697 if (rEigenValuesMatrix.size1() != size || rEigenValuesMatrix.size2() != size)
1698 rEigenValuesMatrix.resize(size, size, false);
1699
1700 const TMatrixType2 identity_matrix = IdentityMatrix(size);
1701 noalias(rEigenVectorsMatrix) = identity_matrix;
1702 noalias(rEigenValuesMatrix) = rA;
1703
1704 // Auxiliar values
1705 TMatrixType2 aux_A, aux_V_matrix, rotation_matrix;
1706 double a, u, c, s, gamma, teta;
1707 IndexType index1, index2;
1708
1709 aux_A.resize(size,size,false);
1710 aux_V_matrix.resize(size,size,false);
1711 rotation_matrix.resize(size,size,false);
1712
1713 for(IndexType iterations = 0; iterations < MaxIterations; ++iterations) {
1714 is_converged = true;
1715
1716 a = 0.0;
1717 index1 = 0;
1718 index2 = 1;
1719
1720 for(IndexType i = 0; i < size; ++i) {
1721 for(IndexType j = (i + 1); j < size; ++j) {
1722 if((std::abs(rEigenValuesMatrix(i, j)) > a ) && (std::abs(rEigenValuesMatrix(i, j)) > Tolerance)) {
1723 a = std::abs(rEigenValuesMatrix(i,j));
1724 index1 = i;
1725 index2 = j;
1726 is_converged = false;
1727 }
1728 }
1729 }
1730
1731 if(is_converged) {
1732 break;
1733 }
1734
1735 // Calculation of Rotation angle
1736 gamma = (rEigenValuesMatrix(index2, index2)-rEigenValuesMatrix(index1, index1)) / (2 * rEigenValuesMatrix(index1, index2));
1737 u = 1.0;
1738
1739 if(std::abs(gamma) > Tolerance && std::abs(gamma)< (1.0/Tolerance)) {
1740 u = gamma / std::abs(gamma) * 1.0 / (std::abs(gamma) + std::sqrt(1.0 + gamma * gamma));
1741 } else {
1742 if (std::abs(gamma) >= (1.0/Tolerance)) {
1743 u = 0.5 / gamma;
1744 }
1745 }
1746
1747 c = 1.0 / (std::sqrt(1.0 + u * u));
1748 s = c * u;
1749 teta = s / (1.0 + c);
1750
1751 // Rotation of the Matrix
1752 noalias(aux_A) = rEigenValuesMatrix;
1753 aux_A(index2, index2) = rEigenValuesMatrix(index2,index2) + u * rEigenValuesMatrix(index1, index2);
1754 aux_A(index1, index1) = rEigenValuesMatrix(index1,index1) - u * rEigenValuesMatrix(index1, index2);
1755 aux_A(index1, index2) = 0.0;
1756 aux_A(index2, index1) = 0.0;
1757
1758 for(IndexType i = 0; i < size; ++i) {
1759 if((i!= index1) && (i!= index2)) {
1760 aux_A(index2, i) = rEigenValuesMatrix(index2, i) + s * (rEigenValuesMatrix(index1, i)- teta * rEigenValuesMatrix(index2, i));
1761 aux_A(i, index2) = rEigenValuesMatrix(index2, i) + s * (rEigenValuesMatrix(index1, i)- teta * rEigenValuesMatrix(index2, i));
1762 aux_A(index1, i) = rEigenValuesMatrix(index1, i) - s * (rEigenValuesMatrix(index2, i) + teta * rEigenValuesMatrix(index1, i));
1763 aux_A(i, index1) = rEigenValuesMatrix(index1, i) - s * (rEigenValuesMatrix(index2, i) + teta * rEigenValuesMatrix(index1, i));
1764 }
1765 }
1766
1767 noalias(rEigenValuesMatrix) = aux_A;
1768
1769 // Calculation of the eigeneigen vector matrix V
1770 noalias(rotation_matrix) = identity_matrix;
1771 rotation_matrix(index2, index1) = -s;
1772 rotation_matrix(index1, index2) = s;
1773 rotation_matrix(index1, index1) = c;
1774 rotation_matrix(index2, index2) = c;
1775
1776 noalias(aux_V_matrix) = ZeroMatrix(size, size);
1777
1778 for(IndexType i = 0; i < size; ++i) {
1779 for(IndexType j = 0; j < size; ++j) {
1780 for(IndexType k = 0; k < size; ++k) {
1781 aux_V_matrix(i, j) += rEigenVectorsMatrix(i, k) * rotation_matrix(k, j);
1782 }
1783 }
1784 }
1785 noalias(rEigenVectorsMatrix) = aux_V_matrix;
1786 }
1787
1788 KRATOS_WARNING_IF("MathUtils::EigenSystem", !is_converged) << "Spectral decomposition not converged " << std::endl;
1789
1790 return is_converged;
1791 }
